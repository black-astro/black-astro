public partial class MainViewModel : ObservableObject
{
    [ObservableProperty] private string _keyword = "";
    [ObservableProperty] private bool _isBusy;

    public ObservableCollection<OrderRow> Rows { get; } = new();

    [RelayCommand(CanExecute = nameof(CanSearch))]
    private async Task SearchAsync(CancellationToken ct)
    {
        IsBusy = true;
        try
        {
            var rows = await _repo.FindAsync(Keyword, ct);
            Rows.Clear();
            foreach (var r in rows) Rows.Add(r);
        }
        finally { IsBusy = false; }
    }

    private bool CanSearch() => !IsBusy && Keyword.Length >= 2;
}

DESK 04

비동기 UI — 멈추지 않게

무거운 작업을 그대로 실행하면 창이 하얗게 굳습니다. 가장 흔한 사고입니다.

기본

UI 스레드에서 직접
var data = _api.GetAll();     // 5초
Grid.ItemsSource = data;
// 5초간 "응답 없음"
await 로 넘기기
var data = await _api.GetAllAsync(ct);
Grid.ItemsSource = data;
// 그동안 창이 살아 있다
// CPU 작업은 스레드로 넘긴다
var result = await Task.Run(() => HeavyParse(bytes), ct);

// 진행률 보고 — UI 스레드로 자동 복귀합니다
var progress = new Progress<int>(p => Bar.Value = p);
await ImportAsync(path, progress, ct);

// 다른 스레드에서 UI 를 만져야 할 때만
Application.Current.Dispatcher.Invoke(() => Status.Text = "완료");

취소와 중복 실행 막기

private CancellationTokenSource? _cts;

private async Task SearchAsync()
{
    _cts?.Cancel();                       // 이전 검색 취소
    _cts = new CancellationTokenSource();
    var ct = _cts.Token;

    try
    {
        await Task.Delay(300, ct);        // 디바운스 — 타이핑이 멈춘 뒤 실행
        var rows = await _repo.FindAsync(Keyword, ct);
        if (ct.IsCancellationRequested) return;
        Apply(rows);
    }
    catch (OperationCanceledException) { /* 정상 취소 */ }
}
.Result·.Wait() 는 UI 에서 데드락을 일으킵니다. UI 스레드가 결과를 기다리는데, 그 결과를 만들려면 UI 스레드로 돌아와야 하기 때문입니다. 끝까지 await 하세요.
UI 프레임워크의 규칙은 하나입니다 — 컨트롤은 만든 스레드에서만 만질 수 있고, 그 스레드는 계속 비어 있어야 합니다.

어떤 작업을 어떻게 넘길까

작업쓸 것스레드를 쓰나흔한 실수
DB 조회 · HTTP 호출 · 파일 IOawait XxxAsync(ct)안 씁니다(IO 완료 대기)Task.Run(() => File.ReadAllText()) — 스레드만 낭비
큰 CSV 파싱 · 이미지 처리 · 압축await Task.Run(...)스레드풀 하나UI 스레드에서 직접 → 굳음
진행률 표시IProgress<T>—매 행마다 보고 → 보고가 더 느림
주기적 갱신(1초마다)PeriodicTimer—System.Timers.Timer 는 UI 스레드가 아님
다른 스레드에서 UI 갱신Dispatcher.InvokeAsync—동기 Invoke 남발 → 느려짐
여러 개를 동시에await Task.WhenAll(...)상황에 따라순차 await → 합계 시간이 그대로 더해짐

진행률 · 취소 · 버튼 잠금을 한 묶음으로

public partial class ImportViewModel : ObservableObject
{
    [ObservableProperty] private int _percent;
    [ObservableProperty] private string _status = "대기";
    [ObservableProperty] private bool _isBusy;          // 버튼 IsEnabled 에 반대로 바인딩

    private CancellationTokenSource? _cts;

    [RelayCommand(CanExecute = nameof(CanImport))]
    private async Task ImportAsync()
    {
        _cts = new CancellationTokenSource();
        IsBusy = true;
        ImportCommand.NotifyCanExecuteChanged();        // 버튼 즉시 잠금

        // 보고를 100ms 로 묶는다 — 매 행 보고하면 UI 가 그것 때문에 느려집니다
        var last = 0L;
        var progress = new Progress<(int done, int total)>(p =>
        {
            var now = Environment.TickCount64;
            if (now - last < 100 && p.done != p.total) return;
            last = now;
            Percent = p.done * 100 / p.total;
            Status  = $"{p.done:N0} / {p.total:N0} 행";
        });

        try
        {
            var rows = await Task.Run(
                () => CsvParser.Parse(_path, progress, _cts.Token), _cts.Token);
            Status = $"완료 — {rows.Count:N0} 행";
        }
        catch (OperationCanceledException) { Status = "취소했습니다"; }
        catch (Exception ex)
        {
            Status = "실패";
            _dialog.Error("가져오기에 실패했습니다.", ex.Message);   // 사용자에게 보이기
            _log.LogError(ex, "import failed: {Path}", _path);
        }
        finally
        {
            IsBusy = false;
            ImportCommand.NotifyCanExecuteChanged();
            _cts.Dispose(); _cts = null;
        }
    }

    private bool CanImport() => !IsBusy;

    [RelayCommand]
    private void Cancel() => _cts?.Cancel();
}
버튼 잠금을 빼먹으면 사용자가 두 번 누릅니다. 느린 작업일수록 반드시 두 번 눌립니다 — 그리고 데이터가 두 번 들어갑니다. IsBusy 하나로 버튼 잠금 · 진행률 표시 · 취소 버튼 노출을 한꺼번에 묶으세요.

사내 도구가 느려지는 이유는 대개 조회 결과를 전부 화면에 올려서입니다.

가상화 켜기

<DataGrid ItemsSource="{Binding Rows}"

가상화를 깨는 것들 — ScrollViewer 안에 DataGrid 를 넣으면 높이가 무한이 되어 전 행을 그립니다. Height="Auto" 와 Grouping 도 같은 문제를 일으킵니다. 목록은 * 높이를 가진 Grid 행에 두세요.

그래도 느리면

  • DB 에서 필터·정렬·페이징 — 10만 행을 가져오지 않는 것이 근본 해결입니다
  • ObservableCollection 대량 추가는 느립니다 — 목록을 새로 만들어 한 번에 교체하세요
  • 컨버터에서 무거운 계산 금지 — 셀마다 호출됩니다
  • 이미지 열은 DecodePixelWidth 로 축소해 로드
// 한 번에 교체 (알림이 한 번만 발생)

// 또는 일시적으로 바인딩 해제

사용자는 10만 행을 보지 않습니다. 검색·필터를 잘 만들어 100행 이하로 좁히게 하는 것이 성능 최적화보다 효과가 큽니다.
가상화는 켜는 것보다 깨뜨리지 않는 것이 어렵습니다 — 부모가 "무한 높이"를 주는 순간 조용히 꺼집니다.

// ① DB 에서 잘라 오기 — 가장 효과가 큽니다
public async Task<PagedResult<OrderRow>> SearchAsync(
    OrderFilter f, int page, int size, CancellationToken ct)
{
    var q = _db.Orders.AsNoTracking()                 // 읽기 전용 → 추적 비용 0
        .Where(o => o.CreatedAt >= f.From && o.CreatedAt < f.To);

    if (f.Keyword is { Length: > 0 } k)
        q = q.Where(o => o.CustomerName.Contains(k));

    var total = await q.CountAsync(ct);

    var rows = await q.OrderByDescending(o => o.CreatedAt)
        .Skip((page - 1) * size).Take(size)
        .Select(o => new OrderRow(                    // ★ 필요한 열만
            o.Id, o.CustomerName, o.Amount, o.CreatedAt))
        .ToListAsync(ct);

    return new(rows, total, page, size);
}

public sealed record VersionInfo(string Latest, string MinSupported, string Url, string Sha256);

async Task CheckUpdateAsync()
{
    var info = await _http.GetFromJsonAsync<VersionInfo>(UpdateUrl);
    var cur  = Assembly.GetExecutingAssembly().GetName().Version!;

    if (IsOlder(cur, Version.Parse(info!.MinSupported)))
    {
        // 서버 API 가 깨지는 변경 — 업데이트 전에는 못 쓰게
        await ForceUpdateAsync(info);
        return;
    }
    if (IsOlder(cur, Version.Parse(info.Latest)))
        ShowUpdateBanner(info);        // 작업을 방해하지 않는 배너
}

DB 직접 연결의 위험

클라이언트에 DB 접속 정보를 넣지 마세요. exe 는 얼마든지 뜯어볼 수 있습니다. 접속 문자열을 넣는 순간 사내 DB 가 전 직원에게 열립니다.
  • API 서버를 한 겹 두세요 — 권한·감사 로그를 그곳에서 처리
  • 불가피하면 읽기 전용 계정 + 필요한 뷰만 노출
  • Windows 통합 인증을 쓰면 비밀번호를 앱이 들고 있지 않아도 됩니다
  • 감사 로그 — 누가 언제 무엇을 조회했는지 남기세요
API 서버는 🔷 ASP.NET Core 탭으로 바로 만들 수 있습니다. 데스크탑 앱 + 작은 API 조합이 사내 도구의 가장 안전한 형태입니다.

// ① 창 위치 복원 — 없는 모니터로 가지 않게
protected override void OnSourceInitialized(EventArgs e)
{
    base.OnSourceInitialized(e);
    var s = Settings.Default;
    if (s.WinWidth <= 0) return;

    var rect = new Rect(s.WinLeft, s.WinTop, s.WinWidth, s.WinHeight);
    var visible = SystemParameters.VirtualScreenWidth > 0 &&
                  rect.Left > SystemParameters.VirtualScreenLeft - rect.Width + 80 &&
                  rect.Top  >= SystemParameters.VirtualScreenTop &&
                  rect.Left < SystemParameters.VirtualScreenLeft +
                              SystemParameters.VirtualScreenWidth - 80;

    if (!visible) { WindowStartupLocation = WindowStartupLocation.CenterScreen; return; }
    Left = rect.Left; Top = rect.Top; Width = rect.Width; Height = rect.Height;
}

// ② 전역 예외 — 조용히 죽지 않게
public App()
{
    DispatcherUnhandledException += (s, e) =>
    {
        Log.Fatal(e.Exception, "unhandled (ui)");
        MessageBox.Show($"예기치 못한 오류가 발생했습니다.\n\n" +
            $"로그: {LogPath}\n담당자에게 이 파일을 보내 주세요.",
            "오류", MessageBoxButton.OK, MessageBoxImage.Error);
        e.Handled = true;                  // 앱을 살려 둘지는 상황에 따라
    };
    TaskScheduler.UnobservedTaskException += (s, e) =>
    {
        Log.Error(e.Exception, "unhandled (task)");
        e.SetObserved();
    };
    AppDomain.CurrentDomain.UnhandledException += (s, e) =>
        Log.Fatal(e.ExceptionObject as Exception, "unhandled (domain)");
}

private static Mutex? _mutex;
protected override void OnStartup(StartupEventArgs e)
{
    _mutex = new Mutex(true, @"Global\MyTool", out var isNew);
    if (!isNew)
    {
        MessageBox.Show("이미 실행 중입니다.");
        Shutdown(); return;
    }
    base.OnStartup(e);
}
