DESK 03

데이터 바인딩 · MVVM

WPF 의 핵심. 화면과 데이터를 연결해 두면 값만 바꿔도 화면이 따라옵니다.

ViewModel

dotnet add package CommunityToolkit.Mvvm
public partial class MainViewModel : ObservableObject
{
    [ObservableProperty] private string _keyword = "";
    [ObservableProperty] private bool _isBusy;

    public ObservableCollection<OrderRow> Rows { get; } = new();

    [RelayCommand(CanExecute = nameof(CanSearch))]
    private async Task SearchAsync(CancellationToken ct)
    {
        IsBusy = true;
        try
        {
            var rows = await _repo.FindAsync(Keyword, ct);
            Rows.Clear();
            foreach (var r in rows) Rows.Add(r);
        }
        finally { IsBusy = false; }
    }

    private bool CanSearch() => !IsBusy && Keyword.Length >= 2;
}

XAML 에서 연결

<Window.DataContext>
  <vm:MainViewModel/>
</Window.DataContext>

<TextBox Text="{Binding Keyword, UpdateSourceTrigger=PropertyChanged}"/>

<Button Content="검색" Command="{Binding SearchCommand}"/>
<!-- CanExecute 가 false 면 버튼이 자동으로 비활성화됩니다 -->

<ProgressBar IsIndeterminate="True"
             Visibility="{Binding IsBusy,
                          Converter={StaticResource BoolToVis}}"/>

<DataGrid ItemsSource="{Binding Rows}" AutoGenerateColumns="False">
  <DataGrid.Columns>
    <DataGridTextColumn Header="주문번호" Binding="{Binding OrderNo}"/>
    <DataGridTextColumn Header="금액" Binding="{Binding Amount, StringFormat=N0}"/>
  </DataGrid.Columns>
</DataGrid>
MVVM 을 쓰는 진짜 이유는 테스트입니다. 화면 없이 ViewModel 만 테스트할 수 있어 "버튼을 눌렀을 때 무엇이 일어나는가"를 자동으로 검증할 수 있습니다.
바인딩 오류는 조용히 실패합니다. 속성 이름을 틀려도 화면만 비어 있을 뿐 예외가 없습니다. 출력 창의 바인딩 오류를 항상 확인하고, 개발 중에는 PresentationTraceSources.TraceLevel=High 로 자세히 보세요.
DESK 04

비동기 UI — 멈추지 않게

무거운 작업을 그대로 실행하면 창이 하얗게 굳습니다. 가장 흔한 사고입니다.

기본

UI 스레드에서 직접
var data = _api.GetAll();     // 5초
Grid.ItemsSource = data;
// 5초간 "응답 없음"
await 로 넘기기
var data = await _api.GetAllAsync(ct);
Grid.ItemsSource = data;
// 그동안 창이 살아 있다
// CPU 작업은 스레드로 넘긴다
var result = await Task.Run(() => HeavyParse(bytes), ct);

// 진행률 보고 — UI 스레드로 자동 복귀합니다
var progress = new Progress<int>(p => Bar.Value = p);
await ImportAsync(path, progress, ct);

// 다른 스레드에서 UI 를 만져야 할 때만
Application.Current.Dispatcher.Invoke(() => Status.Text = "완료");

취소와 중복 실행 막기

private CancellationTokenSource? _cts;

private async Task SearchAsync()
{
    _cts?.Cancel();                       // 이전 검색 취소
    _cts = new CancellationTokenSource();
    var ct = _cts.Token;

    try
    {
        await Task.Delay(300, ct);        // 디바운스 — 타이핑이 멈춘 뒤 실행
        var rows = await _repo.FindAsync(Keyword, ct);
        if (ct.IsCancellationRequested) return;
        Apply(rows);
    }
    catch (OperationCanceledException) { /* 정상 취소 */ }
}
.Result·.Wait() 는 UI 에서 데드락을 일으킵니다. UI 스레드가 결과를 기다리는데, 그 결과를 만들려면 UI 스레드로 돌아와야 하기 때문입니다. 끝까지 await 하세요.
DESK 05

대용량 목록 — 10만 행을 끊김 없이

사내 도구가 느려지는 이유는 대개 조회 결과를 전부 화면에 올려서입니다.

가상화 켜기

<DataGrid ItemsSource="{Binding Rows}"
          EnableRowVirtualization="True"
          EnableColumnVirtualization="True"
          VirtualizingPanel.IsVirtualizing="True"
          VirtualizingPanel.VirtualizationMode="Recycling"
          ScrollViewer.IsDeferredScrollingEnabled="True"/>
가상화를 깨는 것들 — ScrollViewer 안에 DataGrid 를 넣으면 높이가 무한이 되어 전 행을 그립니다. Height="Auto" 와 Grouping 도 같은 문제를 일으킵니다. 목록은 * 높이를 가진 Grid 행에 두세요.

그래도 느리면

  • DB 에서 필터·정렬·페이징 — 10만 행을 가져오지 않는 것이 근본 해결입니다
  • ObservableCollection 대량 추가는 느립니다 — 목록을 새로 만들어 한 번에 교체하세요
  • 컨버터에서 무거운 계산 금지 — 셀마다 호출됩니다
  • 이미지 열은 DecodePixelWidth 로 축소해 로드
// 한 번에 교체 (알림이 한 번만 발생)
Rows = new ObservableCollection<Row>(loaded);
OnPropertyChanged(nameof(Rows));

// 또는 일시적으로 바인딩 해제
Grid.ItemsSource = null;
Grid.ItemsSource = loaded;
사용자는 10만 행을 보지 않습니다. 검색·필터를 잘 만들어 100행 이하로 좁히게 하는 것이 성능 최적화보다 효과가 큽니다.

SQLite

dotnet add package Microsoft.EntityFrameworkCore.Sqlite
public sealed class LocalDb : DbContext
{
    public DbSet<CacheRow> Rows => Set<CacheRow>();

    protected override void OnConfiguring(DbContextOptionsBuilder b)
    {
        var dir = Path.Combine(
            Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
            "MyTool");
        Directory.CreateDirectory(dir);
        b.UseSqlite($"Data Source={Path.Combine(dir, "local.db")}");
    }
}

// 시작 시 스키마 반영

// 두 번 실행 막기 (설정 파일·DB 동시 접근 사고 방지)
private static Mutex? _mutex;

protected override void OnStartup(StartupEventArgs e)
{
    _mutex = new Mutex(true, @"Global\MyTool.SingleInstance", out var isNew);
    if (!isNew)
    {
        MessageBox.Show("이미 실행 중입니다.");
        Shutdown();
        return;
    }
    base.OnStartup(e);
}
// 로그인 시 자동 시작 — 사용자가 켜게 하세요 (기본값 끄기)
const string RunKey = @"Software\Microsoft\Windows\CurrentVersion\Run";

void SetAutoStart(bool on)
{
    using var key = Registry.CurrentUser.OpenSubKey(RunKey, writable: true)!;
    if (on) key.SetValue("MyTool", $"\"{Environment.ProcessPath}\" --tray");
    else    key.DeleteValue("MyTool", throwOnMissingValue: false);
}

서명이 없으면 막힙니다

서명하지 않은 exe 는 SmartScreen 이 막고 백신이 격리합니다. 사내 배포라면 최소한 정보보안팀에 해시를 등록해 예외 처리를 받으세요. 외부 배포라면 코드 서명 인증서가 사실상 필수입니다.
  • MSIX — 설치·업데이트·제거가 깔끔하고 서명이 전제입니다
  • 사내 배포는 인트라넷 공유 폴더 + 자동 업데이트 조합이 흔합니다
  • 32/64비트 — 요즘은 x64 만으로 충분합니다(ARM64 는 필요 시)
  • .NET 런타임 버전을 사내 표준으로 정해 두면 용량을 아낄 수 있습니다
같은 고민을 다른 스택에서는 — 🟨 JS·TS 가이드의 Electron 코드 서명, ☕ 자바 가이드의 jpackage 가 같은 자리를 다룹니다.
DESK 10

자동 업데이트 · 크래시 수집

사내 PC 50대의 버전이 제각각이 되는 순간 "재현이 안 되는 버그"가 시작됩니다.

업데이트 확인

public sealed record VersionInfo(string Latest, string MinSupported, string Url, string Sha256);

async Task CheckUpdateAsync()
{
    var info = await _http.GetFromJsonAsync<VersionInfo>(UpdateUrl);
    var cur  = Assembly.GetExecutingAssembly().GetName().Version!;

    if (IsOlder(cur, Version.Parse(info!.MinSupported)))
    {
        // 서버 API 가 깨지는 변경 — 업데이트 전에는 못 쓰게
        await ForceUpdateAsync(info);
        return;
    }
    if (IsOlder(cur, Version.Parse(info.Latest)))
        ShowUpdateBanner(info);        // 작업을 방해하지 않는 배너
}
강제 업데이트 스위치를 첫 버전부터 넣으세요. 서버 API 를 바꿔야 할 때 유일한 탈출구입니다. 내려받은 파일은 해시를 검증한 뒤 교체하세요.

크래시 · 로그

protected override void OnStartup(StartupEventArgs e)
{
    AppDomain.CurrentDomain.UnhandledException += (_, ev) =>
        Fatal((Exception)ev.ExceptionObject);

    DispatcherUnhandledException += (_, ev) =>
    {
        Fatal(ev.Exception);
        ev.Handled = true;             // 창은 살려 두고 안내
        MessageBox.Show("오류가 발생했습니다. 로그가 저장되었습니다.");
    };

    TaskScheduler.UnobservedTaskException += (_, ev) => Fatal(ev.Exception);
}

static void Fatal(Exception ex)
{
    var path = Path.Combine(LogDir, $"crash_{DateTime.Now:yyyyMMdd_HHmmss}.log");
    File.WriteAllText(path, ex.ToString());
}
로그를 사용자가 보내기 쉽게 만드세요. "로그 폴더 열기" 버튼 하나만 있어도 지원 시간이 크게 줄어듭니다. 가능하면 버튼 하나로 서버에 전송하게 하세요.

사내 시스템 연동

데스크탑 앱을 쓰는 진짜 이유. 로컬 장비와 사내 서버에 붙습니다.

자주 나오는 요구

요구방법
바코드·시리얼 장비SerialPort · 벤더 SDK(P/Invoke)
라벨 프린터ZPL 문자열을 직접 전송
사내 DB 직접 조회읽기 전용 계정 + 뷰
공유 폴더 감시FileSystemWatcher
사내 인증(AD)Windows 통합 인증
 
// 폴더 감시 — 파일이 들어오면 자동 처리
var w = new FileSystemWatcher(dropDir, "*.csv") { EnableRaisingEvents = true };
w.Created += async (_, e) =>
{
    await WaitUntilReadyAsync(e.FullPath);   // ★ 복사가 끝날 때까지 대기
    await ImportAsync(e.FullPath);
};
파일 생성 이벤트는 "복사 완료"가 아닙니다. 큰 파일은 쓰는 중에 이벤트가 옵니다. 열어 보고 잠겨 있으면 잠시 뒤 재시도하는 로직이 반드시 필요합니다.

DB 직접 연결의 위험

클라이언트에 DB 접속 정보를 넣지 마세요. exe 는 얼마든지 뜯어볼 수 있습니다. 접속 문자열을 넣는 순간 사내 DB 가 전 직원에게 열립니다.
  • API 서버를 한 겹 두세요 — 권한·감사 로그를 그곳에서 처리
  • 불가피하면 읽기 전용 계정 + 필요한 뷰만 노출
  • Windows 통합 인증을 쓰면 비밀번호를 앱이 들고 있지 않아도 됩니다
  • 감사 로그 — 누가 언제 무엇을 조회했는지 남기세요
API 서버는 🔷 ASP.NET Core 탭으로 바로 만들 수 있습니다. 데스크탑 앱 + 작은 API 조합이 사내 도구의 가장 안전한 형태입니다.
