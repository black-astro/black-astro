자바에서 배치라고 하면 Spring Batch + Quartz 가 거의 정답이었습니다. .NET 에는 그 자리를 통째로 채우는 프레임워크가 없습니다. 대신 조각들이 있고, 조각을 고르는 기준이 있습니다. 이 섹션이 그 기준입니다.

먼저 사실부터 — Spring Batch 에 1:1 대응하는 것은 없습니다. "닷넷 배치 프레임워크"를 검색하면 나오는 것들(Hangfire · Quartz.NET · Coravel)은 전부 스케줄러이거나 작업 큐입니다. Worker Service — 상주하는 배치 프로세스 만들기

스프링에서 @Scheduled 를 단 빈이나 ApplicationRunner 로 하던 일입니다. .NET 에서는 BackgroundService 한 클래스가 그 자리를 다 맡습니다. 중요한 것은 어떻게 멈추는가 와 두 번 뜨지 않게 하는가 입니다.

Worker 수명주기 — 종료 신호를 받고도 30초. 그 안에 끝나거나, 체크포인트를 남기고 나가야 합니다.

public sealed class SettleWorker(
    ILogger<SettleWorker> log,
    IServiceScopeFactory scopes) : BackgroundService
{
    // 5분마다 한 바퀴. PeriodicTimer 가 Task.Delay 보다 정확합니다
    private readonly PeriodicTimer _tick =
        new(TimeSpan.FromMinutes(5));

    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        log.LogInformation("워커 기동 {Machine}", Environment.MachineName);

        // 첫 바퀴를 바로 돌지, 5분 뒤에 돌지 — 의도적으로 고르세요
        while (await _tick.WaitForNextTickAsync(ct))
        {
            // ★ 스코프를 매 바퀴 새로 엽니다 (아래 note 참고)
            using var scope = scopes.CreateScope();
            var svc = scope.ServiceProvider
                           .GetRequiredService<ISettleService>();
            try
            {
                await svc.RunOnceAsync(ct);
            }
            catch (OperationCanceledException) when (ct.IsCancellationRequested)
            {
                throw;                       // 종료 신호는 그대로 올려보냅니다
            }
            catch (Exception ex)
            {
                // ★ 삼키지 않으면 프로세스가 죽습니다
                log.LogError(ex, "한 바퀴 실패 — 다음 주기에 다시 시도합니다");
            }
        }
    }

    public override async Task StopAsync(CancellationToken ct)
    {
        log.LogInformation("종료 신호 수신 — 정리 시작");
        _tick.Dispose();
        await base.StopAsync(ct);
    }
}

SQL Server 애플리케이션 락 — 실제 코드

public sealed class SqlAppLock : IAsyncDisposable
{
    private readonly SqlConnection _cn;
    private SqlAppLock(SqlConnection cn) => _cn = cn;

    /// <summary>못 잡으면 null — 이미 다른 서버가 돌고 있다는 뜻</summary>
    public static async Task<SqlAppLock?> TryAcquireAsync(
        string cs, string name, TimeSpan wait, CancellationToken ct)
    {
        var cn = new SqlConnection(cs);
        await cn.OpenAsync(ct);

        await using var cmd = cn.CreateCommand();
        cmd.CommandText = "sp_getapplock";
        cmd.CommandType = CommandType.StoredProcedure;
        cmd.Parameters.AddWithValue("@Resource", name);
        cmd.Parameters.AddWithValue("@LockMode", "Exclusive");
        cmd.Parameters.AddWithValue("@LockOwner", "Session");
        cmd.Parameters.AddWithValue("@LockTimeout", (int)wait.TotalMilliseconds);

        var rv = cmd.Parameters.Add("@rv", SqlDbType.Int);
        rv.Direction = ParameterDirection.ReturnValue;
        await cmd.ExecuteNonQueryAsync(ct);

        // 0 = 즉시 획득, 1 = 대기 후 획득, 음수 = 실패
        if ((int)rv.Value < 0) { await cn.DisposeAsync(); return null; }
        return new SqlAppLock(cn);
    }

    // 커넥션을 닫으면 락도 풀립니다 — 서버가 죽어도 자동 해제
    public ValueTask DisposeAsync() => _cn.DisposeAsync();
}

Windows Mutex 는 서버 간 중복을 못 막습니다. new Mutex(true, @"Global\SettleBatch") 는 그 머신 안에서만 유일합니다. 이중화한 배치 서버 2대는 서로를 전혀 모릅니다. "뮤텍스 걸었는데 두 번 돌았다"는 사고가 여기서 나옵니다.

종료 신호를 끝까지 전파하기 🔴 고급

CancellationToken 은 인자로 계속 넘기지 않으면 아무 일도 안 합니다. 자바의 Thread.interrupt() 와 달리 스레드를 건드리는 것이 아니라 "멈춰 달라"는 값일 뿐이라, 받는 쪽이 확인해야 멈춥니다. 아래 세 곳을 빠뜨리면 Ctrl+C 를 눌러도 안 죽는 배치가 됩니다. 

// ① 모든 async 호출에 토큰을 넘긴다 — 하나라도 빠지면 거기서 멈춥니다
await db.SaveChangesAsync(ct);
await http.GetAsync(url, ct);
await stream.ReadAsync(buf, ct);
await foreach (var row in ReadCsvAsync(path).WithCancellation(ct)) { ... }

// ② 루프 안에서 스스로 확인한다 (CPU 바운드 구간엔 await 가 없으므로)
foreach (var chunk in chunks)
{
    ct.ThrowIfCancellationRequested();     // 여기서 던져 나갑니다
    Process(chunk);
}

// ③ 긴 배치는 "안전 지점"에서만 나간다 — 청크 경계
foreach (var chunk in chunks)
{
    await ProcessAndCommitAsync(chunk, CancellationToken.None); // 이 안은 끝까지
    await SaveCheckpointAsync(chunk.LastId, CancellationToken.None);

    if (ct.IsCancellationRequested)        // 커밋 직후에만 빠져나옵니다
    {

        break;                             // 다음 실행이 여기서 이어받습니다
    }

왜 ③에서 CancellationToken.None 을 쓰나요? 청크 한 개를 처리하는 도중에 취소되면 DB 는 커밋됐는데 체크포인트는 안 남는 어중간한 상태가 됩니다. 그래서 "한 청크는 무조건 끝까지, 대신 청크 경계에서만 나간다"로 설계합니다. ShutdownTimeout 을 청크 하나의 최대 소요 시간보다 길게 잡아야 하는 이유이기도 합니다. Spring Batch 의 StepExecution 종료 처리도 같은 원리로 청크 경계에서만 멈춥니다.
Windows 서비스 · systemd 로 등록하는 방법은 🖧 서버 구축 탭(h04 · h05) 에서 sc.exe create · 유닛 파일 전문과 함께 다룹니다. 여기서는 AddWindowsService() / AddSystemd() 한 줄이면 같은 실행 파일이 콘솔로도 서비스로도 뜬다는 것만 기억하세요 — 개발 중에는 dotnet run, 운영에서는 서비스입니다.
정리 — BackgroundService 는 싱글턴이니 바퀴마다 스코프를 열고, 종료 토큰을 끝까지 인자로 넘기며, 이중화했다면 sp_getapplock 으로 한 대만 돌게 합니다.

청크 처리 — Spring Batch 를 300줄로 다시 짓기

Reader-Processor-Writer, 커밋 단위, 재시작 지점. Spring Batch 가 대신 해 주던 것을 직접 만드는 섹션입니다. 만들고 나면 오히려 "이게 다였나" 싶어집니다 — 대신 내 도메인에 정확히 맞습니다.

// ★ 400만 행이어도 메모리에 1건씩만 올라옵니다
public static async IAsyncEnumerable<TradeRow> ReadAsync(
    string path,
    long skipLines,                              // 체크포인트에서 이어받기
    [EnumeratorCancellation] CancellationToken ct = default)
{
    using var fs = new FileStream(path, FileMode.Open, FileAccess.Read,
                                  FileShare.Read, 64 * 1024,
                                  FileOptions.SequentialScan);
    using var sr = new StreamReader(fs, Encoding.UTF8);

    long lineNo = 0;
    while (await sr.ReadLineAsync(ct) is { } line)
    {
        lineNo++;
        if (lineNo <= skipLines) continue;       // 이미 처리한 구간은 건너뜀
        if (line.Length == 0) continue;

        yield return TradeRow.Parse(line, lineNo);
    }
}
[EnumeratorCancellation] 를 빠뜨리면 호출부의 .WithCancellation(ct) 가 조용히 무시됩니다. 컴파일 경고도 안 납니다(분석기를 켜면 CA2016 이 잡아 줍니다). IAsyncEnumerable 을 쓸 때 가장 흔한 실수입니다.

Chunk — 청크 자르기 확장 메서드

public static async IAsyncEnumerable<List<T>> ChunkAsync<T>(
    this IAsyncEnumerable<T> src, int size,
    [EnumeratorCancellation] CancellationToken ct = default)
{
    var buf = new List<T>(size);
    await foreach (var item in src.WithCancellation(ct))
    {
        buf.Add(item);
        if (buf.Count < size) continue;
        yield return buf;
        buf = new List<T>(size);     // ★ 재사용하지 말 것 — 아래 warn
    }
    if (buf.Count > 0) yield return buf;
}
buf.Clear() 로 재사용하면 버그입니다. 소비자가 그 리스트를 비동기로 붙잡고 있는 사이 생산자가 지워 버립니다. 실제로 "가끔 몇 백 건이 사라지는" 사고가 이 한 줄에서 납니다. GC 압박이 걱정되면 ArrayPool<T> 로 명시적으로 빌리고 반납하세요 (f06).

public sealed class ChunkRunner(
    IDbContextFactory<AppDbContext> dbf,
    IBatchJournal journal,
    ILogger<ChunkRunner> log)
{
    public async Task<BatchResult> RunAsync(
        string job, DateOnly bizDate, string step,
        Func<string?, IAsyncEnumerable<TradeRow>> readFrom,
        int chunkSize, int skipLimit, CancellationToken ct)
    {
        var run  = await journal.StartAsync(job, bizDate, ct);
        var last = await journal.LoadCheckpointAsync(job, bizDate, step, ct);
        long read = 0, wrote = 0, skipped = 0;

        log.LogInformation("재시작 지점 {Key} 부터 시작", last?.LastKey ?? "(처음)");

        try
        {
            await foreach (var chunk in readFrom(last?.LastKey)
                                        .ChunkAsync(chunkSize, ct))
            {
                read += chunk.Count;

                // ── ② 변환 · 검증 : 실패 건은 버리지 말고 모읍니다
                var good = new List<Trade>(chunk.Count);
                var bad  = new List<(TradeRow Row, string Why)>();
                foreach (var row in chunk)
                {
                    if (Validate(row) is { } why) bad.Add((row, why));
                    else good.Add(Map(row));
                }

                await using var db = await dbf.CreateDbContextAsync(ct);
                await using var tx = await db.Database
                                             .BeginTransactionAsync(ct);

                await db.BulkInsertAsync(good, ct);          // f05
                await journal.AppendSkipsAsync(db, run.RunId, bad, ct);
                await journal.SaveCheckpointAsync(
                    db, job, bizDate, step,
                    lastKey: chunk[^1].Key, done: read, ct);  // ★ 같은 tx

                await tx.CommitAsync(ct);                    // 여기서 원자적 확정

                wrote   += good.Count;
                skipped += bad.Count;

                if (skipped > skipLimit)
                    throw new BatchAbortException(
                        $"스킵 한도 초과 {skipped} > {skipLimit}");

                if (read % 100_000 == 0)
                    log.LogInformation("진행 {Read}건 · 적재 {Wrote} · 스킵 {Skip}",
                                        read, wrote, skipped);

                if (ct.IsCancellationRequested)              // 청크 경계에서만 탈출
                {
                    await journal.StopAsync(run.RunId, read, wrote, skipped, ct);
                    return BatchResult.Stopped(read, wrote, skipped);
                }
            }

            await journal.SucceedAsync(run.RunId, read, wrote, skipped, ct);
            return BatchResult.Success(read, wrote, skipped);
        }
        catch (Exception ex)
        {
            // 체크포인트는 이미 커밋되어 있으므로 다음 실행이 이어받습니다
            await journal.FailAsync(run.RunId, ex, read, wrote, skipped,
                                    CancellationToken.None);
            throw;
        }
    }
}

public static async Task BulkCopyAsync(
    string cs, IEnumerable<Trade> rows, CancellationToken ct)
{
    await using var cn = new SqlConnection(cs);
    await cn.OpenAsync(ct);

    using var bulk = new SqlBulkCopy(cn, SqlBulkCopyOptions.TableLock, null)
    {
        DestinationTableName = "dbo.Trade_Stage",
        BatchSize            = 10_000,   // 한 번에 서버로 넘길 행 수
        BulkCopyTimeout      = 0,        // 0 = 무제한 (배치에서는 이게 맞습니다)
        EnableStreaming      = true      // ★ 전체를 메모리에 안 올립니다
    };

    // 열 이름이 다르면 반드시 매핑 — 순서만 믿으면 언젠가 깨집니다
    bulk.ColumnMappings.Add(nameof(Trade.TradeNo),  "TRADE_NO");
    bulk.ColumnMappings.Add(nameof(Trade.TradedAt), "TRADED_AT");
    bulk.ColumnMappings.Add(nameof(Trade.Amount),   "AMT");

    // FastMember 로 객체 목록을 IDataReader 로 감쌉니다
    using var reader = ObjectReader.Create(rows,
        nameof(Trade.TradeNo), nameof(Trade.TradedAt), nameof(Trade.Amount));

    await bulk.WriteToServerAsync(reader, ct);
}

public static async IAsyncEnumerable<List<Trade>> ReadByKeysetAsync(
    AppDbContext db, long startAfterId, int size,
    [EnumeratorCancellation] CancellationToken ct = default)
{
    var lastId = startAfterId;                 // 체크포인트에서 받은 값
    while (true)
    {
        var page = await db.Trades
            .AsNoTracking()                    // ★ 배치 읽기는 항상 이것
            .Where(t => t.Id > lastId)         // ★ OFFSET 대신 부등호
            .OrderBy(t => t.Id)                // ★ 인덱스와 같은 순서
            .Take(size)
            .ToListAsync(ct);

        if (page.Count == 0) yield break;

        yield return page;
        lastId = page[^1].Id;                  // 다음 페이지의 시작점
    }
}

static Encoding DetectEncoding(string path)
{
    Span<byte> head = stackalloc byte[4];
    using (var fs = File.OpenRead(path)) fs.ReadExactly(head);

    // ① BOM 이 있으면 그 말을 믿습니다
    if (head[0] == 0xEF && head[1] == 0xBB && head[2] == 0xBF)
        return new UTF8Encoding(false);
    if (head[0] == 0xFF && head[1] == 0xFE) return Encoding.Unicode;
    if (head[0] == 0xFE && head[1] == 0xFF) return Encoding.BigEndianUnicode;

    // ② BOM 이 없으면 — UTF-8 로 엄격하게 디코딩을 시도해 봅니다
    var strict = new UTF8Encoding(false, throwOnInvalidBytes: true);
    try
    {
        using var sr = new StreamReader(path, strict, false);
        for (int i = 0; i < 200 && sr.ReadLine() is not null; i++) { }
        return strict;                         // 200줄이 멀쩡하면 UTF-8
    }
    catch (DecoderFallbackException)
    {
        return Encoding.GetEncoding(949);      // 아니면 CP949 로 봅니다

// PUT /uploads/{uploadId}/chunks/{index}
app.MapPut("/uploads/{id}/chunks/{index:int}", async (
    string id, int index, HttpRequest req, IUploadStore store,
    CancellationToken ct) =>
{
    var meta = await store.GetAsync(id, ct);
    if (meta is null) return Results.NotFound();

    // 이미 받은 청크면 그냥 성공 — ★ 멱등성
    if (meta.Received.Contains(index)) return Results.Ok(meta.Progress());

    var path = store.ChunkPath(id, index);
    await using (var fs = File.Create(path))
        await req.Body.CopyToAsync(fs, 1024 * 1024, ct);

    // 체크섬으로 무결성 확인 (클라이언트가 헤더로 보냅니다)
    if (req.Headers.TryGetValue("X-Chunk-SHA256", out var want))
    {
        var got = Convert.ToHexString(
            await SHA256.HashDataAsync(File.OpenRead(path), ct));
        if (!got.Equals(want!, StringComparison.OrdinalIgnoreCase))
        { File.Delete(path); return Results.StatusCode(422); }
    }

    await store.MarkReceivedAsync(id, index, ct);
    return Results.Ok(meta.Progress());
});

async Task ScanFolderAsync(CancellationToken ct)
{
    foreach (var path in Directory.EnumerateFiles(_in, "*.dat"))
    {
        // ① 배타 열기 시도 — 송신측이 아직 쓰고 있으면 실패합니다 (Windows)
        if (!TryOpenExclusive(path, out var fs)) continue;
        fs!.Dispose();

        // ② 크기가 두 번 연속 같은지 (리눅스·NFS 에서는 이 방법이 필요)
        var len1 = new FileInfo(path).Length;
        await Task.Delay(1000, ct);
        if (new FileInfo(path).Length != len1) continue;   // 아직 자라는 중

        // ③ 내가 맡는다 — work 폴더로 이동. 실패하면 남이 가져간 것
        var work = Path.Combine(_work, Path.GetFileName(path));
        try { File.Move(path, work, overwrite: false); }
        catch (IOException) { continue; }                  // 정상 — 경쟁에서 짐

        await HandleAsync(work, ct);
    }
}

static bool TryOpenExclusive(string path, out FileStream? fs)
{
    try { fs = new FileStream(path, FileMode.Open, FileAccess.Read, FileShare.None); return true; }
    catch (IOException) { fs = null; return false; }       // 잠겨 있음
}

// 트레일러 검증 — 이 30줄이 "데이터가 잘렸다"를 100% 잡습니다
public static async Task<FileStats> VerifyAsync(
    string path, Encoding enc, CancellationToken ct)
{
    string? header = null, trailer = null;
    long body = 0; decimal sum = 0m;

    using var sr = new StreamReader(path, enc);
    while (await sr.ReadLineAsync(ct) is { } line)
    {
        if (line.Length == 0) continue;
        switch (line[0])
        {
            case 'H': header  = line; break;
            case 'T': trailer = line; break;
            default:
                body++;
                sum += decimal.Parse(line.Split('|')[3],
                                     CultureInfo.InvariantCulture);
                break;
        }
    }

    if (header  is null) throw new BatchAbortException("헤더 레코드 없음 — 파일이 잘렸습니다");
    if (trailer is null) throw new BatchAbortException("트레일러 없음 — 전송이 완료되지 않았습니다");

    var t = trailer.Split('|');
    var declaredCount = long.Parse(t[1]);
    var declaredSum   = decimal.Parse(t[2], CultureInfo.InvariantCulture);

    if (declaredCount != body)
        throw new BatchAbortException($"건수 불일치 — 선언 {declaredCount:N0} / 실제 {body:N0}");
    if (declaredSum != sum)
        throw new BatchAbortException($"합계 불일치 — 선언 {declaredSum:N2} / 실제 {sum:N2}");

    return new FileStats(body, sum);
}

public async Task RunPipelineAsync(string path, CancellationToken ct)
{
    // 채널 두 개로 3단을 잇습니다
    var raw  = Channel.CreateBounded<TradeRow>(new BoundedChannelOptions(2_000)
                   { FullMode = BoundedChannelFullMode.Wait, SingleWriter = true });
    var done = Channel.CreateBounded<Trade>(new BoundedChannelOptions(2_000)
                   { FullMode = BoundedChannelFullMode.Wait });

    // ── ① 읽기 (IO 바운드) — 1개면 충분합니다. 디스크는 하나니까요
    var read = Task.Run(async () =>
    {
        try
        {
            await foreach (var r in ReadCsvAsync(path).WithCancellation(ct))
                await raw.Writer.WriteAsync(r, ct);
        }
        catch (Exception ex) { raw.Writer.Complete(ex); throw; }
        raw.Writer.Complete();
    }, ct);

    // ── ② 변환 (CPU 바운드) — 코어 수만큼
    var workers = Environment.ProcessorCount;
    var xform = Enumerable.Range(0, workers).Select(_ => Task.Run(async () =>
    {
        await foreach (var r in raw.Reader.ReadAllAsync(ct))
        {
            if (Validate(r) is { } why) { await skips.WriteAsync((r, why), ct); continue; }
            await done.Writer.WriteAsync(Map(r), ct);
        }
    }, ct)).ToArray();

    // 변환기가 전부 끝나야 done 을 닫을 수 있습니다
    _ = Task.Run(async () => { await Task.WhenAll(xform); done.Writer.Complete(); }, ct);

    // ── ③ 쓰기 (IO 바운드) — DB 커넥션 수에 맞춥니다
    var write = Task.Run(async () =>
    {
        var buf = new List<Trade>(5_000);
        await foreach (var t in done.Reader.ReadAllAsync(ct))
        {
            buf.Add(t);

            buf = new List<Trade>(5_000);
        }
        if (buf.Count > 0) await bulk.CopyAsync(buf, ct);
    }, ct);

}

[DisableConcurrentExecution(timeoutInSeconds: 4 * 60 * 60)]
[AutomaticRetry(Attempts = 0)]        // ★ 재시도는 우리가 통제합니다
public sealed class SettleJob(
    ISftpGateway sftp, IFileVerifier verify, ChunkRunner chunks,
    ISettleAggregator agg, IResultWriter writer, IBatchJournal journal,
    INotifier notify, ILogger<SettleJob> log)
{
    public async Task RunAsync(DateOnly? forDate, CancellationToken ct)
    {
        var bizDate = forDate ?? KstToday().AddDays(-1);
        var sw = Stopwatch.StartNew();

        // 다른 서버가 돌고 있으면 조용히 물러납니다 (f02)
        await using var mine = await SqlAppLock.TryAcquireAsync(
            _cs, $"batch:settle:{bizDate:yyyyMMdd}", TimeSpan.Zero, ct);
        if (mine is null) { log.LogInformation("다른 서버가 처리 중"); return; }

        var run = await journal.StartAsync("daily-settle", bizDate, ct);
        try
        {
            // ── ① 수신 : SFTP → 로컬 (임시 확장자 → rename)
            var local = await sftp.DownloadAsync(
                $"/out/TRD_{bizDate:yyyyMMdd}.dat", ct);
            log.LogInformation("수신 완료 {Bytes:N0} 바이트 ({Sec:F0}초)",
                new FileInfo(local).Length, sw.Elapsed.TotalSeconds);

            // ── ② 검증 : 헤더 · 트레일러 · 건수 · 합계 (f08)
            var stats = await verify.VerifyAsync(local, Cp949, ct);
            log.LogInformation("검증 통과 {Rows:N0}행 합계 {Sum:N2}",
                stats.Rows, stats.Sum);

            // ── ③④ 청크 처리 + 벌크 적재 (f04 · f05)
            var r = await chunks.RunAsync(
                job: "daily-settle", bizDate, step: "load",
                readFrom: key => TradeFileReader.ReadAsync(local, key, Cp949),
                chunkSize: 5_000, skipLimit: 10_000, ct);

            if (r.Kind == ResultKind.Stopped)      // 종료 신호로 중단
            { log.LogWarning("중단됨 — 다음 실행이 이어받습니다"); return; }

            // ── ⑤ 집계 : 스테이징 → 본 테이블 MERGE + 수수료 계산
            var summary = await agg.AggregateAsync(bizDate, ct);

            // ── ⑥ 송신 : 결과 파일 생성 후 SFTP 업로드
            var outPath = await writer.WriteAsync(bizDate, summary, ct);
            await sftp.UploadAsync(outPath, $"/in/STL_{bizDate:yyyyMMdd}.dat", ct);

            await journal.SucceedAsync(run.RunId, r.Read, r.Wrote, r.Skipped, ct);
            await notify.SendAsync(Alert.Info("정산 완료",
                $"{bizDate:yyyy-MM-dd} · {r.Wrote:N0}건 · {sw.Elapsed.TotalMinutes:F0}분"), ct);
        }
        catch (Exception ex)
        {
            await journal.FailAsync(run.RunId, ex, CancellationToken.None);
            await notify.SendAsync(Alert.High("정산 실패",
                $"{bizDate:yyyy-MM-dd}\n{ex.Message}"), CancellationToken.None);
            throw;                                  // Hangfire 대시보드에 빨갛게
        }
    }
}

public async Task<string> WriteAsync(
    DateOnly bizDate, IAsyncEnumerable<SettleRow> rows, CancellationToken ct)
{
    var tmp   = Path.Combine(_out, $"STL_{bizDate:yyyyMMdd}.dat.tmp");
    var final = tmp[..^4];                       // .tmp 떼기

    long n = 0; decimal sum = 0m;

    // ★ 대외 규약: CP949 · CRLF · BOM 없음
    await using (var fs = File.Create(tmp))
    await using (var sw = new StreamWriter(fs, Cp949) { NewLine = "\r\n" })
    {
        await sw.WriteLineAsync(
            $"H|SETTLE|{DateTime.UtcNow.AddHours(9):yyyyMMddHHmmss}|{bizDate:yyyyMMdd}");

        await foreach (var r in rows.WithCancellation(ct))
        {
            // 고정폭: 가맹점(10) 건수(9) 총액(15) 수수료(15) 정산액(15)
            await sw.WriteLineAsync(
                $"D{r.MerchantId,10}{r.TxCount,9:D9}" +
                $"{(long)(r.GrossAmt * 100),15:D15}" +
                $"{(long)(r.FeeAmt   * 100),15:D15}" +
                $"{(long)(r.NetAmt   * 100),15:D15}");
            n++; sum += r.NetAmt;
        }

        await sw.WriteLineAsync($"T|{n}|{sum:F2}");
    }

    File.Move(tmp, final, overwrite: true);      // ★ 원자적 공개 (f08)
    return final;
}
