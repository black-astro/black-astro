Minimal API 와 Controller 중 무엇을? 요즘 기본은 Minimal API 입니다 — 가볍고 빠릅니다. 필터·모델 바인딩을 복잡하게 쓰거나 팀이 익숙하다면 Controller 도 여전히 좋은 선택입니다. 섞어 써도 됩니다.
게임 클라이언트가 부를 API 는 응답을 작게. 모바일 환경에서는 왕복 횟수와 크기가 체감 속도를 정합니다. 화면 하나에 필요한 데이터는 한 번에 묶어 주세요.
미들웨어는 줄이 아니라 터널입니다 — 요청이 들어갈 때 한 번, 응답이 나올 때 한 번, 같은 코드를 두 번 지납니다.

등록 순서 — 이 순서가 정답에 가깝습니다

Program.cs 미들웨어 순서

var app = builder.Build();

app.UseExceptionHandler("/error");   // 1. 가장 바깥 — 아래 전부를 감싼다
app.UseHsts();                       // 2. 브라우저에 "https 로만 와라"
app.UseHttpsRedirection();           // 3. http → https
app.UseStaticFiles();                // 4. 정적 파일은 여기서 끝 (뒤로 안 감)
app.UseRouting();                    // 5. 어느 엔드포인트인지 결정
app.UseCors("game");                 // 6. 라우팅 뒤, 인증 앞
app.UseRateLimiter();                // 7. 인증 뒤가 더 좋을 때도 있다(아래 note)
app.UseAuthentication();             // 8. 토큰을 읽어 User 를 채운다
app.UseAuthorization();              // 9. 그 User 로 권한을 본다
app.MapPlayers().MapShop();          // 10. 끝점

app.Run();
UseAuthentication 이 UseAuthorization 보다 먼저. 순서를 뒤집으면 항상 401 이 납니다 — 권한을 볼 시점에 User 가 아직 비어 있기 때문입니다. 증상은 "토큰이 맞는데 로그인이 안 돼요" 로 나타납니다.

직접 만들어 보면 구조가 보입니다

// 요청 하나에 걸린 시간을 재는 최소 미들웨어
app.Use(async (ctx, next) =>
{
    var sw = Stopwatch.StartNew();
    await next(ctx);                       // ★ 여기서 안쪽 전부가 실행된다
    sw.Stop();
    // next() 가 돌아온 뒤 = 응답이 나가는 길
    ctx.Response.Headers["X-Elapsed-ms"] = sw.ElapsedMilliseconds.ToString();
});

// next() 를 부르지 않으면 거기서 요청이 끝난다 — 이것이 "단락(short-circuit)"
app.Use(async (ctx, next) =>
{
    if (ctx.Request.Headers["X-Client-Ver"] != "1.4.2")
    {
        ctx.Response.StatusCode = 426;     // Upgrade Required
        await ctx.Response.WriteAsJsonAsync(new { need = "1.4.2" });
        return;                            // next 호출 없음 → 뒤로 안 감
    }
    await next(ctx);
});
강제 업데이트가 이 한 조각으로 끝납니다. 구버전 클라이언트를 엔드포인트에 닿기 전에 잘라 내면 DB 도 안 건드리고 로그도 안 더럽힙니다. 점검 모드도 같은 자리에 둡니다.

public class GameDb(DbContextOptions<GameDb> o) : DbContext(o)
{
    public DbSet<Player> Players => Set<Player>();
    public DbSet<Item>   Items   => Set<Item>();

    protected override void OnModelCreating(ModelBuilder b)
    {
        b.Entity<Player>(e =>
        {
            e.HasKey(x => x.Id);
            e.HasIndex(x => x.Nickname).IsUnique();
            e.Property(x => x.Nickname).HasMaxLength(20).IsRequired();
            e.HasMany(x => x.Items).WithOne(i => i.Owner)
             .OnDelete(DeleteBehavior.Cascade);
        });
    }
}

string IssueToken(Player p)
{
    var claims = new[]
    {
        new Claim(JwtRegisteredClaimNames.Sub, p.Id.ToString()),
        new Claim("nick", p.Nickname),
        new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString()),
    };
    var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);
    var token = new JwtSecurityToken(issuer, audience, claims,
                    expires: DateTime.UtcNow.AddMinutes(30), signingCredentials: creds);
    return new JwtSecurityTokenHandler().WriteToken(token);
}

{
    var provider = await VerifyExternalAsync(d.Provider, d.IdToken);   // ★ 서버 검증
    if (provider is null) return Results.Unauthorized();

    var me = await db.Players.FindAsync(user.PlayerId());
    if (await db.Players.AnyAsync(x => x.ExternalId == provider.Id))
        return Results.Conflict("이미 연결된 계정입니다");

    await db.SaveChangesAsync();
    return Results.NoContent();

기기 식별자만으로 계정을 만들면 기기를 잃을 때 끝입니다. 계정 연동을 강하게 유도하고, 소셜 로그인의 idToken 은 반드시 서버가 제공자에 검증해야 합니다 — 클라이언트가 준 값을 믿으면 남의 계정을 가로챌 수 있습니다.
토큰 수명·리프레시 회전은 ☕ 자바 가이드의 Security 탭과 동일한 원리입니다 — 액세스는 짧게, 리프레시는 한 번 쓰면 교체.
서명만 확인하면 되므로 서버를 여러 대로 늘려도 세션 공유가 필요 없습니다.

검증 · 에러 처리

클라이언트가 보낸 값은 전부 의심합니다. 그리고 에러 형식을 하나로 통일합니다.

입력 검증

// FluentValidation
public class BuyItemValidator : AbstractValidator<BuyItemDto>
{
    public BuyItemValidator()
    {
        RuleFor(x => x.ItemId).GreaterThan(0);
        RuleFor(x => x.Count).InclusiveBetween(1, 99);
        RuleFor(x => x.Nickname).MaximumLength(20)
            .Matches("^[가-힣a-zA-Z0-9]+$").WithMessage("특수문자는 쓸 수 없습니다");
    }
}

app.MapPost("/shop/buy", async (BuyItemDto d, IValidator<BuyItemDto> v, IShop shop) =>
{
    var r = await v.ValidateAsync(d);
    if (!r.IsValid) return Results.ValidationProblem(r.ToDictionary());

    // ★ 가격은 서버가 조회 — 클라이언트가 보낸 금액은 절대 쓰지 않는다
    return await shop.BuyAsync(d.ItemId, d.Count) switch
    {
        BuyResult.Ok o        => Results.Ok(o),
        BuyResult.NoGold      => Results.BadRequest(new { code = "NO_GOLD" }),
        BuyResult.SoldOut     => Results.Conflict(new { code = "SOLD_OUT" }),
        _                     => Results.Problem(),
    };
});

방식지연방향맞는 곳대가

SignalR 의 확장 문제는 대부분 "메시지가 반만 도착한다"는 증상으로 옵니다 — 서버 대수만큼 확률이 갈립니다.

게임 백엔드 설계 — 재화 · 인벤토리

여기서 실수하면 아이템이 복사되거나 사라집니다. 트랜잭션과 멱등성이 전부입니다.

구매 — 한 트랜잭션으로

public async Task<BuyResult> BuyAsync(int playerId, int itemId, int count,
                                       string idempotencyKey, CancellationToken ct)
{
    // ① 같은 요청이 두 번 오면 이전 결과를 그대로 (네트워크 재시도 대비)
    if (await _db.Receipts.FindAsync([idempotencyKey], ct) is {} done)
        return done.Result;

    await using var tx = await _db.Database.BeginTransactionAsync(ct);

    // ② 가격은 서버가 조회 — 클라이언트 값은 쓰지 않는다
    var item = await _db.ItemDefs.FindAsync([itemId], ct)
               ?? throw new NotFoundException();
    var cost = item.Price * count;

    // ③ 재화 차감은 조건부 UPDATE 로 (동시 요청에도 음수가 되지 않게)
    var affected = await _db.Players
        .Where(p => p.Id == playerId && p.Gold >= cost)
        .ExecuteUpdateAsync(s => s.SetProperty(p => p.Gold, p => p.Gold - cost), ct);
    if (affected == 0) return BuyResult.NoGold;

    // ④ 지급 · 영수증 · 로그를 같은 트랜잭션에
    _db.Inventories.Add(new Inventory { PlayerId = playerId, ItemId = itemId, Count = count });
    _db.Receipts.Add(new Receipt { Key = idempotencyKey, Result = BuyResult.Ok });
    _db.GoldLogs.Add(new GoldLog { PlayerId = playerId, Delta = -cost, Reason = "buy" });

    await _db.SaveChangesAsync(ct);
    await tx.CommitAsync(ct);
    return BuyResult.Ok;
}

지켜야 할 원칙

  • 재화 변동은 반드시 로그 — 문의·환불·치트 조사의 유일한 근거입니다
  • 조건부 UPDATE — WHERE gold >= cost 로 DB 가 판정하게
  • 멱등 키 — 같은 요청이 두 번 와도 한 번만 처리
  • 지급과 차감은 같은 트랜잭션 — 중간에 죽으면 둘 다 취소
  • 수량 상한 — 인벤토리·우편함 한도를 서버가 강제
  • 시간은 서버 기준 — 클라이언트 시각으로 일일 보상을 주면 조작됩니다
"읽고 → 계산하고 → 쓰기"는 동시 요청에서 깨집니다. 골드 100 인 플레이어가 동시에 두 번 구매하면 둘 다 "충분함"으로 읽어 골드가 음수가 됩니다. 조건부 UPDATE 나 낙관적 동시성(RowVersion)으로 막으세요.
게임 백엔드에서 사고가 나는 곳은 대부분 재화가 오가는 이 한 지점입니다.

gRPC — 서버끼리는 이걸로

클라이언트에게는 REST·SignalR 이 편하지만, 서버끼리(매치메이커 ↔ 게임 서버 ↔ 백엔드)는 gRPC 가 표준입니다 — 계약이 코드로 강제되고, HTTP/2 바이너리라 빠릅니다.

① 계약부터 — .proto 파일

// Protos/match.proto
syntax = "proto3";
option csharp_namespace = "Game.Match";

service Matchmaker {
  // 단발 호출 — 요청 하나, 응답 하나
  rpc FindMatch (FindMatchRequest) returns (MatchTicket);
  // 서버 스트리밍 — 매칭 진행 상황을 밀어 준다
  rpc WatchTicket (MatchTicket) returns (stream TicketUpdate);
}

message FindMatchRequest {
  int64 player_id = 1;      // 번호는 저장 형식 — 바꾸면 호환이 깨집니다
  int32 mmr = 2;
  string region = 3;
}
message MatchTicket  { string ticket_id = 1; }
message TicketUpdate { string state = 1; string server_addr = 2; }
proto 가 곧 문서이자 타입입니다. REST 처럼 "필드 이름이 뭐였더라"가 없습니다 — 빌드하면 C# 클래스와 클라이언트 코드가 자동 생성되고, 다른 언어(Go·Python) 서버와도 같은 파일로 통신합니다.

② 서버 구현 — 생성된 베이스를 상속

// dotnet add package Grpc.AspNetCore
// csproj: <Protobuf Include="Protos\match.proto" GrpcServices="Server" />

public class MatchmakerService : Matchmaker.MatchmakerBase
{
    public override async Task<MatchTicket> FindMatch(
        FindMatchRequest req, ServerCallContext ctx)
    {
        var id = await _queue.EnqueueAsync(req.PlayerId, req.Mmr, req.Region);
        return new MatchTicket { TicketId = id };
    }

    public override async Task WatchTicket(MatchTicket t,
        IServerStreamWriter<TicketUpdate> stream, ServerCallContext ctx)
    {
        await foreach (var u in _queue.Watch(t.TicketId, ctx.CancellationToken))
            await stream.WriteAsync(u);      // 상태가 바뀔 때마다 밀어 준다
    }
}

// Program.cs
builder.Services.AddGrpc();
app.MapGrpcService<MatchmakerService>();
// ③ 호출하는 쪽 (게임 서버 등)
// csproj: GrpcServices="Client" 로 같은 proto 등록
using var ch = GrpcChannel.ForAddress("https://match.internal:5001");
var client = new Matchmaker.MatchmakerClient(ch);   // 채널은 재사용!
var ticket = await client.FindMatchAsync(
    new FindMatchRequest { PlayerId = 1001, Mmr = 1500, Region = "kr" },
    deadline: DateTime.UtcNow.AddSeconds(3));        // 데드라인 = 타임아웃

상황선택

public record GrantRequest(
    long   PlayerId,
    string ItemCode,
    int    Amount,
    string Reason);          // ★ 사유 필수 — 감사 로그에 남는다

admin.MapPost("/grant", async (GrantRequest req, ClaimsPrincipal op,
                               GameDb db, IAuditLog audit) =>
{
    // ① 상한선 — 실수로 0 하나 더 붙는 사고를 기계가 막는다
    if (req.Amount is <= 0 or > 10_000)
        return Results.BadRequest(new { error = "amount 1~10000" });
    if (string.IsNullOrWhiteSpace(req.Reason))
        return Results.BadRequest(new { error = "reason required" });

    // ② 지급은 유저 API 와 같은 서비스 로직을 태운다 (지급 이력 포함)
    await _wallet.GrantAsync(req.PlayerId, req.ItemCode, req.Amount,
                             source: $"admin:{op.Identity!.Name}");

    // ③ 감사 로그 — 조회만 빼고 전부 남긴다
    await audit.WriteAsync(op.Identity!.Name!, "grant", req);
    return Results.Ok();
}).RequireAuthorization("role:grant");
대량·고액 작업은 2인 승인으로. 전체 유저 보상 우편처럼 파급이 큰 명령은 한 명이 요청 → 다른 한 명이 승인해야 실행되게 만드세요. "요청 저장 → 승인 시 실행" 테이블 하나면 구현됩니다.
