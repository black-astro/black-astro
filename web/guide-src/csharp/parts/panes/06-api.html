API 03

EF Core — DB 연동

C# 클래스로 테이블을 다룹니다. 느려지는 지점이 정해져 있으니 그것만 조심하면 됩니다.

기본

public class GameDb(DbContextOptions<GameDb> o) : DbContext(o)
{
    public DbSet<Player> Players => Set<Player>();
    public DbSet<Item>   Items   => Set<Item>();

    protected override void OnModelCreating(ModelBuilder b)
    {
        b.Entity<Player>(e =>
        {
            e.HasKey(x => x.Id);
            e.HasIndex(x => x.Nickname).IsUnique();
            e.Property(x => x.Nickname).HasMaxLength(20).IsRequired();
            e.HasMany(x => x.Items).WithOne(i => i.Owner)
             .OnDelete(DeleteBehavior.Cascade);
        });
    }
}

builder.Services.AddDbContext<GameDb>(o =>
    o.UseNpgsql(builder.Configuration.GetConnectionString("Db")));
dotnet ef migrations add Init
dotnet ef database update

느려지는 세 가지

N+1 쿼리

foreach (var p in players)
    Console.WriteLine(p.Items.Count);

Include 로 한 번에
var players = await db.Players
    .Include(p => p.Items)

  • 필요한 컬럼만 — Select(p => new { p.Id, p.Nickname })
  • 페이징 필수 — Skip/Take 없이 전체를 가져오지 마세요
  • 인덱스 — 조회 조건에 쓰는 컬럼에는 반드시

API 04

인증 — JWT · 게스트 로그인

게임은 게스트 시작 → 나중에 계정 연동이 표준입니다. 그 흐름을 만듭니다.

토큰 발급과 검증

builder.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(o =>
    {
        o.TokenValidationParameters = new()
        {
            ValidateIssuer = true, ValidIssuer = cfg["Jwt:Issuer"],
            ValidateAudience = true, ValidAudience = cfg["Jwt:Audience"],
            ValidateLifetime = true,
            ClockSkew = TimeSpan.FromSeconds(30),      // 기본 5분은 너무 길다
            IssuerSigningKey = new SymmetricSecurityKey(
                Encoding.UTF8.GetBytes(cfg["Jwt:Key"]!)),
        };
    });
app.UseAuthentication();
app.UseAuthorization();

string IssueToken(Player p)
{
    var claims = new[]
    {
        new Claim(JwtRegisteredClaimNames.Sub, p.Id.ToString()),
        new Claim("nick", p.Nickname),
        new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString()),
    };
    var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);
    var token = new JwtSecurityToken(issuer, audience, claims,
                    expires: DateTime.UtcNow.AddMinutes(30), signingCredentials: creds);
    return new JwtSecurityTokenHandler().WriteToken(token);
}

게스트 → 계정 연동

// ① 첫 실행 — 기기 식별자로 게스트 계정 생성
app.MapPost("/auth/guest", async (GuestDto d, GameDb db) =>
{
    var p = await db.Players.FirstOrDefaultAsync(x => x.DeviceId == d.DeviceId)
            ?? db.Players.Add(new Player { DeviceId = d.DeviceId }).Entity;
    await db.SaveChangesAsync();
    return Results.Ok(new { token = IssueToken(p), refresh = IssueRefresh(p) });
});

// ② 나중에 소셜 계정 연결 (기기 분실 대비)
app.MapPost("/auth/link", async (LinkDto d, ClaimsPrincipal user, GameDb db) =>
{
    var provider = await VerifyExternalAsync(d.Provider, d.IdToken);   // ★ 서버 검증
    if (provider is null) return Results.Unauthorized();

    var me = await db.Players.FindAsync(user.PlayerId());
    if (await db.Players.AnyAsync(x => x.ExternalId == provider.Id))
        return Results.Conflict("이미 연결된 계정입니다");

    me!.ExternalId = provider.Id;
    await db.SaveChangesAsync();
    return Results.NoContent();
}).RequireAuthorization();
기기 식별자만으로 계정을 만들면 기기를 잃을 때 끝입니다. 계정 연동을 강하게 유도하고, 소셜 로그인의 idToken 은 반드시 서버가 제공자에 검증해야 합니다 — 클라이언트가 준 값을 믿으면 남의 계정을 가로챌 수 있습니다.
토큰 수명·리프레시 회전은 ☕ 자바 가이드의 Security 탭과 동일한 원리입니다 — 액세스는 짧게, 리프레시는 한 번 쓰면 교체.
API 05

검증 · 에러 처리

클라이언트가 보낸 값은 전부 의심합니다. 그리고 에러 형식을 하나로 통일합니다.

입력 검증

// FluentValidation
public class BuyItemValidator : AbstractValidator<BuyItemDto>
{
    public BuyItemValidator()
    {
        RuleFor(x => x.ItemId).GreaterThan(0);
        RuleFor(x => x.Count).InclusiveBetween(1, 99);
        RuleFor(x => x.Nickname).MaximumLength(20)
            .Matches("^[가-힣a-zA-Z0-9]+$").WithMessage("특수문자는 쓸 수 없습니다");
    }
}

app.MapPost("/shop/buy", async (BuyItemDto d, IValidator<BuyItemDto> v, IShop shop) =>
{
    var r = await v.ValidateAsync(d);
    if (!r.IsValid) return Results.ValidationProblem(r.ToDictionary());

    // ★ 가격은 서버가 조회 — 클라이언트가 보낸 금액은 절대 쓰지 않는다
    return await shop.BuyAsync(d.ItemId, d.Count) switch
    {
        BuyResult.Ok o        => Results.Ok(o),
        BuyResult.NoGold      => Results.BadRequest(new { code = "NO_GOLD" }),
        BuyResult.SoldOut     => Results.Conflict(new { code = "SOLD_OUT" }),
        _                     => Results.Problem(),
    };
});

에러 형식 통일

app.UseExceptionHandler(x => x.Run(async ctx =>
{
    var ex = ctx.Features.Get<IExceptionHandlerFeature>()?.Error;
    var (status, code) = ex switch
    {
        NotFoundException      => (404, "NOT_FOUND"),
        DomainRuleException d  => (400, d.Code),
        _                      => (500, "INTERNAL"),
    };

    log.Error(ex, "요청 실패 {Path}", ctx.Request.Path);
    ctx.Response.StatusCode = status;
    await ctx.Response.WriteAsJsonAsync(new
    {
        code,
        traceId = Activity.Current?.Id ?? ctx.TraceIdentifier,   // ★ 문의 대응용
    });
}));

백그라운드 작업

public class DailyResetService(IServiceScopeFactory scopes, ILogger<DailyResetService> log)
    : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            var next = NextUtcMidnight();
            await Task.Delay(next - DateTimeOffset.UtcNow, ct);

            using var scope = scopes.CreateScope();          // ★ Scoped 를 쓰려면 필수
            var db = scope.ServiceProvider.GetRequiredService<GameDb>();
            try { await ResetDailyAsync(db, ct); }
            catch (Exception e) { log.Error(e, "일일 초기화 실패"); }
        }
    }
}
builder.Services.AddHostedService<DailyResetService>();
서버가 여러 대면 같은 작업이 여러 번 돕니다. 일일 보상 지급이 3대에서 3번 실행되면 사고입니다. 분산 락이나 전용 스케줄러(Hangfire·Quartz)로 한 대만 실행하도록 보장하세요(⚡ 대규모 트래픽 탭).
API 08

게임 백엔드 설계 — 재화 · 인벤토리

여기서 실수하면 아이템이 복사되거나 사라집니다. 트랜잭션과 멱등성이 전부입니다.

구매 — 한 트랜잭션으로

public async Task<BuyResult> BuyAsync(int playerId, int itemId, int count,
                                       string idempotencyKey, CancellationToken ct)
{
    // ① 같은 요청이 두 번 오면 이전 결과를 그대로 (네트워크 재시도 대비)
    if (await _db.Receipts.FindAsync([idempotencyKey], ct) is {} done)
        return done.Result;

    await using var tx = await _db.Database.BeginTransactionAsync(ct);

    // ② 가격은 서버가 조회 — 클라이언트 값은 쓰지 않는다
    var item = await _db.ItemDefs.FindAsync([itemId], ct)
               ?? throw new NotFoundException();
    var cost = item.Price * count;

    // ③ 재화 차감은 조건부 UPDATE 로 (동시 요청에도 음수가 되지 않게)
    var affected = await _db.Players
        .Where(p => p.Id == playerId && p.Gold >= cost)
        .ExecuteUpdateAsync(s => s.SetProperty(p => p.Gold, p => p.Gold - cost), ct);
    if (affected == 0) return BuyResult.NoGold;

    // ④ 지급 · 영수증 · 로그를 같은 트랜잭션에
    _db.Inventories.Add(new Inventory { PlayerId = playerId, ItemId = itemId, Count = count });
    _db.Receipts.Add(new Receipt { Key = idempotencyKey, Result = BuyResult.Ok });
    _db.GoldLogs.Add(new GoldLog { PlayerId = playerId, Delta = -cost, Reason = "buy" });

    await _db.SaveChangesAsync(ct);
    await tx.CommitAsync(ct);
    return BuyResult.Ok;
}

지켜야 할 원칙

  • 재화 변동은 반드시 로그 — 문의·환불·치트 조사의 유일한 근거입니다
  • 조건부 UPDATE — WHERE gold >= cost 로 DB 가 판정하게
  • 멱등 키 — 같은 요청이 두 번 와도 한 번만 처리
  • 지급과 차감은 같은 트랜잭션 — 중간에 죽으면 둘 다 취소
  • 수량 상한 — 인벤토리·우편함 한도를 서버가 강제
  • 시간은 서버 기준 — 클라이언트 시각으로 일일 보상을 주면 조작됩니다
"읽고 → 계산하고 → 쓰기"는 동시 요청에서 깨집니다. 골드 100 인 플레이어가 동시에 두 번 구매하면 둘 다 "충분함"으로 읽어 골드가 음수가 됩니다. 조건부 UPDATE 나 낙관적 동시성(RowVersion)으로 막으세요.

  • 레벨 기준 — Error 는 사람이 조치해야 하는 것만. 나머지는 Warning 이하로
  • 보관 기간 — 재화 로그는 길게(분쟁), 일반 로그는 짧게(비용)
traceId 를 클라이언트 에러 화면에 보여 주세요. 사용자가 문의할 때 그 값을 알려 주면 수억 줄의 로그에서 그 요청 하나를 바로 찾을 수 있습니다.
로그가 병목이 되는 경우가 있습니다. 초당 수만 건이면 디스크 I/O 가 응답을 느리게 합니다. 비동기 싱크·샘플링을 쓰고, 디버그 로그를 운영에서 켜 두지 마세요.
API 10

테스트 — 단위 · 통합

밸런스와 규칙은 자주 바뀝니다. 바꿔도 안 깨진다는 확신이 개발 속도입니다.

단위 테스트

public class DamageTests
{
    [Theory]
    [InlineData(100, 20, 0.0, 80)]
    [InlineData(100, 20, 0.5, 90)]     // 방어율 50%
    [InlineData(100, 999, 0.0, 0)]     // 체력은 0 미만이 되지 않는다
    public void 데미지_계산(int hp, int atk, double def, int expected)
    {
        var result = Combat.Apply(hp, atk, def);
        Assert.Equal(expected, result);
    }
}

통합 테스트 — 진짜 DB 로

public class ApiFixture : WebApplicationFactory<Program>, IAsyncLifetime
{
    private readonly PostgreSqlContainer _db =
        new PostgreSqlBuilder().WithImage("postgres:17-alpine").Build();

    public async Task InitializeAsync() => await _db.StartAsync();

    protected override void ConfigureWebHost(IWebHostBuilder b) =>
        b.UseSetting("ConnectionStrings:Db", _db.GetConnectionString());

public class ShopTests(ApiFixture f) : IClassFixture<ApiFixture>
{

    public async Task 골드가_부족하면_구매_실패()

게임 백엔드에서 특히 중요한 테스트 — 동시 요청(같은 아이템을 두 번 살 때), 멱등성(같은 요청 두 번), 경계값(재화 0·인벤토리 만석). 전부 실제 사고가 나는 지점입니다.
API 11

배포 — Docker · 무중단

한 대에서 여러 대로. 배포 중에도 서비스가 멈추지 않게 만듭니다.

Dockerfile

FROM mcr.microsoft.com/dotnet/sdk:10.0 AS build
WORKDIR /src
COPY *.sln .
COPY src/GameApi/*.csproj src/GameApi/
RUN dotnet restore                      # ★ 의존성 레이어를 먼저 (캐시)
COPY . .
RUN dotnet publish src/GameApi -c Release -o /app --no-restore

FROM mcr.microsoft.com/dotnet/aspnet:10.0
WORKDIR /app
COPY --from=build /app .
ENV ASPNETCORE_URLS=http://+:8080
EXPOSE 8080
USER $APP_UID                           # 루트로 돌리지 않는다
ENTRYPOINT ["dotnet", "GameApi.dll"]
// 헬스체크 — 로드밸런서가 이걸 보고 트래픽을 넣습니다
builder.Services.AddHealthChecks()
    .AddDbContextCheck<GameDb>()
    .AddRedis(cfg.GetConnectionString("Redis")!);

app.MapHealthChecks("/health/live");    // 살아 있나 (프로세스)
app.MapHealthChecks("/health/ready");   // 받을 준비가 됐나 (DB·캐시 포함)

무중단의 조건

  • 그레이스풀 셧다운 — 종료 신호를 받으면 진행 중 요청을 마치고 종료
  • readiness 와 liveness 분리 — 준비 안 됐는데 트래픽이 오면 에러가 납니다
  • DB 마이그레이션은 하위 호환으로 — 컬럼 삭제는 다음 배포에서
  • 무상태 — 세션을 메모리에 두면 확장할 수 없습니다
  • 롤백 절차를 미리 문서로
builder.Services.Configure<HostOptions>(o =>
    o.ShutdownTimeout = TimeSpan.FromSeconds(30));   // 마무리 시간 확보
배포 인프라 전반(리버스 프록시·로드밸런싱·HTTPS·무중단 전략)은 🌐 웹서버 가이드에서 언어와 무관하게 다룹니다.
API 12

백엔드 체크리스트

게임 백엔드를 열기 전에 확인할 것들을 모았습니다.

보안

  • ☐ 모든 계산을 서버가 — 가격·데미지·보상·확률
  • ☐ 인증 없는 엔드포인트가 없는지 (기본을 RequireAuthorization 으로)
  • ☐ 남의 데이터 조회 차단 — 경로의 id 를 그대로 믿지 않기
  • ☐ 속도 제한 — 로그인·구매·랭킹 조회
  • ☐ HTTPS 강제 · 인증서 자동 갱신
  • ☐ 의존 패키지 취약점 점검(dotnet list package --vulnerable)

운영

  • ☐ DB 백업·복구를 실제로 복구해 봤는지
  • ☐ 재화 로그 보관 정책
  • ☐ 점검 모드·강제 업데이트 스위치
  • ☐ 운영 도구 — 계정 조회·아이템 지급·밴
  • ☐ 대시보드 — 요청 수 · 에러율 · p99 · DB 커넥션
  • ☐ 첫 주 모니터링 당번과 연락 체계
다음 단계 — 동시접속이 늘어난 뒤의 이야기는 ⚡ 대규모 트래픽 탭에서 이어집니다. 실시간 전투가 필요하면 🌐 게임 서버, DB 튜닝은 🗄️ 데이터베이스 가이드로.
