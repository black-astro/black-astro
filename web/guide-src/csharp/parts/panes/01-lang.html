C# 02

변수 · 타입 · var

C# 은 정적 타입 언어입니다. 타입을 적는 순간 컴파일러가 실수를 대신 잡아 줍니다.

기본형과 var

int    hp    = 100;          // 32비트 정수
long   gold  = 9_000_000_000; // 64비트 (밑줄로 자릿수 구분 가능)
float  speed = 5.5f;          // Unity 에서 가장 많이 쓰는 실수형
double ratio = 0.75;          // 기본 실수형 (정밀도 ↑)
decimal price = 19_900m;      // ★ 돈 계산은 무조건 decimal
bool   alive = true;
char   grade = 'S';
string name  = "용사";

// var — '타입을 안 쓰는 것'이 아니라 '컴파일러가 추론'하는 것
var count = 10;               // int 로 확정 (나중에 문자열 대입 불가)
var items = new List<string>();

// 상수
const int MaxLevel = 99;      // 컴파일 시점 고정
readonly int seed;            // 생성자에서 한 번만 대입 가능
돈에 float · double 을 쓰면 틀립니다. 0.1 + 0.2 != 0.3 이기 때문입니다. 금액·수량 정산은 decimal, 게임의 좌표·속도처럼 오차가 허용되는 값은 float 이 정답입니다.

값 타입 vs 참조 타입 — 이 구분이 성능을 가릅니다

값 타입 (struct)
  • int · float · bool · Vector3
  • 스택에 놓이고 복사됩니다
  • GC 부담이 없습니다 — 게임 루프에서 중요
참조 타입 (class)
  • string · 배열 · List · 내가 만든 class
  • 힙에 놓이고 주소가 복사됩니다
  • 많이 만들면 GC 가 돌며 프레임이 튑니다
struct Point { public int X, Y; }
class   Node  { public int Value; }

var p1 = new Point { X = 1 };
var p2 = p1;  p2.X = 99;        // 복사본만 바뀐다
Console.WriteLine(p1.X);        // 1

var n1 = new Node { Value = 1 };
var n2 = n1;  n2.Value = 99;    // 같은 객체를 가리킨다
Console.WriteLine(n1.Value);    // 99  ← 함께 바뀐다
Unity 의 Vector3·Quaternion 은 struct 입니다. 그래서 transform.position.x = 1; 이 컴파일되지 않습니다 — 복사본을 고쳐 봐야 원본이 안 바뀌기 때문입니다. transform.position = new Vector3(1, y, z); 처럼 통째로 넣어야 합니다.

C# 04

조건 · 반복 · switch 식

C# 의 switch 는 값을 돌려주는 식입니다. 여기부터 코드가 짧아집니다.

반복문 네 가지

for (int i = 0; i < 10; i++) { }

foreach (var e in enemies) { }        // 가장 많이 씁니다

while (hp > 0) { }
do { } while (retry);

// 인덱스가 함께 필요하면
foreach (var (item, i) in items.Select((x, i) => (x, i))) { }

// 범위·인덱스 문법 (C# 8+)
var last  = arr[^1];       // 마지막
var first3 = arr[..3];     // 앞 3개
var mid   = arr[2..5];     // 2~4
foreach 중에 컬렉션을 고치면 예외가 납니다. 지울 대상은 따로 모았다가 루프 밖에서 지우거나, for 문을 뒤에서 앞으로 돌리세요. 게임에서 "죽은 적 제거" 코드가 여기서 자주 터집니다.

switch 식 · 패턴 매칭

// 값을 돌려주는 switch — if/else 사슬이 사라집니다
string Grade(int score) => score switch
{
    >= 90 => "S",
    >= 80 => "A",
    >= 70 => "B",
    _      => "F",          // _ 는 나머지 전부
};

// 타입으로 분기
decimal Damage(IWeapon w) => w switch
{
    Sword s when s.IsCritical => s.Power * 2,
    Sword s                   => s.Power,
    Bow { Arrows: 0 }         => 0,          // 속성 패턴
    Bow b                     => b.Power * 0.8m,
    null                      => throw new ArgumentNullException(nameof(w)),
    _                         => 1,
};

// 튜플 패턴 — 가위바위보
string Result(int me, int you) => (me, you) switch
{
    (0, 2) or (1, 0) or (2, 1) => "승",
    var (a, b) when a == b     => "무",
    _                          => "패",
};
모든 경우를 다 적으면 _ 없이도 됩니다. enum 을 switch 할 때 _ 를 빼 두면, 나중에 enum 에 값을 추가했을 때 컴파일러가 경고해 줍니다 — 빠뜨린 분기를 사람이 찾지 않아도 됩니다.
C# 05

클래스 · 구조체 · record

셋 다 "데이터 묶음"이지만 쓰는 자리가 다릅니다. 고르는 기준을 정리합니다.

클래스 — 기본 선택지

public class Player
{
    // 자동 속성
    public string Name { get; }              // 읽기 전용
    public int    Hp   { get; private set; } // 밖에서는 못 바꿈
    public int    Level { get; set; } = 1;   // 기본값

    // 주 생성자 (C# 12+)
    public Player(string name, int hp) => (Name, Hp) = (name, hp);

    public void TakeDamage(int dmg)
    {
        Hp = Math.Max(0, Hp - dmg);
        if (Hp == 0) Died?.Invoke(this);
    }

    public event Action<Player>? Died;       // 이벤트 (c09 에서)

    public override string ToString() => $"{Name}(Lv.{Level})";
}

record 와 struct 는 언제

// record — '값으로 비교되는' 데이터 묶음 (DTO·설정·이벤트)
public record ItemDto(int Id, string Name, int Price);

var a = new ItemDto(1, "칼", 100);
var b = new ItemDto(1, "칼", 100);
Console.WriteLine(a == b);           // True ← 내용이 같으면 같다
var c = a with { Price = 200 };      // 일부만 바꾼 복사본

// struct — 작고 많이 만들어지는 값 (좌표·색·ID)
public readonly record struct Tile(int X, int Y);   // 값 타입 record

C# 06

상속 · 인터페이스

게임 코드가 커질 때 가장 먼저 무너지는 곳입니다. 상속은 얕게, 인터페이스는 작게.

문법

public abstract class Enemy               // 추상 — 직접 생성 불가
{
    public abstract int Hp { get; }       // 자식이 반드시 구현
    public virtual void OnSpawn() { }     // 자식이 바꿔도 되고 안 바꿔도 됨
    protected void Log(string m) => Console.WriteLine(m);
}

public sealed class Slime : Enemy         // sealed — 더는 상속 못 함(성능↑)
{
    public override int Hp => 30;
    public override void OnSpawn()
    {
        base.OnSpawn();                   // 부모 것도 실행
        Log("슬라임 등장");
    }
}

public interface IDamageable
{
    void TakeDamage(int amount);
    bool IsAlive => true;                 // 기본 구현 (C# 8+)
}

게임에서는 상속보다 조합

상속으로 밀어붙이면
Enemy
 └ FlyingEnemy
    └ FlyingShootingEnemy
       └ FlyingShootingBossEnemy
"날면서 폭발하는 보스"는 어디에?
기능을 부품으로
class Enemy {
  IMovement move;    // 걷기·날기
  IAttack   attack;  // 근접·원거리
  IDeath    death;   // 폭발·드롭
}
Unity 는 이 조합 방식을 엔진 차원에서 강제합니다. GameObject 에 컴포넌트를 붙여 능력을 더하는 구조가 정확히 이것입니다 (🎮 Unity 기초 탭 02 컴포넌트). 스크립트를 짤 때도 상속 3단계를 넘기지 않는 것을 규칙으로 두세요.

인터페이스로 프로그래밍한다 — 실전 감각

// "무엇이든 맞을 수 있는 것"을 향해 코드를 짭니다
void Explode(Vector3 center, float radius, int dmg)
{
    foreach (var hit in FindInRadius(center, radius))
        if (hit is IDamageable target)     // 적·상자·벽 무엇이든
            target.TakeDamage(dmg);
}

// 새 오브젝트(드럼통)를 추가할 때
public class Barrel : IDamageable          // 인터페이스만 구현하면
{
    public void TakeDamage(int dmg) => ExplodeChain();
}
// ★ Explode() 는 한 글자도 고치지 않습니다 —
//    "확장에는 열려 있고 수정에는 닫혀 있다"의 실제 모습

언제 abstract, 언제 interface

질문답
공유할 코드·상태가 있다abstract class — 절반 만든 부모
약속(시그니처)만 필요하다interface — 계약서
여러 능력을 동시에 가져야 한다interface — 클래스 상속은 1개뿐

인터페이스 이름은 능력으로 짓습니다 — IDamageable(맞을 수 있는) · IInteractable(상호작용 가능한) · ISaveable(저장되는). "~할 수 있는" 으로 읽히면 잘 지은 것입니다.
메서드 하나짜리 클래스를 위해 인터페이스+팩토리를 만들지 마세요. 추상화는 두 번째 구현이 등장할 때 도입해도 늦지 않습니다 — 미리 만든 추상화의 절반은 예측이 빗나갑니다.
C# 07

컬렉션 완전정복

무엇을 고르느냐가 곧 성능입니다. 넣기·찾기·순회 중 무엇이 잦은지로 결정합니다.

기본 네 가지

// List — 순서가 있고 인덱스로 접근 (가장 많이 씁니다)
var enemies = new List<Enemy>(capacity: 64);   // 크기를 알면 미리 잡기
enemies.Add(e);  enemies.RemoveAt(0);

// Dictionary — 키로 즉시 찾기 O(1)
var byId = new Dictionary<int, Item>();
if (byId.TryGetValue(id, out var item)) { }    // ★ 있으면 꺼내기

// HashSet — 중복 없이 담고 '있는지'만 확인
var visited = new HashSet<int>();
if (!visited.Add(nodeId)) return;              // 이미 있었다면 false

// Queue / Stack — 순서 규칙
var q = new Queue<Job>();  q.Enqueue(j);  q.Dequeue();
var s = new Stack<Node>(); s.Push(n);     s.Pop();
ContainsKey 뒤에 [key] 로 또 꺼내지 마세요. 해시 계산을 두 번 합니다. TryGetValue 한 번이면 됩니다.

고르는 표 · 흔한 실수

하려는 일고를 것비고
순서대로 담고 순회List가장 무난
ID 로 즉시 찾기DictionaryO(1)
중복 제거 · 포함 검사HashSetO(1)
먼저 온 것부터 처리Queue작업 큐

List.Remove 는 O(n) 입니다. 매 프레임 수백 개를 지우면 그것만으로 느려집니다. 게임에서는 지우는 대신 비활성화하고 재사용하세요 — 오브젝트 풀링(🕹️ Unity 실전 탭)의 출발점입니다.
초기 크기를 주면 재할당이 사라집니다. new List<T>(1000) 처럼 예상 크기를 넘기는 것만으로 내부 배열 복사가 없어져 GC 압력이 눈에 띄게 줄어듭니다.
C# 08

제네릭 · null 안전

제네릭은 타입을 매개변수로 받는 것, null 안전은 NullReferenceException 을 컴파일 시점으로 옮기는 것입니다.

제네릭

// 타입만 다르고 로직이 같은 코드를 하나로
public class Pool<T> where T : class, new()
{
    private readonly Stack<T> _idle = new();

    public T Rent() => _idle.Count > 0 ? _idle.Pop() : new T();
    public void Return(T item) => _idle.Push(item);
}

var bullets = new Pool<Bullet>();

// 제약(where) 종류
// where T : class      참조 타입만
// where T : struct     값 타입만
// where T : new()      기본 생성자 있어야
// where T : IComparable<T>   인터페이스 구현

Unity 의 Update() 안에서 LINQ 를 쓰지 마세요. Where·OrderBy 는 내부적으로 객체를 만들어 매 프레임 쓰레기를 남깁니다. 초기화 때 한 번이나 이벤트가 있을 때만 쓰고, 루프 안에서는 평범한 for 문이 정답입니다.

읽는 훈련 — 실무 LINQ 는 이렇게 생겼습니다

// "부서별 평균 급여 상위 3개 부서" — SQL 로 치면 GROUP BY + ORDER BY + LIMIT
var top = employees
    .Where(e => e.IsActive)                 // ① 재직자만
    .GroupBy(e => e.Dept)                   // ② 부서로 묶고
    .Select(g => new                        // ③ 그룹마다 요약을 만들고
    {
        Dept = g.Key,
        Avg  = g.Average(e => e.Salary),
        Count = g.Count(),
    })
    .Where(x => x.Count >= 3)               // ④ 3명 이상 부서만
    .OrderByDescending(x => x.Avg)          // ⑤ 평균 높은 순
    .Take(3)                                // ⑥ 상위 3개
    .ToList();                              // ⑦ 여기서 비로소 실행

// 위에서 아래로 "거르고 → 묶고 → 요약하고 → 정렬하고 → 자른다"
// 같은 일을 for 문으로 짜면 30줄 + 중간 Dictionary 두 개가 필요합니다

WHEREWhere

JOINJoin — 또는 Dictionary 로 O(1) 매칭
