돈에 float · double 을 쓰면 틀립니다. 0.1 + 0.2 != 0.3 이기 때문입니다. 금액·수량 정산은 decimal, 게임의 좌표·속도처럼 오차가 허용되는 값은 float 이 정답입니다.

값 타입 vs 참조 타입 — 이 구분이 성능을 가릅니다

값 타입 (struct)
  • int · float · bool · Vector3
  • 스택에 놓이고 복사됩니다
  • GC 부담이 없습니다 — 게임 루프에서 중요
참조 타입 (class)
  • string · 배열 · List · 내가 만든 class
  • 힙에 놓이고 주소가 복사됩니다
  • 많이 만들면 GC 가 돌며 프레임이 튑니다
struct Point { public int X, Y; }
class   Node  { public int Value; }

var p1 = new Point { X = 1 };
var p2 = p1;  p2.X = 99;        // 복사본만 바뀐다
Console.WriteLine(p1.X);        // 1

var n1 = new Node { Value = 1 };
var n2 = n1;  n2.Value = 99;    // 같은 객체를 가리킨다
Console.WriteLine(n1.Value);    // 99  ← 함께 바뀐다
Unity 의 Vector3·Quaternion 은 struct 입니다. 그래서 transform.position.x = 1; 이 컴파일되지 않습니다 — 복사본을 고쳐 봐야 원본이 안 바뀌기 때문입니다. transform.position = new Vector3(1, y, z); 처럼 통째로 넣어야 합니다.

조건 · 반복 · switch 식

C# 의 switch 는 값을 돌려주는 식입니다. 여기부터 코드가 짧아집니다.

반복문 네 가지

switch 식 · 패턴 매칭

// 값을 돌려주는 switch — if/else 사슬이 사라집니다
string Grade(int score) => score switch
{
    >= 90 => "S",
    >= 80 => "A",
    >= 70 => "B",
    _      => "F",          // _ 는 나머지 전부
};

// 타입으로 분기
decimal Damage(IWeapon w) => w switch
{
    Sword s when s.IsCritical => s.Power * 2,
    Sword s                   => s.Power,
    Bow { Arrows: 0 }         => 0,          // 속성 패턴
    Bow b                     => b.Power * 0.8m,
    null                      => throw new ArgumentNullException(nameof(w)),
    _                         => 1,
};

// 튜플 패턴 — 가위바위보
string Result(int me, int you) => (me, you) switch
{
    (0, 2) or (1, 0) or (2, 1) => "승",
    var (a, b) when a == b     => "무",
    _                          => "패",
};
모든 경우를 다 적으면 _ 없이도 됩니다. enum 을 switch 할 때 _ 를 빼 두면, 나중에 enum 에 값을 추가했을 때 컴파일러가 경고해 줍니다 — 빠뜨린 분기를 사람이 찾지 않아도 됩니다.

클래스 · 구조체 · record

셋 다 "데이터 묶음"이지만 쓰는 자리가 다릅니다. 고르는 기준을 정리합니다.

클래스 — 기본 선택지

public class Player
{
    // 자동 속성
    public string Name { get; }              // 읽기 전용
    public int    Hp   { get; private set; } // 밖에서는 못 바꿈
    public int    Level { get; set; } = 1;   // 기본값

    // 주 생성자 (C# 12+)
    public Player(string name, int hp) => (Name, Hp) = (name, hp);

    public void TakeDamage(int dmg)
    {
        Hp = Math.Max(0, Hp - dmg);
        if (Hp == 0) Died?.Invoke(this);
    }

    public event Action<Player>? Died;       // 이벤트 (c09 에서)

    public override string ToString() => $"{Name}(Lv.{Level})";
}

static T Max<T>(T a, T b) where T : IComparable<T>
    => a.CompareTo(b) >= 0 ? a : b;                   // ✓

// 제네릭 수학 (.NET 7+) — int·float·decimal 을 하나로
static T Sum<T>(ReadOnlySpan<T> xs) where T : INumber<T>
{
    var acc = T.Zero;
    foreach (var x in xs) acc += x;
    return acc;
}
Sum<int>([1, 2, 3]);        // 6
Sum<double>([1.5, 2.5]);    // 4.0

{

    private readonly T _prefab;

    public T Rent()
    {

        x.gameObject.SetActive(true);

    {
        x.gameObject.SetActive(false);
        _idle.Push(x);
    }

// ✗ 클라이언트 시계로 판정 — 폰 시계를 돌리면 쿨타임·출석이 뚫립니다
if (DateTime.Now >= nextRewardAt) GiveReward();

// ✓ 판정은 서버 시각으로만 — 클라이언트는 표시만 합니다
public sealed class RewardService(TimeProvider clock)
{
    public bool TryClaim(Player p)
    {
        var now = clock.GetUtcNow();                  // 서버 UTC
        if (now < p.NextRewardAtUtc) return false;    // 아직입니다
        p.NextRewardAtUtc = NextResetUtc(now);
        Give(p);
        return true;
    }

    // "매일 KST 오전 5시 리셋" — 규칙은 로컬로 정의하고 UTC 로 저장
    static DateTimeOffset NextResetUtc(DateTimeOffset utcNow)
    {
        var kst   = TimeZoneInfo.FindSystemTimeZoneById("Asia/Seoul");
        var local = TimeZoneInfo.ConvertTime(utcNow, kst);
        var reset = new DateTimeOffset(local.Year, local.Month, local.Day,
                                       5, 0, 0, local.Offset);
        if (local >= reset) reset = reset.AddDays(1); // 오늘 5시가 지났으면 내일
        return reset.ToUniversalTime();
    }
}

// 클라이언트는 서버가 준 시각과의 차이만 그립니다
var remain = serverNextResetUtc - serverNowUtc;      // 응답에 함께 받은 값
timerText = $"{remain.Hours:D2}:{remain.Minutes:D2}:{remain.Seconds:D2}";
이벤트 시작·쿨타임·랭킹 마감은 전부 서버 UTC 로 판정하세요. 클라이언트 시계는 사용자가 마음대로 바꿀 수 있는 입력값입니다. DB 에는 UTC(또는 Unix 초)로 저장하고, "한국 시간 오전 5시" 같은 운영 규칙만 위처럼 변환 함수 한 곳에 모아 두면 서머타임이 있는 지역에 서버를 옮겨도 코드가 바뀌지 않습니다.
서비스가 국경을 넘는 순간 UTC 로 저장하지 않은 대가를 전부 치르게 됩니다.

열거형과 플래그 — 상태를 안전하게

"상태를 문자열로 들고 다니는 코드"는 오타 하나에 무너집니다. enum 은 상태에 이름과 타입을 줍니다.

enum 기본 — 게임 상태 예제

// ★ 번호를 명시하세요 — 저장된 값과의 계약입니다
public enum GameState
{
    Title    = 0,
    Loading  = 1,
    Playing  = 2,
    Paused   = 3,
    GameOver = 4,
}

// switch 식과 조합 — c04 의 규칙: _ 를 빼면
// 값을 추가했을 때 빠진 분기를 컴파일러가 경고합니다
string Bgm(GameState s) => s switch
{
    GameState.Title    => "bgm_title",
    GameState.Loading  => "bgm_none",
    GameState.Playing  => "bgm_field",
    GameState.Paused   => "bgm_field",
    GameState.GameOver => "bgm_gameover",
};

// 상태 전환도 표로 — 잘못된 전환을 타입이 막아 줍니다
bool CanChange(GameState from, GameState to) => (from, to) switch
{
    (GameState.Title,   GameState.Loading) => true,
    (GameState.Loading, GameState.Playing) => true,
    (GameState.Playing, GameState.Paused or GameState.GameOver) => true,
    (GameState.Paused,  GameState.Playing) => true,
    _ => false,
};

// 전체 값 순회 — 디버그 메뉴·드롭다운 만들 때
foreach (var s in Enum.GetValues<GameState>())
    AddDebugButton(s.ToString());

[Flags] — 하나의 값에 여러 상태를

// 비트 하나가 상태 하나 — 반드시 2의 거듭제곱으로
[Flags]
public enum BuffState
{
    None    = 0,
    Poison  = 1 << 0,   // 1
    Stun    = 1 << 1,   // 2
    Shield  = 1 << 2,   // 4
    Haste   = 1 << 3,   // 8
    Debuffs = Poison | Stun,   // 조합에도 이름을 붙일 수 있습니다
}

var buff = BuffState.Poison | BuffState.Shield;   // 켜기: OR

// 검사 — 비트 AND (뜨거운 경로에서는 HasFlag 보다 이쪽)
if ((buff & BuffState.Poison) != 0) TickPoison();
if (buff.HasFlag(BuffState.Stun)) return;         // 읽기는 이쪽이 편합니다

buff &= ~BuffState.Poison;     // 끄기: AND NOT
buff ^= BuffState.Haste;       // 뒤집기: XOR

// [Flags] 를 붙이면 ToString 이 조합을 읽어 줍니다
Console.WriteLine(buff);       // "Shield, Haste" (없으면 그냥 숫자)
[Flags] 는 표시용 애트리뷰트일 뿐, 값 검증을 해 주지 않습니다. 2의 거듭제곱을 안 지키면 Poison | Stun == 3 == 다른 값 같은 충돌이 조용히 일어납니다. 값이 5개를 넘보기 시작하면 1 << n 표기로만 적는 것을 규칙으로 두세요.

파싱 — 밖에서 온 값은 믿지 않습니다

// 설정 파일·패킷·DB 에서 문자열로 들어올 때
if (Enum.TryParse<GameState>(input, ignoreCase: true, out var state))
    Apply(state);

// ★ 함정: TryParse 는 아무 숫자나 통과시킵니다
Enum.TryParse<GameState>("99", out var bad);   // true! bad == (GameState)99

// 숫자 입력 가능성이 있으면 정의 여부까지 확인
if (Enum.TryParse<GameState>(input, true, out var s)
    && Enum.IsDefined(s))
    Apply(s);

// 캐스팅도 검증이 없습니다 — DB 의 int 를 받을 때 주의
var fromDb = (GameState)row.State;             // 99 여도 예외 없음
if (!Enum.IsDefined(fromDb)) fromDb = GameState.Title;

var opt = new JsonSerializerOptions

[Flags] 조합값에는 Enum.IsDefined 가 false 를 돌려줍니다. Poison | Shield == 5 는 "정의된 이름"이 아니기 때문입니다. 플래그 검증은 (value & ~All) == 0 처럼 허용 비트 밖이 비었는지로 확인하세요.

 숫자 저장문자열 저장
용량·속도작고 빠름 — 패킷·랭킹에큼 — 수백만 행이면 체감
이름 변경(rename)안전 — 번호만 유지하면 됨기존 데이터가 못 읽힘

어느 쪽이든 "지우지 않는다"가 진짜 규칙입니다. 숫자로 저장한다면 번호를 반드시 명시하고 재사용하지 않기, 문자열로 저장한다면 이름을 바꾸지 않기. 이미 저장된 값은 코드보다 오래 삽니다 — 옛 값을 만났을 때의 기본값 처리(IsDefined 검사)까지가 한 세트입니다.
enum 에 로직이 자꾸 붙으면 클래스로 승격할 때입니다. switch 가 대여섯 군데로 흩어지기 시작하면 상태별 동작을 가진 상태 클래스(전략 패턴)가 더 맞습니다 — enum 은 데이터가 단순한 이름표일 때 가장 좋습니다.
