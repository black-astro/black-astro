미들웨어 vs 서블릿 필터 — 닮았지만 결정적으로 다릅니다 🟡 중급

좌우 대조 — 흩어진 순서 vs 눈에 보이는 순서. 이 차이 하나가 운영 난이도를 가릅니다.

직접 만든 미들웨어 — 상관 관계 ID

public sealed class CorrelationIdMiddleware(RequestDelegate next)
{
    private const string Header = "X-Correlation-Id";

    public async Task InvokeAsync(HttpContext ctx, ILogger<CorrelationIdMiddleware> log)
    {
        // ── 들어가는 길 ──────────────────────────
        var id = ctx.Request.Headers[Header].FirstOrDefault()
                 ?? Guid.NewGuid().ToString("N");
        ctx.Items["cid"] = id;

        using (log.BeginScope(new Dictionary<string, object> { ["cid"] = id }))
        {
            ctx.Response.OnStarting(() =>      // 응답 헤더는 "시작 직전"에
            {
                ctx.Response.Headers[Header] = id;
                return Task.CompletedTask;
            });

            await next(ctx);                    // ← 여기가 chain.doFilter()
        }
        // ── 나오는 길 ────────────────────────────
        // 여기서는 이미 응답이 나가는 중이라 헤더를 못 고칩니다
    }
}

// 등록 — 확장 메서드로 감싸는 것이 관행입니다
public static class CorrelationIdExtensions
{
    public static IApplicationBuilder UseCorrelationId(this IApplicationBuilder b)
        => b.UseMiddleware<CorrelationIdMiddleware>();
}

app.UseCorrelationId();   // UseSerilogRequestLogging 보다 앞에

Spring 필터로 쓰면

@Component
@Order(Ordered.HIGHEST_PRECEDENCE)
public class CorrelationIdFilter extends OncePerRequestFilter {

    @Override
    protected void doFilterInternal(HttpServletRequest req,
                                    HttpServletResponse res,
                                    FilterChain chain)
            throws IOException, ServletException {
        String id = Optional.ofNullable(req.getHeader("X-Correlation-Id"))
                            .orElse(UUID.randomUUID().toString());
        MDC.put("cid", id);
        res.setHeader("X-Correlation-Id", id);
        try {
            chain.doFilter(req, res);
        } finally {
            MDC.remove("cid");     // 스레드풀 재사용 때문에 반드시 지워야 합니다
        }
    }
}
MDC.remove 에 해당하는 실수가 .NET 엔 없습니다. BeginScope 는 using 이 끝나면 자동으로 걷히고, 스코프는 스레드가 아니라 AsyncLocal 로 비동기 흐름을 따라 다닙니다. await 뒤에 스레드가 바뀌어도 상관 ID 가 살아 있습니다.
OnStarting 을 쓰는 이유. await next() 뒤에서 Response.Headers 를 건드리면 "Headers are read-only, response has already started" 예외가 납니다. 응답이 이미 소켓으로 나가기 시작했기 때문입니다. 엔터프라이즈 코드에서 가장 자주 보는 초보 예외 1위입니다.

기동·종료에 훅 걸기

// Spring 의 @PostConstruct / @PreDestroy / ApplicationRunner 자리
public sealed class WarmupService(
    IServiceProvider sp,
    IHostApplicationLifetime life,
    ILogger<WarmupService> log) : IHostedService
{
    public async Task StartAsync(CancellationToken ct)
    {
        // 기동 시 캐시 예열 — 첫 사용자가 느린 것을 막습니다
        using var scope = sp.CreateScope();     // ★ Scoped 는 스코프를 열어야 씁니다
        var db = scope.ServiceProvider.GetRequiredService<PayrollDbContext>();
        await db.Departments.AsNoTracking().LoadAsync(ct);
        log.LogInformation("부서 캐시 예열 완료");

        life.ApplicationStopping.Register(() =>
            log.LogInformation("종료 신호 수신 — 새 요청을 받지 않습니다"));
    }

    public Task StopAsync(CancellationToken ct) => Task.CompletedTask;
}

생성자에서 비동기를 못 씁니다. @PostConstruct 안에서 DB 를 읽던 코드는 IHostedService.StartAsync 로 옮겨야 합니다. 생성자에서 .Result 나 .Wait() 를 쓰면 기동 중 데드락이 납니다 — 실제로 겪습니다.
순서가 필요하면 등록 순서를 씁니다. AddHostedService 는 등록한 순서대로 StartAsync 를 부르고, 종료할 때는 역순으로 StopAsync 를 부릅니다. 마이그레이션 실행기를 먼저, 배치 워커를 나중에 등록하면 순서가 맞습니다.
Kestrel 은 앱의 일부입니다 — 이 문장이 전환의 핵심입니다. 톰캣에 WAR 를 올리던 감각으로 보면 "웹 서버를 어디에 설치하지?" 가 계속 걸립니다. 설치하지 않습니다. dotnet Payroll.Api.dll 을 실행하는 순간 그 프로세스가 곧 웹 서버입니다. Nginx 나 IIS 는 여전히 앞에 둘 수 있지만 그것은 서블릿 컨테이너가 아니라 리버스 프록시 역할입니다 (🔷 h01·h06 에서 끝까지 다룹니다).
정리 — Program.cs 는 빈 설정 파일 + 필터 등록 + main() 을 합친 파일입니다. Build() 앞은 등록, 뒤는 사용. 미들웨어는 쓴 순서대로 들어갔다가 역순으로 나옵니다. 기동·종료 훅이 필요하면 IHostedService 를 씁니다.

DI 컨테이너 — 수명 세 가지가 사고의 90%입니다

Spring 은 기본이 싱글턴이고 웹 요청 스코프는 일부러 켜야 합니다. .NET 은 반대로 Scoped(요청당 하나)를 기본으로 쓰는 문화입니다. 이 차이를 모르고 넘어오면 DbContext 를 싱글턴으로 등록하는 사고가 반드시 한 번은 납니다. 수명이 겹치면 무슨 일이 생기는지를 그림으로 먼저 보겠습니다.

간트 — 수명은 길이입니다. 긴 막대가 짧은 막대를 붙잡는 순간 버그가 됩니다.
수명Spring 대응언제 쓰나주의
Singleton기본 스코프 (singleton)설정 · 시계 · HttpClient 팩토리 · 캐시 · 순수 계산기스레드 안전해야 합니다. 여러 요청이 동시에 들어옵니다
Scoped@RequestScopeDbContext · 리포지토리 · 서비스 · 현재 사용자 정보배경 작업에서 쓰려면 CreateScope() 를 직접 열어야 합니다
Transient@Scope("prototype")가벼운 검증기 · 빌더 · 전략 객체IDisposable 이면 스코프가 끝날 때까지 안 버려집니다

// Payroll.Domain/Approvals/ApprovalDocument.cs
// using 문에 EF Core 도 ASP.NET Core 도 없습니다.
namespace Payroll.Domain.Approvals;

public sealed class ApprovalDocument
{
    private readonly List<ApprovalStep> _steps = [];

    public Guid Id { get; private set; } = Guid.CreateVersion7();
    public string Title { get; private set; } = "";
    public decimal Amount { get; private set; }
    public DocumentStatus Status { get; private set; } = DocumentStatus.Draft;
    public IReadOnlyList<ApprovalStep> Steps => _steps;

    // 업무 규칙이 클래스 안에 있습니다 — 밖에서 상태를 못 바꿉니다
    public Result Approve(Guid approverId, DateTimeOffset now)
    {
        if (Status is not DocumentStatus.InProgress)
            return Result.Fail("진행 중인 문서만 결재할 수 있습니다");

        var step = _steps.FirstOrDefault(s => s.ApproverId == approverId
                                           && s.State == StepState.Waiting);
        if (step is null) return Result.Fail("결재 차례가 아닙니다");

        step.Approve(now);
        if (_steps.All(s => s.State == StepState.Approved))
            Status = DocumentStatus.Completed;

        return Result.Ok();
    }
}

// Payroll.Application/Approvals/ApproveDocument.cs
public sealed record ApproveDocumentCommand(Guid DocumentId, Guid ApproverId)
    : IRequest<Result>;

public sealed class ApproveDocumentHandler(
    IApprovalRepository repo,
    IUnitOfWork uow,
    IClock clock) : IRequestHandler<ApproveDocumentCommand, Result>
{
    public async Task<Result> Handle(ApproveDocumentCommand cmd, CancellationToken ct)
    {
        var doc = await repo.FindWithStepsAsync(cmd.DocumentId, ct);
        if (doc is null) return Result.Fail("문서 없음");

        var r = doc.Approve(cmd.ApproverId, clock.Now);
        if (r.IsFailure) return r;

        await uow.SaveChangesAsync(ct);   // 여기가 트랜잭션 경계
        return Result.Ok();
    }
}

파이프라인 동작 = AOP 자리

// Spring 의 @Transactional · @Validated 애스펙트에 해당합니다
public sealed class ValidationBehavior<TReq, TRes>(
    IEnumerable<IValidator<TReq>> validators)
    : IPipelineBehavior<TReq, TRes> where TReq : notnull
{
    public async Task<TRes> Handle(TReq req,
        RequestHandlerDelegate<TRes> next, CancellationToken ct)
    {
        var ctx = new ValidationContext<TReq>(req);
        var failures = validators
            .Select(v => v.Validate(ctx))
            .SelectMany(r => r.Errors)
            .Where(f => f is not null).ToList();

        if (failures.Count > 0) throw new ValidationException(failures);
        return await next(ct);
    }
}

public readonly record struct Result(bool IsSuccess, string? Error)

ENT 05

EF Core 10 vs JPA — 영속성 컨텍스트가 어디로 갔나

JPA 를 제대로 써 본 사람은 EF Core 를 아주 빨리 익힙니다. 영속성 컨텍스트 = ChangeTracker, flush = SaveChanges, JPQL = LINQ 로 거의 그대로 옮겨집니다. 다른 것은 지연 로딩이 기본이 아니라는 점과 변경 감지 방식 두 가지입니다. 그 두 가지가 N+1 문제의 얼굴을 바꿔 놓습니다. (EF Core 기초 사용법은 🔷 w03 에 있습니다. 여기서는 운영에서 터지는 것들만 다룹니다.)

상태 기계 — SaveChanges 가 모든 전이의 방아쇠입니다. 그 전까지는 메모리에서만 일어납니다.

결재·재고·좌석처럼 여러 사람이 같은 행을 동시에 건드리는 업무에서는 "마지막에 저장한 사람이 이깁니다" 가 곧 데이터 사고입니다. 팀장 두 명이 같은 문서를 동시에 열어 한 명은 반려, 한 명은 승인하면 버전 토큰이 없을 때 둘 다 성공하고 결과는 하나만 남습니다.

비관적 락이 필요할 때도 있습니다. db.Documents.FromSql($"SELECT * FROM documents WHERE id = {id} FOR UPDATE") 같은 식으로 SELECT ... FOR UPDATE 를 직접 씁니다. 짧은 트랜잭션 안에서만 쓰세요 — 야간 배치와 겹치면 데드락으로 갑니다 (🔷 e06).
웹에서는 "긴 낙관적 잠금" 이 정답입니다. 화면을 열 때 받은 Version 을 클라이언트가 들고 있다가 저장 요청에 같이 보냅니다. 서버는 그 값을 엔티티에 세팅하고 저장합니다 — entry.Property(x => x.Version).OriginalValue = dto.Version; 사용자가 커피를 마시고 온 30분 동안 DB 커넥션을 잡고 있을 수는 없습니다.

트랜잭션 — @Transactional 이 없는 세계에서 경계 긋기

Spring 개발자가 .NET 에서 가장 크게 당황하는 지점입니다. 애너테이션 하나로 열리던 트랜잭션이 없습니다. 대신 SaveChanges() 한 번이 곧 트랜잭션 하나라는 아주 단순한 규칙이 있고, 그 규칙만으로 부족할 때 BeginTransaction 과 TransactionScope 가 있습니다. 그리고 분산 트랜잭션은 사실상 없습니다 — 그 자리를 아웃박스 패턴이 메웁니다.

기본 — SaveChanges 가 곧 트랜잭션

// 이 안의 변경은 전부 한 트랜잭션입니다. 명시적 선언이 필요 없습니다.
doc.Approve(userId, now);              // Modified
db.AuditLogs.Add(new AuditLog(...));   // Added
db.Notifications.Add(new Notify(...)); // Added

await db.SaveChangesAsync(ct);
// → BEGIN; UPDATE documents...; INSERT audit_logs...;
//   INSERT notifications...; COMMIT;
// 하나라도 실패하면 전부 롤백됩니다.
이 사실 하나가 @Transactional 의 80%를 대체합니다. 유스케이스 핸들러가 SaveChanges 를 한 번만 부르도록 짜면 트랜잭션 경계 고민이 사라집니다. "핸들러 하나 = 트랜잭션 하나" 를 팀 규칙으로 두세요.
SaveChanges 를 루프 안에서 부르지 마세요. Spring 에서 saveAll 을 부르던 자리에 반복 저장을 넣으면 트랜잭션이 N개가 됩니다 — 중간에 실패하면 절반만 반영된 상태가 남습니다. 모으고 마지막에 한 번 부르는 것이 정답입니다.

명시적 트랜잭션이 필요한 세 경우

@Transactional 전파 속성은 어떻게 대응하나 🔴 실무

Spring 전파뜻.NET 대응
REQUIRED (기본)있으면 참여, 없으면 시작기본값 — 같은 DbContext 를 쓰면 자동으로 참여
REQUIRES_NEW기존 것을 멈추고 새로 시작TransactionScope(Required New) 또는 별도 DbContext (권장)
SUPPORTS있으면 참여, 없으면 그냥기본 동작과 같음 (신경 쓸 것 없음)
NOT_SUPPORTED트랜잭션 없이 실행TransactionScope(Suppress) · 새 커넥션
MANDATORY없으면 예외db.Database.CurrentTransaction is null 검사
NEVER있으면 예외같은 방식으로 직접 검사
NESTED세이브포인트tx.CreateSavepointAsync("s1") / RollbackToSavepointAsync

        while (!ct.IsCancellationRequested)
        {
            await using var db = await factory.CreateDbContextAsync(ct);

            // 여러 인스턴스가 같은 행을 집지 않도록 잠급니다
            var batch = await db.OutboxMessages
                .Where(m => m.PublishedAt == null && m.Attempts < 10)
                .OrderBy(m => m.OccurredAt)
                .Take(100)
                .FromSql($@"SELECT * FROM outbox_messages
                            WHERE published_at IS NULL
                            ORDER BY occurred_at LIMIT 100
                            FOR UPDATE SKIP LOCKED")
                .ToListAsync(ct);

            foreach (var m in batch)
            {
                try
                {
                    await bus.PublishAsync(m.Type, m.Payload, ct);
                    m.MarkPublished(DateTimeOffset.UtcNow);
                }
                catch (Exception ex)
                {
                    m.MarkFailed(ex.Message);      // attempts++ · last_error
                    log.LogWarning(ex, "아웃박스 발행 실패 {Id}", m.Id);
                }
            }
            await db.SaveChangesAsync(ct);
            await Task.Delay(TimeSpan.FromSeconds(5), ct);
        }
    }
}
FOR UPDATE SKIP LOCKED 가 핵심입니다. 인스턴스 3대가 동시에 돌면 같은 메시지를 세 번 보냅니다. 이 힌트를 주면 이미 잡힌 행은 건너뜁니다. SQL Server 라면 WITH (UPDLOCK, READPAST) 입니다. 그래도 소비 쪽은 반드시 멱등하게 짜세요 — 중복은 언젠가 옵니다 (🔷 f10).
보상 트랜잭션 — 되돌릴 수 없는 일이 섞일 때. "결재 승인 → 회계 시스템 전표 생성 → 메일 발송" 처럼 남의 시스템이 끼면 롤백이 불가능합니다. 이때는 각 단계를 성공/실패로 기록하고, 실패 시 앞 단계를 되돌리는 반대 작업을 정의합니다 — 전표 생성 실패 시 승인 취소 대신 "전표 오류" 상태로 두고 담당자 화면에 띄우는 편이 현실적인 경우가 많습니다. 업무 담당자와 합의된 보상 규칙이 기술보다 중요합니다.
정리 — SaveChanges 한 번이 트랜잭션 하나입니다. 핸들러 하나 = SaveChanges 하나 를 규칙으로 두면 대부분 끝납니다. REQUIRES_NEW 는 DbContext 를 하나 더, 재시도를 켰다면 실행 전략으로 감싸고, DB 와 메시지를 함께 보내야 하면 아웃박스를 씁니다.

        SELECT  d.dept_code                              AS DeptCode,

public sealed class BusinessExceptionHandler(
    IProblemDetailsService pds,
    ILogger<BusinessExceptionHandler> log) : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(
        HttpContext ctx, Exception ex, CancellationToken ct)
    {
        var (status, title) = ex switch
        {
            ValidationException      => (400, "입력값이 올바르지 않습니다"),
            NotFoundException        => (404, "대상을 찾을 수 없습니다"),
            BusinessRuleException    => (409, "업무 규칙 위반"),
            UnauthorizedAccessException => (403, "권한이 없습니다"),
            DbUpdateConcurrencyException => (409, "다른 사용자가 먼저 처리했습니다"),
            _                        => (0, "")            // 우리가 모르는 예외
        };
        if (status == 0)
        {
            log.LogError(ex, "처리되지 않은 예외");
            return false;      // 다음 핸들러(기본 500)로 넘깁니다
        }

        ctx.Response.StatusCode = status;
        return await pds.TryWriteAsync(new ProblemDetailsContext
        {
            HttpContext = ctx,
            ProblemDetails = new ProblemDetails
            {
                Title  = title,
                Detail = ex.Message,
                Status = status,
                Type   = $"https://wiki.acme/errors/{ex.GetType().Name}"
            }
        });
    }
}

public sealed class ApprovalLimitHandler(IUserGradeReader grades)
    : AuthorizationHandler<ApprovalLimitRequirement, ApprovalDocument>
{
    protected override async Task HandleRequirementAsync(
        AuthorizationHandlerContext ctx,
        ApprovalLimitRequirement req,
        ApprovalDocument resource)          // ★ 자원을 함께 받습니다
    {
        var userId = ctx.User.GetUserId();
        var grade  = await grades.GetAsync(userId);

        var limit = grade switch
        {
            Grade.TeamLead   => 10_000_000m,
            Grade.Director   => 100_000_000m,
            Grade.Executive  => decimal.MaxValue,
            _                => 0m
        };

        if (resource.Amount <= limit && resource.DepartmentId == grade.DeptId)
            ctx.Succeed(req);               // 부르지 않으면 자동으로 실패입니다
    }
}

namespace Payroll.Domain.Approvals;

public enum DocumentStatus { Draft, InProgress, Completed, Rejected, Canceled }
public enum StepState      { Waiting, Approved, Rejected, Skipped }

public sealed class ApprovalDocument : IHasDomainEvents
{
    private readonly List<ApprovalStep> _steps = [];
    private readonly List<IDomainEvent> _events = [];

    public Guid   Id           { get; private set; } = Guid.CreateVersion7();
    public string Title        { get; private set; } = "";
    public decimal Amount      { get; private set; }
    public Guid   DrafterId    { get; private set; }
    public Guid   DepartmentId { get; private set; }
    public DocumentStatus Status { get; private set; } = DocumentStatus.Draft;
    public DateTimeOffset CreatedAt { get; private set; }
    public DateTimeOffset? CompletedAt { get; private set; }
    public uint Version { get; private set; }              // 낙관적 동시성 (🔷 e05)

    public IReadOnlyList<ApprovalStep> Steps => _steps;
    public IReadOnlyList<IDomainEvent> Events => _events;

    private ApprovalDocument() { }                          // EF Core 용

    public static ApprovalDocument Draft(
        string title, decimal amount, Guid drafterId, Guid deptId, DateTimeOffset now)
    {
        if (string.IsNullOrWhiteSpace(title)) throw new ArgumentException("제목 필수");
        if (amount <= 0) throw new ArgumentException("금액은 0보다 커야 합니다");
        return new ApprovalDocument
        {
            Title = title.Trim(), Amount = amount,
            DrafterId = drafterId, DepartmentId = deptId, CreatedAt = now
        };
    }

    // 결재선을 붙이고 상신 — 여기까지가 한 덩어리입니다
    public Result Submit(IReadOnlyList<Guid> approverIds, DateTimeOffset now)
    {
        if (Status is not DocumentStatus.Draft) return Result.Fail("이미 상신된 문서입니다");
        if (approverIds.Count is 0) return Result.Fail("결재선이 비어 있습니다");
        if (approverIds.Count > 5)  return Result.Fail("결재선은 5단계까지입니다");

        for (var i = 0; i < approverIds.Count; i++)
            _steps.Add(ApprovalStep.Create(Id, i + 1, approverIds[i]));

        Status = DocumentStatus.InProgress;
        _events.Add(new DocumentSubmitted(Id, DrafterId, approverIds[0], now));
        return Result.Ok();
    }

    public Result Approve(Guid approverId, DateTimeOffset now)
    {
        if (Status is not DocumentStatus.InProgress)
            return Result.Fail("진행 중인 문서만 결재할 수 있습니다");

        var current = _steps.Where(s => s.State == StepState.Waiting)
                            .MinBy(s => s.Order);
        if (current is null)              return Result.Fail("대기 중인 단계가 없습니다");
        if (current.ApproverId != approverId) return Result.Fail("결재 차례가 아닙니다");

        current.Approve(now);

        var next = _steps.Where(s => s.State == StepState.Waiting).MinBy(s => s.Order);
        if (next is null)
        {
            Status = DocumentStatus.Completed;
            CompletedAt = now;
            _events.Add(new DocumentCompleted(Id, DepartmentId, Amount, now));
        }
        else _events.Add(new StepArrived(Id, next.ApproverId, now));

        return Result.Ok();
    }

    public Result Reject(Guid approverId, string reason, DateTimeOffset now)
    {
        if (string.IsNullOrWhiteSpace(reason)) return Result.Fail("반려 사유는 필수입니다");
        var current = _steps.Where(s => s.State == StepState.Waiting).MinBy(s => s.Order);
        if (current?.ApproverId != approverId) return Result.Fail("결재 차례가 아닙니다");

        current.Reject(reason, now);
        Status = DocumentStatus.Rejected;
        _events.Add(new DocumentRejected(Id, DrafterId, reason, now));
        return Result.Ok();
    }

    public IReadOnlyList<IDomainEvent> PopEvents()
    {
        var copy = _events.ToArray();
        _events.Clear();

public sealed class AuditSaveChangesInterceptor(
    ICurrentUser user, IClock clock) : SaveChangesInterceptor
{
    public override ValueTask<InterceptionResult<int>> SavingChangesAsync(
        DbContextEventData e, InterceptionResult<int> r, CancellationToken ct = default)
    {
        var ctx = e.Context!;
        var now = clock.Now;

        // ① 도메인 이벤트 → 아웃박스 행 (🔷 e06)
        foreach (var ev in ctx.ChangeTracker.Entries<IHasDomainEvents>()
                              .SelectMany(x => x.Entity.PopEvents()))
            ctx.Add(OutboxMessage.From(ev, now));

        // ② 변경된 엔티티 → 감사 로그
        foreach (var entry in ctx.ChangeTracker.Entries()
                     .Where(x => x.State is EntityState.Added
                                          or EntityState.Modified
                                          or EntityState.Deleted))
        {
            if (entry.Entity is OutboxMessage or AuditLog) continue;   // 무한 루프 방지
            ctx.Add(new AuditLog(
                Guid.CreateVersion7(), user.EmployeeNo,
                $"{entry.Entity.GetType().Name}.{entry.State}",
                entry.Entity.GetType().Name,
                entry.Properties.First(p => p.Metadata.IsPrimaryKey())
                     .CurrentValue?.ToString() ?? "",
                user.Ip, user.CorrelationId, now));
        }
        return base.SavingChangesAsync(e, r, ct);
    }
}

public sealed record ApproveCommand(Guid DocumentId, Guid ApproverId)
    : IRequest<Result>;

public sealed class ApproveHandler(
    PayrollDbContext db,
    IAuthorizationService auth,
    ICurrentUser user,
    IClock clock,
    ApprovalMetrics metrics) : IRequestHandler<ApproveCommand, Result>
{
    public async Task<Result> Handle(ApproveCommand c, CancellationToken ct)
    {
        var doc = await db.ApprovalDocuments
            .Include(d => d.Steps)                 // N+1 방지 (🔷 e05)
            .FirstOrDefaultAsync(d => d.Id == c.DocumentId, ct);
        if (doc is null) return Result.Fail("문서 없음");

        // 자원 기반 권한 — 금액 한도 · 부서 검사 (🔷 e10)
        var authz = await auth.AuthorizeAsync(user.Principal, doc, "CanApprove");
        if (!authz.Succeeded) return Result.Fail("결재 권한이 없습니다");

        var r = doc.Approve(c.ApproverId, clock.Now);
        if (r.IsFailure) return r;

        // 문서 변경 + 결재선 변경 + 감사 로그 + 아웃박스 = 한 트랜잭션
        await db.SaveChangesAsync(ct);

        if (doc.Status is DocumentStatus.Completed)
            metrics.Completed(doc.DepartmentId.ToString(),
                (clock.Now - doc.CreatedAt).TotalHours);

        return Result.Ok();
    }
}

항목확인
시간대컨테이너는 UTC — 저장은 UTC, 표시만 KST
한글 인코딩로그·파일 출력에 UTF-8 확인 (🔷 f06)
메모리 한도한도를 주면 GC 가 맞춰 동작합니다 (🔷 h11)
마이그레이션앱이 아니라 배포 잡이 실행 (🔷 e05)
프록시 뒤UseForwardedHeaders 필수 (🔷 h06)
헬스체크ready 를 LB 에 연결 (🔷 e11)
시크릿이미지에 넣지 말고 환경 변수 (🔷 e09)
시간대는 업무 시스템의 단골 사고입니다. "3월 2일 결재분" 을 UTC 기준으로 집계하면 한국 시각 오전 9시 이전 건이 전날로 잡힙니다. 저장은 DateTimeOffset(UTC), 집계 기준일은 명시적으로 KST 로 변환 — TimeZoneInfo.FindSystemTimeZoneById("Asia/Seoul"). 리눅스와 윈도우에서 같은 ID 로 동작합니다(.NET 6+).
다음 단계 — 이 API 를 실제 사내 서버(IIS 또는 Ubuntu)에 올리는 절차는 🖧 서버 구축 탭에서 화면 단계별로 다룹니다 (🔷 h03·h05·h14). 야간 정산 배치와 대용량 파일 연계는 🗂️ 배치 탭입니다 (🔷 f13).
여기까지 왔다면 남은 것은 반복입니다. 결재 하나를 이렇게 만들 수 있으면 휴가 신청 · 구매 요청 · 출장 정산 은 같은 뼈대에 규칙만 갈아 끼우는 일입니다. Spring Boot 로 이 시스템을 만들 수 있었다면 이제 .NET 10 으로도 만들 수 있습니다 — 다른 것은 등록 · 트랜잭션 · 호스팅 세 가지뿐이었고, 그 셋을 전부 지나왔습니다.
정리 — 도메인에 규칙을 넣고(세터 금지), 매핑은 설정 클래스로 빼고, 인터셉터가 감사·아웃박스를 자동으로 채우고, 핸들러는 SaveChanges 를 한 번만 부릅니다. 통합 테스트는 진짜 DB + 동시성까지 검증하고, 배포는 멀티스테이지 이미지 + 외부 마이그레이션 잡입니다.
