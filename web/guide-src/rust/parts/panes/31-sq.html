use std::cmp::Reverse;
use std::collections::{BinaryHeap, VecDeque};

fn main() {
    let mut stack = vec![1, 2, 3];                 // 스택 = Vec
    let mut queue = VecDeque::from([1, 2, 3]);     // 큐 = VecDeque
    let mut maxh  = BinaryHeap::from([1, 5, 3]);   // 최대 힙
    let mut minh  = BinaryHeap::from([Reverse(4), Reverse(2)]); // 최소 힙

    println!("{:?}", stack.pop());        // Some(3)  ← 마지막에 넣은 것
    println!("{:?}", queue.pop_front());  // Some(1)  ← 처음 넣은 것
    println!("{:?}", maxh.pop());         // Some(5)  ← 가장 큰 것
    println!("{:?}", minh.pop());         // Some(Reverse(2)) ← 가장 작은 것

    stack.clear(); queue.clear();
    println!("{:?} {:?}", stack.pop(), queue.pop_front()); // None None
}

SQ 02

스택 — Vec 의 끝이 꼭대기, pop() 은 Option

Vec<T> 는 힙 버퍼 하나와 길이(len)·용량(cap) 두 숫자입니다. 끝에 넣고(push) 끝에서 빼면(pop) 다른 원소를 하나도 옮기지 않으니 그대로 스택입니다. 다른 언어와 결정적으로 다른 점은 하나 — 빈 스택에서 pop() 해도 아무 일도 터지지 않고 None 이 옵니다. 대신 그 None 을 처리하지 않으면 값을 쓸 수 없습니다.

메모리 배치 — 스택 연산은 len 숫자 하나를 올리고 내리는 일입니다. 그래서 별도 Stack 타입이 필요 없습니다.

기본 API — push · pop · last · last_mut

fn main() {
    let mut stack: Vec<i32> = Vec::new();
    stack.push(1);
    stack.push(2);
    stack.push(3);

    println!("{:?}", stack.last());      // Some(3)  ← peek (빌려 보기)
    if let Some(top) = stack.last_mut() {
        *top *= 10;                      // 꼭대기를 그 자리에서 수정
    }
    println!("{stack:?}");               // [1, 2, 30]

    println!("{:?}", stack.pop());       // Some(30)
    println!("{:?}", stack.pop());       // Some(2)
    println!("{:?}", stack.pop());       // Some(1)
    println!("{:?}", stack.pop());       // None   ← 패닉이 아닙니다
    println!("{}", stack.is_empty());    // true
}
last() 는 Option<&T> — 값을 꺼내지 않고 빌려 봅니다. 그래서 last() 로 얻은 참조를 들고 있는 동안에는 push 를 못 합니다(빌림 규칙 · 🔐 o02). 들여다본 값을 계속 쓰려면 stack.last().copied() 나 .cloned() 로 복사본을 받으세요.

fn main() {
    let mut stack = vec!["a", "b", "c"];

    // ★ 가장 많이 쓰는 형태 — 빌 때까지 꼭대기부터
    while let Some(x) = stack.pop() {
        print!("{x} ");                  // c b a
    }
    println!();

    // 순회 순서 — Vec 의 iter() 는 '바닥부터' 입니다
    let s = vec![1, 2, 3];               // 3 이 꼭대기
    let bottom_up: Vec<_> = s.iter().collect();
    let top_down:  Vec<_> = s.iter().rev().collect();
    println!("{bottom_up:?} {top_down:?}"); // [1, 2, 3] [3, 2, 1]

    // 1.86+ — 조건이 맞을 때만 꺼내기
    let mut v = vec![1, 2, 3, 4];
    println!("{:?}", v.pop_if(|x| *x % 2 == 0)); // Some(4)
    println!("{:?}", v.pop_if(|x| *x % 2 == 0)); // None (3 은 홀수라 남음)
}

fn eval_rpn(tokens: &[&str]) -> Result<i64, String> {
    let mut st: Vec<i64> = Vec::new();
    for &t in tokens {
        match t {
            "+" | "-" | "*" | "/" => {
                // 꼭대기가 오른쪽 피연산자 — 순서 주의 ★
                let (Some(b), Some(a)) = (st.pop(), st.pop()) else {
                    return Err(format!("피연산자 부족: {t}"));
                };
                st.push(match t {
                    "+" => a + b,
                    "-" => a - b,
                    "*" => a * b,
                    _ => {
                        if b == 0 { return Err("0으로 나눔".into()); }
                        a / b
                    }
                });
            }
            n => st.push(n.parse::<i64>().map_err(|e| format!("{n}: {e}"))?),
        }
    }
    match st.as_slice() {
        [x] => Ok(*x),                        // 정확히 하나 남아야 정상
        _ => Err(format!("식이 잘못됨: 남은 값 {}개", st.len())),
    }
}

fn main() {
    println!("{:?}", eval_rpn(&["3", "4", "+", "2", "*"]));  // Ok(14)
    println!("{:?}", eval_rpn(&["10", "3", "-", "4", "/"])); // Ok(1)
    println!("{:?}", eval_rpn(&["1", "+"]));  // Err("피연산자 부족: +")
    println!("{:?}", eval_rpn(&["4", "0", "/"])); // Err("0으로 나눔")
    println!("{:?}", eval_rpn(&["1", "2"]));  // Err("식이 잘못됨: 남은 값 2개")
}

fn top_two(s: &mut Vec<i32>) -> Option<(i32, i32)> {
    let a = s.pop()?;

/// 꼭대기만 만질 수 있는 스택. Vec 의 나머지 메서드는 숨깁니다.
#[derive(Debug)]
pub struct Stack<T> {
    items: Vec<T>,
}

impl<T> Stack<T> {
    pub fn new() -> Self { Self { items: Vec::new() } }
    pub fn with_capacity(n: usize) -> Self { Self { items: Vec::with_capacity(n) } }
    pub fn push(&mut self, x: T) { self.items.push(x) }
    pub fn pop(&mut self) -> Option<T> { self.items.pop() }
    pub fn peek(&self) -> Option<&T> { self.items.last() }
    pub fn len(&self) -> usize { self.items.len() }
    pub fn is_empty(&self) -> bool { self.items.is_empty() }
}

impl<T> Default for Stack<T> {          // clippy::new_without_default 대응
    fn default() -> Self { Self::new() }
}

// 꼭대기부터 도는 반복자만 공개 — 순서 혼동을 원천 차단
impl<T> Stack<T> {
    pub fn iter_top_down(&self) -> impl Iterator<Item = &T> {
        self.items.iter().rev()
    }
}

재귀 대신 명시적 스택 — 깊은 트리에서 스택 오버플로 방지

struct Node {
    val: i32,
    children: Vec<Node>,
}

// 재귀 버전 — 깊이 10만짜리 사슬이면 스택(리눅스 메인 8MB · 윈도우 1MB)을 넘칠 수 있음
fn sum_rec(n: &Node) -> i64 {
    n.val as i64 + n.children.iter().map(sum_rec).sum::<i64>()
}

// 반복 버전 — 힙에 있는 Vec 이 호출 스택 역할
fn sum_iter(root: &Node) -> i64 {
    let mut stack: Vec<&Node> = vec![root];
    let mut total = 0i64;
    while let Some(n) = stack.pop() {
        total += n.val as i64;
        stack.extend(n.children.iter()); // &Node 만 쌓음 — 복사 없음
    }
    total
}

fn main() {
    let tree = Node { val: 1, children: vec![
        Node { val: 2, children: vec![] },
        Node { val: 3, children: vec![Node { val: 4, children: vec![] }] },
    ]};
    println!("{} {}", sum_rec(&tree), sum_iter(&tree)); // 10 10
}
방문 순서까지 재귀와 같게 하려면 자식을 거꾸로 쌓습니다 — stack.extend(n.children.iter().rev()). 그래야 첫 자식이 꼭대기에 와서 먼저 나옵니다. 또 하나, 깊은 트리는 drop 도 재귀라 해제할 때 넘칠 수 있습니다 — 그때는 Drop 을 직접 구현해 자식을 std::mem::take 로 빼서 같은 스택 방식으로 풉니다.

SQ 03

큐 (FIFO) — VecDeque 의 뒤로 넣고 앞으로 뺍니다

큐는 std::collections::VecDeque<T> 하나로 끝납니다. push_back 으로 넣고 pop_front 로 꺼내면 둘 다 O(1) 입니다. 문제는 Vec 으로도 "되는 것처럼 보인다"는 점입니다 — v.remove(0) 은 컴파일도 되고 테스트도 통과하지만, 꺼낼 때마다 나머지 전부를 한 칸씩 당깁니다. 파이썬의 list.pop(0), JS 의 Array.shift() 와 똑같은 함정입니다.

선 그래프 — remove(0) 큐는 n² 로 휘고 VecDeque 는 바닥에 붙어 있습니다. 소요 시간이 아니라 옮김 횟수(계산값)입니다.

기본 큐 — push_back · pop_front · front

use std::collections::VecDeque;

fn main() {
    let mut q: VecDeque<&str> = VecDeque::new();
    q.push_back("a");
    q.push_back("b");
    q.push_back("c");

    println!("{:?}", q.front());      // Some("a")  ← peek
    println!("{:?}", q.pop_front());  // Some("a")
    println!("{q:?}");                // ["b", "c"]
    println!("{} {:?}", q.len(), q.get(1)); // 2 Some("c")
    println!("{}", q[0]);             // b   ← 인덱싱도 O(1)

    while let Some(x) = q.pop_front() {
        print!("{x} ");               // b c
    }
    println!();
    println!("{:?}", q.pop_front());  // None
}
만드는 법 네 가지 — VecDeque::new() · VecDeque::with_capacity(n) · VecDeque::from([1, 2, 3]) · iter.collect::<VecDeque<_>>(). 이미 있는 Vec 은 VecDeque::from(v) 로 바꾸면 버퍼를 그대로 물려받아 O(1) 입니다(복사 없음).

let mut q: Vec<Job> = Vec::new();

let mut head = 0;

    //   끝까지 돌아도 이미 처리한 원소의 메모리가 그대로 남습니다
}

use std::collections::VecDeque;

let mut q: VecDeque<Job> = VecDeque::new();
q.push_back(job);            // 뒤에 넣기
let next = q.pop_front();    // ✅ O(1) · Option 으로 안전
q.push_front(urgent);        // ✅ 앞에 넣기도 O(1) (새치기)

// 원소를 처리하며 새 원소를 넣어도 문제없음
while let Some(job) = q.pop_front() {   // job 은 '꺼낸 값' — 빌림 아님
    for child in job.spawn_children() {
        q.push_back(child);             // ✅ q 는 지금 아무도 빌리지 않음
    }
}
// 'head 인덱스 + Vec' 을 제대로 만든 것이 바로 VecDeque 의 원형 버퍼입니다 (SQ 04)
Vec 에서 앞을 빼도 되는 유일한 경우 — 원소가 수십 개 이하로 항상 작다는 게 보장될 때입니다. 그때는 오히려 remove(0) 의 memmove 가 캐시 안에서 끝나 충분히 빠릅니다. 하지만 "항상 작다"는 가정은 운영에서 자주 깨지므로, 큐라는 의도가 있으면 처음부터 VecDeque 로 쓰는 것이 코드를 읽는 사람에게도 정직합니다.

실무에선 이렇게 — 큐를 구조체에 넣고, 한꺼번에 넣고 빼기 🔴 실무

작업 큐 구조체 — 소유권 그대로

use std::collections::VecDeque;

#[derive(Debug)]
struct Job { id: u32, retries: u8 }

struct Dispatcher {
    pending: VecDeque<Job>,
    done: Vec<u32>,
}

impl Dispatcher {
    fn submit_all(&mut self, jobs: impl IntoIterator<Item = Job>) {
        self.pending.extend(jobs);              // 뒤에 한꺼번에
    }

    fn run(&mut self) {
        while let Some(mut job) = self.pending.pop_front() {
            if job.id % 3 == 0 && job.retries == 0 {
                job.retries += 1;
                self.pending.push_back(job);    // 실패 → 맨 뒤로 다시
                continue;
            }
            self.done.push(job.id);
        }
    }
}

fn main() {
    let mut d = Dispatcher { pending: VecDeque::new(), done: Vec::new() };
    d.submit_all((1..=6).map(|id| Job { id, retries: 0 }));
    d.run();
    println!("{:?}", d.done);   // [1, 2, 4, 5, 3, 6]
}
왜 빌림 에러가 안 나나 — pop_front() 는 값을 소유권째 꺼냅니다. job 은 더 이상 큐 안에 있지 않으니 self.pending.push_back(job) 을 해도 겹치는 빌림이 없습니다. "꺼내서 처리하고 필요하면 다시 넣는다" 가 Rust 큐 코드의 기본 리듬입니다.

한꺼번에 — extend · drain · retain · split_off

use std::collections::VecDeque;

fn main() {
    let mut q: VecDeque<u32> = (1..=10).collect();

    // 앞에서 최대 3개를 배치로 꺼내기 — 남은 게 적어도 안전
    let n = q.len().min(3);
    let batch: Vec<u32> = q.drain(..n).collect();
    println!("{batch:?}");          // [1, 2, 3]

    q.retain(|&x| x % 2 == 0);      // 조건에 맞는 것만 남김 (순서 유지)
    println!("{q:?}");              // [4, 6, 8, 10]

    let tail = q.split_off(2);      // 2번째부터 떼어 새 덱으로
    println!("{q:?} {tail:?}");     // [4, 6] [8, 10]

    q.extend([100, 200]);           // 뒤에 여러 개
    q.append(&mut VecDeque::from([7])); // 다른 덱을 통째로 이어 붙임
    println!("{q:?}");              // [4, 6, 100, 200, 7]

    // Vec 으로 돌려받기 — 앞이 버퍼 시작이 아니면 O(n) 이동은 있지만 재할당은 없음
    let v: Vec<u32> = q.into();
    println!("{v:?}");              // [4, 6, 100, 200, 7]
}
drain(..3) 을 그냥 쓰면 원소가 2개뿐일 때 패닉합니다(범위가 len 을 넘음). 위처럼 len().min(k) 로 자르거나, 반복자로 std::iter::from_fn(|| q.pop_front()).take(k) 처럼 꺼내세요.

SQ 04

덱의 속 — 원형 버퍼와 as_slices() 의 두 조각

VecDeque 는 버퍼 한 장 위에서 시작 위치(head)와 길이(len)만 움직입니다. 앞에서 빼면 head 가 한 칸 오른쪽으로, 뒤가 끝에 닿으면 버퍼 처음으로 감겨(wrap) 들어갑니다. 그래서 논리적으로는 한 줄인데 메모리에서는 두 토막일 수 있고, 이것이 as_slices() 가 슬라이스를 두 개 돌려주는 이유입니다. 이 구조를 알면 어떤 연산이 O(1) 이고 어떤 게 O(n) 인지 외울 필요가 없습니다.

메모리 배치 — 덱은 head 와 len 두 숫자로 버퍼 위를 돕니다. 감겨 있으면 두 조각, 그게 as_slices() 의 정체입니다.

그림을 코드로 — 직접 확인하기

use std::collections::VecDeque;

fn main() {
    let mut d: VecDeque<i32> = VecDeque::with_capacity(4);
    d.extend([1, 2, 3, 4]);
    for _ in 0..3 { d.pop_front(); }   // 1 2 3 꺼냄 → head 가 3 으로
    d.extend([5, 6, 7]);               // 뒤가 끝이라 물리 0·1·2 로 감김

    println!("{d:?}");                 // [4, 5, 6, 7]  ← 논리 순서
    println!("{:?}", d.as_slices());   // ([4], [5, 6, 7])
    println!("cap={}", d.capacity());  // cap=4

    d.push_back(8);                    // 가득 → 용량 8, 짧은 쪽만 이동
    println!("{:?}", d.as_slices());   // ([4], [5, 6, 7, 8])
    println!("cap={}", d.capacity());  // cap=8

    d.make_contiguous();               // 한 줄로 펴기
    println!("{:?}", d.as_slices());   // ([4, 5, 6, 7, 8], [])
}
// as_slices 의 경계와 capacity 값은 '현재 구현'의 결과입니다.
// 문서가 보장하는 것은 두 조각을 이어 붙이면 논리 순서라는 것뿐 —
// 코드가 경계 위치에 기대면 안 됩니다.

두 조각을 다루는 세 가지 방법

use std::collections::VecDeque;
use std::io::Write;

fn dump(d: &VecDeque<u8>, out: &mut impl Write) -> std::io::Result<()> {
    // ① 조각마다 처리 — 복사·재배치 0 ★ 가장 좋음
    let (a, b) = d.as_slices();
    out.write_all(a)?;
    out.write_all(b)?;
    Ok(())
}

fn main() {
    let mut d: VecDeque<i32> = VecDeque::from([3, 1, 2]);
    d.push_front(9);

    // ② 반복자 — 두 조각을 알아서 이어 줍니다
    let s: i32 = d.iter().sum();
    println!("{s}");                     // 15

    // ③ 정렬처럼 '연속된 한 장'이 꼭 필요할 때만 펴기
    d.make_contiguous().sort();          // &mut [i32] 를 돌려줌
    println!("{d:?}");                   // [1, 2, 3, 9]
    println!("{:?}", d.binary_search(&3)); // Ok(2)

    let mut out = Vec::new();
    dump(&VecDeque::from(*b"hi"), &mut out).unwrap();
    println!("{out:?}");                 // [104, 105]
}

덱 연산 한 바퀴

use std::collections::VecDeque;

fn main() {
    let mut d: VecDeque<i32> = (1..=6).collect(); // [1, 2, 3, 4, 5, 6]

    d.rotate_left(2);                  // [3, 4, 5, 6, 1, 2]
    d.rotate_right(1);                 // [2, 3, 4, 5, 6, 1]

    let mid: Vec<i32> = d.range(1..4).copied().collect();
    println!("{mid:?}");               // [3, 4, 5]

    let taken: Vec<i32> = d.drain(..2).collect();
    println!("{taken:?} {d:?}");       // [2, 3] [4, 5, 6, 1]

    d.truncate(2);                     // ★ '앞쪽' 2개를 남김
    d.swap(0, 1);
    println!("{d:?}");                 // [5, 4]

    if let Some(b) = d.back_mut() { *b = 40; }
    d.push_front(0);
    println!("{:?} {:?}", d.front(), d.back()); // Some(0) Some(40)
}
두 가지 패닉 — rotate_left(k) 는 k > len 이면 패닉합니다(나머지 연산을 대신 해 주지 않음). 바퀴 수가 클 수 있으면 d.rotate_left(k % d.len()) 로 — 단, len == 0 이면 0 으로 나눔이니 먼저 확인. range(a..b) · drain(a..b) 도 범위가 len 을 넘으면 패닉입니다.

우선순위 큐 — BinaryHeap 은 최대 힙, 최소는 Reverse

BinaryHeap<T> 는 T: Ord 인 값 중 가장 큰 것을 먼저 꺼냅니다. 자바 PriorityQueue · 파이썬 heapq · C# PriorityQueue 는 모두 가장 작은 것이 먼저라 옮겨 오는 사람마다 한 번씩 넘어집니다. 비교 함수를 넘기는 생성자가 없고, 순서는 타입의 Ord 구현이 정한다는 것도 Rust 만의 방식입니다. 그래서 최소 힙은 std::cmp::Reverse 로 감싸고, 여러 키는 Ord 를 직접 구현합니다.

트리와 배열 — "트리"는 칸 번호 계산일 뿐입니다. 배열 순서는 정렬 순서가 아니라는 것도 여기서 보입니다.

기본 — 최대 힙 · Reverse 로 최소 힙

use std::cmp::Reverse;
use std::collections::BinaryHeap;

fn main() {
    let mut h = BinaryHeap::from(vec![3, 1, 4, 1, 5, 9, 2, 6]); // O(n) 정리
    println!("{:?}", h.peek());          // Some(9)
    println!("{:?}", h.pop());           // Some(9)
    println!("{:?}", h.pop());           // Some(6)
    println!("{}", h.len());             // 6

    // 최소 힙 — 값을 Reverse 로 감싸 넣고, 꺼낼 때 패턴으로 벗김
    let mut mh = BinaryHeap::new();
    for x in [3, 1, 4, 1, 5] { mh.push(Reverse(x)); }
    while let Some(Reverse(x)) = mh.pop() {
        print!("{x} ");                  // 1 1 3 4 5
    }
    println!();

    // 정렬해서 다 꺼내기 — ★ 오름차순입니다 (pop 순서와 반대)
    let sorted = BinaryHeap::from(vec![5, 2, 8, 1]).into_sorted_vec();
    println!("{sorted:?}");              // [1, 2, 5, 8]

    // iter() · {:?} 는 '배열 순서' — 정렬돼 있지 않습니다
    let h2 = BinaryHeap::from(vec![3, 1, 4, 1, 5, 9, 2, 6]);
    println!("{:?}", h2.into_vec());     // [9, 6, 4, 1, 5, 3, 2, 1] (구현 세부)
}

Ord 직접 구현 — 우선순위 + 들어온 순서

use std::cmp::Ordering;
use std::collections::BinaryHeap;

#[derive(Debug, PartialEq, Eq)]
struct Task { pri: u8, seq: u64, name: String }

impl Ord for Task {
    fn cmp(&self, other: &Self) -> Ordering {
        self.pri.cmp(&other.pri)                     // 우선순위 높을수록 '큼'
            .then_with(|| other.seq.cmp(&self.seq))  // 같으면 먼저 온 것이 '큼'
    }
}
impl PartialOrd for Task {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))                        // ★ 반드시 cmp 에 위임
    }
}

fn main() {
    let mut q = BinaryHeap::new();
    let mut seq = 0u64;
    for (pri, name) in [(1, "정리"), (3, "결제"), (3, "환불"), (2, "메일")] {
        q.push(Task { pri, seq, name: name.to_string() });
        seq += 1;
    }
    while let Some(t) = q.pop() { print!("{} ", t.name); }
    println!();                                      // 결제 환불 메일 정리
}
같은 우선순위의 순서는 보장되지 않습니다 — 힙은 안정(stable)하지 않아서, seq 없이 pri 만 비교하면 "결제"와 "환불" 중 무엇이 먼저 나올지 아무도 약속하지 않습니다. FIFO 가 필요하면 단조 증가하는 순번을 두 번째 키로 넣는 것이 정석입니다. 여기서는 seq 가 유일해서 derive 한 PartialEq(이름까지 비교)와 Ord 가 어긋나지 않습니다.

f64 는 Ord 가 아니다 — total_cmp 로 감싸기

use std::cmp::Ordering;
use std::collections::BinaryHeap;

// let h: BinaryHeap<f64>;  ❌ the trait bound `f64: Ord` is not satisfied
//   NaN 때문에 f64 는 PartialOrd 까지만 구현합니다

#[derive(Debug, Clone, Copy)]
struct Score(f64);

impl PartialEq for Score {
    fn eq(&self, o: &Self) -> bool { self.0.total_cmp(&o.0).is_eq() }
}
impl Eq for Score {}
impl PartialOrd for Score {
    fn partial_cmp(&self, o: &Self) -> Option<Ordering> { Some(self.cmp(o)) }
}
impl Ord for Score {
    fn cmp(&self, o: &Self) -> Ordering { self.0.total_cmp(&o.0) } // 1.62+
}

fn main() {
    let mut h: BinaryHeap<Score> =
        [0.5, -0.0, 2.5, 0.0, -1.0].into_iter().map(Score).collect();
    while let Some(Score(x)) = h.pop() { print!("{x} "); }
    println!();                  // 2.5 0.5 0 -0 -1   ← -0.0 < +0.0 로 구분
}
// total_cmp 의 순서: 음의 NaN < -∞ < … < -0.0 < +0.0 < … < +∞ < 양의 NaN
//   NaN 이 섞여도 패닉하지 않고 어딘가에 '정해진 자리'를 받습니다
// 크레이트로 해결: ordered-float 의 OrderedFloat<f64> · NotNan<f64>
//   NotNan 은 만들 때 NaN 을 거부합니다 — 점수·거리처럼 NaN 이 '버그'인 값에
PartialEq 를 derive 하지 않은 이유 — derive 하면 == 는 IEEE 규칙(NaN ≠ NaN, -0.0 == 0.0)을 쓰고 cmp 는 total_cmp 규칙을 써서 둘이 어긋납니다. Eq 와 Ord 가 서로 다른 말을 하면 힙·정렬·BTreeMap 이 이상하게 동작합니다. 같은 비교 함수 하나에서 전부 파생시키세요.

실무에선 이렇게 — top-k · 꼭대기 수정 · 다익스트라 🔴 실무

top-k 와 peek_mut

use std::cmp::Reverse;
use std::collections::BinaryHeap;
use std::collections::binary_heap::PeekMut;

/// 가장 큰 k 개 — 크기 k 의 '최소' 힙을 유지 (메모리 O(k))
fn top_k(nums: &[i32], k: usize) -> Vec<i32> {
    let mut h = BinaryHeap::with_capacity(k + 1);
    for &x in nums {
        h.push(Reverse(x));
        if h.len() > k { h.pop(); }            // 가장 작은 것을 버림
    }
    let mut out: Vec<i32> = h.into_iter().map(|Reverse(x)| x).collect();
    out.sort_unstable_by(|a, b| b.cmp(a));
    out
}

fn main() {
    println!("{:?}", top_k(&[5, 1, 9, 3, 7, 9, 2], 3)); // [9, 9, 7]

    // peek_mut — 꼭대기를 그 자리에서 고치면, drop 될 때 제자리로 내려감
    let mut h = BinaryHeap::from(vec![10, 7, 3]);
    if let Some(mut top) = h.peek_mut() {
        *top = 1;                               // 10 → 1
    }
    println!("{:?}", h.peek());                 // Some(7)

    // 조건이 맞을 때만 꺼내기 — peek 후 pop 을 따로 하지 않음
    if let Some(top) = h.peek_mut() {
        if *top > 5 {
            let v = PeekMut::pop(top);
            println!("꺼냄 {v}");               // 꺼냄 7
        }
    }
    println!("{:?}", h.into_sorted_vec());     // [1, 3]
}
한 번만 뽑는다면 힙이 아니라 v.select_nth_unstable(i) 가 평균 O(n) 으로 더 빠릅니다. 힙 top-k 는 스트림처럼 값이 계속 들어올 때 쓰는 도구입니다.

다익스트라 — 완전한 예제

use std::cmp::Reverse;
use std::collections::BinaryHeap;

fn dijkstra(adj: &[Vec<(usize, u64)>], start: usize) -> Vec<Option<u64>> {
    let mut dist: Vec<Option<u64>> = vec![None; adj.len()];
    let mut heap = BinaryHeap::new();
    dist[start] = Some(0);
    heap.push(Reverse((0u64, start)));          // (거리, 노드) — 튜플은 사전순 비교

    while let Some(Reverse((d, u))) = heap.pop() {
        if dist[u].is_some_and(|best| d > best) {
            continue;                           // ★ 낡은 항목 — 더 짧은 길을 이미 찾음
        }
        for &(v, w) in &adj[u] {
            let nd = d + w;
            if dist[v].is_none_or(|best| nd < best) {   // 1.82+
                dist[v] = Some(nd);
                heap.push(Reverse((nd, v)));    // decrease-key 대신 '새로 넣기'
            }
        }
    }
    dist
}

fn main() {
    let mut adj: Vec<Vec<(usize, u64)>> = vec![Vec::new(); 6];
    for (u, v, w) in [(0, 1, 4), (0, 2, 1), (2, 1, 2), (1, 3, 1), (2, 3, 5), (3, 4, 3)] {
        adj[u].push((v, w));
    }
    for (i, d) in dijkstra(&adj, 0).iter().enumerate() {
        match d {
            Some(d) => println!("{i}: {d}"),
            None => println!("{i}: 도달 불가"),
        }
    }
}
// 0: 0
// 1: 3        (0→2→1)
// 2: 1
// 3: 4        (0→2→1→3)
// 4: 7
// 5: 도달 불가
힙 안의 원소는 고칠 수 없습니다 — 우선순위를 바꾸는 decrease_key 가 없습니다. 그래서 새 항목을 하나 더 넣고, 꺼낼 때 낡은 것을 건너뛰는(lazy deletion) 위 방식이 표준입니다. continue 줄을 빼먹으면 답은 맞아도 같은 노드를 여러 번 펼쳐 느려집니다.

한 단계 더 — 넣은 뒤에 순서가 바뀌면 🧬 전문가

힙 안에서 키가 바뀌는 세 가지 경로

① &mut 로 직접 고치기 → 불가능
   BinaryHeap 은 iter_mut 을 주지 않습니다. 고칠 수 있는 건
   peek_mut 로 받은 꼭대기뿐이고, 그건 drop 때 다시 정리됩니다.

② 내부 가변성 (Cell · RefCell · Arc<Mutex>) 으로 키를 바꾸기
   컴파일은 됩니다. 하지만 공식 문서의 표현으로 '논리 오류' —
   pop 순서가 틀어지거나 엉뚱한 값이 나옵니다.
   메모리 안전(UB)은 깨지지 않고 그 힙 안에서만 망가집니다.

③ Ord 구현이 일관되지 않음 (a<b 이고 b<a 등)
   ②와 같은 결과. 랜덤·시간에 의존하는 cmp 가 대표적입니다.

→ 키는 넣을 때 '값으로 굳혀서' 넣습니다 (우선순위를 복사해 튜플로).
→ 바뀌는 우선순위가 핵심이면 priority-queue 크레이트

스레드 안전 큐 — 채널이 기본, 직접 만들면 Mutex + Condvar

자바라면 ArrayBlockingQueue 를 꺼낼 자리에서 Rust 는 채널을 씁니다. 보내는 쪽(Sender)과 받는 쪽(Receiver)이 처음부터 갈라져 있어 "누가 넣고 누가 빼나"가 타입에 드러나고, 보내는 쪽이 모두 사라지면 받는 쪽 루프가 저절로 끝납니다 — 종료 신호를 따로 만들 일이 거의 없습니다. 채널 기초(mpsc · crossbeam 맛보기)는 ⚡ 성능 탭 e07에 있으니, 여기서는 용량 · 종료 · 워커 풀 · 직접 구현을 깊게 봅니다.

use std::sync::mpsc;
use std::thread;

fn main() {
    let (tx, rx) = mpsc::sync_channel::<String>(2);   // 2개까지 쌓임
    let mut handles = Vec::new();
    for id in 0..3 {
        let tx = tx.clone();                           // 생산자마다 사본
        handles.push(thread::spawn(move || {
            for n in 0..2 {
                // 가득 차면 여기서 멈춤 = 배압. 수신자가 사라졌으면 Err
                tx.send(format!("p{id}-{n}")).unwrap();
            }
        }));                                           // 스레드 끝 → 사본 drop
    }
    drop(tx);           // ★ 원본도 버려야 '모든 송신자 사라짐'이 성립

    let mut got: Vec<String> = rx.iter().collect();   // 송신자 0 + 비면 끝
    got.sort();                                        // 도착 순서는 매번 다름
    println!("{} {:?}", got.len(), got);
    // 6 ["p0-0", "p0-1", "p1-0", "p1-1", "p2-0", "p2-1"]
    for h in handles { h.join().unwrap(); }
}

Mutex<VecDeque> + Condvar 두 개 — 완전한 예제

use std::collections::VecDeque;
use std::sync::{Condvar, Mutex};
use std::thread;

struct Inner<T> { buf: VecDeque<T>, closed: bool }

pub struct BoundedQueue<T> {
    inner: Mutex<Inner<T>>,
    not_empty: Condvar,
    not_full: Condvar,
    cap: usize,
}

impl<T> BoundedQueue<T> {
    pub fn new(cap: usize) -> Self {
        assert!(cap > 0);
        Self {
            inner: Mutex::new(Inner { buf: VecDeque::with_capacity(cap), closed: false }),
            not_empty: Condvar::new(),
            not_full: Condvar::new(),
            cap,
        }
    }

    /// 가득 차면 기다림. 닫혔으면 값을 Err 로 돌려줌.
    pub fn push(&self, item: T) -> Result<(), T> {
        let g = self.inner.lock().unwrap();
        let mut g = self.not_full
            .wait_while(g, |s| s.buf.len() >= self.cap && !s.closed)
            .unwrap();                       // 깨어나도 조건을 다시 확인 (가짜 깨움 대비)
        if g.closed { return Err(item); }
        g.buf.push_back(item);
        drop(g);                             // 락을 먼저 풀고 깨움
        self.not_empty.notify_one();
        Ok(())
    }

    /// 비면 기다림. 닫혔고 비었으면 None.
    pub fn pop(&self) -> Option<T> {
        let g = self.inner.lock().unwrap();
        let mut g = self.not_empty
            .wait_while(g, |s| s.buf.is_empty() && !s.closed)
            .unwrap();
        let item = g.buf.pop_front();        // 닫혔어도 남은 것은 꺼내 줌 (drain)
        drop(g);
        if item.is_some() { self.not_full.notify_one(); }
        item
    }

    pub fn close(&self) {
        self.inner.lock().unwrap().closed = true;
        self.not_empty.notify_all();         // 잠든 모두를 깨워 종료를 알림
        self.not_full.notify_all();
    }
}

fn main() {
    let q = BoundedQueue::<u64>::new(8);
    let q = &q;                                  // 스코프 스레드에 빌려줄 참조
    let (count, sum) = thread::scope(|s| {       // 1.63+ — Arc 가 필요 없음
        let consumers: Vec<_> = (0..3).map(|_| s.spawn(move || {
            let (mut n, mut sum) = (0usize, 0u64);
            while let Some(x) = q.pop() { n += 1; sum += x; }
            (n, sum)
        })).collect();
        let producers: Vec<_> = (0..2).map(|_| s.spawn(move || {
            for i in 1..=100u64 { q.push(i).unwrap(); }
        })).collect();

        for p in producers { p.join().unwrap(); }
        q.close();                               // 생산 끝 → 닫기
        consumers.into_iter()
            .map(|h| h.join().unwrap())
            .fold((0, 0), |(a, b), (n, s)| (a + n, b + s))
    });
    println!("소비 {count}개, 합계 {sum}");      // 소비 200개, 합계 10100
}

async 의 큐 — tokio::sync::mpsc 와 배압 · 종료

async 코드에서 스레드용 채널(std::sync::mpsc)의 recv() 를 부르면 런타임 워커 스레드 자체가 멈춥니다. 그래서 tokio 에는 .await 로 기다리는 채널이 따로 있습니다. 핵심은 하나 — 용량을 정한 channel(n) 의 send().await 가 곧 배압입니다. 소비자가 느리면 생산자가 자연히 늦춰집니다. 채널 네 종류와 액터 패턴은 🧵 서비스 탭 h03에서 다뤘으니, 여기서는 큐로서의 성질 · 워커 풀 · 깔끔한 종료에 집중합니다.

스윔레인 — 취소 신호는 입구에만 보내고, 출구는 채널이 닫히는 것으로 자연히 끝나게 합니다. 그래야 유실이 0 입니다.

생산자-소비자 + 정상 종료 — 완전한 예제

// Cargo.toml: tokio = { version = "1", features = ["full"] }
//             tokio-util = "0.7"
use std::time::Duration;
use tokio::sync::mpsc;
use tokio_util::sync::CancellationToken;

#[tokio::main]
async fn main() {
    let (tx, mut rx) = mpsc::channel::<u32>(4);   // 4칸 — 차면 send 가 대기
    let token = CancellationToken::new();

    let prod_token = token.clone();
    let producer = tokio::spawn(async move {
        let mut sent = 0u32;
        loop {
            tokio::select! {
                _ = prod_token.cancelled() => break,
                r = tx.send(sent) => {
                    if r.is_err() { break; }        // 수신자가 사라짐
                    sent += 1;
                }
            }
        }
        sent                // 여기서 tx 가 drop → 채널이 '닫힘'
    });

    let consumer = tokio::spawn(async move {
        let mut done = 0u32;
        while let Some(_job) = rx.recv().await {    // 닫히고 비면 None
            tokio::time::sleep(Duration::from_millis(1)).await; // 일하는 척
            done += 1;
        }
        done
    });

    tokio::time::sleep(Duration::from_millis(50)).await;
    token.cancel();                                  // 종료 신호는 입구에만
    let sent = producer.await.unwrap();
    let done = consumer.await.unwrap();
    println!("보냄 {sent} · 처리 {done} · 유실 {}", sent - done);
    // 보냄 N · 처리 N · 유실 0   (N 은 실행마다 다름)
}

수신자 하나를 워커 넷이 나눠 받기

use std::sync::Arc;
use tokio::sync::{mpsc, Mutex};
use tokio::task::JoinSet;

#[tokio::main]
async fn main() {
    let (tx, rx) = mpsc::channel::<u64>(16);
    let rx = Arc::new(Mutex::new(rx));          // tokio 의 Mutex (await 가능)
    let mut set = JoinSet::new();

    for _ in 0..4 {
        let rx = Arc::clone(&rx);
        set.spawn(async move {
            let mut sum = 0u64;
            loop {
                // ★ 락은 이 문장 끝에서 풀립니다 — 처리 중에는 안 잡고 있음
                let job = rx.lock().await.recv().await;
                let Some(n) = job else { break };   // 닫히고 비면 종료
                sum += n * n;                        // 처리는 락 밖에서
            }
            sum
        });
    }

    for n in 1..=100 { tx.send(n).await.unwrap(); }
    drop(tx);                                    // 입구 닫기

    let mut total = 0;
    while let Some(r) = set.join_next().await { total += r.unwrap(); }
    println!("{total}");                         // 338350
}

use std::time::Duration;
use tokio::sync::mpsc::{error::SendTimeoutError, Sender};

#[derive(Debug)]

/// HTTP 핸들러에서 — 0.2초 안에 자리가 안 나면 503 으로 돌려보냄
async fn enqueue(tx: &Sender<String>, job: String) -> Enqueue {
    match tx.send_timeout(job, Duration::from_millis(200)).await {
        Ok(()) => Enqueue::Ok,
        Err(SendTimeoutError::Timeout(_job)) => Enqueue::Busy, // 값은 돌려받음
        Err(SendTimeoutError::Closed(_job)) => Enqueue::Busy,  // 종료 중
    }
}

fn balanced(s: &str) -> bool {
    let mut st = Vec::new();
    for c in s.chars() {
        match c {
            '(' | '[' | '{' => st.push(c),
            ')' | ']' | '}' => {
                let want = match c { ')' => '(', ']' => '[', _ => '{' };
                if st.pop() != Some(want) { return false; } // 비어도 여기서 걸림
            }
            _ => {}
        }
    }
    st.is_empty()                       // 열린 채로 끝나면 false
}

fn next_greater(a: &[i32]) -> Vec<Option<i32>> {
    let mut ans = vec![None; a.len()];
    let mut st: Vec<usize> = Vec::new(); // 값이 아니라 인덱스 ★
    for (i, &x) in a.iter().enumerate() {
        while let Some(&top) = st.last() {
            if a[top] >= x { break; }
            ans[top] = Some(x);
            st.pop();
        }
        st.push(i);
    }
    ans
}

fn main() {
    for s in ["{[()]}", "([)]", "((", "a(b)c", ")("] {
        print!("{} ", balanced(s));     // true false false true false
    }
    println!();
    println!("{:?}", next_greater(&[2, 1, 2, 4, 3]));
    // [Some(4), Some(2), Some(4), None, None]
}
while let Some(&top) = st.last() 가 빌림 에러가 안 나는 이유 — 패턴의 &top 이 usize 를 복사해 오므로 st 를 빌린 참조가 그 줄에서 끝납니다. 그래서 같은 반복 안에서 st.pop() 을 해도 됩니다.

use std::collections::VecDeque;

fn window_max(a: &[i32], k: usize) -> Vec<i32> {
    assert!(k > 0);
    let mut dq: VecDeque<usize> = VecDeque::new(); // 값이 줄어드는 인덱스들
    let mut out = Vec::new();
    for i in 0..a.len() {
        if dq.front().is_some_and(|&f| f + k <= i) { dq.pop_front(); } // 창 밖
        while dq.back().is_some_and(|&b| a[b] <= a[i]) { dq.pop_back(); }
        dq.push_back(i);
        if i + 1 >= k { out.push(a[dq[0]]); }      // 맨 앞이 창의 최댓값
    }
    out
}

fn bfs_grid(grid: &[&str]) -> Vec<Vec<Option<u32>>> {
    let g: Vec<&[u8]> = grid.iter().map(|r| r.as_bytes()).collect();
    let (h, w) = (g.len(), g[0].len());
    let mut dist = vec![vec![None; w]; h];
    let mut q = VecDeque::from([(0usize, 0usize)]);
    dist[0][0] = Some(0);
    while let Some((r, c)) = q.pop_front() {
        let d = dist[r][c].expect("큐에 든 칸은 거리가 정해져 있음");
        for (dr, dc) in [(0isize, 1isize), (1, 0), (0, -1), (-1, 0)] {
            // 0 에서 -1 을 하면 usize 언더플로 → checked 로 None (1.66+)
            let (Some(nr), Some(nc)) =
                (r.checked_add_signed(dr), c.checked_add_signed(dc)) else { continue };
            if nr >= h || nc >= w || g[nr][nc] == b'#' || dist[nr][nc].is_some() {
                continue;
            }
            dist[nr][nc] = Some(d + 1);
            q.push_back((nr, nc));
        }
    }
    dist
}

fn main() {
    println!("{:?}", window_max(&[1, 3, -1, -3, 5, 3, 6, 7], 3)); // [3, 3, 5, 5, 6, 7]
    for row in bfs_grid(&["S..#", ".#..", "...#", "#..."]) {
        let cells: Vec<String> =
            row.iter().map(|d| d.map_or("#".into(), |v| v.to_string())).collect();
        println!("{}", cells.join(" "));
    }
}

0-1 BFS — 비용 0 은 앞으로, 1 은 뒤로

use std::collections::VecDeque;

/// '#' 을 지나려면 벽을 하나 부숴야 함(비용 1), '.' 은 0.
/// 왼쪽 위 → 오른쪽 아래까지 부술 벽의 최소 개수.
fn min_walls(grid: &[&str]) -> u32 {
    let g: Vec<&[u8]> = grid.iter().map(|r| r.as_bytes()).collect();
    let (h, w) = (g.len(), g[0].len());
    let mut dist = vec![vec![u32::MAX; w]; h];
    let mut dq = VecDeque::from([(0usize, 0usize)]);
    dist[0][0] = 0;
    while let Some((r, c)) = dq.pop_front() {
        for (dr, dc) in [(0isize, 1isize), (1, 0), (0, -1), (-1, 0)] {
            let (Some(nr), Some(nc)) =
                (r.checked_add_signed(dr), c.checked_add_signed(dc)) else { continue };
            if nr >= h || nc >= w { continue; }
            let cost = u32::from(g[nr][nc] == b'#');
            let nd = dist[r][c] + cost;
            if nd < dist[nr][nc] {
                dist[nr][nc] = nd;
                if cost == 0 { dq.push_front((nr, nc)) }   // ★ 덱의 양끝을 다 씀
                else { dq.push_back((nr, nc)) }
            }
        }
    }
    dist[h - 1][w - 1]
}

fn main() {
    println!("{}", min_walls(&["..#.", "##.#", "...#", "#.#."])); // 2
}
왜 덱인가 — 가중치가 0과 1뿐이면 덱의 앞쪽에는 거리 d, 뒤쪽에는 d+1 만 있게 유지할 수 있어 BinaryHeap 없이 다익스트라와 같은 답을 O(V+E) 로 얻습니다. Rust 에서는 VecDeque 한 개로 끝납니다.

undo / redo — 스택 두 개

#[derive(Debug, Clone)]
enum Edit {
    Insert { pos: usize, text: String },
    Delete { pos: usize, text: String },
}

struct Editor { buf: String, undo: Vec<Edit>, redo: Vec<Edit> }

impl Editor {
    fn apply(&mut self, e: &Edit) {
        match e {
            Edit::Insert { pos, text } => self.buf.insert_str(*pos, text),
            Edit::Delete { pos, text } => self.buf.replace_range(*pos..*pos + text.len(), ""),
        }
    }
    fn inverse(e: &Edit) -> Edit {
        match e.clone() {
            Edit::Insert { pos, text } => Edit::Delete { pos, text },
            Edit::Delete { pos, text } => Edit::Insert { pos, text },
        }
    }
    fn exec(&mut self, e: Edit) {
        self.apply(&e);
        self.undo.push(e);
        self.redo.clear();                     // ★ 새 편집이 오면 redo 는 무효
    }
    fn undo(&mut self) -> bool {
        let Some(e) = self.undo.pop() else { return false };
        self.apply(&Self::inverse(&e));
        self.redo.push(e);
        true
    }
    fn redo(&mut self) -> bool {
        let Some(e) = self.redo.pop() else { return false };
        self.apply(&e);
        self.undo.push(e);
        true
    }
}

fn main() {
    let mut ed = Editor { buf: String::new(), undo: vec![], redo: vec![] };
    ed.exec(Edit::Insert { pos: 0, text: "hello".into() });
    ed.exec(Edit::Insert { pos: 5, text: " world".into() });
    ed.exec(Edit::Delete { pos: 0, text: "hello".into() });
    println!("{:?}", ed.buf);                  // " world"
    ed.undo(); println!("{:?}", ed.buf);       // "hello world"
    ed.undo(); println!("{:?}", ed.buf);       // "hello"
    ed.redo(); println!("{:?}", ed.buf);       // "hello world"
    ed.exec(Edit::Insert { pos: 11, text: "!".into() });
    println!("{:?} redo={}", ed.buf, ed.redo.len()); // "hello world!" redo=0
}
pos 는 바이트 위치입니다 — insert_str · replace_range 는 문자 경계가 아니면 패닉합니다. 한글을 다루는 편집기라면 위치를 char_indices 로 구하세요. 이력에 상한이 필요하면 undo 를 VecDeque 로 바꾸고 넘칠 때 pop_front() 로 가장 오래된 것을 버립니다.
정리 — 스택 문제는 Vec<usize> 에 인덱스를, 큐 문제는 VecDeque 에 인덱스나 좌표를 넣고 while let Some(..) = pop..() 으로 꺼냅니다. 빌림 에러가 나면 ① 큐를 빌려 돌지 말고 꺼내서 쓰기 ② &mut 대신 인덱스 ③ self 메서드 대신 필드 직접 — 이 세 가지로 거의 다 풀립니다.

성능과 메모리 — 연속 메모리가 거의 항상 이깁니다

복잡도표만 보면 LinkedList 의 양끝 연산도 O(1) 이라 VecDeque 와 같아 보입니다. 실제로 갈리는 것은 원소 하나에 할당이 몇 번 일어나는가와 다음 원소가 메모리의 어디에 있는가입니다. Rust 는 값을 컨테이너 안에 그대로 넣으므로(자바처럼 박싱하지 않음) 연속 배열의 장점이 특히 큽니다. 벤치마크 수치는 기계마다 다르니 자릿수만 보는 예시로 읽고, 직접 재는 법은 ⚡ 성능 탭 e01 의 criterion 을 쓰세요.

막대(로그 눈금) — 연속 버퍼 · 원소별 할당 · O(n²) 가 각각 한 자릿수 이상씩 차이 납니다. 예시 값이므로 순서와 자릿수만 믿으세요.
메모리 배치 — 같은 원소 하나에 8B 대 24B+, 그리고 연속이냐 흩어졌냐. 속도 차이의 대부분이 여기서 나옵니다.

use std::collections::VecDeque;

fn main() {
    // ① 개수를 알면 미리 — 늘어나며 복사하는 일이 0
    let mut q: VecDeque<u32> = VecDeque::with_capacity(1_000);
    let before = q.capacity();
    for i in 0..1_000 { q.push_back(i); }
    println!("{}", q.capacity() == before);   // true — 재할당 없음

    // ② 루프마다 새로 만들지 말고 비워서 다시
    let mut frontier: Vec<usize> = Vec::new();
    for round in 0..3 {
        frontier.clear();                     // 용량은 남음
        frontier.extend(0..round * 10);
    }

    // ③ 잠깐 커졌다가 평소 작아지면 상한을 두고 줄이기
    q.clear();
    q.shrink_to(64);                          // 최소 64 는 남김
    println!("{}", q.capacity() >= 64);       // true
}

값 vs Box — 기준표

상황넣는 형태
원소가 작다 (수십~백여 바이트)값 그대로 VecDeque<Job>
원소가 매우 크다 (KB 단위)VecDeque<Box<Job>> — 이동이 포인터 8B
종류가 여럿 (트레이트 객체)VecDeque<Box<dyn Task>> — 피할 수 없음
enum 인데 한 변형만 큼큰 변형 안쪽만 Box (clippy::large_enum_variant)
여러 곳이 같은 값을 공유Arc<Job> — 큐에는 포인터만
자바 습관 주의 — 자바의 ArrayDeque<Job> 는 사실 참조의 배열이라 늘 "Box 버전"입니다. Rust 로 옮기며 습관처럼 Box 를 씌우면 장점(연속 메모리 · 할당 0)을 스스로 버리는 셈입니다. 기본은 값, 크기를 재 보고 문제가 될 때만 Box.

흔한 실수 — 컴파일은 되는데 틀린 것과 컴파일러가 막아 주는 것

Rust 는 빈 컨테이너 접근과 데이터 경쟁을 타입으로 막아 주지만, O(n) 큐 · 방향을 착각한 힙 · 무한히 자라는 채널 · 필요 이상 오래 잡은 락은 컴파일러가 모릅니다. 여기 모은 열 가지는 리뷰에서 실제로 자주 보이는 것들입니다. 앞쪽 다섯은 "돌기는 도는데 틀린" 것, 뒤쪽 다섯은 "컴파일 에러가 나서 헤매는" 것입니다.

① 레이트 리미터 — 슬라이딩 윈도우 로그

use std::collections::VecDeque;

struct SlidingWindow {
    limit: usize,
    window_ms: u64,
    hits: VecDeque<u64>,          // 허용된 요청 시각 (오래된 것이 앞)
}

impl SlidingWindow {
    fn new(limit: usize, window_ms: u64) -> Self {
        Self { limit, window_ms, hits: VecDeque::with_capacity(limit) }
    }

    /// 시각을 인자로 받음 → 테스트에서 시간을 고정할 수 있음
    fn allow(&mut self, now_ms: u64) -> bool {
        while self.hits.front().is_some_and(|&t| t + self.window_ms <= now_ms) {
            self.hits.pop_front();     // 창 밖으로 나간 기록 버림
        }
        if self.hits.len() < self.limit {
            self.hits.push_back(now_ms);
            true
        } else {
            false
        }
    }
}

fn main() {
    let mut rl = SlidingWindow::new(3, 1000);         // 1초에 3번
    let log: Vec<String> = [0, 100, 200, 300, 900, 1000, 1100, 1250, 2000]
        .into_iter()
        .map(|t| format!("{t}:{}", if rl.allow(t) { "O" } else { "X" }))
        .collect();
    println!("{}", log.join(" "));
    // 0:O 100:O 200:O 300:X 900:X 1000:O 1100:O 1250:O 2000:O
}
운영 코드로 바꿀 때 — u64 밀리초 대신 std::time::Instant 와 Duration, 사용자별이면 HashMap<UserId, SlidingWindow>(오래 안 쓰인 키 정리 필수), 서버가 여러 대면 같은 알고리즘을 Redis 정렬 집합(ZSET)으로 옮깁니다. 한 프로세스 안이라면 governor 크레이트(GCRA)가 메모리를 더 적게 씁니다.

② 최근 N개 — 넘치면 앞을 버림

use std::collections::VecDeque;

struct Recent<T> { cap: usize, buf: VecDeque<T> }

impl<T> Recent<T> {
    fn new(cap: usize) -> Self {
        assert!(cap > 0);
        Self { cap, buf: VecDeque::with_capacity(cap) }
    }
    /// 넣고, 넘쳤으면 밀려난 가장 오래된 값을 돌려줌
    fn push(&mut self, x: T) -> Option<T> {
        let evicted = if self.buf.len() == self.cap { self.buf.pop_front() } else { None };
        self.buf.push_back(x);
        evicted
    }
    fn newest_first(&self) -> impl Iterator<Item = &T> {
        self.buf.iter().rev()
    }
}

fn main() {
    let mut r = Recent::new(3);
    for x in 1..=5 {
        if let Some(old) = r.push(x) { print!("버림 {old} · "); }
    }
    println!();                                          // 버림 1 · 버림 2 ·
    println!("{:?}", r.newest_first().collect::<Vec<_>>()); // [5, 4, 3]
}
// 더 짧게 — 최신을 앞에 넣고 truncate 로 뒤(오래된 것)를 잘라 냄
let mut log: VecDeque<String> = VecDeque::with_capacity(101);
log.push_front(line);    // 최신이 맨 앞
log.truncate(100);       // truncate 는 '앞쪽 100개'를 남김 → 오래된 것이 잘림
용량은 처음에 한 번 — with_capacity(cap) 로 만들면 가득 찬 뒤로는 pop_front 와 push_back 이 같은 버퍼를 돌며 할당이 다시 일어나지 않습니다. 최근 로그 · 최근 지연 시간 표본 · 채팅 스크롤백이 전부 이 모양입니다.

③ 재시도 큐 — 가장 이른 시각부터 (지수 백오프)

use std::cmp::Ordering;
use std::collections::BinaryHeap;
use std::time::{Duration, Instant};

struct Due<J> { at: Instant, seq: u64, attempt: u32, job: J }

// 비교는 (시각, 순번)만 — job 은 Ord 가 아니어도 됨
impl<J> PartialEq for Due<J> {
    fn eq(&self, o: &Self) -> bool { (self.at, self.seq) == (o.at, o.seq) }
}
impl<J> Eq for Due<J> {}
impl<J> PartialOrd for Due<J> {
    fn partial_cmp(&self, o: &Self) -> Option<Ordering> { Some(self.cmp(o)) }
}
impl<J> Ord for Due<J> {
    // 이른 것이 '큼' → 최대 힙이 가장 이른 것부터 꺼냄
    fn cmp(&self, o: &Self) -> Ordering { (o.at, o.seq).cmp(&(self.at, self.seq)) }
}

struct RetryQueue<J> { heap: BinaryHeap<Due<J>>, seq: u64 }

impl<J> RetryQueue<J> {
    fn new() -> Self { Self { heap: BinaryHeap::new(), seq: 0 } }
    fn schedule(&mut self, job: J, attempt: u32, now: Instant) {
        let backoff = Duration::from_millis(100 * 2u64.pow(attempt.min(10))); // 100·200·400…
        self.seq += 1;
        self.heap.push(Due { at: now + backoff, seq: self.seq, attempt, job });
    }
    fn pop_ready(&mut self, now: Instant) -> Option<(J, u32)> {
        if self.heap.peek()?.at > now { return None; }   // 가장 이른 것도 아직
        self.heap.pop().map(|d| (d.job, d.attempt))
    }
    fn next_wakeup(&self) -> Option<Instant> { self.heap.peek().map(|d| d.at) }
}

fn main() {
    let t0 = Instant::now();
    let mut rq = RetryQueue::new();
    rq.schedule("a", 0, t0);    // +100ms
    rq.schedule("b", 2, t0);    // +400ms
    rq.schedule("c", 1, t0);    // +200ms

    let now = t0 + Duration::from_millis(250);
    while let Some((job, attempt)) = rq.pop_ready(now) {
        print!("{job}({attempt}) ");                    // a(0) c(1)
    }
    println!();
    println!("{:?}", rq.next_wakeup().map(|t| t - t0)); // Some(400ms)
}
next_wakeup 이 핵심 — 워커는 "가장 이른 시각까지만" 자면 됩니다 (tokio::time::sleep_until 또는 recv_timeout). 1초마다 전부 훑는 폴링이 사라집니다. 실제로는 백오프에 무작위 흔들림(jitter)을 더해 재시도가 한꺼번에 몰리지 않게 합니다.

④ 이벤트 버퍼 — 500건 또는 1초마다 한 번에 저장

use std::time::Duration;
use tokio::sync::mpsc;
use tokio::time::{interval, MissedTickBehavior};

const MAX_BATCH: usize = 500;

async fn batcher(mut rx: mpsc::Receiver<String>) {
    let mut buf: Vec<String> = Vec::with_capacity(MAX_BATCH);
    let mut tick = interval(Duration::from_secs(1));
    tick.set_missed_tick_behavior(MissedTickBehavior::Delay);
    loop {
        let room = MAX_BATCH - buf.len();     // ★ 미리 계산 (아래 &mut buf 와 겹치지 않게)
        tokio::select! {
            n = rx.recv_many(&mut buf, room) => {
                if n == 0 { break; }                       // 닫혔고 비었음
                if buf.len() >= MAX_BATCH { flush(&mut buf).await; }
            }
            _ = tick.tick() => {
                if !buf.is_empty() { flush(&mut buf).await; } // 느린 시간대엔 1초마다
            }
        }
    }
    if !buf.is_empty() { flush(&mut buf).await; }        // 종료 전 마지막 비우기
}

async fn flush(buf: &mut Vec<String>) {
    println!("{}건 저장", buf.len());        // 실제로는 DB 에 한 번에 INSERT
    buf.clear();
}
room 을 따로 빼는 이유 — rx.recv_many(&mut buf, MAX_BATCH - buf.len()) 로 한 줄에 쓰면 첫 인자에서 buf 를 가변으로 빌린 상태로 둘째 인자에서 다시 읽게 되어 E0502 입니다. 또 room 이 0 이면 recv_many 가 즉시 0 을 돌려줘 "닫힘"과 구분이 안 되므로, 가득 차는 순간 바로 flush 해서 0 이 되지 않게 했습니다. recv_many 는 select! 안에서 써도 안전한(cancel-safe) 메서드입니다.

⑤ 동시 실행 개수 제한 — 세마포어가 곧 큐

use std::sync::Arc;
use tokio::sync::Semaphore;
use tokio::task::JoinSet;

async fn fetch(url: String) -> usize { url.len() }   // 실제로는 HTTP 요청

#[tokio::main]
async fn main() {
    let limit = Arc::new(Semaphore::new(8));          // 동시에 최대 8개
    let mut set = JoinSet::new();

    for i in 1..=20 {
        let url = format!("https://example.com/{i}");
        // 자리가 날 때까지 여기서 대기 → 생성 속도가 처리 속도에 묶임 (배압)
        let permit = Arc::clone(&limit).acquire_owned().await.unwrap();
        set.spawn(async move {
            let n = fetch(url).await;
            drop(permit);                             // 끝나면 자리 반납
            n
        });
    }

    let (mut count, mut bytes) = (0, 0);
    while let Some(r) = set.join_next().await {
        count += 1;
        bytes += r.unwrap();
    }
    println!("{count}건 · {bytes}바이트");             // 20건 · 431바이트
}

⑥ 예약 실행 · 만료 — DelayQueue

// Cargo.toml: tokio-util = { version = "0.7", features = ["time"] }
//             tokio-stream = "0.1"
use std::time::Duration;
use tokio_stream::StreamExt;               // DelayQueue 는 Stream
use tokio_util::time::DelayQueue;

#[tokio::main]
async fn main() {
    let mut dq: DelayQueue<&str> = DelayQueue::new();
    let a = dq.insert("세션 A 만료", Duration::from_millis(30));
    let _b = dq.insert("세션 B 만료", Duration::from_millis(10));
    let c = dq.insert("세션 C 만료", Duration::from_millis(20));

    dq.reset(&a, Duration::from_millis(50)); // 활동이 있었으니 연장
    let _ = dq.remove(&c);                    // 로그아웃 → 취소

    while let Some(expired) = dq.next().await {
        println!("{}", expired.into_inner());
    }
    // 세션 B 만료
    // 세션 A 만료      (비면 None 으로 루프가 끝남)
}
BinaryHeap 재시도 큐(③)와 무엇이 다른가 — DelayQueue 는 tokio 타이머 휠 위에 있어 넣을 때 받은 키로 바로 연장·취소할 수 있고, 깨어날 시각을 직접 계산할 필요가 없습니다. 세션 만료 · 캐시 TTL · 타임아웃처럼 "넣은 뒤 바뀌는 시각"이 있으면 DelayQueue, 넣고 나면 안 바뀌면 힙이 더 단순합니다.
정리 — 레이트 리미터와 최근 N개는 VecDeque 의 앞에서 버리고 뒤에 넣기, 재시도는 시각을 키로 한 최소 힙, 배치 저장은 recv_many + 시간 상한, 동시 실행 제한은 세마포어, 바뀌는 만료 시각은 DelayQueue. 시간을 인자로 받게 짜면 이 모든 코드를 sleep 없이 테스트할 수 있습니다.
