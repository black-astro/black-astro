한 장 지도 — C 에는 표준 스택·큐가 없다

C 표준 라이브러리가 주는 자료구조 도구는 qsort 와 bsearch 가 사실상 전부입니다. 그러니 "무슨 타입을 쓰나"가 아니라 "어떤 모양으로 만드나"를 먼저 정해야 합니다. 답을 가르는 질문은 셋뿐입니다 — 누가 같이 쓰나, 순서가 우선순위인가, 최대 개수를 아는가. 동적 배열·연결 리스트의 기본은 ⚙️ C 실전 탭 c14 에 있으니, 여기서는 그 위에 스택·큐·덱을 세웁니다.

//
// 1. 실패할 수 있는 연산은 bool 을 돌려주고 값은 out 으로.
//      bool ring_pop(Ring *r, int *out);   // false 면 *out 손대지 않음

// 2. 용량은 2의 거듭제곱, 컴파일 타임에 검사 (SQ 03 · SQ 09)
// 3. 인덱스는 부호 없는 타입 (uint32_t / size_t) — 넘쳐도 정의된 동작
// 4. 메모리는 init 에서 한 번. push/pop 안에서는 malloc 금지
// 5. 동시성 계약을 글로:
//      "ring_push 는 ISR 하나에서만, ring_pop 은 main 하나에서만"

자바와 비교하면 — 자바의 offer/poll 이 "특수값(null) 반환", add/remove 가 "예외" 쌍인 것처럼, C 에서는 bool + out 매개변수가 예외 없는 언어의 정석입니다. C 에는 예외가 없으니 반환값을 무시하지 못하게 C23 [[nodiscard]] 를 붙이면 더 좋습니다 C23.

왜 C 에선 직접 만드는 게 정답일 때가 많은가 🟡 중급

제네릭 컨테이너가 없는 대신 얻는 것

다른 언어의 범용 큐는 "어떤 상황에서도 동작"하도록 만들어져 있습니다 — 성장, 예외, 박싱, 락. C 로 직접 만들면 상황에 딱 맞는 것만 남길 수 있습니다.

  • 크기가 컴파일 타임에 고정 → 링커 맵에서 RAM 사용량이 바이트 단위로 보입니다
  • 원소가 값으로 박혀 있음 → 포인터 추적 없이 연속 메모리 (SQ 09)
  • 동시성 모델이 하나 → ISR 하나·메인 하나면 락이 필요 없습니다 (SQ 07)
  • 실패 정책을 고름 → 가득 차면 버릴지, 가장 오래된 것을 덮을지 (SQ 12 로그 링)

그래도 가져다 쓰는 편이 나은 경우

  • RTOS 위 — 블로킹·타임아웃·우선순위 상속까지 RTOS 큐가 이미 합니다. 직접 만든 condvar 큐보다 정확합니다 (FreeRTOS 탭 r05)
  • 리눅스 사용자 공간의 MPMC 고성능 큐 — 다중 생산자·다중 소비자 락프리는 직접 만들지 마세요. liblfds · concurrencykit 같은 검증된 구현을 씁니다
  • GLib 기반 앱 — GAsyncQueue 가 메인 루프와 잘 맞물립니다
직접 만들 때의 최소 조건 — 경계 테스트(0개, 1개, 가득, 가득−1, 인덱스 넘침 직전)를 단위 테스트로 고정해 두세요. 링 버퍼 버그의 대부분은 이 다섯 점 중 하나에서 납니다. 테스트 프레임워크는 ⚙️ C 실전 탭 c24 의 Unity 를 그대로 씁니다.
정리 — C 에는 표준 스택·큐가 없으므로 배열 스택 · 2의 거듭제곱 링 · 이진 힙 · 침입형 리스트 네 모양을 손에 익히고, 같이 쓰는 주체(스레드·ISR)에 따라 뮤텍스 큐나 SPSC 링을 고르면 실무의 거의 전부가 덮입니다.
SQ 02

스택 — 배열과 top 하나, 그리고 타입 일반화 세 가지

스택은 C 로 만들 수 있는 가장 단순한 자료구조입니다. 배열 하나와 정수 하나면 끝납니다. 단순한 만큼 차이는 규약에서 납니다 — top 이 "마지막 원소"인지 "다음 빈칸"인지, 빈 스택에서 pop 하면 무엇을 돌려주는지, 늘어날 때 실패하면 기존 데이터가 살아 있는지. 그리고 int 전용을 넘어 아무 타입이나 담는 방법이 C 에는 세 가지 있습니다.

스택의 상태는 top 하나가 전부입니다. 칸에 남은 값은 top 위에 있는 한 없는 것입니다.

고정 크기 스택 — 전체 코드 (malloc 0회)

#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

#define STACK_CAP 4

typedef struct {
    int    items[STACK_CAP];
    size_t top;                 // 다음에 넣을 칸 = 현재 개수
} IntStack;

static void stack_init(IntStack *s)        { s->top = 0; }
static bool stack_empty(const IntStack *s) { return s->top == 0; }
static bool stack_full(const IntStack *s)  { return s->top == STACK_CAP; }

static bool stack_push(IntStack *s, int v) {
    if (stack_full(s)) return false;         // 오버플로 — 거절
    s->items[s->top++] = v;                  // 쓰고 나서 올림
    return true;
}
static bool stack_pop(IntStack *s, int *out) {
    if (stack_empty(s)) return false;        // 언더플로 — *out 그대로
    *out = s->items[--s->top];               // 내리고 나서 읽음
    return true;
}
static bool stack_peek(const IntStack *s, int *out) {
    if (stack_empty(s)) return false;
    *out = s->items[s->top - 1];
    return true;
}

int main(void) {
    IntStack s;
    stack_init(&s);
    for (int i = 1; i <= 5; i++)
        if (!stack_push(&s, i * 10))
            printf("push %d 실패: 가득 참\n", i * 10);
    int v;
    while (stack_pop(&s, &v)) printf("pop %d\n", v);
    if (!stack_pop(&s, &v)) printf("빈 스택 pop: false\n");
    return 0;
}
// 출력:
// push 50 실패: 가득 참
// pop 40
// pop 30
// pop 20
// pop 10
// 빈 스택 pop: false

top 규약 두 가지 — 하나로 통일하세요

top = 다음 빈칸 (권장)top = 마지막 원소
초기값0-1 → 부호 있는 타입 필요
pusha[top++] = va[++top] = v
popv = a[--top]v = a[top--]
개수toptop + 1
가득 참top == CAPtop == CAP - 1

"다음 빈칸" 규약은 size_t 를 그대로 쓸 수 있고(−1 이 필요 없음), 개수와 top 이 같아 off-by-one 이 생길 자리가 하나 줄어듭니다. 교과서의 top = -1 은 int 를 강요하고, top == CAP - 1 같은 "−1 붙은 비교"를 낳습니다.

빈 스택 처리 세 갈래 — ① bool 반환(위 코드, 라이브러리용) · ② assert(!stack_empty(s))(빈 pop 이 곧 로직 버그인 내부 코드) · ③ 특수값 반환(-1, INT_MIN) — ③ 은 정상 값과 겹치는 순간 버그이므로 피합니다. assert 는 NDEBUG 빌드에서 사라진다는 점도 기억하세요.

최대 크기를 모를 때 — realloc 2배 성장 🟡 중급

동적 스택 — 실패해도 기존 데이터를 지킵니다

#include <stdlib.h>
#include <stdint.h>
#include <stdbool.h>

typedef struct { int *data; size_t len, cap; } DynStack;

#define DYNSTACK_INIT { NULL, 0, 0 }

bool dstack_push(DynStack *s, int v) {
    if (s->len == s->cap) {
        size_t ncap = s->cap ? s->cap * 2 : 8;
        if (ncap > SIZE_MAX / sizeof *s->data)     // 곱셈 넘침 검사
            return false;
        int *p = realloc(s->data, ncap * sizeof *p);
        if (!p) return false;   // ★ s->data 는 아직 유효 — 원소 그대로
        s->data = p;            //   성공한 뒤에만 덮어씁니다
        s->cap  = ncap;
    }
    s->data[s->len++] = v;
    return true;
}

bool dstack_pop(DynStack *s, int *out) {
    if (s->len == 0) return false;
    *out = s->data[--s->len];
    return true;
}

void dstack_free(DynStack *s) {
    free(s->data);
    *s = (DynStack)DYNSTACK_INIT;     // 해제 후 재사용해도 안전
}

// 1000 번 push 하면: realloc 8 회 (cap 8→16→…→1024)
// 옮겨진 원소 최악 8+16+…+512 = 1016 개 < 2n → 분할 상환 O(1)

줄일 때는 1/4 에서 절반으로 — 진동 방지

// pop 뒤에 호출 (선택 사항)
static void dstack_shrink(DynStack *s) {
    // len 이 cap 의 1/4 이하일 때만 절반으로.
    // 1/2 에서 줄이면 경계에서 push/pop 반복 시
    // 매번 realloc 이 일어납니다 (스래싱)
    if (s->cap > 8 && s->len <= s->cap / 4) {
        size_t ncap = s->cap / 2;
        int *p = realloc(s->data, ncap * sizeof *p);
        if (p) { s->data = p; s->cap = ncap; }
        // 줄이기 실패는 무시해도 됩니다 — 큰 버퍼 그대로 써도 정답
    }
}
2배를 고정하지 않아도 됩니다 — 1.5배(cap + cap/2)는 해제된 이전 블록을 나중에 재사용할 여지가 생겨 메모리 효율이 조금 낫다는 이유로 쓰이기도 합니다. 핵심은 곱하기라는 점이고, +16 같은 더하기 성장은 O(n²) 입니다. 근거와 그래프는 c14 에 있습니다.
임베디드에서는 쓰지 않습니다 — 2 KB SRAM 에서 realloc 은 단편화로 "남은 메모리는 충분한데 할당 실패"를 만듭니다. 최대치를 계산해 고정 크기로 두세요 (SQ 09).

아무 타입이나 담기 — void* · 매크로 생성 · _Generic 🔴 고급

하나가 정답이 아닙니다 — 경계를 넘으면 void*, 안쪽 뜨거운 경로면 매크로 생성입니다.

① void* + 원소 크기 — memcpy 로 복사

#include <string.h>   // memcpy
#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    unsigned char *data;   // 바이트 단위로 계산하려고 char 계열
    size_t len, cap, esz;  // esz = 원소 하나의 바이트 수
} GStack;

bool gstack_init(GStack *s, size_t esz, size_t cap) {
    s->data = malloc(cap * esz);     // (cap * esz 넘침 검사 생략)
    s->len = 0; s->cap = cap; s->esz = esz;
    return s->data != NULL;
}
bool gstack_push(GStack *s, const void *elem) {
    if (s->len == s->cap) return false;
    memcpy(s->data + s->len * s->esz, elem, s->esz);
    s->len++;
    return true;
}
bool gstack_pop(GStack *s, void *out) {
    if (s->len == 0) return false;
    s->len--;
    memcpy(out, s->data + s->len * s->esz, s->esz);
    return true;
}

// 사용 (함수 안에서) — 값이 아니라 주소를 넘깁니다
typedef struct { short x, y; } Pt;
GStack st;
gstack_init(&st, sizeof(Pt), 64);        // ★ sizeof(Pt*) 로 쓰면 조용히 망가짐
Pt p = { 3, 4 };
gstack_push(&st, &p);
Pt q;
gstack_pop(&st, &q);                     // q = {3, 4}

② 매크로로 타입별 생성 + ③ _Generic 디스패치

#include <stdbool.h>
#include <stddef.h>

// ② T 하나에 구조체 + 함수 세 개를 찍어냅니다
#define DEFINE_STACK(T, CAP)                                      \
    typedef struct { T items[CAP]; size_t top; } T##_stack;       \
    static inline void T##_stack_init(T##_stack *s) { s->top = 0; } \
    static inline bool T##_stack_push(T##_stack *s, T v) {        \
        if (s->top == (CAP)) return false;                        \
        s->items[s->top++] = v; return true; }                    \
    static inline bool T##_stack_pop(T##_stack *s, T *out) {      \
        if (s->top == 0) return false;                            \
        *out = s->items[--s->top]; return true; }

DEFINE_STACK(int, 32)       // int_stack, int_stack_push ...
DEFINE_STACK(double, 16)    // double_stack ...

// ③ C11 _Generic — 스택 포인터의 타입을 보고 함수를 고릅니다
#define stack_push(s, v) _Generic((s),          \
        int_stack *:    int_stack_push,         \
        double_stack *: double_stack_push)(s, v)
#define stack_pop(s, out) _Generic((s),         \
        int_stack *:    int_stack_pop,          \
        double_stack *: double_stack_pop)(s, out)

void demo(void) {
    int_stack a;    int_stack_init(&a);
    double_stack b; double_stack_init(&b);
    stack_push(&a, 7);       // → int_stack_push
    stack_push(&b, 2.5);     // → double_stack_push
    double d;
    stack_pop(&b, &d);       // d = 2.5
    // stack_pop(&a, &d);    // ✗ 컴파일 에러: double* 를 int* 로 못 넘김
}
토큰 붙이기의 한계 — DEFINE_STACK(unsigned int, 8) 이나 DEFINE_STACK(char *, 8) 은 이름에 공백·* 가 들어가 깨집니다. typedef unsigned int uint; 처럼 한 단어 이름을 먼저 만드세요. _Generic 의 기본 문법은 ⚙️ C 실전 탭 c21 에 있습니다.
정리 — 스택은 top = 다음 빈칸 규약 + bool 반환 + out 매개변수로 만들고, 최대치를 알면 고정 배열, 모르면 성공한 뒤에만 포인터를 덮는 realloc 2배. 타입 일반화는 모듈 경계면 void*, 뜨거운 내부면 매크로 생성(+ 필요하면 _Generic).
SQ 03

큐(FIFO) — 원형 버퍼와 "가득 참 vs 비어 있음" 세 가지 해법

배열로 큐를 만들면 앞에서 뺄 때마다 남은 원소를 당겨야 합니다(memmove, O(n)). 그래서 C 의 큐는 거의 언제나 원형 버퍼입니다 — 쓰는 자리(head)와 읽는 자리(tail)가 배열 위를 돌고, 끝에 닿으면 처음으로 돌아갑니다. 문제는 하나입니다. head == tail 이 "비었다"인지 "꽉 찼다"인지 구분이 안 됩니다. 해법이 셋 있고, 무엇을 고르느냐가 나중에 락 없이 쓸 수 있는지(SQ 07)를 결정합니다.

한 칸 비우기는 용량 하나를 버리고, count 는 공유 변수 하나를 늘리고, 무한 증가 인덱스는 둘 다 안 합니다.
방식비었다가득 찼다쓸 수 있는 칸쓰는 변수ISR ↔ 메인 락 없이
① 한 칸 비우기h == t((h+1) & M) == tN − 1생산자 h · 소비자 t가능
② count 유지count == 0count == NN양쪽이 count불가 (갱신 유실)
③ 무한 증가 인덱스h == th - t == NN생산자 h · 소비자 t가능 — 권장

③ 무한 증가 인덱스 — 권장 구현 (전체)

#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>   // PRIu32
#include <stdbool.h>

#define Q_CAP  8u                       // 반드시 2의 거듭제곱
#define Q_MASK (Q_CAP - 1u)
_Static_assert(Q_CAP != 0 && (Q_CAP & (Q_CAP - 1u)) == 0,
               "Q_CAP must be a power of two");

typedef struct {
    int      buf[Q_CAP];
    uint32_t head;     // 지금까지 넣은 총 개수 (계속 증가, 넘쳐도 됨)
    uint32_t tail;     // 지금까지 뺀 총 개수
} Queue;

static inline uint32_t q_count(const Queue *q) { return q->head - q->tail; }
static inline bool q_empty(const Queue *q) { return q->head == q->tail; }
static inline bool q_full(const Queue *q)  { return q_count(q) == Q_CAP; }

bool q_push(Queue *q, int v) {
    if (q_full(q)) return false;
    q->buf[q->head & Q_MASK] = v;       // 칸 번호는 마스크로
    q->head++;                          // 인덱스는 그냥 +1
    return true;
}
bool q_pop(Queue *q, int *out) {
    if (q_empty(q)) return false;
    *out = q->buf[q->tail & Q_MASK];
    q->tail++;
    return true;
}

int main(void) {
    // 넘침 직전에서 시작해 경계를 일부러 밟습니다 (테스트에도 이렇게)
    Queue q = { .head = 0xFFFFFFFEu, .tail = 0xFFFFFFFEu };
    for (int i = 1; i <= 9; i++)
        if (!q_push(&q, i))
            printf("push %d 거절 (count=%" PRIu32 ")\n", i, q_count(&q));
    printf("head=%" PRIu32 " tail=%" PRIu32 " count=%" PRIu32 "\n",
           q.head, q.tail, q_count(&q));
    int v;
    while (q_pop(&q, &v)) printf("%d ", v);
    printf("\nhead=%" PRIu32 " tail=%" PRIu32 "\n", q.head, q.tail);
    return 0;
}
// 출력:
// push 9 거절 (count=8)
// head=6 tail=4294967294 count=8
// 1 2 3 4 5 6 7 8
// head=6 tail=6

① · ② — 크기가 2의 거듭제곱이 아닐 때

// ① 한 칸 비우기 — N 이 아무 값이어도 됩니다 (N-1 개 저장)
#define N 10
typedef struct { int buf[N]; uint16_t h, t; } Ring1;

static inline uint16_t next(uint16_t i) {
    return (uint16_t)(i + 1 == N ? 0 : i + 1);   // % 대신 비교 — 나눗셈 없음
}
bool r1_push(Ring1 *r, int v) {
    uint16_t nh = next(r->h);
    if (nh == r->t) return false;      // 한 칸 남았으면 '가득'
    r->buf[r->h] = v;
    r->h = nh;
    return true;
}
bool r1_pop(Ring1 *r, int *out) {
    if (r->h == r->t) return false;
    *out = r->buf[r->t];
    r->t = next(r->t);
    return true;
}

// ② count 유지 — 단일 스레드라면 가장 읽기 쉽습니다
typedef struct { int buf[N]; uint16_t h, t, count; } Ring2;

bool r2_push(Ring2 *r, int v) {
    if (r->count == N) return false;
    r->buf[r->h] = v;
    r->h = next(r->h);
    r->count++;           // ★ 소비자도 count-- 를 합니다 → 공유 쓰기
    return true;
}
③ 은 2의 거듭제곱에서만 — 크기가 10 이면 무한 증가 인덱스를 쓸 수 없습니다(아래 계산). 크기를 바꿀 수 없다면 ① 을 쓰고, count 방식(②)은 스레드·ISR 가 끼는 순간 버립니다 — count++ 는 읽기·더하기·쓰기 세 동작이라 양쪽에서 하면 하나가 사라집니다(SQ 07).

무한 증가 인덱스가 넘쳐도 안전한 이유 — 계산으로 확인 🔴 고급

핵심 두 줄 — 부호 없는 산술은 원 위의 산술이고, N 이 2³² 를 나누면 원을 몇 바퀴 돌아도 칸 번호가 이어집니다.

SQ 04

덱 — 양끝이 열린 원형 버퍼, 연결 큐, 그리고 침입형 리스트

덱은 원형 버퍼에 앞쪽 push/pop 을 더한 것입니다. 자바 ArrayDeque, 러스트 VecDeque 가 정확히 이 구조이고, C 에서는 head(맨 앞 칸)와 len(개수) 두 값으로 표현하는 것이 가장 깔끔합니다. 까다로운 곳은 자랄 때입니다 — 데이터가 배열 끝에서 감겨 있으면 realloc 만으로는 순서가 깨집니다. 뒤쪽 절반은 노드를 쓰는 큐 두 가지, 특히 임베디드와 커널이 사랑하는 침입형 리스트입니다.

원형 버퍼를 키울 때는 배열 순서가 아니라 논리 순서로 복사해야 합니다 — memcpy 두 번.

덱 — 전체 코드 (용량 2의 거듭제곱, 자동 성장)

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

typedef struct {
    int   *buf;
    size_t cap;    // 항상 2의 거듭제곱
    size_t head;   // 맨 앞 원소가 있는 칸
    size_t len;    // 원소 개수
} Deque;

#define DQ_AT(d, i) ((d)->buf[((d)->head + (i)) & ((d)->cap - 1)])

static bool dq_init(Deque *d, size_t cap_pow2) {
    d->buf = malloc(cap_pow2 * sizeof *d->buf);
    d->cap = cap_pow2; d->head = 0; d->len = 0;
    return d->buf != NULL;
}

static bool dq_grow(Deque *d) {
    size_t ncap = d->cap * 2;
    int *nb = malloc(ncap * sizeof *nb);
    if (!nb) return false;                  // 실패해도 기존 덱은 멀쩡
    size_t first = d->cap - d->head;        // head 부터 배열 끝까지
    if (first > d->len) first = d->len;
    memcpy(nb, d->buf + d->head, first * sizeof *nb);           // 조각 ①
    memcpy(nb + first, d->buf, (d->len - first) * sizeof *nb); // 조각 ②
    printf("grow %zu->%zu (조각 %zu + %zu)\n",
           d->cap, ncap, first, d->len - first);
    free(d->buf);
    d->buf = nb; d->cap = ncap; d->head = 0;
    return true;
}

static bool dq_push_back(Deque *d, int v) {
    if (d->len == d->cap && !dq_grow(d)) return false;
    d->buf[(d->head + d->len) & (d->cap - 1)] = v;
    d->len++;
    return true;
}
static bool dq_push_front(Deque *d, int v) {
    if (d->len == d->cap && !dq_grow(d)) return false;
    d->head = (d->head - 1) & (d->cap - 1);  // 0-1 = SIZE_MAX → & → cap-1
    d->buf[d->head] = v;
    d->len++;
    return true;
}
static bool dq_pop_front(Deque *d, int *out) {
    if (d->len == 0) return false;
    *out = d->buf[d->head];
    d->head = (d->head + 1) & (d->cap - 1);
    d->len--;
    return true;
}
static bool dq_pop_back(Deque *d, int *out) {
    if (d->len == 0) return false;
    d->len--;
    *out = DQ_AT(d, d->len);
    return true;
}

int main(void) {
    Deque d;
    if (!dq_init(&d, 4)) return 1;
    dq_push_back(&d, 1);   dq_push_back(&d, 2);
    dq_push_front(&d, 0);  dq_push_front(&d, -1);
    printf("buf=[%d,%d,%d,%d] head=%zu len=%zu\n",
           d.buf[0], d.buf[1], d.buf[2], d.buf[3], d.head, d.len);
    dq_push_back(&d, 3);                  // 가득 → 두 조각 복사
    printf("head=%zu len=%zu cap=%zu\n", d.head, d.len, d.cap);
    printf("앞→뒤:");
    for (size_t i = 0; i < d.len; i++) printf(" %d", DQ_AT(&d, i));
    int a, b;
    dq_pop_front(&d, &a);
    dq_pop_back(&d, &b);
    printf("\npop_front=%d pop_back=%d len=%zu\n", a, b, d.len);
    free(d.buf);
    return 0;
}
// 출력:
// buf=[1,2,-1,0] head=2 len=4
// grow 4->8 (조각 2 + 2)
// head=0 len=5 cap=8
// 앞→뒤: -1 0 1 2 3
// pop_front=-1 pop_back=3 len=3
연산비용

size_t 의 0 − 1 은 안전합니다 — 부호 없는 넘침은 정의된 동작이라 SIZE_MAX 가 되고, 마스크를 걸면 cap - 1 입니다. 같은 코드를 int head 로 쓰고 % 를 쓰면 -1 % 4 == -1 이 되어 배열 앞을 씁니다. 덱은 부호 없는 인덱스 + 마스크가 가장 짧고 안전한 조합입니다.

연결 큐 — head 에서 빼고 tail 에 붙입니다

typedef struct Node { int v; struct Node *next; } Node;
typedef struct { Node *head, *tail; } LQueue;   // {NULL, NULL} = 빈 큐

bool lq_push(LQueue *q, int v) {
    Node *n = malloc(sizeof *n);
    if (!n) return false;
    n->v = v;
    n->next = NULL;
    if (q->tail) q->tail->next = n;   // 비어 있지 않으면 꼬리 뒤에
    else         q->head = n;         // 비어 있었으면 머리도 n
    q->tail = n;
    return true;
}

bool lq_pop(LQueue *q, int *out) {
    Node *n = q->head;
    if (!n) return false;
    *out = n->v;
    q->head = n->next;
    if (!q->head) q->tail = NULL;     // ★ 마지막을 뺐으면 꼬리도 비움
    free(n);
    return true;
}
// ★ 줄을 빠뜨리면: tail 이 해제된 노드를 가리킨 채 남고,
//   다음 push 가 그 해제된 메모리에 next 를 씁니다 (use-after-free)

침입형 리스트 — list_head 와 container_of 로 할당 없는 큐 🔴 고급

침입형 = 고리를 객체 안에. 큐는 고리만 잇고, container_of 가 고리에서 객체로 되돌아갑니다.

리눅스 커널 방식 — 전체 코드 (malloc 0회)

#include <stdio.h>
#include <stddef.h>     // offsetof
#include <stdbool.h>

struct list_head { struct list_head *next, *prev; };

#define LIST_HEAD_INIT(name) { &(name), &(name) }
#define container_of(ptr, type, member) \
    ((type *)((char *)(ptr) - offsetof(type, member)))

static inline void list_init(struct list_head *h) { h->next = h->prev = h; }
static inline bool list_empty(const struct list_head *h) { return h->next == h; }

// h 바로 앞 = 꼬리에 끼움 → 큐의 push
static inline void list_add_tail(struct list_head *n, struct list_head *h) {
    n->prev = h->prev;
    n->next = h;
    h->prev->next = n;
    h->prev = n;
}
// 어디에 있든 O(1) 로 빼고 자기 자신을 가리키게
static inline void list_del_init(struct list_head *n) {
    n->prev->next = n->next;
    n->next->prev = n->prev;
    list_init(n);
}

typedef struct {
    int id;
    const char *name;
    struct list_head node;          // ★ 큐에 걸리는 고리
} Job;

static Job jobs[3] = { {1, "read"}, {2, "parse"}, {3, "send"} };
static struct list_head ready = LIST_HEAD_INIT(ready);
static struct list_head done  = LIST_HEAD_INIT(done);

static Job *job_pop(struct list_head *q) {
    if (list_empty(q)) return NULL;
    struct list_head *n = q->next;          // 맨 앞
    list_del_init(n);
    return container_of(n, Job, node);      // 고리 → 객체
}

int main(void) {
    for (int i = 0; i < 3; i++)
        list_add_tail(&jobs[i].node, &ready);
    Job *j;
    while ((j = job_pop(&ready)) != NULL) {
        printf("run %d %s\n", j->id, j->name);
        list_add_tail(&j->node, &done);     // 같은 고리로 다른 큐에
    }
    printf("ready empty=%d, done first=%s\n", list_empty(&ready),
           container_of(done.next, Job, node)->name);
    return 0;
}
// 출력:
// run 1 read
// run 2 parse
// run 3 send
// ready empty=1, done first=read

왜 임베디드 · 커널은 이 방식을 쓰나

  • 할당이 없다 — 객체가 정적 배열이든 스택이든 메모리 풀이든, 큐에 넣는 순간 메모리가 늘지 않습니다. ISR 에서도 넣고 뺄 수 있습니다
  • O(1) 삭제 — 객체 포인터만 있으면 큐의 어디에 있든 바로 뺍니다. 타임아웃으로 대기 큐에서 작업을 취소할 때 결정적입니다
  • 한 객체, 여러 큐 — 고리를 두 개 두면(run_node, all_node) 실행 큐와 전체 목록에 동시에 걸립니다
  • 타입 무관 — 리스트 코드는 list_head 만 알고, 타입은 container_of 가 되돌립니다. void* 캐스트가 없습니다
어디서같은 아이디어
리눅스 커널struct list_head · list_entry(= container_of)
BSD · glibc<sys/queue.h> 의 TAILQ_ENTRY 매크로
FreeRTOSTCB 안의 ListItem_t + pvOwner 역포인터
Zephyrsys_dlist_t · CONTAINER_OF
대가 — 객체의 수명을 큐가 관리하지 않습니다. 큐에 걸린 채로 객체를 해제(또는 스택 변수가 함수 밖으로 사라짐)하면 리스트가 죽은 메모리를 가리킵니다. 해제 전에 list_del_init 를 부르는 규칙을 지키고, 디버그 빌드에서는 해제 시 !list_empty(&obj->node) 면 assert 로 잡으세요.
정리 — 덱은 head + len + 마스크로 양끝 O(1), 자랄 때는 memcpy 두 번으로 펴서 복사합니다. 노드 큐가 필요하면 malloc 노드 대신 침입형 list_head — 할당 없고, O(1) 삭제, ISR 에서도 안전합니다.
SQ 05

우선순위 큐 — 배열 이진 힙과 비교 함수 포인터, 그리고 타이머 큐

"가장 급한 것부터" 꺼내는 큐는 C 에서도 배열 위의 이진 힙이 정답입니다. 트리를 포인터로 만들지 않습니다 — i 번 칸의 부모는 (i-1)/2, 자식은 2i+1 · 2i+2 라는 산수가 곧 트리입니다. 순서 규칙은 qsort 와 같은 비교 함수 포인터로 받습니다. 그리고 임베디드에서 힙을 가장 많이 쓰는 곳, 소프트웨어 타이머를 32비트 tick 넘침까지 고려해 만들어 봅니다.

힙은 정렬된 배열이 아닙니다 — 루트만 최소가 보장되고, 꺼낼 때마다 log n 단계로 다시 맞춥니다.

고정 용량 이진 힙 — 비교 함수 포인터

#include <stddef.h>
#include <stdbool.h>

// qsort 와 같은 규약: 음수면 a 가 먼저 나와야 함
typedef int (*heap_cmp)(const void *a, const void *b);

typedef struct {
    void   **a;          // 원소 포인터 배열 (원소 자체는 호출자 소유)
    size_t   len, cap;
    heap_cmp cmp;
} Heap;

static void swap_p(void **x, void **y) { void *t = *x; *x = *y; *y = t; }

static void sift_up(Heap *h, size_t i) {
    while (i > 0) {
        size_t p = (i - 1) / 2;
        if (h->cmp(h->a[i], h->a[p]) >= 0) break;  // 부모가 먼저 → 멈춤
        swap_p(&h->a[i], &h->a[p]);
        i = p;
    }
}
static void sift_down(Heap *h, size_t i) {
    for (;;) {
        size_t l = 2 * i + 1, r = l + 1, m = i;
        if (l < h->len && h->cmp(h->a[l], h->a[m]) < 0) m = l;
        if (r < h->len && h->cmp(h->a[r], h->a[m]) < 0) m = r;
        if (m == i) break;
        swap_p(&h->a[i], &h->a[m]);
        i = m;
    }
}

bool heap_push(Heap *h, void *item) {
    if (h->len == h->cap) return false;      // 고정 용량 — 임베디드형
    h->a[h->len] = item;
    sift_up(h, h->len);
    h->len++;
    return true;
}
void *heap_peek(const Heap *h) { return h->len ? h->a[0] : NULL; }
void *heap_pop(Heap *h) {
    if (h->len == 0) return NULL;
    void *top = h->a[0];
    h->a[0] = h->a[--h->len];               // 마지막을 루트로
    if (h->len) sift_down(h, 0);
    return top;
}

다중 키 비교 + 도착 순번 — 같은 우선순위의 순서

#include <stdio.h>
#include <stdint.h>

typedef struct { const char *name; int pri; uint32_t seq; } Task;

static int task_cmp(const void *x, const void *y) {
    const Task *a = x, *b = y;
    if (a->pri != b->pri)                       // 1순위: 작은 pri 먼저
        return (a->pri > b->pri) - (a->pri < b->pri);
    return (a->seq > b->seq) - (a->seq < b->seq); // 2순위: 먼저 온 것
}

int main(void) {
    static Task t[] = { {"log",3}, {"irq",0}, {"ui",2},
                        {"net",1}, {"tick",0}, {"save",2} };
    void *slots[8];
    Heap h = { slots, 0, 8, task_cmp };
    uint32_t seq = 0;
    for (size_t i = 0; i < 6; i++) { t[i].seq = seq++; heap_push(&h, &t[i]); }

    for (size_t i = 0; i < h.len; i++) {        // 배열 그대로 — 정렬 아님
        const Task *k = h.a[i];
        printf("%s(%d) ", k->name, k->pri);
    }
    printf("\n");
    Task *k;
    while ((k = heap_pop(&h)) != NULL) printf("%s(%d) ", k->name, k->pri);
    printf("\n");
    return 0;
}
// 출력:
// irq(0) tick(0) ui(2) log(3) net(1) save(2)
// irq(0) tick(0) net(1) ui(2) save(2) log(3)
return a->pri - b->pri; 금지 — 값이 크면 뺄셈이 넘칩니다(INT_MIN - 1 은 UB). (a > b) - (a < b) 가 -1/0/1 을 넘침 없이 만드는 C 관용구입니다.
seq 를 빼면? — 같은 우선순위 1 인 A·B·C·D·E 를 넣고 0 인 z 를 넣은 뒤 다 꺼내면 z C E D A B 순서가 나옵니다(실제로 돌린 결과). 힙은 안정적이지 않습니다. "같으면 먼저 온 것"이 요구 사항이면 순번을 마지막 키로 넣어야 합니다.

소프트웨어 타이머 큐 — tick 넘침에 안전한 비교 🔴 고급

tick 비교는 값이 아니라 차이로 — (int32_t)(a − b) 의 부호가 "어느 쪽이 먼저"입니다.

타이머 힙 — 위의 Heap 그대로 재사용

#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>

typedef struct { uint32_t deadline; const char *name; } SwTimer;

// a 가 b 보다 먼저 울려야 하면 음수
static int timer_cmp(const void *x, const void *y) {
    const SwTimer *a = x, *b = y;
    int32_t d = (int32_t)(a->deadline - b->deadline);   // ★ 차이의 부호
    return (d > 0) - (d < 0);
}

int main(void) {
    uint32_t now = 0xFFFFFF00u;          // 부팅 약 49.7일 — 넘침 256 ms 전
    static SwTimer t[] = { {0,"blink"}, {0,"adc"}, {0,"wdt"}, {0,"uart"} };
    const uint32_t after[] = { 500, 100, 1000, 300 };
    void *slots[8];
    Heap h = { slots, 0, 8, timer_cmp };
    for (int i = 0; i < 4; i++) {
        t[i].deadline = now + after[i];  // 부호 없는 덧셈 — 넘쳐도 정의됨
        heap_push(&h, &t[i]);
    }

    now += 400;                          // 400 ms 경과 → now == 144
    SwTimer *top;
    while ((top = heap_peek(&h)) != NULL &&
           (int32_t)(now - top->deadline) >= 0) {   // ★ now >= deadline 금지
        heap_pop(&h);
        printf("fire %s (deadline=%" PRIu32 ")\n", top->name, top->deadline);
    }
    top = heap_peek(&h);
    printf("next %s in %" PRId32 " ms\n", top->name,
           (int32_t)(top->deadline - now));
    return 0;
}
// 출력:
// fire adc (deadline=4294967140)
// fire uart (deadline=44)
// next blink in 100 ms

조건과 대안

  • 조건 — 대기 중인 모든 마감이 서로 2³¹ tick 이내여야 합니다. 1 ms tick 이면 2³¹ ms ≈ 24.8일. 그보다 긴 타이머는 "하루짜리를 여러 번"으로 쪼갭니다
  • 캐스트 — 범위 밖 uint32_t → int32_t 변환은 표준상 구현 정의지만 GCC · Clang · Arm 컴파일러는 모두 2의 보수로 잘라 냅니다. 리눅스 커널의 time_after() 도 같은 식입니다
  • ISR 에서 타이머 추가 — 힙은 여러 칸을 고치므로 SPSC 가 아닙니다. ISR 은 "추가 요청"만 SPSC 링(SQ 07)에 넣고, 힙 조작은 메인 루프 한 곳에서
타이머 자료구조추가가장 이른 것쓰는 곳
정렬된 연결 리스트O(n)O(1)FreeRTOS 지연·타이머 리스트 (개수 적음)
이진 힙O(log n)O(1) · 꺼내기 O(log n)libevent · 사용자 공간 이벤트 루프
타이머 휠O(1)tick 마다 슬롯 하나리눅스 커널 · 네트워크 타임아웃 수만 개
타이머 휠 한 줄 요약 — 슬롯 256 개짜리 원형 배열의 각 칸이 침입형 리스트(SQ 04)입니다. slot = (now + delay) & 255 에 끼우고, tick 마다 현재 슬롯 하나만 훑습니다. 255 tick 보다 먼 것은 "남은 바퀴 수"를 같이 적거나 상위 휠에 둡니다. 타이머가 수십 개면 힙으로 충분합니다.
정리 — 우선순위 큐는 배열 이진 힙 + 비교 함수 포인터, 같은 우선순위의 순서가 필요하면 순번을 마지막 키로. 타이머 큐의 비교는 (int32_t)(a − b) — tick 넘침을 지나도 순서가 맞습니다.
SQ 06

스레드 안전 큐 — 뮤텍스 + 조건 변수, 타임아웃과 깔끔한 종료

여러 스레드가 넣고 여러 스레드가 빼는(N:M) 큐는 뮤텍스 하나 + 조건 변수 둘이 표준 해법입니다. 기본형(while 로 다시 확인하는 이유, closed 플래그)은 ⚙️ C 실전 탭 c19 에 있으니, 여기서는 실무에서 바로 부딪히는 세 가지를 더합니다 — 타임아웃 대기(시계가 바뀌어도 흔들리지 않게), 닫은 뒤 남은 것 비우기(drain), 그리고 포인터를 싣는 범용 메시지.

조건 변수는 "알림"일 뿐 상태가 아닙니다 — 깨어나면 반드시 큐를 다시 봅니다.

bq.c — 유한 · 블로킹 · 타임아웃 · drain 종료 (전체 코드)

// cc -std=c17 -O2 -pthread bq.c
#define _POSIX_C_SOURCE 200809L
#include <pthread.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>
#include <time.h>
#include <errno.h>

#define BQ_CAP 4u                 // 일부러 작게 — 배압이 실제로 걸리도록
_Static_assert((BQ_CAP & (BQ_CAP - 1u)) == 0, "BQ_CAP: power of two");

typedef enum { BQ_OK, BQ_TIMEOUT, BQ_CLOSED } BqStatus;

typedef struct {
    void           *slot[BQ_CAP];
    uint32_t        head, tail;   // 무한 증가 인덱스 (SQ 03)
    bool            closed;
    pthread_mutex_t mu;
    pthread_cond_t  not_empty, not_full;
} BQueue;

int bq_init(BQueue *q) {
    q->head = q->tail = 0;
    q->closed = false;
    pthread_condattr_t ca;
    pthread_condattr_init(&ca);
    // ★ 벽시계(REALTIME)가 NTP 로 바뀌어도 타임아웃이 흔들리지 않게
    pthread_condattr_setclock(&ca, CLOCK_MONOTONIC);   // macOS 에는 없음
    int rc = pthread_mutex_init(&q->mu, NULL);
    if (!rc) rc = pthread_cond_init(&q->not_empty, &ca);
    if (!rc) rc = pthread_cond_init(&q->not_full, &ca);
    pthread_condattr_destroy(&ca);
    return rc;
}

// 가득이면 기다림(배압). 닫힌 큐에는 넣지 않음
BqStatus bq_push(BQueue *q, void *item) {
    pthread_mutex_lock(&q->mu);
    while (q->head - q->tail == BQ_CAP && !q->closed)
        pthread_cond_wait(&q->not_full, &q->mu);
    if (q->closed) {
        pthread_mutex_unlock(&q->mu);
        return BQ_CLOSED;
    }
    q->slot[q->head & (BQ_CAP - 1u)] = item;
    q->head++;
    pthread_cond_signal(&q->not_empty);
    pthread_mutex_unlock(&q->mu);
    return BQ_OK;
}

static void deadline_in(struct timespec *ts, long ms) {
    clock_gettime(CLOCK_MONOTONIC, ts);
    ts->tv_sec  += ms / 1000;
    ts->tv_nsec += (ms % 1000) * 1000000L;
    if (ts->tv_nsec >= 1000000000L) { ts->tv_sec++; ts->tv_nsec -= 1000000000L; }
}
// 비면 최대 timeout_ms 기다림.
// 닫혀도 남은 것은 다 꺼내 준 뒤에야 BQ_CLOSED (drain)
BqStatus bq_pop(BQueue *q, void **out, long timeout_ms) {
    struct timespec dl;
    deadline_in(&dl, timeout_ms);       // 절대 시각 — 가짜 깨어남에도 고정
    pthread_mutex_lock(&q->mu);
    while (q->head == q->tail && !q->closed) {
        int rc = pthread_cond_timedwait(&q->not_empty, &q->mu, &dl);
        if (rc == ETIMEDOUT && q->head == q->tail) {
            pthread_mutex_unlock(&q->mu);
            return BQ_TIMEOUT;
        }
    }
    if (q->head == q->tail) {           // 닫혔고, 다 비었음
        pthread_mutex_unlock(&q->mu);
        return BQ_CLOSED;
    }
    *out = q->slot[q->tail & (BQ_CAP - 1u)];
    q->tail++;
    pthread_cond_signal(&q->not_full);
    pthread_mutex_unlock(&q->mu);
    return BQ_OK;
}

void bq_close(BQueue *q) {
    pthread_mutex_lock(&q->mu);
    q->closed = true;
    pthread_cond_broadcast(&q->not_empty);   // 기다리는 소비자 전원
    pthread_cond_broadcast(&q->not_full);    // 기다리는 생산자 전원
    pthread_mutex_unlock(&q->mu);
}

// ───── 사용: 생산자 2 · 소비자 2 ─────
static BQueue q;
static long total[2];
static int  got[2];

static void *producer(void *arg) {
    intptr_t base = (intptr_t)arg;               // 100 또는 200
    for (intptr_t i = 1; i <= 5; i++)
        bq_push(&q, (void *)(base + i));         // 작은 정수는 포인터에 실어 보냄
    return NULL;
}
static void *consumer(void *arg) {
    int id = (int)(intptr_t)arg;
    void *item;
    BqStatus st;
    while ((st = bq_pop(&q, &item, 500)) != BQ_CLOSED) {
        if (st == BQ_TIMEOUT) continue;          // 하트비트 같은 주기 작업 자리
        total[id] += (long)(intptr_t)item;       // 자기 칸만 씀 — 경쟁 없음
        got[id]++;
    }
    return NULL;
}

int main(void) {
    if (bq_init(&q) != 0) return 1;
    pthread_t p[2], c[2];
    for (intptr_t i = 0; i < 2; i++)
        pthread_create(&c[i], NULL, consumer, (void *)i);
    pthread_create(&p[0], NULL, producer, (void *)(intptr_t)100);
    pthread_create(&p[1], NULL, producer, (void *)(intptr_t)200);
    pthread_join(p[0], NULL);
    pthread_join(p[1], NULL);
    bq_close(&q);                    // ★ 생산자가 모두 끝난 뒤에 닫기
    pthread_join(c[0], NULL);
    pthread_join(c[1], NULL);
    printf("consumed=%d sum=%ld\n", got[0] + got[1], total[0] + total[1]);
    return 0;
}
// 출력 (소비자별 몫은 매번 다르지만 합은 고정):
// consumed=10 sum=1530
타임아웃은 절대 시각으로 한 번만 계산 — 루프 안에서 매번 "지금 + 500 ms" 를 다시 계산하면 가짜 깨어남이 일어날 때마다 마감이 뒤로 밀려 영원히 기다릴 수 있습니다. 위 코드는 dl 을 루프 밖에서 고정합니다.

SQ 07

락 없는 큐 — SPSC 링의 acquire/release, 그리고 ISR ↔ 메인 루프

C 에는 async/await 도 채널도 없습니다. 대신 C 의 "비동기"는 두 가지입니다 — 다른 스레드, 그리고 인터럽트. 둘 다 생산자와 소비자가 정확히 하나씩이면 락 없이 큐를 만들 수 있습니다(SPSC: single-producer single-consumer). 조건은 SQ 03 의 무한 증가 인덱스(각자 자기 인덱스만 씀)와, 이 섹션의 주제인 메모리 순서입니다. volatile 만으로 되는 경우와 안 되는 경우를 정확히 가릅니다.

SPSC 링의 순서 규칙은 두 쌍뿐입니다 — head: release→acquire(데이터 공개), tail: release→acquire(칸 반납).

spsc.c — C11 stdatomic SPSC 링 (전체 코드, 스레드 둘로 검증)

// cc -std=c17 -O2 -pthread spsc.c
#include <stdatomic.h>
#include <pthread.h>
#include <stdint.h>
#include <stdbool.h>
#include <stdio.h>

#define SPSC_CAP  1024u
#define SPSC_MASK (SPSC_CAP - 1u)
_Static_assert((SPSC_CAP & SPSC_MASK) == 0, "SPSC_CAP: power of two");

typedef struct {
    // 서로 다른 코어가 쓰는 변수는 캐시 줄(64 B)을 나눠 둡니다
    _Alignas(64) _Atomic uint32_t head;   // 생산자만 씀
    _Alignas(64) _Atomic uint32_t tail;   // 소비자만 씀
    _Alignas(64) int buf[SPSC_CAP];
} Spsc;

// 생산자 스레드 하나에서만 호출
bool spsc_push(Spsc *q, int v) {
    uint32_t h = atomic_load_explicit(&q->head, memory_order_relaxed); // 내 것
    uint32_t t = atomic_load_explicit(&q->tail, memory_order_acquire); // ⑥
    if (h - t == SPSC_CAP) return false;           // 가득
    q->buf[h & SPSC_MASK] = v;                      // ①
    atomic_store_explicit(&q->head, h + 1, memory_order_release);     // ②
    return true;
}

// 소비자 스레드 하나에서만 호출
bool spsc_pop(Spsc *q, int *out) {
    uint32_t t = atomic_load_explicit(&q->tail, memory_order_relaxed); // 내 것
    uint32_t h = atomic_load_explicit(&q->head, memory_order_acquire); // ③
    if (h == t) return false;                        // 비었음
    *out = q->buf[t & SPSC_MASK];                    // ④
    atomic_store_explicit(&q->tail, t + 1, memory_order_release);     // ⑤
    return true;
}
// ───── 검증: 100만 개를 넘겨 합을 맞춰 봅니다 ─────
#define N 1000000

static Spsc q;   // 정적 → head = tail = 0 으로 초기화

static void *producer(void *arg) {
    (void)arg;
    for (int i = 1; i <= N; i++)
        while (!spsc_push(&q, i))
            ;                          // 가득이면 돌며 기다림 (실무는 yield/sleep)
    return NULL;
}

static void *consumer(void *arg) {
    long long *sum = arg;
    int got = 0, v;
    while (got < N)
        if (spsc_pop(&q, &v)) { *sum += v; got++; }
    return NULL;
}

int main(void) {
    long long sum = 0;
    pthread_t p, c;
    pthread_create(&c, NULL, consumer, &sum);
    pthread_create(&p, NULL, producer, NULL);
    pthread_join(p, NULL);
    pthread_join(c, NULL);
    printf("sum=%lld\n", sum);         // 1 + 2 + … + 1000000
    return 0;
}
// 출력:
// sum=500000500000
읽기·쓰기순서이유
자기 인덱스 읽기relaxed나만 쓰는 값 — 경쟁 없음
head 저장 ②release① 데이터 쓰기를 앞에 묶음
head 읽기 ③acquire④ 데이터 읽기를 뒤에 묶음
tail 저장 ⑤release④ 읽기가 끝난 뒤 칸 반납
tail 읽기 ⑥acquire반납된 칸에만 ① 쓰기
⑥ 의 acquire 를 relaxed 로 바꾸면 — "칸이 비었다"는 걸 보고 덮어쓰기 시작했는데, 소비자의 ④ 읽기가 아직 끝나지 않은 것처럼 재배치될 수 있습니다(읽기-후-쓰기 위험). 테스트에선 거의 안 나타나고 ARM 서버에서 드물게 터지는 종류라, 순서를 "필요할 때 낮추는" 최적화는 하지 마세요. 기본형(atomic_load/atomic_store = seq_cst)도 정답이고, 조금 느릴 뿐입니다.

ISR ↔ 메인 루프 — 락 없이 되는 조건과 volatile 의 한계 🔴 고급

락 없이 되는 네 가지 조건 — 하나라도 깨지면 인터럽트를 끄세요

  1. 생산자 하나 · 소비자 하나 — 같은 큐에 push 하는 ISR 이 둘이면(우선순위가 달라 서로 끼어들면) 같은 칸에 씁니다
  2. 각 인덱스는 한쪽만 씀 — head 는 ISR, tail 은 main. count 방식(SQ 03 ②)은 탈락
  3. 인덱스 읽기·쓰기가 명령 하나 — Cortex-M 의 정렬된 32비트는 OK. AVR 의 16비트는 두 명령 → 읽는 도중 ISR 이 바꾸면 반쪽짜리 값(찢어진 읽기)
  4. 데이터 쓰기 → 인덱스 갱신 순서가 지켜짐 — 컴파일러와 CPU 둘 다에게
환경④ 를 지키는 최소 도구
AVR (8비트, 캐시 없음)uint8_t 인덱스 + 컴파일러 배리어
Cortex-M0/M3/M4 (단일 코어)C11 atomic (위 코드) 또는 배리어
Cortex-M7 + D-캐시ISR↔main 은 위와 같음. DMA 버퍼는 캐시 정리 필요
듀얼 코어 (RP2040 등)반드시 C11 atomic (DMB 포함)

AVR 판 — uint8_t 인덱스 + 컴파일러 배리어

#include <avr/io.h>
#include <avr/interrupt.h>
#include <stdint.h>
#include <stdbool.h>

#define RX_N 64u           // 2의 거듭제곱 · uint8 무한 증가면 ≤ 128
_Static_assert((RX_N & (RX_N - 1u)) == 0 && RX_N <= 128u, "RX_N");
#define BARRIER() __asm__ __volatile__("" ::: "memory")

static uint8_t rx_buf[RX_N];
static volatile uint8_t rx_head;   // ISR 만 씀 — 8비트라 읽기가 명령 하나
static volatile uint8_t rx_tail;   // main 만 씀

ISR(USART_RX_vect) {               // 생산자: 이 ISR 하나뿐
    uint8_t b = UDR0;
    uint8_t h = rx_head;
    if ((uint8_t)(h - rx_tail) == RX_N) return;   // 가득 — 버림
    rx_buf[h & (RX_N - 1u)] = b;
    BARRIER();                     // ① 데이터가 ② head 보다 먼저
    rx_head = (uint8_t)(h + 1);
}

bool rx_get(uint8_t *out) {        // 소비자: main 루프 하나뿐
    uint8_t t = rx_tail;
    if (rx_head == t) return false;
    BARRIER();                     // head 를 본 뒤에 데이터 읽기
    *out = rx_buf[t & (RX_N - 1u)];
    BARRIER();                     // 다 읽은 뒤에 칸 반납
    rx_tail = (uint8_t)(t + 1);
    return true;
}
UART 레지스터 · 보레이트까지 포함한 완성형 드라이버는 🔷 AVR 탭 a07, STM32 의 순환 DMA 수신은 🟦 STM32 탭 m07, 인터럽트 자체의 규칙은 🔩 하드웨어 탭 h08 에 있습니다. 이 코드는 "왜 이 모양이어야 하는가"를 보여 주는 최소형입니다.
volatile 만 쓴 코드무엇이 깨지나고치는 법
인덱스만 volatile, buf 는 일반 배열컴파일러가 buf 쓰기를 head 갱신 뒤로 옮길 수 있음 — volatile 끼리만 순서가 보장됨배리어 또는 C11 release/acquire
AVR 에서 volatile uint16_t headmain 이 하위 바이트를 읽은 직후 ISR 이 head 를 0x00FF→0x0100 으로 바꾸면 0x01FF 를 읽음uint8_t 인덱스, 또는 ATOMIC_BLOCK 으로 읽기
volatile uint8_t count 를 양쪽에서 ++/--count++ = 읽기·더하기·쓰기 — 사이에 ISR 의 count-- 가 끼면 하나 유실count 를 없애고 무한 증가 인덱스
듀얼 코어에서 volatile 인덱스다른 코어가 쓰기 순서를 다르게 관찰 (CPU 재배치)C11 atomic — 하드웨어 배리어(DMB) 포함
정리 — 생산자·소비자가 하나씩이면 무한 증가 인덱스 + release/acquire 로 락 없는 큐가 됩니다. ISR ↔ main 에서는 인덱스가 명령 하나로 읽히는 폭인지(AVR 은 uint8_t), 데이터 → 인덱스 순서를 배리어로 지키는지 확인하세요 — volatile 은 그중 절반만 합니다.
SQ 08

알고리즘 패턴 — C 다운 코드로 쓰는 여섯 가지

코딩 테스트와 펌웨어 양쪽에서 같은 패턴이 반복됩니다. C 로 쓸 때의 요령은 하나입니다 — 최대 크기를 먼저 계산해서 고정 배열로. "각 원소가 한 번만 들어간다"를 증명할 수 있으면 원형 버퍼조차 필요 없고, 길이 n 짜리 배열에 인덱스 두 개면 됩니다. 아래 출력은 같은 알고리즘을 그대로 옮겨 실제로 돌린 값입니다.

① 괄호 짝 검사 — 문자 스택

#include <stdio.h>
#include <string.h>

// 짝이 맞으면 -1, 아니면 문제가 드러난 위치
static long check_brackets(const char *s) {
    char st[256];                         // 깊이 상한 — 넘으면 거절
    size_t top = 0, n = strlen(s);
    for (size_t i = 0; i < n; i++) {
        char c = s[i];
        if (c == '(' || c == '[' || c == '{') {
            if (top == sizeof st) return (long)i;
            st[top++] = c;
        } else if (c == ')' || c == ']' || c == '}') {
            char want = c == ')' ? '(' : c == ']' ? '[' : '{';
            if (top == 0 || st[--top] != want) return (long)i;
        }
    }
    return top ? (long)n : -1;            // 남은 여는 괄호
}

int main(void) {
    const char *t[] = { "{a[i] = (b + c)}", "f(x[1)]", "((a)", "x)" };
    for (int i = 0; i < 4; i++)
        printf("%-18s %ld\n", t[i], check_brackets(t[i]));
    return 0;
}
// 출력:
// {a[i] = (b + c)}   -1
// f(x[1)]            5
// ((a)               4
// x)                 1

② 단조 스택 — 오른쪽의 다음 큰 수

// 스택에는 '아직 답을 못 찾은' 인덱스가 값이 줄어드는 순서로 쌓입니다
static void next_greater(const int *a, int n, int *res) {
    int st[64], top = 0;                  // n ≤ 64 가정 (호출부에서 검사)
    for (int i = 0; i < n; i++) res[i] = -1;
    for (int i = 0; i < n; i++) {
        while (top > 0 && a[st[top - 1]] < a[i])
            res[st[--top]] = a[i];        // a[i] 가 그들의 '다음 큰 수'
        st[top++] = i;
    }
}

int main(void) {
    int a[] = { 2, 7, 3, 5, 4, 6, 8 }, res[7];
    next_greater(a, 7, res);
    for (int i = 0; i < 7; i++) printf("%d ", res[i]);
    printf("\n");
    return 0;
}
// 출력:
// 7 8 5 6 6 8 -1
왜 O(n) 인가 — 각 인덱스는 스택에 정확히 한 번 들어가고 최대 한 번 나옵니다. 안쪽 while 이 있어도 전체 반복은 2n 을 넘지 않습니다. 같은 이유로 스택 크기 상한은 n 입니다.
단조 덱은 앞에서 답을 읽고, 뒤에서 쓸모없는 것을 버립니다 — 각 인덱스가 한 번씩만 드나들어 O(n).

③ 슬라이딩 윈도우 최댓값 — 원형도 필요 없는 덱

// 각 인덱스는 덱에 한 번 들어가고 한 번 나옴
// → h, t 가 절대 n 을 넘지 않으니 길이 n 배열이면 충분
static void window_max(const int *a, int n, int k, int *out) {
    int dq[64], h = 0, t = 0, o = 0;      // n ≤ 64 가정
    for (int i = 0; i < n; i++) {
        if (h < t && dq[h] <= i - k) h++;            // 창 밖으로 나간 앞
        while (h < t && a[dq[t - 1]] <= a[i]) t--;   // 나보다 작은 뒤 버림
        dq[t++] = i;
        if (i >= k - 1) out[o++] = a[dq[h]];
    }
}

int main(void) {
    int a[] = { 1, 3, -1, -3, 5, 3, 6, 7 }, out[6];
    window_max(a, 8, 3, out);
    for (int i = 0; i < 6; i++) printf("%d ", out[i]);
    printf("\n");
    return 0;
}
// 출력:
// 3 3 5 5 6 7
펌웨어에서 — "최근 1초 동안의 최대 전류" 같은 피크 검출이 이 패턴입니다. 샘플이 끝없이 들어오면 n 이 무한이니 그때는 SQ 04 의 원형 덱(용량 = 창 크기 k 이상의 2의 거듭제곱)으로 바꿉니다.

④ BFS · ⑤ 0-1 BFS — 격자 최단 거리

#define ROWS 5
#define COLS 8
static const int DR[4] = { -1, 1, 0, 0 }, DC[4] = { 0, 0, -1, 1 };

// ④ 칸마다 큐에 한 번만 → 선형 배열 큐 (원형 불필요)
static int bfs(const char *const g[ROWS]) {
    int dist[ROWS * COLS], q[ROWS * COLS], head = 0, tail = 0, s = 0, goal = 0;
    for (int i = 0; i < ROWS * COLS; i++) {
        dist[i] = -1;
        if (g[i / COLS][i % COLS] == 'S') s = i;
        if (g[i / COLS][i % COLS] == 'G') goal = i;
    }
    dist[s] = 0; q[tail++] = s;
    while (head < tail) {
        int cur = q[head++], r = cur / COLS, c = cur % COLS;
        for (int d = 0; d < 4; d++) {
            int nr = r + DR[d], nc = c + DC[d];
            if (nr < 0 || nr >= ROWS || nc < 0 || nc >= COLS) continue;
            if (g[nr][nc] == '#' || dist[nr * COLS + nc] != -1) continue;
            dist[nr * COLS + nc] = dist[cur] + 1;
            q[tail++] = nr * COLS + nc;
        }
    }
    return dist[goal];
}

// ⑤ 벽(#)을 부수면 비용 1, 빈칸은 0 → 0 은 앞에, 1 은 뒤에
typedef struct { int a[128]; unsigned head, len; } IntDeque;  // 40칸 × 최대 2회 ≤ 128
static void pf(IntDeque *d, int v) { d->head = (d->head - 1u) & 127u; d->a[d->head] = v; d->len++; }
static void pb(IntDeque *d, int v) { d->a[(d->head + d->len) & 127u] = v; d->len++; }
static int  popf(IntDeque *d) { int v = d->a[d->head]; d->head = (d->head + 1u) & 127u; d->len--; return v; }

static int zero_one_bfs(const char *const g[ROWS]) {
    int dist[ROWS * COLS], s = 0, goal = 0;
    IntDeque dq = { .head = 0, .len = 0 };
    for (int i = 0; i < ROWS * COLS; i++) {
        dist[i] = 1 << 30;
        if (g[i / COLS][i % COLS] == 'S') s = i;
        if (g[i / COLS][i % COLS] == 'G') goal = i;
    }
    dist[s] = 0; pf(&dq, s);
    while (dq.len) {
        int cur = popf(&dq), r = cur / COLS, c = cur % COLS;
        for (int d = 0; d < 4; d++) {
            int nr = r + DR[d], nc = c + DC[d];
            if (nr < 0 || nr >= ROWS || nc < 0 || nc >= COLS) continue;
            int id = nr * COLS + nc, w = g[nr][nc] == '#';
            if (dist[cur] + w < dist[id]) {
                dist[id] = dist[cur] + w;
                if (w) pb(&dq, id); else pf(&dq, id);
            }
        }
    }
    return dist[goal];
}

int main(void) {
    const char *const g1[ROWS] = { "S.#.....", ".##.###.", "....#...",
                                   ".#.##.#.", ".#....#G" };
    const char *const g2[ROWS] = { "S..#....", ".#.#.##.", ".#.#..#.",
                                   "...#.#..", ".#.#...G" };
    printf("BFS 최단 거리: %d\n", bfs(g1));
    printf("0-1 BFS 부술 벽: %d\n", zero_one_bfs(g2));
    return 0;
}
// 출력:
// BFS 최단 거리: 15
// 0-1 BFS 부술 벽: 1

⑥ undo / redo — 오래된 기록을 버리는 링 스택 🟡 중급

크기가 정해진 undo — 스택인데 바닥이 움직입니다

#include <stdio.h>
#include <stdint.h>
#include <stdbool.h>

#define HIST 4u                       // 2의 거듭제곱 — 넘치면 가장 오래된 것 버림
typedef struct { int v[HIST]; uint32_t top, bottom; } RingStack;

static void rs_push(RingStack *s, int x) {
    s->v[s->top & (HIST - 1u)] = x;
    s->top++;
    if (s->top - s->bottom > HIST) s->bottom++;   // 바닥이 한 칸 올라감
}
static bool rs_pop(RingStack *s, int *out) {
    if (s->top == s->bottom) return false;
    s->top--;
    *out = s->v[s->top & (HIST - 1u)];
    return true;
}

static RingStack undo_s, redo_s;
static int value = 0;

static void set_value(int v) {
    rs_push(&undo_s, value);
    redo_s.top = redo_s.bottom;       // ★ 새 동작은 redo 를 비웁니다
    value = v;
}
static bool undo(void) {
    int p;
    if (!rs_pop(&undo_s, &p)) return false;
    rs_push(&redo_s, value);
    value = p;
    return true;
}
static bool redo(void) {
    int n;
    if (!rs_pop(&redo_s, &n)) return false;
    rs_push(&undo_s, value);
    value = n;
    return true;
}

int main(void) {
    for (int v = 10; v <= 50; v += 10) set_value(v);
    printf("set x5 -> %d\n", value);
    int k = 0;
    while (undo()) k++;
    printf("undo %d회 -> %d\n", k, value);
    redo(); redo();
    printf("redo x2 -> %d\n", value);
    set_value(99);
    bool r = redo();
    printf("set 99 -> %d, redo=%d\n", value, r);
    return 0;
}
// 출력:
// set x5 -> 50
// undo 4회 -> 10        ← 기록이 4칸이라 처음 값 0 까지는 못 감
// redo x2 -> 30

SQ 09

성능과 메모리 — malloc 없는 정적 풀과 바이트 단위 예산

C 의 큐 성능은 대부분 연산이 아니라 메모리에서 갈립니다 — 원소가 연속해 있나, 노드마다 할당하나, 크기를 몇 바이트로 잡았나. 임베디드에서는 한발 더 나가 RAM 이 바이트 단위 예산입니다. 이 섹션은 원소당 실제 바이트를 계산하고, 큐·풀의 크기를 컴파일 타임에 검사하는 방법을 정리합니다.

원소가 작을수록 포인터와 malloc 머리가 본체보다 커집니다 — int 4 바이트에 32 바이트를 쓰는 것이 노드 큐입니다.

임베디드 — 정적 풀과 크기 예산 🟡 중급

큐 크기는 "넉넉히"가 아니라 예산입니다 — 합계와 비율을 표로 들고 있어야 줄일 곳이 보입니다.

정적 풀 — 빈 칸 목록 자체가 스택

#include <stdint.h>
#include <stddef.h>

#define POOL_N 16u
typedef struct { uint8_t data[60]; uint16_t len; } Packet;   // 62 B

static Packet  pool[POOL_N];                 // .bss — 부팅 때 0
static uint8_t free_stk[POOL_N];             // 빈 칸 번호들의 스택
static uint8_t free_top;

void pool_init(void) {
    for (uint8_t i = 0; i < POOL_N; i++)
        free_stk[i] = (uint8_t)(POOL_N - 1u - i);   // 0 번이 맨 위에 오도록
    free_top = POOL_N;
}
Packet *pool_alloc(void) {                   // O(1), 단편화 없음
    return free_top ? &pool[free_stk[--free_top]] : NULL;
}
void pool_free(Packet *p) {
    free_stk[free_top++] = (uint8_t)(p - pool);     // 포인터 → 칸 번호
}

// 큐에는 62 B 사본 대신 1 B 칸 번호만 넣습니다
//   생산자: Packet *p = pool_alloc(); 채움; q_push(&q, idx)
//   소비자: q_pop(&q, &idx); 처리; pool_free(&pool[idx])
// 메모리: 풀 16×62 = 992 B + 빈칸 스택 16 B + 인덱스 큐 16 B
ISR 이 alloc, main 이 free 하면 free_top 을 양쪽이 고칩니다 — SQ 07 의 count 와 같은 문제입니다. 이때는 빈 칸 목록도 SPSC 링으로 만들어 "빈 칸 링(main→ISR) + 찬 칸 링(ISR→main)" 두 개로 순환시킵니다. FreeRTOS 탭 r05 의 free/full 큐 두 개가 같은 모양입니다.

컴파일 타임에 못 박기 — _Static_assert

#include <stdint.h>
// (POOL_N · Packet 은 왼쪽 카드의 정의를 그대로 씁니다)

#define IS_POW2(x)  ((x) != 0 && ((x) & ((x) - 1)) == 0)

C 버전별 — C11 은 _Static_assert(식, "메시지") 키워드 + <assert.h> 의 static_assert 매크로, C23 은 static_assert 가 키워드가 되고 메시지를 생략할 수 있습니다. 구조체 레이아웃 검사는 🅲 C 기초 탭 c06 의 패딩 이야기와 짝입니다.
필드 순서로 큐를 줄이기 — Cortex-M 에서 {uint8_t type; uint32_t ts; uint16_t len;} 는 패딩 때문에 12 B, {uint32_t ts; uint16_t len; uint8_t type;} 는 8 B 입니다. 32 칸 큐면 384 B → 256 B. (AVR 은 정렬이 1 이라 둘 다 7 B)
정리 — 원소가 작을수록 노드 큐는 손해가 커서(int 하나에 32 B) 배열 링이 기본입니다. 임베디드에서는 정적 풀 + 1 바이트 인덱스 큐로 큰 데이터를 옮기고, 2의 거듭제곱·인덱스 폭·RAM 예산을 _Static_assert 로 못 박습니다.

}
C++ 의 pop() 이 값을 돌려주지 않는 이유는 예외 안전성 때문입니다(복사 중 예외가 나면 원소를 잃음). C 에는 예외가 없어 pop(s, &out) 한 번에 해도 됩니다.

자바 · 파이썬 개발자가 C 에서 놀라는 지점

  • 용량을 정해야 한다 — "가득 차면 어떻게 하나"(거절 · 덮어쓰기 · 대기)를 설계 단계에서 고릅니다
  • 빈 pop 이 예외가 아니다 — 검사하지 않으면 쓰레기 값이 정상처럼 돌아옵니다
  • 제네릭이 없다 — void* · 매크로 · _Generic 중 하나를 고릅니다 (SQ 02)
  • volatile 의 뜻이 다르다 — 자바의 volatile(가시성 + 순서)에 해당하는 것은 C 의 _Atomic 입니다. C 의 volatile 은 하드웨어 레지스터용
  • GC 가 없다 — 큐에 포인터를 넣으면 "누가 해제하나"를 큐 API 문서에 적습니다(소유권 이전)
같은 이름, 같은 내부 — 자바 ArrayDeque · 러스트 VecDeque 는 SQ 04 의 덱과 거의 같은 코드입니다 (2의 거듭제곱 용량 · 마스크 · 자랄 때 두 조각 복사). 반대로 C++ std::deque 와 파이썬 deque 는 고정 크기 블록을 이어 붙인 구조라 원소 주소가 성장 중에도 유지됩니다.
정리 — C 의 링·힙·list_head 는 다른 언어의 ArrayDeque · PriorityQueue · Boost.Intrusive 에 해당합니다. 옮길 때는 빈 pop 동작과 힙 방향(C++·Rust 최대, Java·Python 최소)부터 확인하세요.
SQ 12

실무 사례 — 이벤트 큐 · 명령 큐 · 로그 링 · 재귀 없는 탐색

펌웨어의 메인 루프는 거의 언제나 같은 모양입니다 — 인터럽트가 큐에 넣고, 메인 루프가 꺼내 처리합니다. 그 위에 상태 머신이 얹히고, 사람이 보내는 명령은 줄 단위로 파싱되어 명령 큐로, 무슨 일이 있었는지는 로그 링으로 갑니다. 네 가지 모두 이 탭의 링 버퍼 하나로 만들어지며, 출력은 같은 로직을 그대로 옮겨 돌린 값입니다.

ISR 은 무슨 일이 있었는지만 큐에 넣고, 무엇을 할지는 상태 머신이 한 곳에서 정합니다.

① 이벤트 큐 + 상태 머신 (PC 에서 시뮬레이션)

#include <stdio.h>
#include <stdint.h>
#include <stdbool.h>

typedef enum { EV_BTN, EV_TICK } EvType;
typedef struct { uint8_t type, arg; } Event;      // 2 바이트

#define EQ_N 16u
static Event    eq[EQ_N];
static uint32_t eq_h, eq_t;  // 실기에서는 SQ 07 의 atomic · 배리어 판으로

static bool ev_post(Event e) {                    // ISR 쪽 (생산자 하나)
    if (eq_h - eq_t == EQ_N) return false;        // 가득 — 버리고 알림
    eq[eq_h & (EQ_N - 1u)] = e;
    eq_h++;
    return true;
}
static bool ev_get(Event *e) {                    // main 쪽 (소비자 하나)
    if (eq_h == eq_t) return false;
    *e = eq[eq_t & (EQ_N - 1u)];
    eq_t++;
    return true;
}

typedef enum { ST_IDLE, ST_RUN, ST_PAUSE, ST_DONE } State;
static const char *const ST_NAME[] = { "IDLE", "RUN", "PAUSE", "DONE" };
static State st = ST_IDLE;
static int ticks;

static void dispatch(Event e) {
    State prev = st;
    switch (st) {
    case ST_IDLE:  if (e.type == EV_BTN) { ticks = 0; st = ST_RUN; } break;
    case ST_RUN:   if (e.type == EV_BTN) st = ST_PAUSE;
                   else if (++ticks == 3) st = ST_DONE;
                   break;
    case ST_PAUSE: if (e.type == EV_BTN) st = ST_RUN; break;   // TICK 무시
    case ST_DONE:  if (e.type == EV_BTN) st = ST_IDLE; break;
    }
    if (st != prev)
        printf("%-5s --%s--> %s (ticks=%d)\n", ST_NAME[prev],
               e.type == EV_BTN ? "BTN" : "TICK", ST_NAME[st], ticks);
}

int main(void) {
    const uint8_t script[] = { EV_BTN, EV_TICK, EV_TICK, EV_BTN,
                               EV_TICK, EV_BTN, EV_TICK, EV_BTN };
    for (size_t i = 0; i < sizeof script; i++)    // ISR 들이 넣었다고 가정
        ev_post((Event){ script[i], 0 });
    Event e;
    while (ev_get(&e)) dispatch(e);              // 메인 루프 한 바퀴
    return 0;
}
// 출력:
// IDLE  --BTN--> RUN (ticks=0)
// RUN   --BTN--> PAUSE (ticks=2)
// PAUSE --BTN--> RUN (ticks=2)
// RUN   --TICK--> DONE (ticks=3)
// DONE  --BTN--> IDLE (ticks=3)

② 명령 파서 — 바이트 → 줄 → 명령 큐 → 실행

#include <stdio.h>
#include <string.h>
#include <stdint.h>
#include <inttypes.h>
#include <stdbool.h>

enum { CMD_LED, CMD_PWM, CMD_RESET };
typedef struct { uint8_t op; int16_t arg; } Cmd;

#define CQ_N 4u
static Cmd cq[CQ_N];
static uint32_t cq_h, cq_t, cq_drop;

static void cq_push(Cmd c) {
    if (cq_h - cq_t == CQ_N) { cq_drop++; return; }   // 넘침은 세어 둠
    cq[cq_h++ & (CQ_N - 1u)] = c;
}
static bool cq_pop(Cmd *c) {
    if (cq_h == cq_t) return false;
    *c = cq[cq_t++ & (CQ_N - 1u)];
    return true;
}

static void parse_line(const char *s) {
    char name[8];
    int arg = 0;
    int n = sscanf(s, "%7s %d", name, &arg);
    if (n < 1) return;                                 // 빈 줄
    if (!strcmp(name, "LED") && n == 2 && (arg == 0 || arg == 1))
        cq_push((Cmd){ CMD_LED, (int16_t)arg });
    else if (!strcmp(name, "PWM") && n == 2 && arg >= 0 && arg <= 100)
        cq_push((Cmd){ CMD_PWM, (int16_t)arg });
    else if (!strcmp(name, "RESET") && n == 1)
        cq_push((Cmd){ CMD_RESET, 0 });
    else
        printf("ERR %s\n", s);                         // 검증은 넣기 전에
}

// UART 링(SQ 07)에서 꺼낸 바이트를 하나씩 먹임
static void feed(char c) {
    static char line[32];
    static size_t len;
    if (c == '\r') return;
    if (c == '\n') { line[len] = '\0'; parse_line(line); len = 0; return; }
    if (len < sizeof line - 1) line[len++] = c;       // 긴 줄은 잘라 냄
}

static void execute_all(void) {
    Cmd c;
    while (cq_pop(&c)) {
        switch (c.op) {
        case CMD_LED:   printf("LED %s\n", c.arg ? "on" : "off"); break;
        case CMD_PWM:   printf("PWM %d%%\n", c.arg); break;
        case CMD_RESET: printf("RESET\n"); break;
        }
    }
}

int main(void) {
    const char *rx = "LED 1\r\nPWM 75\nFOO 3\nPWM 300\nRESET\n";
    for (const char *p = rx; *p; p++) feed(*p);
    execute_all();
    printf("dropped=%" PRIu32 "\n", cq_drop);
    return 0;
}
// 출력:
// ERR FOO 3
// ERR PWM 300
// LED on
// PWM 75%
// RESET
// dropped=0

③ 로그 링 — 가득 차면 가장 오래된 것을 덮음

#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>

#define LOG_N 4u                     // 실기에서는 32~256
typedef struct { uint32_t t; uint16_t code, arg; } LogRec;
_Static_assert(sizeof(LogRec) == 8, "LogRec");

static LogRec   logs[LOG_N];
static uint32_t log_n;               // 지금까지 쓴 총 개수

void log_put(uint32_t t, uint16_t code, uint16_t arg) {
    // 거절하지 않습니다 — 최근 것이 더 중요하므로 덮어씀
    logs[log_n & (LOG_N - 1u)] = (LogRec){ t, code, arg };
    log_n++;
}
void log_dump(void) {                // 오래된 것 → 최근 것
    uint32_t n = log_n < LOG_N ? log_n : LOG_N;
    printf("최근 %" PRIu32 "건 (덮어쓴 것 %" PRIu32 "건)\n", n, log_n - n);
    for (uint32_t i = log_n - n; i != log_n; i++) {   // != 라서 넘침에도 안전
        const LogRec *r = &logs[i & (LOG_N - 1u)];
        printf("  t=%" PRIu32 " code=%u arg=%u\n", r->t, r->code, r->arg);
    }
}

int main(void) {
    for (uint16_t k = 1; k <= 6; k++)
        log_put(k * 100u, k, (uint16_t)(k * k));
    log_dump();
    return 0;
}
// 출력:
// 최근 4건 (덮어쓴 것 2건)
//   t=300 code=3 arg=9
//   t=400 code=4 arg=16
//   t=500 code=5 arg=25
//   t=600 code=6 arg=36
리셋 뒤에도 남기기 — 로그 링을 링커 스크립트의 초기화하지 않는 영역(GCC 관례 .noinit)에 두면 워치독 리셋 후에도 RAM 에 남아, 부팅 직후 "죽기 직전 무엇을 했나"를 꺼내 볼 수 있습니다. 섹션 이름과 지원 여부는 툴체인 · 링커 스크립트마다 다르니 맵 파일로 확인하세요.

④ 재귀 없는 flood fill — 크기가 증명된 명시적 스택

#include <stdio.h>
#include <stdint.h>

#define W 16
#define H 16
static uint8_t grid[H][W];        // 0 빈칸 · 1 벽 · 2 칠함
static uint8_t stk[W * H];        // ★ 칸마다 최대 한 번 push → 256 이면 못 넘침
_Static_assert(W <= 16 && H <= 16, "row/col packed into one uint8_t");

static const int8_t DR[4] = { -1, 1, 0, 0 }, DC[4] = { 0, 0, -1, 1 };

static int flood(int r0, int c0, int *max_depth) {
    int top = 0, filled = 0, maxd = 0;
    if (grid[r0][c0] != 0) return 0;
    grid[r0][c0] = 2;                          // push 할 때 표시 — 중복 방지
    stk[top++] = (uint8_t)(r0 << 4 | c0);
    while (top > 0) {
        uint8_t p = stk[--top];
        int r = p >> 4, c = p & 15;
        filled++;
        for (int d = 0; d < 4; d++) {
            int nr = r + DR[d], nc = c + DC[d];
            if (nr < 0 || nr >= H || nc < 0 || nc >= W) continue;
            if (grid[nr][nc] != 0) continue;
            grid[nr][nc] = 2;
            stk[top++] = (uint8_t)(nr << 4 | nc);
            if (top > maxd) maxd = top;
        }
    }
    *max_depth = maxd;
    return filled;
}

int main(void) {
    int maxd;
    int n = flood(0, 0, &maxd);           // 벽 없는 16×16 — 최악에 가까운 경우
    printf("filled=%d max_stack=%d B\n", n, maxd);
    return 0;
}
// 출력:
// filled=256 max_stack=121 B
// (같은 칸 순서의 재귀판은 깊이 256 — 프레임마다 수~수십 B)
상한의 증명 — 칸을 push 할 때 표시하므로 한 칸은 평생 한 번만 스택에 들어갑니다. 그래서 스택 크기 = 칸 수(256 B)면 어떤 입력에도 넘치지 않습니다. 재귀판은 이 보장이 호출 스택으로 넘어가 SQ 10 그림처럼 조용히 RAM 을 덮습니다.

이벤트 큐에 ISR 이 여럿이면 — 생산자가 둘 이상이 되어 SPSC 가 깨집니다. 선택지는 셋입니다: ① ISR 마다 큐를 따로 두고 main 이 차례로 비우기(가장 단순, 권장), ② push 하는 동안만 인터럽트를 잠깐 끄기(__disable_irq() / cli(), 수 마이크로초 이내), ③ RTOS 를 쓴다면 xQueueSendFromISR — 이 경우는 ⏱️ FreeRTOS 탭 r05 의 큐를 그대로 씁니다.
정리 — 펌웨어의 뼈대는 ISR → 링 → 메인 루프 → 상태 머신입니다. 사례마다 다른 것은 단 하나, 가득 찼을 때의 정책(버림 · 거절 · 덮어쓰기 · 대기)이고, 그 결정을 코드 주석과 API 이름에 드러내는 것이 C 다운 설계입니다.
