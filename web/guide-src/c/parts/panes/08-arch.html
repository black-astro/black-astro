C 에는 패키지도, 클래스도, 접근 제어자도, 의존성 주입 컨테이너도 없습니다. 그래서 코드가 1만 줄을 넘고 사람이 셋을 넘고 보드가 두 종류가 되는 순간, 자바 · 파이썬에서는 언어와 프레임워크가 막아 주던 사고가 그대로 터집니다 — 전역 변수 그물, 헤더 하나 고치면 전부 다시 빌드, #ifdef 미로, 순환 의존. 이 탭은 그 빈자리를 규칙과 빌드 도구로 메우는 법을 다룹니다: 불투명 포인터로 만드는 모듈, 헤더 규칙, 레이어와 의존 방향, 여러 보드 · 여러 제품을 담는 디렉터리, CMake 타깃으로 의존을 강제하는 법, vtable 로 만드는 인터페이스와 테스트 대역, 에러 · 메모리 · 이벤트 규약, MISRA 와 정적 분석, 그리고 리눅스 커널 · Zephyr · ESP-IDF · SQLite · Redis 가 실제로 쓰는 구조까지. 이 탭을 마치면 "팀이 함께 오래 고칠 수 있는" C 프로젝트의 골격을 처음부터 세우고, 무너진 프로젝트를 어디서부터 고칠지 판단할 수 있습니다.

ARCH 01

C 프로젝트가 커지면 무엇이 무너지나 — 규모별 증상

C 프로젝트는 갑자기 망가지지 않습니다. 줄 수 · 사람 수 · 보드 수가 하나씩 늘 때마다 정해진 순서로 금이 갑니다. 증상을 먼저 알아 두면 "지금 우리 프로젝트가 어디쯤인지", 그리고 이 탭의 어느 섹션이 그 처방인지가 보입니다. 헤더/소스 분리 · static · Makefile 기초는 C 기초 탭 c10, 라이브러리 배포는 C 실전 탭 c23, 기본 디렉터리 · CMake 표준형은 c25 에 있습니다 — 여기서는 그 다음 이야기입니다.

규모별 증상 — 전역 → 헤더 → #ifdef → 순환 순서로 무너집니다. 이 순서가 곧 이 탭의 목차입니다.

증상 1 — 전역 변수 그물

❌ 누가 언제 바꾸는지 아무도 모름
/* globals.h — 1만 줄 프로젝트에서 흔히 보는 모습 */
extern volatile uint8_t  g_rx_buf[256];
extern volatile uint16_t g_rx_len;
extern int32_t           g_temp_mc;     /* 센서 모듈이 씀 */
extern uint8_t           g_mode;        /* 7개 파일이 씀 */
extern bool              g_wifi_ok;
extern uint32_t          g_err_flags;   /* 아무나 비트를 켬 */

/* display.c */
if (g_mode == 3 && g_wifi_ok) { ... }   /* 3 이 뭐였더라? */

/* protocol.c — 화면 코드가 모르는 사이에 */
g_mode = 3;
g_rx_len = 0;          /* ISR 이 쓰는 중이었다면? */
왜 치명적인가 — 전역 변수 하나는 "그 변수를 건드리는 모든 파일"을 서로 묶습니다. g_mode 를 7개 파일이 쓰면, 버그 하나를 찾으려고 7개 파일을 다 읽어야 합니다. 서버 개발에 빗대면 모든 클래스가 public static 필드 하나를 직접 고치는 상태입니다. 처방은 k02 — 상태를 모듈 안에 static 으로 가두고 함수로만 접근하게 합니다.

/* common.h — 모든 .c 가 이것 하나만 include 합니다 */

만능 헤더는 처음엔 편합니다. 하지만 display.h 에 매크로 하나만 추가해도 그 헤더를 볼 일도 없는 파일 400개가 다시 컴파일됩니다(아래 그림). #ifdef 미로는 조합 수만큼 다른 프로그램이 숨어 있다는 뜻이라 — 조건 4개면 16가지 — 한 번도 빌드되지 않은 조합이 반드시 생깁니다. 처방은 k03(헤더 규칙) 과 k07(설정 관리).
재빌드 범위 — 헤더를 좁게 쪼갤수록 고칠 때 다시 컴파일되는 범위가 줄어듭니다.

ARCH 02

모듈 = 공개 헤더 + 비공개 소스 — 불투명 포인터로 정보 은닉

C 의 "클래스"는 .h 한 장 + .c 한 장입니다. 헤더는 계약서, 소스는 구현이고, 구현 안의 상태는 파일 범위 static 으로 가둡니다. 한 걸음 더 나가면 구조체의 내용까지 감출 수 있습니다 — 불투명 포인터(opaque pointer). 필드를 못 보면 못 고치고, 못 고치면 모듈 밖에서 깨뜨릴 수 없습니다. static 기초는 C 기초 탭 c10 에 있습니다.

불투명 포인터 — 주소는 넘기되 모양은 숨깁니다. 필드를 바꿔도 ring.c 만 다시 컴파일됩니다.

include/ring.h — 계약서 (전부 공개)

#ifndef RING_H
#define RING_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

#ifdef __cplusplus
extern "C" {
#endif

/* 불완전 타입 — 내용은 ring.c 만 압니다 */
typedef struct ring ring_t;

/* capacity 는 2의 거듭제곱(2^31 이하). 실패하면 NULL */
ring_t *ring_create(size_t capacity);
void    ring_destroy(ring_t *r);          /* NULL 이면 아무것도 안 함 */

bool    ring_push(ring_t *r, uint8_t byte);   /* 가득 차면 false */
bool    ring_pop(ring_t *r, uint8_t *out);    /* 비었으면 false */
size_t  ring_count(const ring_t *r);

#ifdef __cplusplus
}
#endif
#endif /* RING_H */
서버 개발에 빗대면 — 자바의 interface + private 필드를 합친 것입니다. ring_t 를 받은 쪽은 함수로만 다룰 수 있습니다. 필드 순서를 바꾸든 head 를 uint16_t 로 줄이든 ring.h 가 그대로면 다른 파일은 다시 컴파일조차 되지 않습니다 — 라이브러리라면 ABI 도 유지됩니다(c23).

src/ring.c — 구현 (상태는 전부 여기)

#include "ring.h"      /* ★ 자기 헤더를 맨 먼저 (k03) */

#include <stdlib.h>

struct ring {
    uint8_t *buf;
    size_t   mask;      /* capacity - 1 */
    uint32_t head;      /* 지금까지 넣은 총 개수 — 넘쳐도 됨 */
    uint32_t tail;      /* 지금까지 꺼낸 총 개수 */
};

static bool is_pow2(size_t n) { return (n != 0u) && ((n & (n - 1u)) == 0u); }

ring_t *ring_create(size_t capacity) {
    if (!is_pow2(capacity) || capacity > 0x80000000u) { return NULL; }
    ring_t *r = malloc(sizeof *r);
    if (r == NULL) { return NULL; }
    r->buf = malloc(capacity);
    if (r->buf == NULL) { free(r); return NULL; }
    r->mask = capacity - 1u;
    r->head = 0u;
    r->tail = 0u;
    return r;
}

void ring_destroy(ring_t *r) {
    if (r != NULL) { free(r->buf); free(r); }
}

size_t ring_count(const ring_t *r) {
    return (size_t)(uint32_t)(r->head - r->tail);  /* 넘침이 있어도 차이는 정확 */
}

bool ring_push(ring_t *r, uint8_t byte) {
    if (ring_count(r) > r->mask) { return false; }  /* count == capacity */
    r->buf[r->head & r->mask] = byte;
    r->head++;
    return true;
}

bool ring_pop(ring_t *r, uint8_t *out) {
    if (r->head == r->tail) { return false; }
    *out = r->buf[r->tail & r->mask];
    r->tail++;
    return true;
}
head · tail 을 계속 증가시키는 이유 — 부호 없는 정수 뺄셈은 232 로 나눈 나머지라서 head 가 0xFFFFFFFF → 0 으로 넘쳐도 head - tail 은 정확한 개수입니다. 덕분에 "가득"과 "빔"을 구분하려고 칸 하나를 버릴 필요가 없습니다. (이 로직은 node 로 옮겨 0xFFFFFFFE 시작 넘침까지 확인했습니다.) ISR 과 메인이 나눠 쓸 때의 volatile · 순서 보장은 통신 · 드라이버 탭에서 다룹니다.

펌웨어에서는 — malloc 없이 불투명 타입 만들기 🟡 중급

불투명 포인터의 약점은 호출자가 크기를 모르니 모듈이 메모리를 대신 잡아야 한다는 것입니다. PC 라면 malloc 이면 끝이지만, 펌웨어는 대개 "초기화 이후 malloc 금지"(k10) 입니다. 실무에서 쓰는 세 가지 방법이 있습니다.

① 모듈 안 정적 풀 — 가장 흔한 방법

/* ring.h 에 한 줄 추가 */
ring_t *ring_create_static(uint8_t *storage, size_t capacity);

/* ring.c */
#ifndef RING_MAX_INSTANCES
#define RING_MAX_INSTANCES 4u        /* config.h 에서 덮어씀 (k07) */
#endif

static struct ring s_pool[RING_MAX_INSTANCES];
static size_t      s_pool_used;      /* 초기화 단계에서만 증가 */

ring_t *ring_create_static(uint8_t *storage, size_t capacity) {
    if ((storage == NULL) || !is_pow2(capacity) ||
        (s_pool_used >= RING_MAX_INSTANCES)) {
        return NULL;
    }
    ring_t *r = &s_pool[s_pool_used];
    s_pool_used++;
    r->buf  = storage;
    r->mask = capacity - 1u;
    r->head = 0u;
    r->tail = 0u;
    return r;
}

/* app.c — 버퍼는 호출자가 정적으로 */
static uint8_t s_rx_mem[256];
static ring_t *s_rx;

void app_init(void) {
    s_rx = ring_create_static(s_rx_mem, sizeof s_rx_mem);
    FW_ASSERT(s_rx != NULL);         /* 개수 부족은 개발 중에 터지게 (k09) */
}
초기화 때만 부르는 함수입니다 — 해제가 없고 s_pool_used++ 가 원자적이지 않아서 태스크 · ISR 에서 부르면 안 됩니다. RAM 사용량이 링크 시점에 확정되어 arm-none-eabi-size 로 보이는 것이 큰 장점입니다.

② 크기만 공개하는 저장소 블롭 · ③ 정직하게 공개

/* ② ring.h — 크기와 정렬만 알려 주고 내용은 비공개 */
typedef struct {
    uintptr_t priv_[4];        /* 32비트 16 B · 64비트 32 B */
} ring_storage_t;

ring_t *ring_init(ring_storage_t *mem, uint8_t *buf, size_t capacity);

/* ring.c — 크기가 모자라면 컴파일이 멈춤 */
_Static_assert(sizeof(struct ring) <= sizeof(ring_storage_t),
               "ring_storage_t 를 키우세요");
_Static_assert(_Alignof(struct ring) <= _Alignof(ring_storage_t),
               "ring_storage_t 정렬 부족");

/* app.c — 스택에도, 전역에도 둘 수 있음 */
static ring_storage_t s_rx_obj;
static uint8_t        s_rx_mem[256];
static ring_t        *s_rx;

void app_init(void) {
    s_rx = ring_init(&s_rx_obj, s_rx_mem, sizeof s_rx_mem);
}
방법장점단점 · 쓰는 곳
① 정적 풀완전한 은닉, 단순개수 상한을 미리 정해야 함
② 블롭개수 제한 없음, 호출자가 배치크기 상수 관리 · FreeRTOS StaticQueue_t 가 이 방식
③ 헤더에 공개가장 단순, 인라인 가능"건드리지 마" 주석에 의존 · HAL 핸들(UART_HandleTypeDef)
②의 주의점 — 엄밀히는 ring_storage_t 로 선언한 메모리를 struct ring 으로 읽는 것이 엄격한 별칭 규칙의 회색 지대입니다. FreeRTOS 도 같은 방식을 쓰고 configASSERT(sizeof) 로 크기를 확인합니다. 팀 규칙으로 정하고 static assert 두 줄을 빼먹지 않는 것이 핵심입니다.

헤더 규칙 — 자기 완결 · 최소 포함 · 전방 선언으로 의존 끊기

C 에서 "의존"은 곧 #include 입니다. 헤더가 헤더를 끌고 오면 의존이 전이되어, 한 파일을 고칠 때 수백 파일이 흔들립니다. 규칙은 다섯 줄이면 됩니다 — 헤더는 혼자 컴파일되게, 필요한 것만 포함, 포인터만 쓰면 전방 선언, 헤더에 정의 금지, 순환 금지. 그리고 다섯 줄 모두 도구로 검사할 수 있습니다.

include 그래프 — 헤더끼리는 고리가 없어야 하고, 무거운 포함은 .c 로 내립니다.

규칙 1 — 자기 완결 헤더 (self-contained)

❌ 누가 먼저 stdint.h 를 포함해 줘야 동작

#define SENSOR_H
int32_t sensor_read(void);     /* int32_t 는 어디서? */
#endif
✅ 필요한 것을 스스로 포함
/* sensor.h */
#ifndef SENSOR_H
#define SENSOR_H
#include <stdint.h>

#endif
/* sensor.c — ★ 자기 헤더를 맨 첫 줄에.
   빠진 include 가 있으면 여기서 바로 에러가 납니다 */

#include <string.h>
#include "drv/i2c.h"

static uart_slot_t s_slot[UART_ID_COUNT];

void uart_set_rx_cb(uart_id_t id, uart_rx_cb_t cb, void *ctx) {

    s_slot[id].ctx = ctx;
}

/* HAL 의 weak 함수를 드라이버가 소유 — main.c 에 두지 않습니다 */
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart) {
    uart_id_t id = bsp_uart_id(huart);
    if (id < UART_ID_COUNT) {
        uart_slot_t *s = &s_slot[id];
        if (s->cb != NULL) { s->cb(s->ctx, s->byte); }
        (void)HAL_UART_Receive_IT(huart, &s->byte, 1u);   /* 다음 바이트 */
    }
}

빌드 구조 — CMake 타깃 = 모듈, 의존 규칙을 컴파일 에러로

k02~k05 의 규칙(내부 헤더 금지 · 레이어 방향 · board.h 격리)은 문서로 두면 3개월 안에 깨집니다. CMake 의 타깃마다 PUBLIC / PRIVATE 를 정확히 적으면 규칙을 어긴 #include 가 파일을 못 찾아 컴파일 에러가 됩니다. 여기에 툴체인 파일 · 프리셋 · 보드 선택 변수 · 링커 스크립트를 더하면, 같은 소스로 PC 테스트 · 개발 보드 · 양산 보드를 명령 한 줄씩으로 빌드합니다. CMake 기본 문법(add_library · target_link_libraries)은 C 실전 탭 c25 에 있습니다.

타깃 의존 그래프 — PRIVATE 가 많을수록 바깥으로 새는 헤더가 줄고, 재빌드 범위도 줄어듭니다.

인터페이스와 의존성 역전 — 함수 포인터 구조체(vtable)와 테스트 대역

k04 의 콜백은 "함수 하나"를 넘겼습니다. 함수 여러 개를 표 하나로 묶어 넘기면 그게 C 의 인터페이스입니다 — C++ 의 가상 함수 표(vtable)를 손으로 만드는 것입니다. 리눅스 커널의 struct file_operations, Zephyr 의 드라이버 API 가 전부 이 모양입니다. 이것으로 앱이 구체 드라이버 대신 인터페이스에 의존하게 만들면(의존성 역전), 하드웨어 없이 PC 에서 앱 로직을 테스트할 수 있습니다.

vtable 메모리 — 인스턴스는 RAM, 표는 const 라서 Flash. 인스턴스가 100개여도 표는 하나입니다.

#define DRV_TEMP_SENSOR_H

typedef struct temp_sensor temp_sensor_t;

typedef struct {
    fw_err_t (*init)(temp_sensor_t *self);
    fw_err_t (*read_mc)(temp_sensor_t *self, int32_t *out_milli_c);
    void     (*sleep)(temp_sensor_t *self);
} temp_sensor_ops_t;

    const temp_sensor_ops_t *ops;   /* ★ 반드시 첫 멤버 */
};

}

    return s->ops->read_mc(s, out);
}
static inline void temp_sensor_sleep(temp_sensor_t *s) {

#endif                            /* 64비트 호스트 테스트에선 72 B */

#define MSG_POOL_N 16u
static msg_t    s_msgs[MSG_POOL_N];        /* 1 KB, .bss */
static msg_t   *s_free;
static uint16_t s_in_use, s_peak;

static inline uint32_t irq_save(void) {
    uint32_t pm = __get_PRIMASK(); __disable_irq(); return pm;
}
static inline void irq_restore(uint32_t pm) { __set_PRIMASK(pm); }

void msg_pool_init(void) {
    s_free = NULL;
    for (size_t i = MSG_POOL_N; i > 0u; i--) {   /* #0 이 맨 앞에 오게 역순 */
        s_msgs[i - 1u].next = s_free;
        s_free = &s_msgs[i - 1u];
    }
}

msg_t *msg_alloc(void) {
    uint32_t pm = irq_save();
    msg_t *m = s_free;
    if (m != NULL) {
        s_free = m->next;
        s_in_use++;
        if (s_in_use > s_peak) { s_peak = s_in_use; }
    }
    irq_restore(pm);
    return m;              /* NULL = 고갈 → 호출자 정책 (버리고 카운트) */
}

void msg_free(msg_t *m) {
    FW_ASSERT((m >= &s_msgs[0]) && (m < &s_msgs[MSG_POOL_N]));
    uint32_t pm = irq_save();
    m->next = s_free;
    s_free  = m;
    s_in_use--;
    irq_restore(pm);
}

uint16_t msg_pool_peak(void) { return s_peak; }  /* 셸 · 로그로 노출 */

typedef struct { uint16_t sig; uint16_t arg; } event_t;
#define EVQ_SIZE 32u                  /* 2의 거듭제곱 */
static event_t  s_q[EVQ_SIZE];
static uint32_t s_head, s_tail;       /* CMSIS 의 __disable_irq 는 컴파일러
                                         메모리 장벽도 겸합니다 */
bool evq_post_from_isr(uint16_t sig, uint16_t arg) {
    uint32_t pm = irq_save();         /* ISR 중첩 대비 — k10 과 같은 함수 */
    bool ok = (s_head - s_tail) < EVQ_SIZE;
    if (ok) {
        s_q[s_head & (EVQ_SIZE - 1u)] = (event_t){ sig, arg };
        s_head++;
    }
    irq_restore(pm);
    return ok;                        /* false = 넘침 → 카운터 증가 */
}

bool evq_get(event_t *out) {
    uint32_t pm = irq_save();
    bool ok = (s_head != s_tail);
    if (ok) { *out = s_q[s_tail & (EVQ_SIZE - 1u)]; s_tail++; }
    irq_restore(pm);
    return ok;
}
bool evq_is_empty(void) { return s_head == s_tail; }   /* 인터럽트 꺼진 채 호출 */

typedef enum { WS_OFF, WS_BOOTING, WS_JOINING, WS_ONLINE, WS_BACKOFF,
               WS_POWERED,            /* 부모 — 현재 상태가 되지는 않음 */
               WS_NONE } wifi_state_t;
enum { EV_POWER_ON = 1, EV_READY, EV_JOINED, EV_TIMEOUT, EV_LINK_LOST, EV_POWER_OFF };

typedef struct wifi wifi_t;
typedef struct {
    wifi_state_t from; uint16_t sig; wifi_state_t to; void (*action)(wifi_t *w);
} wifi_tr_t;

static const wifi_tr_t k_tr[] = {          /* 그림의 화살표 = 표의 한 줄 */
    { WS_OFF,     EV_POWER_ON,  WS_BOOTING, act_power_on },
    { WS_BOOTING, EV_READY,     WS_JOINING, act_join     },
    { WS_BOOTING, EV_TIMEOUT,   WS_BACKOFF, act_backoff  },
    { WS_JOINING, EV_JOINED,    WS_ONLINE,  act_online   },
    { WS_JOINING, EV_TIMEOUT,   WS_BACKOFF, act_backoff  },
    { WS_ONLINE,  EV_LINK_LOST, WS_BACKOFF, act_backoff  },
    { WS_BACKOFF, EV_TIMEOUT,   WS_JOINING, act_join     },
    { WS_POWERED, EV_POWER_OFF, WS_OFF,     act_power_off },  /* ③ 부모 전이 */
};

static const wifi_state_t k_parent[WS_NONE] = {
    [WS_OFF] = WS_NONE,        [WS_BOOTING] = WS_POWERED, [WS_JOINING] = WS_POWERED,
    [WS_ONLINE] = WS_POWERED,  [WS_BACKOFF] = WS_POWERED, [WS_POWERED] = WS_NONE,
};

void wifi_fsm_dispatch(wifi_t *w, const event_t *e) {
    /* ③ HSM: 자기 상태에서 못 찾으면 부모에게 물어봄 */
    for (wifi_state_t s = w->state; s != WS_NONE; s = k_parent[s]) {
        for (size_t i = 0u; i < (sizeof k_tr / sizeof k_tr[0]); i++) {
            if ((k_tr[i].from == s) && (k_tr[i].sig == e->sig)) {
                if (k_tr[i].action != NULL) { k_tr[i].action(w); }
                w->state = k_tr[i].to;
                return;
            }
        }
    }
    /* 어디에도 없음 = 이 상태에선 무시하는 이벤트 */
}

   ONLINE · BACKOFF 에서 POWER_OFF → OFF (부모가 처리) — node 로 같은 표를 돌려 확인 */

액티브 오브젝트(Active Object) — "상태 머신 + 전용 이벤트 큐 + 전용 실행 흐름(태스크)"을 한 덩어리로 묶은 것입니다. 객체끼리는 이벤트만 주고받고 상태를 공유하지 않으므로 뮤텍스가 거의 필요 없습니다. 서버의 액터 모델(Akka · Erlang)과 같은 생각입니다. FreeRTOS 위에서는 "태스크 하나 + xQueueReceive 루프 + dispatch" 로 직접 만들고(r13), 프레임워크로는 Quantum Leaps 의 QP/C(HSM 진입/퇴장 · 시간 이벤트 · 발행/구독, 그래픽 모델링 도구 QM)가 대표적입니다.
위 HSM 은 "부모에게 묻기"만 구현한 최소판입니다. 실제 HSM 은 전이할 때 떠나는 상태들의 exit · 들어가는 상태들의 entry 동작을 올바른 순서로 실행해야 하고(예: POWER_OFF 때 BACKOFF 의 타이머 해제), 이 부분이 직접 짜기 가장 까다롭습니다 — 여기서부터는 프레임워크를 쓰는 편이 낫습니다.
정리 — 플래그 대신 이벤트 큐, 흩어진 변수 대신 명시적 상태 머신. 작으면 switch, 커지면 전이 표, 공통 전이가 많으면 HSM, 여럿이 동시에 돌면 액티브 오브젝트로 갑니다. 핸들러는 기다리지 않고 한 이벤트를 끝까지 처리합니다.

uint8_t total = len + HDR_SIZE; 는 len = 250, HDR_SIZE = 8 이면 int 로 258 을 계산한 뒤 uint8_t 에 잘려 2가 됩니다 — 경고 없이. -Wconversion 과 MISRA 10.3 이 정확히 이 줄을 잡습니다. CERT C 의 INT31-C(정수 변환으로 데이터 손실 금지)도 같은 문제입니다.

/* 외부에서 온 길이 — 네트워크 · UART 입력은 언제나 의심 */
fw_err_t proto_copy_payload(const uint8_t *frame, size_t frame_len,
                            uint8_t *dst, size_t dst_cap, size_t *out_n) {
    if (frame_len < HDR_SIZE) { return FW_ERR_PARAM; }
    size_t n = frame[1];                       /* 공격자가 정하는 값 */
    if ((n > (frame_len - HDR_SIZE)) || (n > dst_cap)) {   /* ARR30-C */
        return FW_ERR_PARAM;
    }
    memcpy(dst, &frame[HDR_SIZE], n);
    *out_n = n;
    return FW_OK;
}
