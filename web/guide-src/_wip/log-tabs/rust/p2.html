LOG 04

tracing-subscriber 실전 — 레이어를 겹쳐 내 구독자를 조립합니다

tracing 은 기록만 하고, 그걸 받아 처리하는 쪽이 구독자(Subscriber) 입니다. tracing-subscriber 는 구독자를 통째로 주는 대신 Registry 위에 레이어를 끼우는 방식을 씁니다 — 포맷 하나, 필터 하나, OTel 하나… 필요한 만큼 겹칩니다.

구독자는 완성품이 아니라 조립품입니다 — with() 한 번이 층 하나입니다.

가장 짧은 것부터 실무형까지

[dependencies]
tracing = "0.1"
tracing-subscriber = { version = "0.3", features = ["env-filter", "json"] }
# "env-filter" 없이는 EnvFilter 를 못 씁니다 ★ 가장 흔한 컴파일 에러
// ① 한 줄 — 실습·예제용
tracing_subscriber::fmt::init();

// ② 필터를 붙인 한 줄 — 작은 CLI 라면 여기까지로 충분합니다
tracing_subscriber::fmt()
    .with_env_filter("info,hyper=warn")
    .init();

// ③ 레이어 합성 — 서비스는 이 모양이 기본값입니다 ★
use tracing_subscriber::{fmt, EnvFilter, prelude::*};

pub fn init_log() {
    let filter = EnvFilter::builder()
        .with_default_directive(tracing::level_filters::LevelFilter::INFO.into())
        .from_env_lossy();          // RUST_LOG 가 이상해도 죽지 않습니다

    let json = std::env::var("LOG_FORMAT").as_deref() == Ok("json");

    tracing_subscriber::registry()
        .with(filter)
        .with(fmt::layer()
            .with_target(true)
            .with_thread_ids(!json)
            .with_ansi(!json && std::io::IsTerminal::is_terminal(&std::io::stdout())))
        .init();
    // json 이면 아래 LOG 05 의 .json() 갈래로 갑니다
}
from_env_lossy() 를 쓰세요 — from_env() 는 RUST_LOG 에 오타가 있으면 에러를 돌려주고, 그걸 unwrap() 하면 서비스가 시작도 못 합니다. lossy 는 이상한 지시자만 버리고 나머지를 살립니다.

fmt::layer() 의 손잡이들

메서드효과
.with_target(true)모듈 경로 표시
.with_thread_ids(true)스레드 번호
.with_thread_names(true)스레드 이름
.with_file(true).with_line_number(true)소스 위치
.with_ansi(false)색 끄기 — 파일·수집기용
.with_span_events(FmtSpan::CLOSE)구간 소요 시간
.compact()한 줄로 촘촘히
.pretty()여러 줄로 읽기 좋게
.json()구조적 로깅 (LOG 05)
.with_writer(w)출력 대상 바꾸기
.with_test_writer()테스트 캡처로
use tracing_subscriber::fmt::format::FmtSpan;

fmt::layer()
    .with_span_events(FmtSpan::NEW | FmtSpan::CLOSE)
    // 출력 예:
    //   INFO handle{req_id=7781}: new
    //   INFO handle{req_id=7781}: close time.busy=1.2ms time.idle=84ms
    //                                     ↑ CPU     ↑ .await 대기 ★
    .with_writer(std::io::stderr)      // 로그는 stderr 가 안전합니다
with_ansi 는 자동이 아닙니다 — 파이프로 넘겨도 색 코드가 그대로 나갑니다. std::io::IsTerminal 로 판단하거나(Rust 1.70+), 운영에서는 무조건 끄세요. ESC[32m 같은 쓰레기가 로그 저장소에 쌓이는 것을 막습니다.

RUST_LOG 문법 전부 — 이 표 하나면 끝납니다 🟡 중급

지시자(directive) 의 문법

RUST_LOG = 지시자[,지시자,...]

지시자 = target[span{field=value}]=level
         └────┘ └──────────────┘ └───┘
          모듈 경로   span 이름·필드    레벨
         (전부 생략 가능 — 레벨만 써도 됩니다)

level = error < warn < info < debug < trace   (또는 off)
        숫자 1~5 로도 씁니다 (1=error … 5=trace)

★ 왼쪽에서 오른쪽으로 읽지 말고 "가장 구체적인 지시자가 이긴다"로 읽으세요
# ── 레벨만 ────────────────────────────────
RUST_LOG=info                    # 전부 info 이상
RUST_LOG=off                     # 전부 끄기

# ── 모듈별 ────────────────────────────────
RUST_LOG=warn,myapp=debug        # 기본 warn, 내 크레이트만 debug ★가장 흔함
RUST_LOG=myapp::db=trace         # 특정 모듈만
RUST_LOG=info,hyper=off,h2=off   # 시끄러운 크레이트 입막음
RUST_LOG=debug,sqlx::query=info  # sqlx 가 쿼리를 전부 찍는 걸 막기

# ── span 이름으로 (tracing 전용) ───────────
RUST_LOG=[request]=debug         # request 라는 span 안에서만 debug
RUST_LOG=myapp[db]=trace         # myapp 크레이트의 db span 안에서만

# ── span 필드 값으로 (tracing 전용) ★ ──────
RUST_LOG=[request{user_id=42}]=trace   # 특정 사용자 요청만 trace
RUST_LOG=[request{path}]=debug         # path 필드가 있기만 하면
필드 값 필터가 진짜 무기입니다 — 운영에서 “특정 고객만 이상하다”일 때 그 고객 요청만 trace 로 켤 수 있습니다. 전체를 trace 로 켰다가 디스크를 채우는 사고를 막아 줍니다.

두 구현의 차이와 실행 중 변경

문법env_loggerEnvFilter

myapp=debug✔✔
a::b=trace✔✔
[span]=debug✗✔
[span{k=v}]✗✔

실행 중 변경✗reload 로
// 실행 중에 레벨을 바꾸기 — reload 레이어
use tracing_subscriber::{reload, EnvFilter, prelude::*};

let (filter, handle) = reload::Layer::new(EnvFilter::new("info"));
tracing_subscriber::registry()
    .with(filter)
    .with(tracing_subscriber::fmt::layer())
    .init();

// 나중에 (예: 관리용 HTTP 엔드포인트에서)
async fn set_level(body: String) -> StatusCode {
    match handle.modify(|f| *f = EnvFilter::new(&body)) {
        Ok(_)  => StatusCode::OK,        // 재배포 없이 debug 로 전환 ★
        Err(_) => StatusCode::BAD_REQUEST,
    }
}
이 엔드포인트는 반드시 보호하세요 — 누구나 부를 수 있으면 trace 로 올려 디스크를 채우는 공격이 됩니다. 내부망 전용 포트에 두거나 관리자 인증 뒤에 두고, 일정 시간 뒤 자동 복귀를 넣는 것이 안전합니다.

레이어별 필터와 “두 곳으로 동시에 쓰기” 🔴 실무

전역 필터 vs 레이어별 필터

use tracing_subscriber::{fmt, filter, EnvFilter, Layer, prelude::*};

// 터미널에는 사람이 읽을 것만, 파일에는 debug 까지 전부
let (file_w, _guard) = tracing_appender::non_blocking(
    tracing_appender::rolling::daily("logs", "app.log"));

tracing_subscriber::registry()
    // ① 콘솔 — info 이상만, 색 있음
    .with(fmt::layer()
        .with_ansi(true)
        .with_filter(EnvFilter::new("info")))
    // ② 파일 — debug 까지, 색 없음, JSON
    .with(fmt::layer()
        .json()
        .with_ansi(false)
        .with_writer(file_w)
        .with_filter(EnvFilter::new("debug,hyper=warn")))
    // ③ 지표용 — 특정 target 만 골라서
    .with(MetricsLayer::new()
        .with_filter(filter::filter_fn(|m| m.target().starts_with("biz::"))))
    .init();
전역 필터와 레이어 필터는 곱해집니다 — .with(EnvFilter) 는 구독자 전체에, .with_filter(...) 는 그 레이어에만 걸립니다. 전역이 info 인데 레이어가 debug 를 원하면 debug 는 안 옵니다 — 전역을 trace 로 열고 레이어별로 좁히는 것이 정석입니다.

조건부 레이어 — 타입이 안 맞을 때

// ✗ 이건 컴파일 안 됩니다 — if 의 두 갈래 타입이 다릅니다
// let layer = if json { fmt::layer().json() } else { fmt::layer() };

// ✔ 방법 1 — Boxed 로 타입을 지웁니다
use tracing_subscriber::Layer;
let layer = if json {
    fmt::layer().json().boxed()
} else {
    fmt::layer().pretty().boxed()
};

// ✔ 방법 2 — Option<Layer> 도 Layer 입니다 (None 이면 없는 셈)
let otel = if otel_on { Some(otel_layer()) } else { None };

tracing_subscriber::registry()
    .with(EnvFilter::from_default_env())
    .with(layer)
    .with(otel)                      // None 이어도 그대로 넘깁니다 ★
    .init();
// 라이브러리에서 구독자를 설치하면 안 됩니다.
// 대신 "레이어를 돌려주는 함수"를 공개하세요 ★
pub fn my_layer<S>() -> impl Layer<S>
where S: tracing::Subscriber + for<'a> tracing_subscriber::registry::LookupSpan<'a>
{ ... }
구독자도 프로세스당 하나입니다. init() 은 두 번째 호출에서 패닉하고 try_init() 은 Err 를 돌려줍니다. 라이브러리는 절대 구독자를 설치하지 마세요 — 앱의 로깅 설정을 통째로 빼앗는 셈입니다.
정리 — 구독자는 registry() + with() 반복으로 조립합니다. 필터는 앞에(뒤 레이어가 아예 안 불립니다), 색은 터미널일 때만, RUST_LOG 는 target[span{field}]=level 한 줄 문법이 전부이고 span 필드 필터가 운영에서 가장 강력한 손잡이입니다.
LOG 05

구조적 로깅 — 문자열이 아니라 필드로 남깁니다

"주문 7781 이 12000원 결제됨" 은 사람만 읽습니다. {order_id: 7781, amount: 12000} 은 기계가 집계하고 정렬하고 알람을 겁니다. 같은 정보를 어떻게 담느냐가 “grep 밖에 못 하는 로그”와 “질의할 수 있는 로그”를 가릅니다.

구조적 로깅의 요점은 message 를 줄이고 필드를 늘리는 것입니다.

필드 문법 — 시길 세 가지

use tracing::info;

let id: u64 = 7781;
let user = User { id: 42, name: "kim".into() };
let path = std::path::Path::new("/tmp/a.csv");

// ① 시길 없음 — 값을 그대로 (숫자 · bool · &str · char 등)
info!(order_id = id, retried = true, "결제 완료");

// ② % — Display 로 문자열화
info!(file = %path.display(), "파일을 읽었습니다");

// ③ ? — Debug 로 문자열화
info!(user = ?user, "사용자 조회");

// ④ 변수 이름을 그대로 필드 이름으로 (축약형) ★
let amount = 12000;
info!(amount, "금액 확인");        // amount = 12000

// ⑤ 점이 들어간 이름은 따옴표로
info!(http.status = 200, "응답");  // ✗ 컴파일 에러
info!("http.status" = 200, "응답"); // ✔

// ⑥ 메시지는 사실 message 라는 필드일 뿐입니다
info!(message = "결제 완료", order_id = id);   // 위 ①과 같습니다
? 를 함부로 쓰지 마세요 — Debug 는 구조체를 통째로 찍습니다. ?req 한 번에 비밀번호와 카드번호가 로그로 새어 나갑니다. 민감한 필드는 #[derive(Debug)] 대신 손으로 Debug 를 구현해 "[redacted]" 을 찍게 하는 것이 가장 확실합니다.

JSON 으로 내보내기

tracing_subscriber::registry()

        .with_current_span(true)    // 현재 span 을 포함
        .with_span_list(true)       // 조상 span 전부를 배열로 ★
        .with_timer(fmt::time::UtcTime::rfc_3339()))
    .init();

좋은 로그 한 줄의 모양 🟡 중급

이렇게 쓰지 마세요
info!("주문 {} 를 사용자 {} 가 {}원에 결제했고 재시도 {}회 만에 성공",
      order.id, user.id, amount, attempt);

// 문제
//  · 집계 불가 — "금액 합계"를 내려면 문자열을 파싱해야 합니다

info!(

    "결제 완료"                 // 메시지는 '사건 이름'만

필드 이름 규칙 — 팀에서 미리 정하세요

필드규칙
*_id식별자는 접미사 통일
outcomesuccess/failure/skipped
duration_ms단위를 이름에 넣습니다
error.kind에러 분류 (문구 아님)

카디널리티가 왜 문제인가 — 로그 저장소는 필드 값의 종류 수만큼 색인을 만듭니다. user_id 처럼 값이 수백만인 필드를 대시보드의 그룹 기준으로 쓰면 저장소가 죽습니다. 조회용 식별자와 집계용 라벨을 구분하세요.

민감 정보를 구조적으로 막기

use std::fmt;

pub struct Secret(String);

// Debug 도 Display 도 절대 원본을 내놓지 않습니다 ★
impl fmt::Debug for Secret {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.write_str("Secret([redacted])")
    }
}
impl fmt::Display for Secret {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.write_str("[redacted]")
    }
}

// 이러면 ?req 로 통째로 찍어도 안전합니다
#[derive(Debug)]
struct LoginReq { user: String, password: Secret }

// 카드번호처럼 일부만 남겨야 하면
fn mask_card(s: &str) -> String {
    format!("****-****-****-{}", &s[s.len().saturating_sub(4)..])
}
검토가 아니라 타입으로 막으세요 — “로그에 비밀번호 찍지 말기”는 코드 리뷰로 지켜지지 않습니다. Secret<T> 같은 래퍼를 만들어 타입 시스템이 강제하게 하면 새로 들어온 사람도 사고를 못 냅니다. secrecy 크레이트가 이 패턴을 구현해 둔 것입니다.
정리 — 메시지는 사건의 이름만 담고, 나머지는 전부 필드로 뺍니다. 시길은 셋 — 없음(값 그대로) · %(Display) · ?(Debug). ? 는 통째로 찍히니 민감 타입은 Debug 를 직접 구현해 막고, 단위는 필드 이름에(duration_ms) 넣는 습관을 팀 규칙으로 정하세요.
LOG 06

크레이트 비교 — 여섯 갈래, 어떤 상황에 무엇을 고르나

러스트 로깅 크레이트는 많아 보이지만 계보는 세 줄기입니다 — log 파사드 위의 구현체들, tracing, 그리고 임베디드 전용 defmt. 아래 표는 강도를 보여 줍니다. 정확한 다운로드 수치가 아니라 상대적 위치로 읽으세요.

기준이 다섯이면 표가 정직합니다 — 한 줄로 좋은 것은 없습니다.

각 크레이트의 실제 코드 🟡 중급

env_logger · fern

// ── env_logger 0.11 — 설정은 빌더로 ──────────────
use env_logger::{Builder, Env, Target};
use std::io::Write;

Builder::from_env(Env::default().default_filter_or("info"))
    .target(Target::Stderr)
    .format(|buf, r| {
        writeln!(buf, "{} [{}] {}: {}",
            chrono::Local::now().format("%H:%M:%S%.3f"),
            r.level(), r.target(), r.args())
    })
    .init();
// ── fern — 체이닝으로 여러 출구를 만듭니다 ─────────
fern::Dispatch::new()
    .level(log::LevelFilter::Info)
    .level_for("myapp::db", log::LevelFilter::Debug)
    .chain(std::io::stderr())                    // 출구 ①
    .chain(fern::log_file("app.log")?)           // 출구 ②
    .format(|out, msg, rec| {
        out.finish(format_args!("[{}] {}", rec.level(), msg))
    })
    .apply()?;
// 장점 — log 만 쓰면서도 '여러 출구'가 쉽습니다
// 한계 — span 이 없습니다. 비동기 문맥은 여전히 손으로 나릅니다

slog · flexi_logger

// ── slog — 로거를 '값으로' 들고 다닙니다 ───────────
let drain = slog_json::Json::default(std::io::stderr()).fuse();
let drain = slog_async::Async::new(drain).build().fuse();
let root = slog::Logger::root(drain, slog::o!("svc" => "pay"));

let req_log = root.new(slog::o!("req_id" => 7781));  // 자식 로거
slog::info!(req_log, "결제 시작"; "amount" => 12000);
// ★ 문맥을 '로거 객체'로 나릅니다 — 함수 인자로 계속 넘겨야 합니다
//   tracing 은 이걸 '현재 span' 으로 바꿔 인자를 없앤 것입니다
// ── flexi_logger — 파일 회전·보관이 기본 탑재 ──────
use flexi_logger::{Logger, Criterion, Naming, Cleanup, FileSpec};

Logger::try_with_str("info, myapp=debug")?
    .log_to_file(FileSpec::default().directory("logs"))
    .rotate(Criterion::Size(10_000_000),   // 10MB 마다
            Naming::Timestamps,
            Cleanup::KeepLogFiles(7))      // 7개만 보관 ★
    .append()
    .start()?;
// 데스크톱 앱·온프렘 설치형에서 '수집기가 없을 때' 좋은 선택입니다
slog 는 왜 밀렸나 — 아이디어(구조적 로깅)는 옳았지만 로거 객체를 인자로 나르는 방식이 비동기와 잘 맞지 않았습니다. tracing 이 같은 아이디어를 암묵적 현재 span 으로 바꾸면서 생태계(tokio·tower·axum·sqlx)가 통째로 그쪽으로 갔습니다.
정리 — 계보는 셋입니다 — log 파사드 + 구현체, tracing, defmt. 새 서비스는 tracing, 새 라이브러리는 log, 임베디드는 defmt 로 시작하면 거의 항상 맞습니다. 채택 정도는 crates.io 다운로드와 lib.rs 순위로 직접 확인하세요 — 남의 숫자를 옮겨 적지 말고요.
