LOG 07

비동기 · Tokio — span 이 .await 를 건너 이어지는 원리

비동기 태스크는 스레드에 붙어 있지 않습니다. .await 에서 양보하고, 다음에 깨어날 때는 다른 워커 스레드일 수 있습니다. 그래서 스레드 로컬에 문맥을 담는 옛 방식은 여기서 깨지고, tracing 은 그것을 구조로 풀었습니다.

span 은 스레드가 아니라 Future 에 붙어 있습니다 — 그래서 워커를 옮겨도 안 끊깁니다.

왜 enter() 를 .await 너머로 들면 안 되나

// ✗ 컴파일은 되지만 틀린 코드입니다
async fn bad(id: u64) {
    let span = tracing::info_span!("job", id);
    let _g = span.enter();          // 가드가 .await 를 넘어 살아 있습니다
    do_io().await;                  // ← 여기서 태스크가 양보합니다
    tracing::info!("끝");
}
// 무슨 일이 나나
//   · 양보한 사이 그 스레드가 '다른 태스크'를 돌립니다
//   · 그런데 현재 span 은 여전히 job{id} 로 남아 있습니다
//   · 남의 태스크 로그에 내 id 가 붙습니다 ★ 문맥 오염
// ✔ 방법 1 — 어트리뷰트 (거의 항상 이걸 씁니다)
#[tracing::instrument(skip_all, fields(id))]
async fn good1(id: u64) {
    do_io().await;
    tracing::info!("끝");
}

// ✔ 방법 2 — Instrument 트레이트로 Future 를 감쌉니다
use tracing::Instrument;
async fn good2(id: u64) {
    async {
        do_io().await;
        tracing::info!("끝");
    }
    .instrument(tracing::info_span!("job", id))   // ★
    .await;
}

// ✔ 방법 3 — 동기 블록만 감쌀 때
span.in_scope(|| { /* .await 없는 코드만 */ });
원리 한 줄 — Instrumented<F> 는 poll() 을 감싸서 폴링 직전에 enter, 폴링이 끝나면 exit 합니다. 즉 span 이 Future 자체에 소유되므로, 어느 스레드에서 폴링되든 문맥이 정확히 그 태스크만 따릅니다.

태스크를 띄울 때 문맥 잇기

use tracing::{Instrument, Span};

// ① spawn 은 문맥을 자동으로 물려주지 않습니다
tokio::spawn(async { work().await });          // ✗ 부모 span 없음

// ② 현재 span 을 명시적으로 물려줍니다 ★
tokio::spawn(work().instrument(Span::current()));

// ③ 새 span 을 만들되 부모를 이어 붙입니다
let parent = Span::current();
tokio::spawn(async move {
    let sp = tracing::info_span!(parent: &parent, "worker", shard = 3);
    async { work().await }.instrument(sp).await;
});

// ④ 아예 부모를 끊고 싶을 때 (배경 잡업 등)
let sp = tracing::info_span!(parent: None, "cron");
// blocking 작업도 마찬가지입니다
let sp = Span::current();
tokio::task::spawn_blocking(move || {
    let _g = sp.enter();          // 여기는 .await 가 없으니 enter 가 안전합니다
    heavy_cpu_work();
}).await?;
“요청 ID 가 배경 작업 로그에 안 붙는다”는 거의 항상 ①번 실수입니다. tokio::spawn 은 완전히 독립된 태스크를 만들기 때문에 .instrument(Span::current()) 를 붙여 주지 않으면 문맥이 끊깁니다.

HTTP 서버에 요청 span 붙이기 🟡 중급

Axum · tower-http 의 TraceLayer

[dependencies]
axum = "0.8"
tower-http = { version = "0.6", features = ["trace", "request-id"] }
tracing = "0.1"
tracing-subscriber = { version = "0.3", features = ["env-filter"] }
use tower_http::trace::TraceLayer;
use tracing::info_span;

let app = Router::new()
    .route("/orders/{id}", get(get_order))
    .layer(
        TraceLayer::new_for_http()
            // 요청마다 span 을 하나 만듭니다 ★
            .make_span_with(|req: &http::Request<_>| {
                let rid = req.headers().get("x-request-id")
                    .and_then(|v| v.to_str().ok())
                    .unwrap_or("-");
                info_span!("http",
                    method = %req.method(),
                    path   = %req.uri().path(),
                    request_id = %rid,
                    status = tracing::field::Empty,   // 나중에 채웁니다
                )
            })
            .on_response(|res: &http::Response<_>, lat: std::time::Duration, sp: &tracing::Span| {
                sp.record("status", res.status().as_u16());
                tracing::info!(duration_ms = lat.as_millis() as u64, "요청 완료");
            })
    );
레이어 순서가 중요합니다 — TraceLayer 를 가장 바깥(가장 마지막 .layer())에 두어야 인증 실패·타임아웃·패닉까지 전부 기록됩니다. 안쪽에 두면 그 앞에서 잘린 요청은 기록이 없습니다.

이렇게 나옵니다

INFO http{method=GET path=/orders/7781 request_id=0f2a}: 요청 시작
DEBUG http{method=GET path=/orders/7781 request_id=0f2a}
      :db{table="orders"}: sqlx: SELECT ... 실행 (4.1ms)
WARN  http{method=GET path=/orders/7781 request_id=0f2a}
      :db{table="orders"}: 느린 쿼리 duration_ms=412
INFO  http{method=GET path=/orders/7781 request_id=0f2a status=200}
      : 요청 완료 duration_ms=418

# 한 요청의 모든 줄이 request_id 로 묶입니다 ★
#   grep 0f2a app.log  →  그 요청에서 일어난 일 전부
// 핸들러 안에서는 아무것도 안 해도 됩니다
#[tracing::instrument(skip(state))]      // 부모는 http span
async fn get_order(State(state): State<App>, Path(id): Path<u64>)
    -> Result<Json<Order>, AppError>
{
    let o = state.repo.find(id).await?;  // repo 안의 로그에도 전부 붙습니다
    Ok(Json(o))
}
요청 ID 는 직접 만들지 마세요 — tower-http 의 SetRequestIdLayer/PropagateRequestIdLayer 가 없으면 만들고 있으면 그대로 전달합니다. 프록시·게이트웨이가 이미 붙여 주는 경우가 많으니 덮어쓰지 않는 것이 중요합니다.

붙이는 법

[build]                       # .cargo/config.toml 에

cargo install --locked tokio-console
tokio-console                       # 기본 http://127.0.0.1:6669

# 보이는 것
#   · 살아 있는 태스크 목록과 각각의 폴링 횟수 · busy/idle 시간
#   · '오래 안 깨어난' 태스크 경고 (누수 후보)

운영에는 켜지 마세요 — tokio_unstable 은 ABI 안정성이 보장되지 않고, 계측 자체가 오버헤드입니다. 개발·부하 시험 환경에서만 켜는 것이 정석이고, 빌드 프로파일을 따로 두어 실수로 배포되지 않게 하세요.

실제로 잡히는 문제들

console 이 보여 주는 것진짜 원인
한 폴링이 수십 ms비동기 안에서 블로킹
(파일 I/O · std::sync::Mutex)
태스크 수가 계속 증가완료되지 않는 태스크 누수
idle 만 길다외부 호출 대기 — 내 잘못 아님
busy 만 길다CPU 작업 — spawn_blocking 으로
세마포어 대기 급증커넥션 풀 고갈
// 가장 흔한 범인 — 비동기 안의 동기 잠금
use std::sync::Mutex;                 // ✗ .await 를 넘어 들면 워커가 멈춥니다
use tokio::sync::Mutex as AsyncMutex; // ✔ 대기 중 양보합니다

// 잠금을 .await 너머로 안 들면 std::sync 가 더 빠릅니다 ★
{
    let mut g = std_mutex.lock().unwrap();
    *g += 1;
}                                      // 여기서 반드시 놓습니다
do_io().await;
tracing 만으로도 절반은 잡힙니다 — FmtSpan::CLOSE 로 time.busy 와 time.idle 을 찍으면 “느린데 CPU 는 안 쓴다(외부 대기)” 와 “CPU 를 다 쓴다(내 코드)” 가 구분됩니다. tokio-console 은 그다음 단계의 도구입니다.
정리 — 비동기에서 문맥이 유지되는 이유는 span 이 스레드가 아니라 Future 에 붙기 때문입니다. async fn 은 #[instrument], 수동 Future 는 .instrument(span), tokio::spawn 은 .instrument(Span::current()) 를 반드시 붙이세요. enter() 가드를 .await 너머로 들면 남의 로그에 내 문맥이 새어 나갑니다.
LOG 08

파일 출력과 로테이션 — WorkerGuard 를 놓치면 로그가 사라집니다

컨테이너가 아니라면(데스크톱 앱 · 온프렘 설치형 · 배치 작업) 파일로 남겨야 합니다. 그때 필요한 두 가지가 회전(rotation) 과 비동기 쓰기 이고, 비동기 쓰기에는 거의 모두가 한 번씩 당하는 함정이 있습니다.

원통 셋은 같은 로거가 만든 날짜별 파일입니다 — 회전은 이름 바꾸기가 아닙니다.

tracing-appender 기본형

[dependencies]
tracing-appender = "0.2"
use tracing_appender::{non_blocking, rolling};
use tracing_subscriber::{fmt, EnvFilter, prelude::*};
use tracing_appender::non_blocking::WorkerGuard;

// ★ guard 를 반드시 돌려주고, main 이 끝까지 들고 있어야 합니다
pub fn init_log() -> WorkerGuard {
    let file = rolling::daily("logs", "app.log");
    let (writer, guard) = non_blocking(file);

    tracing_subscriber::registry()
        .with(EnvFilter::from_default_env())
        .with(fmt::layer().with_writer(std::io::stderr))   // 사람용
        .with(fmt::layer().json().with_ansi(false).with_writer(writer))  // 파일용
        .init();

    guard
}

fn main() {
    let _guard = init_log();     // ★ 이름을 주세요. `let _ =` 이면 즉시 드롭!
    run();
}                                 // 여기서 드롭되며 남은 로그를 전부 비웁니다
가장 많이 나는 사고 — let _ = init_log(); 로 쓰면 WorkerGuard 가 그 줄에서 즉시 드롭되어 워커 스레드가 종료됩니다. 그러면 파일이 만들어지긴 하는데 내용이 비거나 몇 줄만 있습니다. _guard 처럼 이름 있는 변수에 담으세요.

회전 정책과 보관 개수

use tracing_appender::rolling::{RollingFileAppender, Rotation};

// 간단형 — 네 가지가 준비돼 있습니다
rolling::minutely("logs", "app.log");   // 개발·시험용
rolling::hourly  ("logs", "app.log");
rolling::daily   ("logs", "app.log");   // ★ 가장 흔함
rolling::never   ("logs", "app.log");   // 회전 없음

// 빌더형 — 이름과 보관 개수를 지정합니다 ★
let file = RollingFileAppender::builder()
    .rotation(Rotation::DAILY)
    .filename_prefix("app")
    .filename_suffix("log")
    .max_log_files(14)              // 14개를 넘으면 오래된 것부터 지웁니다
    .build("logs")
    .expect("로그 디렉터리를 만들 수 없습니다");

// 만들어지는 파일 이름
//   logs/app.2026-08-31.log
//   logs/app.2026-09-01.log
크기 기준 회전이 필요하면 — tracing-appender 는 시간 기준만 지원합니다. “10MB 마다”가 필요하면 flexi_logger(Criterion::Size) 를 쓰거나, 리눅스라면 logrotate 에 맡기는 것이 가장 단순합니다 (다만 copytruncate 를 써야 파일 핸들이 안 끊깁니다).

non_blocking 이 실제로 하는 일 🟡 중급

채널 + 워커 스레드 + 유실 정책

use tracing_appender::non_blocking::NonBlockingBuilder;

let (writer, guard) = NonBlockingBuilder::default()
    .lossy(true)                 // ★ 기본값 true — 큐가 차면 '버립니다'
    .buffered_lines_limit(128_000)
    .thread_name("log-writer")
    .finish(file);

// lossy(true)  : 큐가 차면 로그를 버리고 앱은 안 막힙니다
//                (버린 개수는 stderr 에 경고로 나옵니다)
// lossy(false) : 큐가 찰 때까지 기다립니다 —
//                디스크가 느리면 '앱이 로그 때문에 멈춥니다' ★
상황고르는 값
온라인 서비스lossy(true) — 지연이 우선
감사 로그 · 금융lossy(false) — 유실 불가
배치 작업lossy(false) 로도 무방
큐 크기초당 줄 수 × 최대 지연 초
감사 로그를 lossy(true) 로 두면 안 됩니다 — “장애 때 로그가 없다”가 정확히 이때 일어납니다. 부하가 몰리는 순간 큐가 차고, 가장 중요한 그 시점의 로그부터 버려집니다.

종료 시 반드시 비우기

#[tokio::main]
async fn main() -> anyhow::Result<()> {
    let _guard = init_log();            // ★ 함수 끝까지 삽니다

    let server = axum::serve(listener, app)
        .with_graceful_shutdown(shutdown_signal());

    if let Err(e) = server.await {
        tracing::error!(error = %e, "서버 종료 실패");
    }

    tracing::info!("정상 종료");
    Ok(())
}                                        // _guard 드롭 → 남은 로그 flush

async fn shutdown_signal() {
    tokio::signal::ctrl_c().await.ok();
    tracing::info!("종료 신호를 받았습니다");
}
// std::process::exit 는 드롭을 건너뜁니다 ★ 절대 주의
fn bad() {
    let _guard = init_log();
    tracing::info!("이 줄은 파일에 안 남을 수 있습니다");
    std::process::exit(1);          // ✗ 소멸자가 안 불립니다
}

// 종료 코드를 주고 싶으면 main 에서 돌려주세요
fn main() -> std::process::ExitCode {
    let _guard = init_log();
    ...
    std::process::ExitCode::from(1)   // ✔ 드롭이 정상적으로 일어납니다
}
컨테이너라면 파일을 쓰지 마세요 — stdout 한 줄기로만 쓰고 수집(Fluent Bit · Vector · Promtail)과 보관은 플랫폼에 맡기는 것이 정석입니다. 파일 로깅은 수집기를 둘 수 없는 환경의 해법입니다.

디스크가 찼을 때 · 로그가 안 보일 때 🔴 실무

점검 순서 (위에서부터)

# ① 그 파일에 정말 쓰고 있나 — 열린 파일 핸들 확인
lsof -p $(pgrep myapp) | grep '\.log'
ls -l --time-style=full-iso logs/       # 갱신 시각을 봅니다

# ② 디스크·아이노드가 남았나 (아이노드 고갈이 의외로 흔합니다)
df -h /var/log ; df -i /var/log

# ③ 권한 — 컨테이너 안의 uid 가 디렉터리를 못 쓰는 경우
id ; ls -ld logs/

# ④ 필터가 다 막고 있나
RUST_LOG=trace ./myapp 2>&1 | head -20

# ⑤ 회전 뒤에도 같은 파일에 쓰고 있나 (logrotate 사고)
#    → copytruncate 를 쓰거나, 앱이 스스로 회전하게 하세요
logrotate 의 고전적 함정 — 기본 동작은 rename 입니다. 프로세스는 이미 열어 둔 파일 핸들을 계속 쓰므로, 새 파일은 비고 지워진 옛 파일에 계속 씁니다(디스크는 안 줄어듭니다). copytruncate 를 쓰거나 postrotate 에서 앱에 시그널을 보내세요.

로그가 디스크를 채우지 않게

// ① 반복되는 경고는 '샘플링' 하거나 '뭉쳐서' 남깁니다
use std::sync::atomic::{AtomicU64, Ordering};
static DROPPED: AtomicU64 = AtomicU64::new(0);

if let Err(e) = send(msg) {
    let n = DROPPED.fetch_add(1, Ordering::Relaxed);
    if n % 1000 == 0 {                        // 1000번에 한 번만
        tracing::warn!(dropped = n + 1, error = %e, "전송 실패 누적");
    }
}

// ② 루프 안의 debug! 는 span 밖으로 빼고 요약만 남깁니다
tracing::info!(rows = n, elapsed_ms = t.elapsed().as_millis() as u64,
               "배치 완료");

// ③ 큰 본문은 길이만 · 해시만
tracing::debug!(body_len = body.len(),
                body_sha = %hex(&sha256(&body)[..8]), "요청 본문");
로그 폭주는 장애의 증상이자 원인입니다 — 외부 API 가 죽으면 초당 수만 줄의 error! 가 쏟아지고, 그게 디스크를 채워 진짜 장애를 만듭니다. 반복 에러에는 누적 카운터 + 주기적 요약을 기본으로 하세요.
정리 — 파일 로깅의 핵심은 셋 — WorkerGuard 를 이름 있는 변수로 살려 두기, 회전 정책과 보관 개수(max_log_files) 정하기, 유실 정책(lossy) 을 의식적으로 고르기. std::process::exit 는 소멸자를 건너뛰므로 마지막 로그가 사라집니다. 컨테이너라면 애초에 stdout 한 줄기가 정답입니다.
LOG 09

성능 — 꺼진 로그는 얼마나 싼가, 켠 로그는 얼마나 비싼가

“로그를 많이 남기면 느려진다”는 말은 절반만 맞습니다. 꺼진 로그는 거의 공짜이고(원자적 비교 한 번), 컴파일 타임에 지우면 완전히 0 입니다. 비싼 것은 켠 상태에서의 포맷·할당·쓰기이고, 그건 측정해서 다뤄야 합니다.

발산 램프 — 가운데(회색)가 기준선이고 양쪽이 이득과 손해입니다.

① 컴파일 타임 제거 — 진짜 0 으로

# log 크레이트 — 코드에서 아예 사라집니다
[dependencies]
log = { version = "0.4", features = ["max_level_debug",
                                     "release_max_level_info"] }
#   max_level_*          : 모든 빌드에 적용
#   release_max_level_*  : --release 빌드에만 적용 ★ 보통 이걸 씁니다

# tracing 도 같은 이름의 피처를 가집니다
tracing = { version = "0.1", features = ["release_max_level_info"] }
// release 빌드에서 이 줄은 '기계어에 없습니다'
log::debug!("무거운 {:?}", expensive_dump());
//          ↑ expensive_dump() 도 호출되지 않습니다 ★

// 확인하는 법 — 심볼이 사라졌는지 봅니다
//   cargo build --release
//   나 cargo asm / cargo-show-asm 으로 함수 본문을 직접 확인
피처는 합쳐집니다(feature unification) — 의존성 중 하나가 max_level_trace 를 켜면 전체가 그 값이 됩니다. 반대로 release_max_level_info 를 켜 두면 운영에서 디버그 로그를 켤 방법이 아예 없어집니다 — 그 대가를 알고 쓰세요.

② 실행 중 필터 — 얼마나 싼가

// log 매크로가 펼쳐지는 모양 (개념적)
if Level::Debug <= STATIC_MAX_LEVEL          // ① 컴파일 타임 상수 비교
   && Level::Debug <= log::max_level() {      // ② 원자적 로드 1회
    /* 여기부터가 진짜 비용 */
}

// tracing 은 한 겹 더 있습니다 — 콜사이트 캐시 ★
//   각 매크로 위치마다 'Interest' 를 한 번 계산해 캐시합니다
//   Never  : 이 위치는 영원히 관심 없음 → 이후 분기조차 거의 없음
//   Always : 무조건 기록
//   Sometimes : 매번 물어봄 (span 필드 필터를 쓸 때)
단계대략의 비용
컴파일 타임에 제거0 (코드가 없음)
꺼진 레벨 (Never 캐시)나노초 단위 · 분기 하나
켠 이벤트 — 필드 기록수백 나노초
포맷 + 문자열 할당마이크로초 단위
동기 파일·터미널 쓰기수십~수백 마이크로초
비동기 큐로 넘기기채널 send 한 번
순서를 기억하세요 — 비용의 대부분은 맨 아래 두 줄에 있습니다. “로그가 느리다”면 십중팔구 동기 쓰기이지 매크로 자체가 아닙니다. 먼저 non_blocking 을 붙이고, 그다음에 레벨을 줄이세요.

use criterion::{black_box, criterion_group, criterion_main, Criterion};

fn bench(c: &mut Criterion) {
    // ① 구독자가 아예 없을 때 (기준선)
    c.bench_function("no subscriber", |b| {
        b.iter(|| tracing::debug!(v = black_box(42), "x"))
    });

    // ② 구독자는 있는데 레벨이 꺼져 있을 때
    let _g = tracing::subscriber::set_default(
        tracing_subscriber::fmt().with_max_level(tracing::Level::INFO).finish());
    c.bench_function("filtered out", |b| {
        b.iter(|| tracing::debug!(v = black_box(42), "x"))
    });

    // ③ 켜져 있고 io::sink 로 버릴 때 (포맷 비용만)
    // ④ 켜져 있고 파일에 쓸 때 (전체 비용)
}
criterion_group!(g, bench);
criterion_main!(g);
네 단계를 따로 재세요 — ①②의 차이가 필터 비용, ②③의 차이가 포맷 비용, ③④의 차이가 I/O 비용입니다. 이렇게 나눠야 “무엇을 줄여야 하는지”가 나옵니다. 보통 ④가 나머지 셋을 합친 것보다 훨씬 큽니다.

정적 디스패치와 레이어 비용

// registry().with(A).with(B) 는 Layered<B, Layered<A, Registry>> 라는
// '구체 타입'을 만듭니다 — 가상 호출이 아니라 정적 디스패치입니다 ★
//   장점: 인라인 가능 · 오버헤드 거의 없음
//   단점: 타입이 길어져 컴파일 에러가 무섭게 나옵니다

// .boxed() 를 쓰면 동적 디스패치로 바뀝니다 (호출당 가상 호출 한 번)
//   조건부 레이어에는 필요하지만, 뜨거운 경로에 여러 개 쌓지는 마세요

// 실측 없이 최적화하지 않는 규칙 ★
//   1) non_blocking 을 붙였는가
//   2) 운영 레벨이 info 인가
//   3) 루프 안에 debug! 가 있는가
//   이 셋을 먼저 확인하면 나머지는 대개 문제가 아닙니다
측정 없이 “로그를 줄이자”고 하지 마세요 — 실제 병목은 거의 항상 동기 쓰기이거나 루프 안의 debug! 입니다. FmtSpan::CLOSE 로 구간 시간을 먼저 보고, 그 다음 criterion 으로 네 단계를 나눠 재는 것이 순서입니다.
정리 — 비용의 계단은 0(컴파일 제거) → 분기 하나(꺼짐) → 필드 기록 → 포맷 → I/O 입니다. 거의 모든 문제는 맨 마지막 I/O에 있으므로 non_blocking 을 먼저 붙이세요. release_max_level_info 는 강력하지만 운영에서 디버그를 켤 길을 없애는 대가가 있습니다.
