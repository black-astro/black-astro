TOOL 02

멀티모듈 · 컨벤션 플러그인

모듈이 열 개가 되면 build.gradle.kts 열 개에 같은 설정을 복사하게 됩니다. 컨벤션 플러그인은 그 중복을 없애는 표준 방법입니다.

모듈 나누기의 핵심은 개수가 아니라 의존 방향입니다 — 아래로만 흐르게 유지하세요.
// build-logic/src/main/kotlin/myapp.kotlin-common.gradle.kts
plugins {
    kotlin("jvm")
    id("io.gitlab.arturbosch.detekt")
}

kotlin {
    jvmToolchain(21)
    compilerOptions { allWarningsAsErrors.set(true) }
}

dependencies {
    val libs = extensions.getByType<VersionCatalogsExtension>().named("libs")
    "implementation"(libs.findLibrary("coroutines-core").get())
    "testImplementation"(libs.findBundle("testing").get())
}

tasks.withType<Test> { useJUnitPlatform() }

// settings.gradle.kts
// includeBuild("build-logic")

// 각 모듈의 build.gradle.kts 는 이렇게 짧아진다
plugins {
    id("myapp.kotlin-common")
}
dependencies {
    implementation(project(":core:domain"))
}

모듈 나누는 기준

  • 도메인 vs 인프라 — 순수 코틀린 모듈은 테스트가 빠릅니다
  • 기능(feature) 단위 — 팀이 나눠 일할 수 있습니다
  • 공개 API 를 좁게 — internal을 적극 쓰면 모듈 밖에서 안 보입니다
  • api 대신 implementation — 의존성 전파를 막아 재컴파일 범위를 줄입니다
처음부터 잘게 쪼개지 마세요. 모듈 하나에 클래스 세 개면 Gradle 오버헤드가 이득보다 큽니다. 빌드가 느려지거나 경계가 필요해질 때 나누는 편이 낫습니다.

모듈 의존성 확인

# 의존성 트리
./gradlew :app:dependencies --configuration runtimeClasspath

# 특정 라이브러리가 어디서 들어오는지
./gradlew :app:dependencyInsight --dependency jackson-databind

# 프로젝트 의존 관계 그래프 (플러그인)
./gradlew projectDependencyGraph

# 버전 충돌 강제 해결
configurations.all {
    resolutionStrategy {
        force("com.fasterxml.jackson.core:jackson-databind:2.18.2")
        failOnVersionConflict()      // 충돌을 에러로 (엄격 모드)
    }
}

    active: true
  ImplicitDefaultLocale:
    active: true

  GlobalCoroutineUsage:         # GlobalScope 금지
    active: true
  SuspendFunWithFlowReturnType:
    active: true
  RedundantSuspendModifier:
    active: true

  TooGenericExceptionCaught:    # catch (e: Exception)
    active: true
  SwallowedException:           # 예외를 삼킴
    active: true

  LongMethod:
    threshold: 60
  CyclomaticComplexMethod:
    threshold: 15
// 커스텀 규칙 — 팀 고유의 금지 사항을 강제한다
class NoRunBlockingInController(config: Config) : Rule(config) {
    override val issue = Issue(
        "NoRunBlockingInController",
        Severity.Defect,
        "컨트롤러에서 runBlocking 을 쓰면 스레드가 막힙니다. suspend 함수를 쓰세요.",
        Debt.TWENTY_MINS,
    )

    override fun visitCallExpression(expression: KtCallExpression) {
        super.visitCallExpression(expression)
        if (expression.calleeExpression?.text == "runBlocking" &&
            expression.containingKtFile.name.endsWith("Controller.kt")
        ) {
            report(CodeSmell(issue, Entity.from(expression), issue.description))
        }
    }
}

// CI 에서 강제
// ./gradlew ktlintCheck detekt --continue
도입 요령 — 기존 프로젝트에 한 번에 켜면 경고가 수천 개 나옵니다. 베이스라인(./gradlew detektBaseline)을 만들어 기존 것은 넘어가고 새 코드만 검사하게 시작하세요. 그다음 조금씩 줄여 나가면 됩니다.

// 실무에서 가장 흔한 조합: JUnit5 + MockK + Kotest 단언 + Turbine
class OrderServiceTest {

    private val repo = mockk<OrderRepository>()
    private val client = mockk<PaymentClient>(relaxed = true)
    private val service = OrderService(repo, client)

    @Test
    fun `재고가 없으면 주문이 실패한다`() = runTest {
        coEvery { repo.findStock(1L) } returns 0

        val result = runCatching { service.place(cmd(productId = 1)) }

        result.isFailure shouldBe true                              // Kotest 단언
        result.exceptionOrNull().shouldBeInstanceOf<OutOfStock>()
        coVerify(exactly = 0) { client.charge(any()) }              // 결제는 시도조차 안 함
    }

    @Test
    fun `상태 흐름은 로딩 후 성공이다`() = runTest {
        coEvery { repo.observe() } returns flowOf(listOf(order()))

        service.state.test {                                        // Turbine
            awaitItem() shouldBe UiState.Loading
            awaitItem().shouldBeInstanceOf<UiState.Success>()
            cancelAndIgnoreRemainingEvents()
        }
    }
}

직렬화 — kotlinx.serialization

리플렉션 없이 컴파일 시점에 직렬화 코드를 생성합니다. 멀티플랫폼·네이티브 이미지에서 Jackson 보다 유리하고, 코틀린 타입을 정확히 다룹니다. 

plugins { kotlin("plugin.serialization") version "2.2.0" }
// implementation("org.jetbrains.kotlinx:kotlinx-serialization-json:1.8.0")

@Serializable
data class Order(
    val id: Long,
    @SerialName("order_no") val orderNo: String,           // 이름 매핑
    val status: OrderStatus,
    @Transient val internalNote: String = "",              // 직렬화 제외
    val items: List<OrderItem> = emptyList(),              // 기본값
)

@Serializable
enum class OrderStatus { @SerialName("new") NEW, @SerialName("paid") PAID }

val json = Json {
    ignoreUnknownKeys = true          // 모르는 필드는 무시 (API 변경에 강해진다)
    encodeDefaults = false            // 기본값과 같으면 출력하지 않는다
    prettyPrint = false
    explicitNulls = false             // null 필드를 생략
    coerceInputValues = true          // null 이 오면 기본값으로

val text = json.encodeToString(order)
val back = json.decodeFromString<Order>(text)

// value class · 도메인 타입을 원시 값으로
object MoneySerializer : KSerializer<Money> {
    override val descriptor = PrimitiveSerialDescriptor("Money", PrimitiveKind.LONG)

TOOL 06

Ktor 서버 — 스프링 대안

젯브레인스가 만든 코틀린 네이티브 웹 프레임워크입니다. 코루틴이 기본이고 가벼워서, 작은 서비스·게이트웨이·KMP 백엔드에 잘 맞습니다. 

fun main() {
    embeddedServer(Netty, port = 8080) { module() }.start(wait = true)
}

fun Application.module() {
    install(ContentNegotiation) { json(Json { ignoreUnknownKeys = true }) }
    install(CallLogging) { level = Level.INFO }
    install(StatusPages) {
        exception<NotFoundException> { call, cause ->
            call.respond(HttpStatusCode.NotFound, ErrorResponse("NOT_FOUND", cause.message))
        }
        exception<Throwable> { call, cause ->
            log.error("처리되지 않은 예외", cause)
            call.respond(HttpStatusCode.InternalServerError, ErrorResponse("INTERNAL", null))
        }
    }
    install(Authentication) {
        jwt("auth-jwt") {
            verifier(jwtVerifier)
            validate { credential -> JWTPrincipal(credential.payload) }
        }
    }
    install(Resources)

    routing {
        get("/health") { call.respond(mapOf("status" to "UP")) }

        authenticate("auth-jwt") {
            route("/api/orders") {
                get {
                    val page = call.request.queryParameters["page"]?.toIntOrNull() ?: 1
                    call.respond(orderService.list(page))      // suspend 함수를 그냥 부른다
                }
                get("/{id}") {
                    val id = call.parameters["id"]?.toLongOrNull()
                        ?: throw BadRequestException("잘못된 id")
                    call.respond(orderService.get(id))
                }
                post {
                    val req = call.receive<CreateOrderRequest>()
                    call.respond(HttpStatusCode.Created, orderService.create(req))
                }
            }
        }

        // SSE · WebSocket 도 기본 지원
        webSocket("/ws/chat") {
            for (frame in incoming) {
                if (frame is Frame.Text) send(Frame.Text("echo: ${frame.readText()}"))
            }
        }
    }
}

스프링과 비교

항목Ktor
철학필요한 것만 install — 명시적
코루틴처음부터 코루틴 기반 (변환 계층이 없다)
기동 속도수백 ms — 스프링보다 훨씬 빠르다
DI내장 없음 → Koin
데이터 접근Exposed · jOOQ 등을 직접 조합
생태계스프링보다 좁다 (배치·시큐리티 등)
멀티플랫폼서버·클라이언트 모두 지원

언제 Ktor 를 고르나

  • 작은 서비스·게이트웨이 — 스프링의 무게가 부담일 때
  • KMP 프로젝트의 백엔드 — 클라이언트와 코드를 공유
  • 기동 속도가 중요 — 서버리스·짧은 수명 컨테이너
  • 코루틴을 온전히 — 트랜잭션·시큐리티 컨텍스트 마찰이 없다
반대로 스프링을 고를 이유 — 팀 경험, JPA·배치·시큐리티 생태계, 사내 표준. 대부분의 기업 서비스는 여전히 스프링이 합리적입니다.
TOOL 07

로깅 · 날짜와 시간

매일 쓰지만 잘못 쓰기도 쉬운 두 가지입니다. 로그는 성능과 보안에, 시간은 정확성에 직결됩니다.

kotlin-logging — 람다 기반

// implementation("io.github.oshai:kotlin-logging-jvm:7.0.3")
private val log = KotlinLogging.logger {}

class OrderService {
    fun place(cmd: Cmd) {
        // 람다라서 레벨이 꺼져 있으면 문자열을 만들지 않는다
        log.debug { "주문 시작 payload=${cmd.toJson()}" }   // 비용 0 (DEBUG 꺼져 있으면)

        // SLF4J 직접 쓸 때는 파라미터 형태로
        // log.info("주문 {}", id)     — "$id" 보다 낫다

        log.error(e) { "주문 실패 id=${cmd.id}" }           // 예외를 첫 인자로
    }
}

// 로그에 담지 말아야 할 것
// · 비밀번호 · 토큰 · 카드번호 · 주민번호
// · 전체 요청/응답 본문 (개인정보 + 용량)
fun String.maskCard() = replace(Regex("""\d{4}-?\d{4}-?\d{4}-?(\d{4})"""), "****-****-****-$1")

HTTP 클라이언트 비교

선택지가 넷입니다. 어떤 스택에 있는지와 멀티플랫폼 계획이 결정 요소입니다.

클라이언트블로킹고르는 이유
Ktor Client논블로킹멀티플랫폼 · 코루틴 기본 · 안드로이드·iOS·서버 공용
Retrofit + OkHttp논블로킹(코루틴)안드로이드 표준 · 인터페이스 선언 방식 · 자료가 가장 많다
Spring WebClient논블로킹스프링 스택 · 커넥션 풀·필터가 잘 갖춰져 있다
Spring RestClient블로킹MVC + 가상 스레드 · 가장 단순한 API
java.net.http.HttpClient둘 다의존성 0 — 간단한 도구·스크립트
// 같은 요청을 네 가지로

// ① Ktor
val order: Order = client.get("orders/$id") { header("X-Trace", traceId) }.body()

// ② Retrofit
@GET("orders/{id}") suspend fun get(@Path("id") id: Long): Order

// ③ WebClient (코루틴)
val order = webClient.get().uri("/orders/{id}", id).retrieve().awaitBody<Order>()

// ④ RestClient (블로킹 — 필요하면 withContext(IO))
val order = restClient.get().uri("/orders/{id}", id).retrieve().body<Order>()!!

// 공통으로 반드시 설정할 것
// · 연결·읽기 타임아웃 (기본값이 무한인 경우가 있다)
// · 커넥션 풀 크기와 maxIdleTime
// · 재시도 정책 (멱등한 요청만)
// · 로깅 (운영에서는 헤더만 · 개발에서는 본문까지)
공통 규칙 하나 — 도메인 예외로 변환하는 어댑터를 반드시 두세요. WebClientResponseException이나 HttpException이 서비스 계층까지 올라오면 클라이언트 라이브러리를 바꾸는 순간 전부 고쳐야 합니다.

Clikt — CLI 프레임워크

// implementation("com.github.ajalt.clikt:clikt:5.0.2")
class Migrate : CliktCommand(help = "주문 데이터를 새 스키마로 옮깁니다") {
    private val source by option("-s", "--source", help = "원본 DB URL").required()
    private val target by option("-t", "--target").required()
    private val batchSize by option("--batch").int().default(1000)
    private val dryRun by option("--dry-run").flag()

    override fun run() = runBlocking {
        echo("마이그레이션 시작 (batch=$batchSize, dryRun=$dryRun)")
        val total = migrate(source, target, batchSize, dryRun)
        echo("완료: ${total}건")
    }
}

class Cli : CliktCommand() { override fun run() = Unit }

fun main(args: Array<String>) = Cli().subcommands(Migrate(), Verify()).main(args)

// 실행 파일로 만들기
// · ./gradlew installDist → bin/ 에 실행 스크립트

코틀린 라이브러리의 호환성 함정

  • inline 함수 본문 변경 — 이미 펼쳐진 코드는 안 바뀝니다. 쓰는 쪽 재컴파일이 필요
  • 기본값 추가·변경 — $default 시그니처가 바뀔 수 있습니다
  • data class 에 프로퍼티 추가 — copy·componentN이 바뀝니다
  • 인터페이스에 메서드 추가 — 기본 구현이 있어도 구현체에 영향
  • sealed 에 자식 추가 — 쓰는 쪽 when이 깨집니다(의도된 동작)
시맨틱 버전을 지키세요 — 호환이 깨지면 메이저를 올리고, @Deprecated로 최소 한 버전은 예고합니다.
