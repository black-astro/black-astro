SCALE 01

트래픽을 숫자로 읽기

“트래픽이 많다”는 말로는 아무것도 결정할 수 없습니다. RPS · 지연 · 동시성 세 숫자로 바꿔야 서버를 몇 대 둘지가 나옵니다.

기본 계산 — 리틀의 법칙

동시 처리 수 = RPS × 평균 응답 시간(초)

예) 초당 2,000 요청 · 평균 응답 200ms
    → 동시에 처리 중인 요청 = 2000 × 0.2 = 400개

이 400 을 무엇으로 감당하는가?
  · MVC 블로킹     : 톰캣 스레드 400개 필요 (메모리 400MB+)
  · MVC + 가상스레드: 가상 스레드 400개 (가볍다) — 단 DB 커넥션은 별개
  · 코루틴/WebFlux : 코루틴 400개 (수백 KB)

그런데 진짜 한계는 대개 여기다
  · DB 커넥션 풀 50개 → DB 쿼리가 50ms 라면 초당 1,000 쿼리가 한계
  · 외부 API rate limit 100 RPS
  · Redis 커넥션 · 파일 디스크립터

    private val failed = registry.counter("orders.failed", "reason", "stock")

캐시 전략 — 2단 구성

가장 효과가 큰 최적화입니다. 로컬 캐시(Caffeine)와 분산 캐시(Redis)를 함께 쓰면 각각의 약점을 서로 메웁니다.

로컬 캐시는 서버마다 값이 다를 수 있다는 점을 감수할 수 있는 데이터에만 씁니다.
@Component
class ProductCache(
    private val redis: RedisCoroutinesCommands<String, String>,
    private val repo: ProductRepository,
    private val json: Json,
) {
    // 1단: 로컬 — 짧은 TTL (서버 간 불일치를 감수할 수 있는 시간)
    private val local: Cache<Long, Product> = Caffeine.newBuilder()
        .maximumSize(10_000)
        .expireAfterWrite(30.seconds.toJavaDuration())
        .recordStats()
        .build()

    suspend fun get(id: Long): Product? {
        local.getIfPresent(id)?.let { return it }                       // ① 로컬

        redis.get("product:$id")?.let { raw ->                          // ② Redis
            val p = json.decodeFromString<Product>(raw)
            local.put(id, p)
            return p
        }

        val p = repo.findAsync(id) ?: return null                       // ③ DB
        redis.setex("product:$id", 300, json.encodeToString(p))         // 5분
        local.put(id, p)
        return p
    }

    // 무효화 — Redis 삭제 + 다른 서버의 로컬 캐시도 비워야 한다
    suspend fun evict(id: Long) {
        redis.del("product:$id")
        local.invalidate(id)
        redis.publish("cache:evict", "product:$id")     // Pub/Sub 으로 전파
    }
}

SCALE 05

캐시 3대 재난

캐시는 잘 돌 때는 조용하다가, 무너질 때 한꺼번에 무너집니다. 세 가지 유형을 알고 대비하면 DB 가 캐시 때문에 죽는 사고를 막을 수 있습니다.

세 가지 모두 정상일 때는 안 보이다가 트래픽이 늘면 한꺼번에 터집니다.
@Component
class SafeCache(
    private val redis: RedisCoroutinesCommands<String, String>,
    private val json: Json,
) {
    private val locks = ConcurrentHashMap<String, Mutex>()

    // ① 스탬피드 방지 — 같은 키는 한 코루틴만 갱신
    suspend fun <T> get(
        key: String,
        ttl: Duration = 5.minutes,
        deserializer: KSerializer<T>,
        loader: suspend () -> T?,
    ): T? {
        redis.get(key)?.let { return decodeOrNull(it, deserializer) }

        val mutex = locks.computeIfAbsent(key) { Mutex() }
        return try {
            mutex.withLock {
                // 이중 검사 — 기다리는 동안 다른 코루틴이 채웠을 수 있다
                redis.get(key)?.let { return@withLock decodeOrNull(it, deserializer) }

                val value = loader()
                // ② 관통 방지 — null 도 짧게 캐시한다
                val ttlSec = if (value == null) 30 else jitter(ttl)
                redis.setex(key, ttlSec, value?.let { json.encodeToString(deserializer, it) } ?: NULL_MARK)
                value
            }
        } finally {
            locks.remove(key)
        }
    }

    // ③ 눈사태 방지 — TTL 에 ±20% 무작위를 섞는다
    private fun jitter(ttl: Duration): Long {
        val base = ttl.inWholeSeconds
        return base + Random.nextLong(-base / 5, base / 5)
    }

    // 캐시에서 꺼낸 문자열 → 값. "없음"으로 저장해 둔 표식이면 null 을 돌려준다
    private fun <T> decodeOrNull(raw: String, deserializer: KSerializer<T>): T? =
        if (raw == NULL_MARK) null else json.decodeFromString(deserializer, raw)

    companion object {
        // 값이 없다는 사실 자체를 캐시할 때 넣는 표식.
        // 실제 JSON 과 절대 겹치지 않도록 일부러 이상한 문자열을 씁니다
        private const val NULL_MARK = "__NULL__"
    }
}

// TTL 이 끝나기 전에 백그라운드로 갱신 (사용자는 항상 캐시 히트)
data class Cached<T>(val value: T, val refreshAt: Instant, val expireAt: Instant)

suspend fun <T> getWithRefreshAhead(key: String, loader: suspend () -> T): T {
    val cached = load<T>(key)
    if (cached != null) {
        if (Clock.System.now() > cached.refreshAt) {
            appScope.launch { runCatching { store(key, loader()) } }   // 비동기 갱신
        }
        return cached.value                                            // 즉시 반환
    }
    return loader().also { store(key, it) }
}
// Caffeine 은 refreshAfterWrite 로 이 동작을 기본 제공한다

// ② Redis 원자 연산 — DB 부하 없이 선착순 처리
suspend fun tryIssueCoupon(couponId: Long, userId: Long): Boolean {
    // 중복 참여 차단 (SET 은 원자적)
    val first = redis.sadd("coupon:$couponId:users", userId.toString()) == 1L
    if (!first) return false

    val left = redis.decr("coupon:$couponId:stock") ?: -1
    if (left < 0) {
        redis.incr("coupon:$couponId:stock")                       // 되돌린다
        redis.srem("coupon:$couponId:users", userId.toString())
        return false
    }
    // 실제 발급은 큐로 넘긴다 (DB 쓰기를 비동기화)
    outbox.publish(CouponIssued(couponId, userId))
    return true
}

// ③ Redisson 분산 락 — 코루틴에서
suspend fun withDistributedLock(key: String, block: suspend () -> Unit) {
    val lock = redisson.getLock(key)
    val acquired = lock.tryLockAsync(3, 10, TimeUnit.SECONDS).awaitSingle()   // 대기 3초, 임대 10초
    check(acquired) { "락 획득 실패: $key" }
    try {
        block()
    } finally {
        if (lock.isHeldByCurrentThread) lock.unlockAsync().awaitSingleOrNull()
    }
}
분산 락의 함정 — ① 임대 시간(TTL)이 지나면 다른 서버가 락을 얻습니다 — 작업이 그보다 오래 걸리면 동시에 실행됩니다. ② 락 안에서 외부 API 를 부르지 마세요 — 임대 시간을 넘기기 쉽습니다. ③ 락은 성능을 희생합니다 — 원자적 UPDATE 로 풀 수 있으면 그쪽이 훨씬 낫습니다. ④ Redis 단일 노드 장애 시 락이 두 번 나갈 수 있습니다(Redlock 논쟁) — 돈이 오가는 처리는 DB 제약조건으로 최종 방어선을 두세요.
SCALE 07

멱등성 — 중복 요청 막기

네트워크는 언제나 불확실합니다. 클라이언트가 같은 요청을 두 번 보내거나, 재시도가 이미 처리된 요청을 또 보냅니다. 두 번 눌러도 한 번만 처리돼야 합니다.

"재시도해도 안전한가"는 클라이언트가 아니라 서버가 보장해야 합니다 — 네트워크는 언제나 두 번 보냅니다.
// 1) 클라이언트가 요청마다 고유 키를 보낸다
@PostMapping("/orders")
suspend fun create(
    @RequestHeader("Idempotency-Key") key: String,
    @RequestBody @Valid req: CreateOrderRequest,
): OrderResponse = idempotent(key, req) { service.place(req.toCommand()) }

// 2) 처리 결과를 키와 함께 저장하고, 같은 키가 오면 저장된 결과를 돌려준다
@Component
class IdempotencyService(
    private val redis: RedisCoroutinesCommands<String, String>,
    private val json: Json,
) {
    suspend fun <T> execute(
        key: String,
        requestHash: String,
        serializer: KSerializer<T>,
        ttl: Duration = 24.hours,
        block: suspend () -> T,
    ): T {
        val redisKey = "idem:$key"

        // 이미 처리됐다면 저장된 응답을 돌려준다
        redis.get(redisKey)?.let { stored ->
            val record = json.decodeFromString<IdemRecord>(stored)
            require(record.requestHash == requestHash) { "같은 키로 다른 요청이 왔습니다" }
            return json.decodeFromString(serializer, record.response)
        }

        // 선점 — SET NX 로 '처리 중' 표시 (동시 요청 차단)
        val acquired = redis.set(redisKey, PROCESSING, SetArgs().nx().ex(60)) != null
        if (!acquired) throw ConflictException("같은 요청이 처리 중입니다")

        return try {
            val result = block()
            redis.setex(redisKey, ttl.inWholeSeconds,
                json.encodeToString(IdemRecord(requestHash, json.encodeToString(serializer, result))))
            result
        } catch (e: Throwable) {
            redis.del(redisKey)        // 실패하면 재시도할 수 있게 지운다
            throw e
        }
    }
}

  • 메시지 소비(Kafka) — 재전달이 기본이라 필수

소비 — 멱등하게

@Component
class OrderEventConsumer(
    private val notifier: Notifier,
    private val processed: ProcessedEventRepository,
) {
    @KafkaListener(topics = ["order-events"], groupId = "notification")
    fun consume(record: ConsumerRecord<String, String>, ack: Acknowledgment) {
        val eventId = record.headers().lastHeader("eventId")?.value()?.decodeToString()
            ?: "${record.topic()}-${record.partition()}-${record.offset()}"

        // ★ 재전달이 기본이므로 반드시 멱등하게
        if (processed.exists(eventId)) { ack.acknowledge(); return }

        val event = json.decodeFromString<OrderEvent>(record.value())
        when (event) {
            is OrderEvent.Placed -> notifier.sendOrderPlaced(event)
            is OrderEvent.Paid -> notifier.sendPaid(event)
            is OrderEvent.Canceled -> notifier.sendCanceled(event)
        }

        processed.save(eventId)
        ack.acknowledge()          // 수동 커밋 — 처리 성공 후에만
    }
}
운영 포인트설정 · 판단

전달 보장기본은 at-least-once(중복 가능) → 소비자가 멱등해야 한다
커밋 시점enable-auto-commit: false + 수동 ack — 처리 전에 커밋되면 유실
재시도·DLQ실패 메시지를 무한 재시도하면 파티션이 멈춥니다 — 몇 번 후 DLQ 로

@Component
class OutboxPublisher(
    private val outboxRepo: OutboxRepository,
    private val kafka: KafkaTemplate<String, String>,
) : DisposableBean {
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO + CoroutineName("outbox"))

    @PostConstruct
    fun start() = scope.launch {
        while (isActive) {
            runCatching { publishBatch() }.onFailure { log.error("아웃박스 발행 실패", it) }
            delay(500)
        }
    }

    @Transactional
    fun publishBatch() {
        // 여러 인스턴스가 함께 돌아도 안전하게 (SKIP LOCKED)
        val events = outboxRepo.findUnpublishedForUpdate(limit = 100)
        events.forEach { e ->
            kafka.send("order-events", e.aggregateId, e.payload).get(3, TimeUnit.SECONDS)
            e.published = true
        }
    }

    override fun destroy() { scope.cancel() }
}

// 분산 환경에서는 2단계 커밋 대신 '보상 트랜잭션'을 쓴다
// 주문 → 재고 차감 → 결제 → 배송 요청
// 결제가 실패하면? 재고를 되돌린다 (보상)

sealed interface SagaStep {
    suspend fun execute(ctx: SagaContext)
    suspend fun compensate(ctx: SagaContext)
}

class ReserveStock(private val client: StockClient) : SagaStep {
    override suspend fun execute(ctx: SagaContext) { client.reserve(ctx.orderId, ctx.items) }
    override suspend fun compensate(ctx: SagaContext) { client.release(ctx.orderId) }
}

suspend fun runSaga(steps: List<SagaStep>, ctx: SagaContext) {
    val done = mutableListOf<SagaStep>()
    try {
        steps.forEach { it.execute(ctx); done += it }
    } catch (e: Exception) {
        done.reversed().forEach {                       // 역순으로 보상
            runCatching { it.compensate(ctx) }
                .onFailure { c -> log.error("보상 실패 — 수동 개입 필요", c) }
        }
        throw e
    }
}

현실적인 조언

  • 사가는 복잡합니다. 서비스를 나누기 전에 정말 나눠야 하는지 먼저 물으세요 — 한 DB 트랜잭션으로 끝나면 그게 최선입니다
  • 보상이 불가능한 작업(메일 발송·외부 결제 승인)은 순서를 뒤로 미룹니다
  • 보상 실패는 반드시 알림으로 — 사람이 개입해야 하는 상태입니다
  • Debezium(CDC)을 쓰면 아웃박스 폴링 없이 DB 변경 로그에서 바로 이벤트를 만들 수 있습니다
최종 일관성을 사용자에게 설명하는 것도 설계입니다 — “주문이 접수되었습니다(처리 중)”처럼 상태를 드러내면 기술적 지연이 자연스러운 UX 가 됩니다.

// ② 부하 차단(load shedding) — 감당 못 할 요청은 빨리 거절한다
@Component
class LoadShedder(private val registry: MeterRegistry) {
    private val inFlight = AtomicInteger()
    private val max = 500

    suspend fun <T> execute(block: suspend () -> T): T {
        if (inFlight.get() >= max) {
            registry.counter("requests.shed").increment()
            throw TooManyRequestsException("잠시 후 다시 시도해 주세요")   // 429
        }
        inFlight.incrementAndGet()
        return try { block() } finally { inFlight.decrementAndGet() }
    }
}
// 느리게 다 받아 주는 것보다 '빨리 거절'하는 편이 전체적으로 낫다
// (대기 큐가 길어지면 이미 포기한 사용자의 요청까지 처리하게 된다)

거절할 때는 Retry-After 헤더를 같이 주세요 — 안 주면 클라이언트가 곧바로 다시 몰려옵니다.

Rate Limit — 토큰 버킷

// Redis Lua 스크립트로 원자적으로 처리
private val script = """
    local key = KEYS[1]
    local limit = tonumber(ARGV[1])
    local window = tonumber(ARGV[2])
    local current = redis.call('INCR', key)
    if current == 1 then redis.call('EXPIRE', key, window) end
    return current
""".trimIndent()

suspend fun allow(userId: Long, limit: Int = 100, windowSec: Int = 60): Boolean {
    val count = redis.eval<Long>(script, ScriptOutputType.INTEGER,
        arrayOf("rate:$userId:${epochMinute()}"), limit.toString(), windowSec.toString())
    return (count ?: 0) <= limit
}

// 초과하면 429 + Retry-After 헤더
@Component
class RateLimitFilter(private val limiter: RateLimiter) : WebFilter {
    override fun filter(exchange: ServerWebExchange, chain: WebFilterChain) = mono {
        val userId = exchange.userId() ?: return@mono chain.filter(exchange).awaitFirstOrNull()
        if (!limiter.allow(userId)) {
            exchange.response.statusCode = HttpStatus.TOO_MANY_REQUESTS
            exchange.response.headers.set("Retry-After", "60")
            return@mono exchange.response.setComplete().awaitFirstOrNull()
        }
        chain.filter(exchange).awaitFirstOrNull()
    }
}

  • 나쁜 알람 — “CPU 80%”(그래서 사용자가 불편한가?), “디스크 사용량 70%”
  • 사용자가 느끼는 것에 걸어야 대응할 이유가 분명해집니다
  • 알람이 자주 울리면 사람이 무시하게 됩니다 — 임계값 관리가 중요

k6 로 시나리오 짜기

// load-test.js
import http from 'k6/http';
import { check, sleep } from 'k6';

export const options = {
  stages: [
    { duration: '2m', target: 200 },   // 서서히 올린다
    { duration: '5m', target: 200 },   // 유지 — 여기서 메모리 누수가 드러난다
    { duration: '2m', target: 1000 },  // 급증 (스파이크)
    { duration: '3m', target: 0 },     // 감소
  ],
  thresholds: {
    http_req_duration: ['p(95)<500', 'p(99)<1500'],
    http_req_failed: ['rate<0.01'],
  },
};

export default function () {
  const res = http.get('https://api.example.com/orders?page=1', {
    headers: { Authorization: `Bearer ${__ENV.TOKEN}` },
  });
  check(res, { 'status 200': (r) => r.status === 200 });
  sleep(1);
}
