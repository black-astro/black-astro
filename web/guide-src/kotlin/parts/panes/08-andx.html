ANDX 01

아키텍처 · 멀티모듈

화면이 열 개를 넘어가면 구조가 필요해집니다. 구글이 권장하는 3계층(UI · Domain · Data)과 기능별 모듈이 지금의 표준입니다.

도메인 계층은 선택입니다 — 여러 리포지터리를 조합하거나 규칙이 복잡할 때만 만드세요.

멀티모듈 구조

:app                        앱 조립 · 네비게이션 그래프
:core:designsystem          테마 · 공통 컴포넌트
:core:common                유틸 · Result · Dispatcher
:core:network               Retrofit · DTO
:core:database              Room · Entity
:core:data                  Repository 구현 (network + database)
:core:model                 도메인 모델 (순수 코틀린)

:feature:orders             주문 화면 + ViewModel
:feature:profile
:feature:search

의존 규칙
  feature → core:data → core:network/database
  feature 끼리는 서로 의존하지 않는다 (앱 모듈이 연결)
모듈을 나누는 이유는 빌드 속도와 경계입니다. 기능 모듈이 서로를 못 부르게 하면 의존성이 스파게티가 되는 것을 구조가 막습니다. (처음부터 다 나눌 필요는 없습니다 — 화면 10개쯤부터 고려)

@Composable
fun DonutChart(values: List<Float>, colors: List<Color>) {
    val total = values.sum().takeIf { it > 0f } ?: return
    Canvas(Modifier.size(160.dp)) {
        var start = -90f
        values.forEachIndexed { i, v ->
            val sweep = v / total * 360f
            drawArc(
                color = colors[i % colors.size],
                startAngle = start,
                sweepAngle = sweep - 2f,
                useCenter = false,
                style = Stroke(width = 28f, cap = StrokeCap.Round),
            )
            start += sweep
        }
    }
}

커스텀 레이아웃

// 태그를 줄바꿈하며 배치하는 FlowRow 를 직접 만들어 보기
@Composable
fun SimpleFlowRow(
    modifier: Modifier = Modifier,
    spacing: Dp = 8.dp,
    content: @Composable () -> Unit,
) {
    Layout(content = content, modifier = modifier) { measurables, constraints ->
        val gap = spacing.roundToPx()
        val placeables = measurables.map { it.measure(constraints.copy(minWidth = 0)) }

        var x = 0; var y = 0; var rowHeight = 0
        val positions = placeables.map { p ->
            if (x + p.width > constraints.maxWidth) { x = 0; y += rowHeight + gap; rowHeight = 0 }
            val pos = IntOffset(x, y)
            x += p.width + gap
            rowHeight = maxOf(rowHeight, p.height)
            pos
        }

        layout(constraints.maxWidth, y + rowHeight) {
            placeables.forEachIndexed { i, p -> p.place(positions[i]) }
        }
    }
}
// 실무에서는 androidx.compose.foundation.layout.FlowRow 를 쓰면 됩니다 —
// 원리를 알면 특수한 배치가 필요할 때 직접 만들 수 있습니다.

Paging — 무한 스크롤

목록이 수천 건이 되면 한 번에 불러올 수 없습니다. Paging 3 는 스크롤에 맞춰 페이지를 요청하고, 캐시하고, 재시도까지 처리합니다.

우리가 직접 쓰는 것은 PagingSource 와 LazyColumn 둘뿐입니다 — 가운데는 라이브러리 몫입니다.
// 1) PagingSource — 어디서 어떻게 가져오나
class OrderPagingSource(
    private val api: OrderApi,
    private val query: String,
) : PagingSource<Int, OrderDto>() {

    override suspend fun load(params: LoadParams<Int>): LoadResult<Int, OrderDto> =
        try {
            val page = params.key ?: 1
            val response = api.list(page = page, size = params.loadSize, q = query)
            LoadResult.Page(
                data = response.items,
                prevKey = if (page == 1) null else page - 1,
                nextKey = if (response.items.isEmpty()) null else page + 1,
            )
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            LoadResult.Error(e)
        }

    override fun getRefreshKey(state: PagingState<Int, OrderDto>): Int? =
        state.anchorPosition?.let { state.closestPageToPosition(it)?.prevKey?.plus(1) }
}

// 2) ViewModel
@HiltViewModel
class OrderListViewModel @Inject constructor(private val api: OrderApi) : ViewModel() {
    private val query = MutableStateFlow("")

    val orders: Flow<PagingData<OrderUi>> = query

        .flatMapLatest { q ->
            Pager(PagingConfig(pageSize = 20, prefetchDistance = 5)) {
                OrderPagingSource(api, q)

        .cachedIn(viewModelScope)        // ★ 회전 시 재요청 방지

// 3) 화면
@Composable
fun OrderList(vm: OrderListViewModel = hiltViewModel()) {
    val items = vm.orders.collectAsLazyPagingItems()

    LazyColumn {
        items(items.itemCount, key = items.itemKey { it.id }) { index ->
            items[index]?.let { OrderCard(it) }
        }

        when (val append = items.loadState.append) {
            is LoadState.Loading -> item { LoadingRow() }
            is LoadState.Error -> item { RetryRow(append.error) { items.retry() } }
            else -> Unit
        }
    }

    when (items.loadState.refresh) {
        is LoadState.Loading -> FullScreenLoading()
        is LoadState.Error -> FullScreenError { items.refresh() }
        else -> Unit
    }
}

백그라운드 작업 — WorkManager

앱이 꺼져 있어도 언젠가는 반드시 실행되어야 하는 작업이 있습니다 — 업로드, 동기화, 정리. 최신 안드로이드는 백그라운드 실행을 강하게 제한하므로 WorkManager 가 사실상 유일한 정답입니다. 

@HiltWorker
class SyncWorker @AssistedInject constructor(
    @Assisted context: Context,
    @Assisted params: WorkerParameters,
    private val repo: OrderRepository,
) : CoroutineWorker(context, params) {      // ★ suspend 로 동작한다

    override suspend fun doWork(): Result = withContext(Dispatchers.IO) {
        try {
            repo.refresh().getOrThrow()
            Result.success()
        } catch (e: IOException) {
            if (runAttemptCount < 3) Result.retry() else Result.failure()
        } catch (e: Exception) {
            Result.failure(workDataOf("error" to e.message))
        }
    }

    // 오래 걸리는 작업이면 알림을 띄워 전면 서비스로 승격
    override suspend fun getForegroundInfo(): ForegroundInfo =
        ForegroundInfo(NOTI_ID, buildNotification("동기화 중"))
}

// 예약
val request = PeriodicWorkRequestBuilder<SyncWorker>(6, TimeUnit.HOURS)
    .setConstraints(

            .setRequiredNetworkType(NetworkType.UNMETERED)   // 와이파이에서만
            .setRequiresBatteryNotLow(true)

    .setBackoffCriteria(BackoffPolicy.EXPONENTIAL, 30, TimeUnit.SECONDS)
    .build()

fun showOrderNotification(context: Context, orderId: Long, title: String) {
    val intent = Intent(Intent.ACTION_VIEW,
        "https://app.example.com/orders/$orderId".toUri(),
        context, MainActivity::class.java)

    val pending = TaskStackBuilder.create(context).run {
        addNextIntentWithParentStack(intent)
        getPendingIntent(orderId.toInt(),
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE)  // ★ 필수
    }

    val noti = NotificationCompat.Builder(context, "order")
        .setSmallIcon(R.drawable.ic_noti)
        .setContentTitle(title)
        .setContentText("탭하면 주문 상세로 이동합니다")
        .setContentIntent(pending)
        .setAutoCancel(true)
        .setPriority(NotificationCompat.PRIORITY_HIGH)
        .build()

    NotificationManagerCompat.from(context).notify(orderId.toInt(), noti)
}

모바일 앱은 사용자 기기 위에서 돌기 때문에 서버와 신뢰 모델이 다릅니다. 토큰 저장, 갱신, 그리고 “앱은 신뢰할 수 없다”는 전제가 핵심입니다. 

// 1) 토큰 저장 — 암호화된 저장소
class TokenStore @Inject constructor(@ApplicationContext context: Context) {
    private val prefs = EncryptedSharedPreferences.create(
        context, "auth",
        MasterKey.Builder(context).setKeyScheme(MasterKey.KeyScheme.AES256_GCM).build(),
        EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
        EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM,
    )
    var accessToken: String?
        get() = prefs.getString("access", null)
        set(v) = prefs.edit { putString("access", v) }
    var refreshToken: String?
        get() = prefs.getString("refresh", null)
        set(v) = prefs.edit { putString("refresh", v) }
}

// 2) 자동 갱신 — OkHttp Authenticator (401 을 받으면 한 번 갱신하고 재시도)
class TokenAuthenticator @Inject constructor(
    private val store: TokenStore,
    private val authApi: Provider<AuthApi>,
) : Authenticator {
    private val mutex = Mutex()

    override fun authenticate(route: Route?, response: Response): Request? {
        if (response.request.header("Authorization") == null) return null
        if (responseCount(response) >= 2) return null          // 무한 루프 방지

        return runBlocking {
            mutex.withLock {                                     // 동시 401 을 한 번만 처리
                val current = store.accessToken
                val requestToken = response.request.header("Authorization")?.removePrefix("Bearer ")
                if (current != null && current != requestToken) {
                    // 다른 요청이 이미 갱신했다
                    return@withLock response.request.newBuilder()
                        .header("Authorization", "Bearer $current").build()
                }
                val refreshed = runCatching {
                    authApi.get().refresh(RefreshRequest(store.refreshToken ?: return@runCatching null))
                }.getOrNull() ?: run { store.clear(); return@withLock null }

                store.accessToken = refreshed.accessToken
                store.refreshToken = refreshed.refreshToken
                response.request.newBuilder()
                    .header("Authorization", "Bearer ${refreshed.accessToken}").build()
            }
        }
    }
}

성능 ① — 재구성 최적화

Compose 앱이 버벅이는 원인의 대부분은 필요 없는 재구성입니다. 원인을 추측하지 말고 측정한 뒤 고칩니다.

목표는 리포트에 skippable 이 찍히는 것입니다 — 그 한 단어가 곧 성능입니다.

측정 — Layout Inspector 와 컴파일러 리포트

    metricsDestination = layout.buildDirectory.dir("compose_metrics")
}
// ./gradlew assembleRelease

//   restartable skippable fun OrderCard(...)      ← 좋다
//   restartable fun OrderCard(...)                 ← skippable 이 아니다 (문제)
“skippable” 이 목표입니다 — 파라미터가 그대로면 그 컴포저블을 건너뜁니다. 불안정한 타입(List·인터페이스·다른 모듈 클래스)이 파라미터에 있으면 skippable 이 깨집니다.

고치는 방법

// ① 불안정 타입 → 안정 타입
@Immutable data class OrderUi(val id: Long, val title: String)
// List → ImmutableList (kotlinx-collections-immutable)
fun OrderList(orders: ImmutableList<OrderUi>)

// ② 상태 읽기를 늦춘다 — 값 대신 람다를 넘긴다
// ❌ 스크롤할 때마다 부모까지 재구성
Header(scrollOffset = listState.firstVisibleItemScrollOffset)
// ✅ 실제로 읽는 곳에서만 재구성
Header(scrollOffset = { listState.firstVisibleItemScrollOffset })

// ③ 람다를 안정적으로
onClick = viewModel::onItemClick        // 매번 새 람다를 만들지 않는다

// ④ 무거운 계산은 캐싱
val formatted = remember(order.total) { currencyFormat.format(order.total) }

// ⑤ 목록에 key 부여 (필수)
items(orders, key = { it.id }) { ... }
과잉 최적화 주의 — 모든 데이터 클래스에 @Immutable을 붙이는 것은 거짓말이 될 수 있습니다(내부에 가변 필드가 있으면 화면이 갱신되지 않습니다). 측정해서 문제가 확인된 곳부터 고치세요.

class FakeOrderRepository : OrderRepository {
    private val state = MutableStateFlow<List<Order>>(emptyList())
    var failNext = false

    override fun observe(): Flow<List<Order>> = state

    override suspend fun refresh(): Result<Unit> =
        if (failNext) Result.failure(IOException())
        else { state.value = listOf(order()); Result.success(Unit) }

    fun emit(orders: List<Order>) { state.value = orders }
}
