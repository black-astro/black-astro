ADV 01

제네릭 — 타입을 인자로 받기

List<String>의 꺾쇠가 제네릭입니다. “여기엔 String만 들어간다”를 컴파일러에게 알려 주는 장치이고, 캐스팅 코드를 지우는 것이 목적입니다. 

// 타입 파라미터 T 를 받는 클래스
class Box<T>(private var item: T) {
    fun get(): T = item
    fun set(v: T) { item = v }
}

val b = Box("hello")     // Box<String> 으로 추론
val s: String = b.get()  // 캐스팅이 필요 없다

// 제네릭 함수 — 함수 이름 '앞'에 <T> 를 적는다
fun <T> firstOrDefault(list: List<T>, default: T): T =
    list.firstOrNull() ?: default

// 여러 개
fun <K, V> Map<K, V>.reverse(): Map<V, K> =
    entries.associate { (k, v) -> v to k }

// 확장 함수의 제네릭
fun <T : Comparable<T>> List<T>.secondLargest(): T? =
    distinct().sortedDescending().getOrNull(1)

제약(bound) — 아무 타입이나 받지 않기

// 상한 하나
fun <T : Number> sum(list: List<T>): Double =
    list.sumOf { it.toDouble() }

// 여러 제약은 where 로
fun <T> process(items: List<T>): List<T>
        where T : Comparable<T>,
              T : CharSequence =
    items.filter { it.isNotEmpty() }.sorted()

// 기본 상한은 Any? — null 을 막으려면 Any 를 적는다
fun <T : Any> requireAll(list: List<T?>): List<T> =
    list.map { it ?: error("null 원소 발견") }
제약을 붙이면 그 타입의 멤버를 쓸 수 있게 됩니다. T : Number라고 적었기 때문에 toDouble()을 부를 수 있는 것입니다 — 제약이 없으면 Any?의 멤버(equals·hashCode·toString)만 보입니다.

실무 형태 — 공통 리포지터리

interface Entity { val id: Long }

interface CrudRepository<T : Entity> {
    fun findById(id: Long): T?
    fun save(entity: T): T
    fun findAll(): List<T>
}

// 공통 동작을 기본 구현으로
fun <T : Entity> CrudRepository<T>.findOrThrow(id: Long): T =
    findById(id) ?: throw NoSuchElementException("없음: $id")

// 사용
class UserRepo : CrudRepository<User> { /* ... */ }
val user: User = userRepo.findOrThrow(1)   // 타입이 User 로 그대로

타입 파라미터의 기본값은 없습니다 — 대신 쓰는 방법

// 코틀린에는 타입 파라미터 기본값이 없다. 오버로딩이나 typealias 로 푼다
typealias StringBox = Box<String>

// 팩토리 함수로 기본형을 제공하는 방식이 더 흔하다
class Cache<K, V>(private val max: Int) { /* ... */ }
fun defaultCache() = Cache<String, Any>(max = 1000)

// 타입 추론이 안 되는 자리에서는 명시적으로 적어 준다
val empty = emptyList<User>()          // 타입을 못 정하므로 필요
val map = mutableMapOf<Long, User>()
ADV 02

변성 — in · out

List<Dog>를 List<Animal> 자리에 넣어도 될까요? 이 질문의 답이 변성(variance)입니다. 코틀린은 이것을 선언하는 쪽에서 한 번 정하게 합니다 — 자바의 와일드카드가 호출부마다 등장하던 것과 다릅니다.

외우는 법 — PECS: Producer는 out(extends), Consumer는 in(super).

out — 꺼내기만 하는 타입

open class Animal(val name: String)
class Dog(name: String) : Animal(name)

// 코틀린 표준 List 선언: public interface List<out E>
val dogs: List<Dog> = listOf(Dog("바둑이"))
val animals: List<Animal> = dogs      // ✅ out 이라 대입된다

// 내가 만드는 생산자 타입에도 붙인다
interface Producer<out T> {
    fun produce(): T
    // fun consume(item: T)   // ❌ 컴파일 에러 — out 은 '인자'로 못 쓴다
}

// 반환 위치에만 T 가 나타나면 out 을 붙일 수 있다
class Repo<out T>(private val items: List<T>) {
    fun all(): List<T> = items
}

in — 넣기만 하는 타입

// Comparator 선언: public interface Comparator<in T>
val animalCmp = Comparator<Animal> { a, b -> a.name.compareTo(b.name) }
val dogCmp: Comparator<Dog> = animalCmp   // ✅ in 이라 대입된다
// 동물을 비교할 줄 알면 개도 비교할 수 있다 — 당연한 이야기다

interface Consumer<in T> {
    fun consume(item: T)
    // fun produce(): T      // ❌ in 은 '반환'에 못 쓴다
}

// 함수 타입이 대표적 — (in P) -> (out R)
// public interface Function1<in P1, out R>
val f: (Animal) -> String = { it.name }
val g: (Dog) -> Any = f              // ✅ 인자는 넓게, 반환은 좁게

사용 지점 변성과 스타 프로젝션

// 선언 지점에 변성을 못 붙이는 타입(MutableList)은 '쓰는 자리'에서 지정한다
fun copy(from: MutableList<out Any>, to: MutableList<Any>) {
    // from 에서는 꺼내기만 가능 (자바의 List<? extends Object>)
    to.addAll(from)
}

fun fill(dest: MutableList<in Dog>, value: Dog) {
    dest.add(value)     // Dog 또는 그 조상 리스트에 넣을 수 있다
}

// 스타 프로젝션 — 타입 인자를 모를 때
fun printSize(list: List<*>) {         // List<out Any?> 와 비슷
    println(list.size)                 // 크기는 알 수 있다
    val first: Any? = list.firstOrNull()  // 꺼내면 Any?
    // list.add(...)  ← 애초에 List 라 없다. MutableList<*> 면 넣기가 막힌다
}

// 실무에서 자주 만나는 자리 — 리플렉션 · 제네릭 컨테이너 검사
fun isEmptyCollection(c: Collection<*>) = c.isEmpty()
실전에서 기억할 한 줄 — 내가 만든 제네릭 타입에서 T가 반환에만 나오면 out, 인자에만 나오면 in을 붙이세요. 컴파일러가 위치를 검사해 주므로 잘못 붙이면 바로 에러가 납니다. 둘 다 나오면 그냥 두면 됩니다 — 대부분의 클래스가 그렇습니다.
ADV 03

타입 소거와 reified

JVM에서는 실행 시점에 List<String>과 List<Int>가 구분되지 않습니다(타입 소거). 코틀린은 inline + reified로 이 한계를 우회하는 유일한 실용적 방법을 제공합니다.

reified 는 마법이 아니라 인라인 복사의 부산물입니다 — 그래서 큰 함수에 붙이면 바이트코드가 불어납니다.
소거 때문에 안 되는 것들
fun <T> check(x: Any): Boolean {
    // return x is T      // ❌ Cannot check for instance of erased type
    return false
}

fun <T> create(): T {
    // return T()         // ❌ 생성자를 부를 수 없다
    TODO()
}

// 실행 시점에는 이렇게 보인다
val a: List<String> = listOf("a")
val b: List<Int> = listOf(1)
println(a.javaClass == b.javaClass)   // true — 둘 다 그냥 ArrayList
inline + reified 로 해결
inline fun <reified T> check(x: Any): Boolean = x is T   // ✅

println(check<String>("hi"))   // true
println(check<Int>("hi"))      // false

// 왜 되나: inline 이라 호출 자리에 코드가 '펼쳐지고',
// 그때 T 가 실제 타입(String)으로 치환된다.
// 즉 컴파일러가 x is String 을 그 자리에 써 넣는 것과 같다

inline fun <reified T> List<*>.filterIsInstance2(): List<T> =
    filter { it is T }.map { it as T }

실무에서 reified 가 빛나는 자리

JSON 역직렬화 · API 클라이언트

// 자바 스타일 — Class 를 인자로 받아야 했다
fun <T> parse(json: String, clazz: Class<T>): T = mapper.readValue(json, clazz)
val u1 = parse(json, User::class.java)

// 코틀린 — 타입만 적으면 된다
inline fun <reified T> parse(json: String): T =
    mapper.readValue(json, object : TypeReference<T>() {})

val u2: User = parse(json)                 // 추론까지 된다
val list = parse<List<User>>(json)         // 제네릭 중첩도 가능

// 실제 라이브러리들이 전부 이 방식이다
// kotlinx.serialization: Json.decodeFromString<User>(text)
// Ktor Client:           client.get(url).body<User>()
// Spring 6:              restClient.get().retrieve().body<User>()

안드로이드 · DI · 로거

// 화면 이동 (안드로이드)
inline fun <reified T : Activity> Context.start() =
    startActivity(Intent(this, T::class.java))
start<DetailActivity>()

// 의존성 조회
inline fun <reified T> Injector.get(): T = get(T::class)
val repo: UserRepo = injector.get()

// 클래스별 로거 — 자바에서 늘 쓰던 상용구가 사라진다
inline fun <reified T> T.logger(): Logger =
    LoggerFactory.getLogger(T::class.java)

class OrderService {
    private val log = logger()
}
제약 — reified는 inline 함수에서만 쓸 수 있습니다. 그래서 클래스의 타입 파라미터에는 붙일 수 없고(class Box<reified T> 불가), 자바에서는 이 함수를 부를 수 없습니다. 또 inline 이라 private 멤버에 접근하는 코드를 공개 inline 함수 안에 둘 수 없습니다.
ADV 04

inline · noinline · crossinline

람다를 인자로 받으면 보통 객체가 하나 생깁니다. 반복문 안이라면 수만 개가 생기죠. inline은 그 비용을 0으로 만드는 대신, 몇 가지 규칙을 따라오게 합니다.

표준 라이브러리의 let · run · apply · forEach · filter가 전부 inline 입니다 — 그래서 공짜로 쓸 수 있습니다.

언제 붙이나 · 언제 안 붙이나

상황판단
람다를 받는 작은 함수✅ 붙인다 (이게 inline 의 존재 이유)
reified 가 필요✅ 필수
람다를 안 받는 함수❌ 이득이 거의 없다 — IDE 도 경고한다
본문이 긴 함수❌ 호출 지점마다 복제되어 코드가 부푼다
공개 API + private 접근❌ 컴파일 에러 (펼쳐진 코드가 private 에 접근 불가)
inline 은 바이너리 호환성에 영향을 줍니다. 라이브러리를 배포한 뒤 inline 함수의 본문을 고치면, 쓰는 쪽을 다시 컴파일해야 반영됩니다 — 이미 펼쳐진 옛 코드가 남아 있기 때문입니다.

noinline · crossinline

// noinline — 이 람다만 펼치지 않는다 (객체로 남겨 저장·전달해야 할 때)
inline fun run2(
    before: () -> Unit,
    noinline after: () -> Unit,     // 나중에 쓰려고 보관해야 한다
) {
    before()
    handlers += after               // 펼쳐진 코드는 저장할 수 없다
}

// crossinline — 펼치되 '비지역 반환'만 금지한다
inline fun async2(crossinline block: () -> Unit) {
    Thread {
        block()      // 다른 실행 흐름 안에서 부른다
        // 여기서 return 이 허용되면 어디로 돌아가야 하나? → 그래서 금지
    }.start()
}

비지역 반환 — inline 만 되는 일

fun findAdmin(users: List<User>): User? {
    users.forEach {                      // forEach 는 inline 이다
        if (it.isAdmin) return it        // ✅ findAdmin 자체를 종료 (비지역 반환)
    }
    return null
}

// inline 이 아니었다면 이 return 은 컴파일 에러다
// (일반 람다에서는 자기 자신만 빠져나올 수 있다)

// 라벨로 람다만 빠져나오기
fun f(users: List<User>) {
    users.forEach {
        if (it.isBanned) return@forEach  // 이번 항목만 건너뛴다
        send(it)
    }
}

성능 — 실제로 얼마나 차이 나나

측정 없이 믿지 마세요

JIT는 대부분의 작은 람다 호출을 인라이닝해 버립니다. 수억 번 도는 루프가 아니면 체감 차이가 없는 경우가 많습니다. inline의 진짜 가치는 세 가지입니다.

  • reified를 쓸 수 있다
  • 비지역 반환이 된다 (제어 흐름 표현력)
  • 안드로이드처럼 객체 생성이 곧 GC 압력인 환경에서 이득

확인하는 법 — 바이트코드 보기

IntelliJ 메뉴
  Tools → Kotlin → Show Kotlin Bytecode
  → [Decompile] 버튼을 누르면 '자바로 치면 이런 코드'가 나온다

무엇을 보나
  · 람다가 클래스로 만들어졌는지 (Function0 · Function1)
  · inline 함수가 정말 펼쳐졌는지
  · data class 가 어떤 equals 를 만들었는지
  · suspend 함수가 어떤 상태 기계로 바뀌었는지  ← 코루틴 탭에서 다시 봅니다
코틀린을 깊이 이해하는 가장 빠른 길이 이 디컴파일 화면입니다. “이 문법이 실제로 무엇으로 바뀌는가”를 눈으로 보면 성능 이야기가 추측에서 사실로 바뀝니다.
ADV 05

위임 프로퍼티

by lazy의 by가 위임입니다. “이 프로퍼티의 읽기·쓰기를 다른 객체에게 맡긴다”는 선언으로, 캐싱·검증·변경 감지·설정 읽기 같은 반복되는 접근자 코드를 한 번만 쓰게 만듭니다.

위임은 "이 프로퍼티의 저장 방식을 통째로 갈아 끼우는" 문법입니다 — lazy 하나만 알아도 절반은 쓴 셈입니다.

표준 위임 넷

import kotlin.properties.Delegates

class Config(map: Map<String, Any>) {
    // 1) lazy — 처음 읽을 때 한 번만 계산
    val heavy: String by lazy { loadFromDisk() }

    // 2) observable — 바뀔 때마다 알림
    var status: String by Delegates.observable("NEW") { _, old, new ->
        log.info("상태 변경 {} → {}", old, new)
    }

    // 3) vetoable — 조건을 통과할 때만 반영
    var age: Int by Delegates.vetoable(0) { _, _, new -> new in 0..150 }

    // 4) 맵 위임 — 키 이름으로 값을 꺼낸다
    val name: String by map
    val port: Int by map
}

val c = Config(mapOf("name" to "서버1", "port" to 8080))
println(c.name)         // 서버1  (map["name"] 를 읽는다)
c.age = 200             // 무시된다 (vetoable 이 거부)
println(c.age)          // 0

notNull · 다른 프로퍼티에 위임

class Service {
    // lateinit 을 못 쓰는 기본 타입에 쓴다
    var port: Int by Delegates.notNull()
}

// 이름을 바꿀 때 — 옛 이름을 남기며 새 이름으로 위임 (2.0+)
class User(val fullName: String) {
    @Deprecated("name 대신 fullName")
    val name: String by this::fullName
}

// 최상위 프로퍼티에도 쓸 수 있다
val version: String by lazy {
    javaClass.getResource("/version.txt")?.readText() ?: "dev"
}
lazy 의 세 가지 모드 — 기본은 SYNCHRONIZED(잠금 사용), PUBLICATION(여러 번 계산될 수 있으나 하나만 채택), NONE(잠금 없음 · 단일 스레드 확실할 때만). 안드로이드 UI 스레드 전용 값이라면 NONE이 가장 빠릅니다.

직접 만들기 — 두 개의 연산자만 있으면 된다

import kotlin.reflect.KProperty

// 읽기 전용이면 getValue 하나, 쓰기까지면 setValue 도
class TrimDelegate {
    private var value: String = ""

    operator fun getValue(thisRef: Any?, property: KProperty<*>): String = value

    operator fun setValue(thisRef: Any?, property: KProperty<*>, newValue: String) {
        value = newValue.trim()
        println("[${property.name}] 저장: '$value'")
    }
}

class Form {
    var title: String by TrimDelegate()
}

val f = Form()
f.title = "   제목  "     // [title] 저장: '제목'
println("'${f.title}'")    // '제목'

// 실무 예 — 환경변수 읽기 위임
class EnvDelegate(private val key: String, private val default: String? = null) {
    operator fun getValue(thisRef: Any?, property: KProperty<*>): String =
        System.getenv(key) ?: default ?: error("환경변수 없음: $key")
}
fun env(key: String, default: String? = null) = EnvDelegate(key, default)

object AppConfig {
    val dbUrl: String by env("DB_URL")
    val port: String by env("PORT", "8080")
}

provideDelegate — 만들 때 검증하기

// 프로퍼티가 '만들어지는 시점'에 개입한다
class RequiredEnv(private val key: String) {
    operator fun provideDelegate(thisRef: Any?, prop: KProperty<*>): EnvDelegate {
        // 앱이 뜰 때 바로 검증 — 나중에 처음 읽을 때가 아니라
        require(System.getenv(key) != null) {
            "환경변수 $key 가 없습니다 (프로퍼티 ${prop.name})"
        }
        return EnvDelegate(key)
    }
}

설정 누락을 기동 시점에 잡을 수 있습니다 — 운영에서 특정 API가 호출될 때야 터지는 것보다 훨씬 낫습니다.

@JvmInline
value class Money(val won: Long) : Comparable<Money> {
    operator fun plus(o: Money) = Money(won + o.won)
    operator fun minus(o: Money) = Money(won - o.won)
    operator fun times(n: Int) = Money(won * n)
    operator fun unaryMinus() = Money(-won)
    override fun compareTo(other: Money) = won.compareTo(other.won)
    override fun toString() = "%,d원".format(won)
}

DSL 만들기

Gradle KTS, Compose, Ktor 라우팅, kotest — 문법처럼 보이는 이 코드들은 전부 수신 객체 람다 + 확장 함수 + 후행 람다 세 재료로 만들어집니다. 원리를 알면 직접 만들 수 있습니다.

Gradle의 dependencies { }도 결국 수신 객체 람다를 받는 평범한 함수입니다.

단계별로 만들어 보기 — HTML 빌더

// 1단계: 노드 구조
@DslMarker
annotation class HtmlDsl                      // 스코프 섞임 방지 (아래 설명)

@HtmlDsl
open class Tag(private val name: String) {
    private val children = mutableListOf<Any>()
    private val attrs = mutableMapOf<String, String>()

    protected fun <T : Tag> add(child: T, block: T.() -> Unit): T {
        child.block()                          // ← 수신 객체 람다 실행
        children += child
        return child
    }

    // 텍스트 노드를 단항 연산자로 (+"안녕")
    operator fun String.unaryPlus() { children += this }

    fun attr(k: String, v: String) { attrs[k] = v }

    override fun toString(): String {
        val a = attrs.entries.joinToString("") { " ${it.key}=\"${it.value}\"" }
        return "<$name$a>${children.joinToString("")}</$name>"
    }
}

class Html : Tag("html") { fun body(block: Body.() -> Unit) = add(Body(), block) }
class Body : Tag("body") {
    fun p(block: P.() -> Unit) = add(P(), block)
    fun ul(block: Ul.() -> Unit) = add(Ul(), block)
}
class P : Tag("p")
class Ul : Tag("ul") { fun li(block: Li.() -> Unit) = add(Li(), block) }
class Li : Tag("li")

// 2단계: 진입점
fun html(block: Html.() -> Unit): Html = Html().apply(block)

// 3단계: 쓰는 쪽 — 문법처럼 보인다
val page = html {
    body {
        p {
            attr("class", "lead")
            +"코틀린 DSL 입니다"
        }
        ul {
            listOf("하나", "둘", "셋").forEach { item ->
                li { +item }               // 반복문도 그냥 쓸 수 있다 (템플릿 엔진과 다른 점)
            }
        }
    }
}
println(page)
// <html><body><p class="lead">코틀린 DSL 입니다</p><ul><li>하나</li>…</ul></body></html>

ADV 08

시퀀스 — 지연 평가

리스트 연산은 단계마다 새 리스트를 만듭니다. Sequence는 원소 하나를 끝까지 흘려보낸 뒤 다음 원소로 갑니다 — 자바 Stream과 같은 방식이고, 무한 수열도 다룰 수 있습니다.

앞에서 끊는 연산(take · first · any)이 있을 때 시퀀스의 이득이 가장 큽니다.

만들기 · 무한 수열

// 컬렉션에서
val s1 = listOf(1, 2, 3).asSequence()

// 원소 나열
val s2 = sequenceOf(1, 2, 3)

// 규칙으로 무한 생성 — null 을 돌려주면 끝난다
val powers = generateSequence(1) { it * 2 }
println(powers.take(10).toList())   // [1,2,4,...,512]

// 파일 한 줄씩 (메모리에 다 올리지 않는다)
File("big.log").useLines { lines ->
    lines.filter { "ERROR" in it }.take(100).toList()
}

// yield 로 직접 만들기 — 코루틴 문법을 쓴다
val fib = sequence {
    var (a, b) = 0 to 1
    while (true) {
        yield(a)                 // 여기서 멈췄다가 다음에 이어서 실행된다
        a = b.also { b = a + b }
    }
}
println(fib.take(8).toList())     // [0,1,1,2,3,5,8,13]

중간 연산과 종단 연산

// 중간(Intermediate) — 시퀀스를 돌려준다. 아직 아무 일도 안 한다
// map · filter · take · drop · distinct · sortedBy(예외) · flatMap · onEach

// 종단(Terminal) — 여기서 비로소 실행된다
// toList · first · sum · count · any · all · forEach · fold · joinToString

val seq = (1..10).asSequence()
    .onEach { println("보는 중: $it") }    // 아직 출력 안 됨
    .filter { it % 2 == 0 }

println("아직 실행 전")
println(seq.toList())                      // 여기서 한꺼번에 실행

// ⚠️ sorted 계열은 '상태를 모아야' 하므로 전부 읽는다 (지연의 이점이 사라진다)
// ⚠️ 시퀀스는 기본적으로 '한 번만' 소비할 수 있는 것이 많다 (constrainOnce)

언제 쓰나 — 판단표

상황선택
원소 수천 개 이하List — 시퀀스는 객체 오버헤드가 있어 오히려 느립니다
수십만 건 + 단계 3개 이상Sequence
first·take(n)으로 앞에서 끊는다Sequence — 이득이 가장 크다
파일·네트워크 스트림Sequence (useLines)
무한 수열Sequence 만 가능
병렬 처리가 필요시퀀스는 병렬이 없습니다 → 자바 parallelStream 또는 코루틴
비동기 스트림 (DB · API)Flow — 코루틴 탭에서 다룹니다
Sequence 와 Flow 의 관계 — 둘 다 “원소를 하나씩 흘려보내는” 모델입니다. 차이는 중간에 기다릴 수 있느냐입니다. Sequence 는 동기(블로킹), Flow는 suspend가 가능해 DB 조회·HTTP 호출을 사이에 넣을 수 있습니다. “동기면 Sequence, 비동기면 Flow”로 외워 두면 됩니다.

ADV 10

typealias · value class

Long 두 개를 인자로 받는 함수는 언젠가 반드시 뒤바뀝니다. value class는 그 사고를 런타임 비용 없이 컴파일 단계에서 막습니다.

value class 는 공짜 타입 안전성입니다 — 다만 박싱되는 네 자리에서는 일반 클래스와 같아집니다.
원시 타입 집착 (primitive obsession)
fun transfer(fromId: Long, toId: Long, amount: Long) { }

// 6개월 뒤 어느 날
transfer(toId, fromId, amount)   // 컴파일 통과. 돈이 반대로 간다 💥

fun send(email: String, name: String) { }
send(name, email)                // 역시 통과
value class — 타입으로 구분
@JvmInline value class AccountId(val value: Long)
@JvmInline value class Money(val won: Long)

fun transfer(from: AccountId, to: AccountId, amount: Money) { }

// transfer(money, from, to)   // ❌ 컴파일 에러
transfer(AccountId(1), AccountId(2), Money(10_000))

// 바이트코드에서는 그냥 long 이다 — 래퍼 객체가 생기지 않는다

value class 의 규칙과 한계

  • 프로퍼티는 하나(val)만. @JvmInline이 필요합니다
  • 함수·계산 프로퍼티는 가질 수 있습니다 (검증 로직도 init에)
  • 박싱되는 경우가 있습니다 — 컬렉션에 담거나(List<Money>), 제네릭·nullable(Money?)로 쓰거나, 인터페이스로 다룰 때
  • 자바에서는 이름이 뒤틀립니다(맹글링) — 자바에서 부를 API 라면 주의
@JvmInline
value class Email(val value: String) {
    init { require("@" in value) { "이메일 형식 오류: $value" } }
    val domain: String get() = value.substringAfter("@")
}

ADV 11

계약 — contracts

if (s.isNullOrBlank()) return 다음 줄에서 s가 non-null로 스마트 캐스트되는 이유가 뭘까요? 컴파일러가 그 함수의 계약을 알고 있기 때문입니다. 이 계약은 직접 선언할 수도 있습니다. 

import kotlin.contracts.*

// 문제: 내가 만든 검사 함수는 스마트 캐스트를 못 만든다
fun isValid(s: String?): Boolean = !s.isNullOrBlank()

fun f(s: String?) {
    if (isValid(s)) {
        // println(s.length)   // ❌ 컴파일러는 s 가 non-null 인지 모른다
    }
}

// 해결: 계약을 선언한다
@OptIn(ExperimentalContracts::class)
fun isValid2(s: String?): Boolean {
    contract { returns(true) implies (s != null) }   // "true 를 돌려주면 s 는 null 이 아니다"
    return !s.isNullOrBlank()
}

fun g(s: String?) {
    if (isValid2(s)) {
        println(s.length)     // ✅ 스마트 캐스트가 된다
    }
}

두 종류의 계약

// 1) returns(...) implies (...) — 반환값이 알려 주는 사실
@OptIn(ExperimentalContracts::class)
fun Any?.isUser(): Boolean {
    contract { returns(true) implies (this@isUser is User) }
    return this is User
}

// 2) callsInPlace — 람다가 언제 몇 번 불리는지
@OptIn(ExperimentalContracts::class)
inline fun <T> measure(block: () -> T): T {
    contract { callsInPlace(block, InvocationKind.EXACTLY_ONCE) }
    val t = System.nanoTime()
    return block().also { println("${System.nanoTime() - t}ns") }
}

// EXACTLY_ONCE 를 선언했기 때문에 이게 컴파일된다
val name: String
measure { name = loadName() }     // val 을 람다 안에서 초기화
println(name)                     // '반드시 한 번' 실행됨을 컴파일러가 안다

프로퍼티 참조 · 애너테이션

// 프로퍼티 참조 — 리플렉션 없이도 이름을 얻는다

// 커스텀 애너테이션 만들고 읽기
@Target(AnnotationTarget.PROPERTY)
@Retention(AnnotationRetention.RUNTIME)     // ← 런타임에 읽으려면 필수
annotation class Masked(val keep: Int = 2)

data class Account(
    @Masked(keep = 4) val cardNo: String,
    val owner: String,
)

fun mask(obj: Any): Map<String, Any?> =
    obj::class.memberProperties.associate { p ->
        val ann = p.annotations.filterIsInstance<Masked>().firstOrNull()
        val v = p.getter.call(obj)
        p.name to if (ann != null && v is String)
            v.takeLast(ann.keep).padStart(v.length, '*') else v
    }

println(mask(Account("1234567812345678", "김현우")))
// {cardNo=************5678, owner=김현우}

ADV 13

애너테이션 · KSP — 빌드 때 코드 만들기

Room·Hilt·Moshi·kotlinx.serialization은 전부 빌드 시점에 코드를 생성합니다. 리플렉션 대신 실제 코드가 생기므로 빠르고, 네이티브 빌드에서도 안전합니다.

KSP는 KAPT의 후계자입니다 — 자바 스텁을 만들지 않아 빌드가 대략 두 배 빠릅니다.

애너테이션 정의 — 세 가지를 정한다

@Target(                            // ① 어디에 붙일 수 있나
    AnnotationTarget.CLASS,
    AnnotationTarget.FUNCTION,
    AnnotationTarget.PROPERTY,
)
@Retention(AnnotationRetention.RUNTIME)  // ② 언제까지 남나
// SOURCE: 컴파일하면 사라짐 (린트용)
// BINARY: 클래스 파일에 남지만 리플렉션 불가
// RUNTIME: 리플렉션으로 읽을 수 있음 (기본값)
@MustBeDocumented
annotation class Audited(val action: String = "")

// ③ 사용 지점 지정 — 코틀린은 프로퍼티가 여러 요소로 컴파일된다
class Dto(

    @get:JsonProperty    // getter 에

ADV 14

함수형 스타일

코틀린은 순수 함수형 언어가 아니지만, 불변 데이터 + 함수 합성 + 값으로서의 실패라는 핵심 도구는 전부 갖고 있습니다. 실무에서 어디까지 가져다 쓰면 이득인지가 이 절의 주제입니다.

함수 합성 · 부분 적용 · 커링

// 합성 — 표준 라이브러리에는 없으니 직접 만든다 (두 줄이면 된다)
infix fun <A, B, C> ((A) -> B).then(g: (B) -> C): (A) -> C = { a -> g(this(a)) }

val trim: (String) -> String = String::trim
val upper: (String) -> String = String::uppercase
val clean = trim then upper
println(clean("  hi  "))        // HI

// 부분 적용 — 인자를 미리 채운 새 함수
fun log(level: String, tag: String, msg: String) = println("[$level][$tag] $msg")
val debug: (String, String) -> Unit = { tag, msg -> log("DEBUG", tag, msg) }
debug("Order", "생성됨")

// 커링
fun sum3(a: Int) = { b: Int -> { c: Int -> a + b + c } }
println(sum3(1)(2)(3))          // 6

// 메모이제이션
fun <A, R> ((A) -> R).memoize(): (A) -> R {
    val cache = mutableMapOf<A, R>()
    return { a -> cache.getOrPut(a) { this(a) } }
}
val slowSquare = { n: Int -> Thread.sleep(100); n * n }.memoize()

성능 관련 오해 셋

  • “코틀린은 자바보다 느리다” — 같은 바이트코드로 갑니다. 차이가 나는 곳은 람다 객체·박싱·컬렉션 복사이고, 전부 피할 수 있습니다
  • “data class 는 무겁다” — 자바에서 손으로 쓰던 equals·hashCode와 같은 코드입니다
  • “전부 asSequence 로 바꾸면 빠르다” — 작은 컬렉션에서는 오히려 느립니다. 측정하고 바꾸세요
