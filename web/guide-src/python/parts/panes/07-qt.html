QT 01

PySide6란 · 설치

PyQt와 기능은 거의 같지만 라이선스가 다릅니다. 업무용이면 PySide6를 고르세요.

이 문서의 enum 표기 — PySide6 공식 문서는 스코프드 enum(Qt.AlignmentFlag.AlignRight) 표기로 전환했습니다. 구 표기(Qt.AlignRight)도 아직 동작하지만 자동완성에 안 뜨고 PyQt6로 옮기면 에러가 나므로 새 코드는 스코프드로 쓰세요.

이벤트 오버라이드

시그널로 안 잡히는 저수준 동작은 이벤트 메서드를 재정의합니다.

이벤트를 재정의할 때 가장 많이 겪는 사고는 super() 를 부르지 않아 기본 동작이 사라지는 것입니다.
from PySide6.QtWidgets import QWidget, QMessageBox
from PySide6.QtCore import Qt

class MainWindow(QWidget):

    def closeEvent(self, event):                 # 창 닫기 ★ 가장 많이 씀
        reply = QMessageBox.question(self, "종료", "정말 종료할까요?")
        if reply == QMessageBox.StandardButton.Yes:
            self.save_settings()
            event.accept()
        else:
            event.ignore()                       # 닫기 취소

    def keyPressEvent(self, event):              # 키 입력
        if event.key() == Qt.Key.Key_Escape:
            self.close()
        elif event.key() == Qt.Key.Key_F5:
            self.refresh()
        elif (event.modifiers() == Qt.KeyboardModifier.ControlModifier
              and event.key() == Qt.Key.Key_S):
            self.save()
        super().keyPressEvent(event)             # ★ 반드시 호출

    def resizeEvent(self, event):
        self.adjust_columns()
        super().resizeEvent(event)

    def dragEnterEvent(self, event):             # 파일 드래그 앤 드롭
        if event.mimeData().hasUrls():
            event.acceptProposedAction()

    def dropEvent(self, event):
        for url in event.mimeData().urls():
            self.load_file(url.toLocalFile())
드래그 앤 드롭 활성화 — self.setAcceptDrops(True)를 __init__에 넣어야 위 두 이벤트가 호출됩니다. “엑셀 파일을 창에 끌어다 놓으면 분석 시작” 같은 UX를 쉽게 만들 수 있습니다.

QThread — UI가 멈추는 진짜 이유

PySide6에서 가장 중요한 규칙입니다. 무거운 작업을 메인 스레드에서 돌리면 창이 하얗게 굳습니다. 직접 비교해 보세요.

왼쪽은 작업 중 스피너가 얼어붙습니다 — 화면을 다시 그릴 틈이 없기 때문입니다. 오른쪽은 작업이 별도 스레드에서 돌아 UI가 계속 반응합니다. 

from PySide6.QtCore import QThread, QObject, Signal, Slot

# ── ① 작업 클래스 (QObject) ───────────────────────────
class Worker(QObject):
    progress = Signal(int)
    finished = Signal(object)
    error    = Signal(str)

    def __init__(self, path):
        super().__init__()
        self.path = path
        self._stop = False

    @Slot()
    def run(self):
        try:
            import pandas as pd
            df = pd.read_csv(self.path)
            results = []
            total = len(df)
            for i, row in df.iterrows():
                if self._stop:            # 중단 요청 확인
                    break
                results.append(heavy_work(row))
                self.progress.emit(int((i+1) / total * 100))
            self.finished.emit(results)
        except Exception as e:
            self.error.emit(str(e))

    def stop(self):
        self._stop = True

# ── ② 메인 창에서 스레드 띄우기 ───────────────────────
def start_job(self, path):
    self.thread = QThread()                       # ★ self. 로 보관!
    self.worker = Worker(path)
    self.worker.moveToThread(self.thread)         # 작업을 스레드로 이사

    self.thread.started.connect(self.worker.run)
    self.worker.progress.connect(self.bar.setValue)
    self.worker.finished.connect(self.on_done)
    self.worker.error.connect(self.on_error)

    # 정리 (누수 방지)
    self.worker.finished.connect(self.thread.quit)
    self.worker.finished.connect(self.worker.deleteLater)
    self.thread.finished.connect(self.thread.deleteLater)

    self.run_btn.setEnabled(False)                # 중복 실행 방지
    self.thread.start()

@Slot(object)
def on_done(self, results):
    self.run_btn.setEnabled(True)
    self.table.setModel(PandasModel(pd.DataFrame(results)))

절대 규칙

  • UI는 오직 메인 스레드에서만 건드립니다. 워커에서 label.setText()를 직접 호출하면 크래시합니다
  • 워커 → UI 전달은 반드시 시그널로
  • QThread와 워커는 self.에 보관하세요. 지역변수면 함수 종료 시 파괴되어 조용히 멈춥니다
  • 스레드 안에서 time.sleep 대신 QThread.msleep()을 쓰면 중단 대응이 쉬워집니다

더 간단한 방법 — QThreadPool

from PySide6.QtCore import QRunnable, QThreadPool, Slot

class Task(QRunnable):
    def __init__(self, fn, *a):
        super().__init__()
        self.fn, self.a = fn, a
    @Slot()
    def run(self):
        self.fn(*self.a)

QThreadPool.globalInstance().start(Task(heavy_work, path))
선택 기준 — 진행률·중단이 필요하면 QThread + Worker, “그냥 백그라운드에서 한 번 돌리기”면 QRunnable이 간편합니다.

자동화와 결합할 때 — PyAutoGUI 루프를 워커에 넣고 진행률을 시그널로 보내면, 사용자는 진행 상황을 보면서 “중단” 버튼도 누를 수 있는 도구가 됩니다.
데스크탑 앱의 완성도는 기능보다 기다리는 동안 창이 살아 있는가에서 갈립니다.

import pandas as pd
from PySide6.QtCore import QAbstractTableModel, Qt, QModelIndex

class PandasModel(QAbstractTableModel):
    """DataFrame 을 QTableView 에 연결하는 어댑터"""

    def __init__(self, df: pd.DataFrame):
        super().__init__()
        self._df = df

    def rowCount(self, parent=QModelIndex()):
        return len(self._df)

    def columnCount(self, parent=QModelIndex()):
        return len(self._df.columns)

    def data(self, index, role=Qt.ItemDataRole.DisplayRole):
        if not index.isValid():
            return None
        value = self._df.iat[index.row(), index.column()]

        if role == Qt.ItemDataRole.DisplayRole:
            if pd.isna(value):
                return ""
            if isinstance(value, float):
                return f"{value:,.2f}"
            return str(value)

        if role == Qt.ItemDataRole.TextAlignmentRole:
            if isinstance(value, (int, float)):
                return int(Qt.AlignmentFlag.AlignRight | Qt.AlignmentFlag.AlignVCenter)

        if role == Qt.ItemDataRole.ForegroundRole:
            from PySide6.QtGui import QColor
            if isinstance(value, (int, float)) and value < 0:
                return QColor("#fb7185")           # 음수는 빨강
        return None

    def headerData(self, section, orientation, role=Qt.ItemDataRole.DisplayRole):
        if role != Qt.ItemDataRole.DisplayRole:
            return None
        if orientation == Qt.Orientation.Horizontal:
            return str(self._df.columns[section])
        return str(self._df.index[section])

    def sort(self, column, order=Qt.SortOrder.AscendingOrder):   # 헤더 클릭 정렬
        self.layoutAboutToBeChanged.emit()
        self._df = self._df.sort_values(
            self._df.columns[column],
            ascending=(order == Qt.SortOrder.AscendingOrder))
        self.layoutChanged.emit()

from pathlib import Path
from PySide6.QtCore import QStandardPaths
from PySide6.QtSql import (QSqlDatabase, QSqlQuery,
                           QSqlTableModel)

def db_path():
    """쓰기 가능한 표준 위치에 DB 파일 준비"""
    base = Path(QStandardPaths.writableLocation(
        QStandardPaths.StandardLocation.AppDataLocation))   # ...\AppData\Roaming\업무도구
    base.mkdir(parents=True, exist_ok=True)
    return base / "app.db"

def init_db():
    db = QSqlDatabase.addDatabase("QSQLITE")
    db.setDatabaseName(str(db_path()))
    if not db.open():
        raise RuntimeError(db.lastError().text())
    QSqlQuery("""CREATE TABLE IF NOT EXISTS 거래처 (
        id   INTEGER PRIMARY KEY AUTOINCREMENT,
        이름  TEXT NOT NULL,
        담당자 TEXT,
        메모  TEXT)""")
    return db

self.model = QSqlTableModel(self)

setFilter 는 SQL 인젝션 통로입니다 — QSqlTableModel.setFilter()는 받은 문자열을 그대로 WHERE 절에 붙입니다. 위처럼 f-string으로 사용자 입력을 끼워 넣으면 ' 하나로 쿼리가 깨지거나 조작될 수 있습니다. 위의 따옴표 제거는 임시방편일 뿐 안전장치가 아닙니다. 값이 사용자 입력에서 온다면 QSqlQuery + prepare() + bindValue()로 파라미터를 바인딩해 조회하고, 그 결과를 모델에 담으세요. 최소한 따옴표는 제거가 아니라 이스케이프('')해야 합니다.

import sqlite3
import pandas as pd

class Store:
    """DB 접근을 한 클래스로 모아두면 관리가 쉽다"""
    def __init__(self, path):
        self.conn = sqlite3.connect(path)
        self.conn.execute("""CREATE TABLE IF NOT EXISTS 이력 (
            id 	INTEGER PRIMARY KEY AUTOINCREMENT,
            파일명 TEXT, 상태 TEXT,
            시각  TEXT DEFAULT CURRENT_TIMESTAMP)""")

    def add(self, 파일명, 상태):
        with self.conn:
            self.conn.execute(
                "INSERT INTO 이력(파일명, 상태) VALUES (?, ?)",
                (파일명, 상태))

    def to_df(self):
        return pd.read_sql(
            "SELECT * FROM 이력 ORDER BY id DESC", self.conn)

# 화면 갱신: 처리 후 df 를 다시 읽어 모델 교체

QML · Qt Quick 이란 — 위젯 말고 또 하나의 길

지금까지 만든 화면은 전부 QWidget 계열이었습니다. PySide6에는 완전히 다른 축이 하나 더 있습니다. QML이라는 전용 언어로 화면을 선언형으로 쓰고, 그리는 일은 GPU가 맡는 Qt Quick 입니다. 먼저 “언제 무엇을 고르는가”부터 정리합니다.

둘 중 하나를 버리는 싸움이 아닙니다 — 화면의 성격이 정해 줍니다.

같은 화면 · 위젯으로 쓰면

class Counter(QWidget):
    def __init__(self):
        super().__init__()
        self.n = 0
        self.label = QLabel("0")          # ① 위젯을 만들고
        self.btn   = QPushButton("+1")

        lay = QVBoxLayout(self)           # ② 레이아웃에 담고
        lay.addWidget(self.label)
        lay.addWidget(self.btn)

        self.btn.clicked.connect(self.up)  # ③ 신호를 잇고

    def up(self):
        self.n += 1
        self.label.setText(str(self.n))   # ④ 직접 화면을 고친다
명령형 — “값이 바뀌었으니 내가 라벨을 다시 써 준다”. 값과 화면을 사람이 손으로 동기화합니다. 화면 요소가 늘수록 setText 호출을 빠뜨려 생기는 버그가 늘어납니다.

첫 QML 앱 — 파이썬 20줄 + main.qml

위젯 앱이 QApplication + QWidget 이었다면, QML 앱은 QGuiApplication + QQmlApplicationEngine 입니다. 뼈대를 한 번 외워 두면 이후 모든 예제가 여기에 얹힙니다.

QML 로드 실패는 조용히 일어납니다 — rootObjects 검사를 습관으로 만드세요.

main.py — 이 20줄이 전부입니다

import sys
from pathlib import Path
from PySide6.QtGui import QGuiApplication
from PySide6.QtQml import QQmlApplicationEngine

if __name__ == "__main__":
    app = QGuiApplication(sys.argv)          # ① 위젯이 없으니 QGui

    engine = QQmlApplicationEngine()         # ②

    qml = Path(__file__).parent / "ui" / "main.qml"
    engine.load(qml)                         # ③ 경로 객체 그대로 됩니다

    if not engine.rootObjects():             # ⑤ 로드 실패 검사 (필수!)
        sys.exit(-1)

    sys.exit(app.exec())                     # ⑥

# main.py 에 30줄만 추가하면 직접 만들 수 있습니다
from PySide6.QtCore import QFileSystemWatcher, QUrl

class HotReload:
    def __init__(self, engine, qml_path):
        self.engine, self.path = engine, qml_path
        self.watcher = QFileSystemWatcher()
        self.watcher.addPath(str(qml_path.parent))   # 폴더 감시
        self.watcher.directoryChanged.connect(self.reload)

    def reload(self, _):
        self.engine.clearComponentCache()   # ★ 캐시를 비워야 새로 읽습니다
        for obj in self.engine.rootObjects():
            obj.deleteLater()
        self.engine.load(self.path)

# 사용
hot = HotReload(engine, qml)   # 변수에 붙들어 둘 것 (GC 방지)
clearComponentCache() 를 빼면 파일을 고쳐도 예전 화면이 그대로 뜹니다. QML 엔진이 한 번 읽은 컴포넌트를 캐시해 두기 때문입니다. 배포 빌드에서는 이 코드가 들어가지 않게 if DEBUG: 로 감싸세요.
Qt Design Studio 도 있습니다 — 디자이너가 그린 화면을 QML 로 뽑아 주는 GUI 도구입니다. 다만 개인 개발자에게는 무겁습니다. 손으로 QML 을 쓰는 편이 배우기 좋고, git diff 도 읽힙니다. 이 문서는 전부 손으로 쓰는 방식으로 갑니다.
정리 — QML 앱의 뼈대는 QGuiApplication → QQmlApplicationEngine → load() → rootObjects 검사 → exec() 다섯 줄입니다. 폴더는 ui/ · backend/ · assets/ 로 처음부터 나누세요.

    onClicked: {
        t.text = "잠깐만요"    // ← 바인딩 파괴
    }
}
// 이후 card.width 를 아무리 바꿔도
// t.text 는 "잠깐만요" 로 굳습니다.
상태를 값으로 두고 바인딩 유지
Item {
    property bool busy: false

    Text {
        id: t
        text: busy ? "잠깐만요"
                   : card.width + "px"
    }
    Button { onClicked: busy = true }
}
// 값만 바꾸면 바인딩은 살아 있습니다.
바인딩 루프 경고를 만나면 — QML Rectangle: Binding loop detected for property "width" 같은 메시지는 A 가 B 를 보고 B 가 다시 A 를 보는 상황입니다. 거의 항상 width: parent.width 와 parent 쪽의 width: childrenRect.width 가 마주 본 경우입니다. 한쪽을 고정값이나 implicitWidth 로 끊으세요.
정리 — QML 문법은 중괄호 트리 + 콜론 프로퍼티 + JS 표현식 셋뿐입니다. 콜론으로 쓴 식은 바인딩이라 자동 갱신되고, 등호로 대입하면 그 자동 갱신이 끊깁니다 — 이것만 기억하세요.

앵커 · 레이아웃 — QML 의 배치는 두 갈래입니다

위젯에서는 QVBoxLayout 하나로 끝났지만 QML 에는 앵커와 레이아웃 두 가지 방식이 있고 섞어 쓰면 안 되는 지점이 있습니다. 여기서 헷갈리면 화면이 겹치거나 사라집니다.

앵커는 변과 변을 잇는 것, 레이아웃은 남는 공간을 나누는 것 — 목적이 다릅니다.

anchors — 12가지 조합이 전부

Rectangle {
    id: parentBox; width: 400; height: 260; color: "#0f1a2e"

    Rectangle {                      // ① 부모를 꽉 채우기
        anchors.fill: parent
        anchors.margins: 12          // 사방 여백
        color: "transparent"
        border.color: "#1e3459"
    }
    Text {                           // ② 가운데
        anchors.centerIn: parent
        text: "가운데"
    }
    Button {                         // ③ 오른쪽 아래
        anchors.right: parent.right
        anchors.bottom: parent.bottom
        anchors.rightMargin: 16
        anchors.bottomMargin: 16
        text: "확인"
    }
    Rectangle {                      // ④ 형제 바로 아래에 붙이기
        anchors.top: title.bottom
        anchors.left: title.left
        anchors.topMargin: 8
        width: 120; height: 2; color: "#3b82f6"
    }
}
쓸 수 있는 변은 left right top bottom · horizontalCenter verticalCenter baseline 그리고 fill centerIn 입니다. 앵커는 부모와 형제에게만 걸 수 있습니다 — 조카나 사촌에게는 못 겁니다.

        Layout.fillHeight: true           // 남는 높이 전부

        delegate: Text { text: "행 " + index }
    }

Layout 첨부 프로퍼티 — 외울 것은 여덟 개

프로퍼티뜻 · 위젯 대응
Layout.fillWidth / fillHeight남는 공간을 차지. addWidget(w, stretch=1) 에 해당
Layout.preferredWidth / Height바라는 크기. 레이아웃이 존중하되 공간이 없으면 줄입니다
Layout.minimumWidth / maximumWidth하한 · 상한. setMinimumSize 에 해당
Layout.alignmentQt.AlignRight | Qt.AlignVCenter 처럼 조합
Layout.margins이 항목 하나만의 여백 (레이아웃 전체는 anchors.margins)
Layout.columnSpan / rowSpanGridLayout 에서 칸 합치기
Layout.row / column격자 위치 직접 지정 (안 쓰면 순서대로 채웁니다)
spacing레이아웃 자신의 프로퍼티. 자식 사이 간격

반응형 — 창 폭에 따라 배치를 바꾸기

ApplicationWindow {
    id: win; width: 900; height: 560; visible: true

    readonly property bool narrow: width < 720   // ① 기준 하나

    GridLayout {
        anchors.fill: parent
        anchors.margins: 16
        columns: win.narrow ? 1 : 2              // ② 칸 수를 바인딩

        Rectangle {
            Layout.fillWidth: true
            Layout.preferredHeight: 200
            color: "#14243d"; radius: 10
        }
        Rectangle {
            Layout.fillWidth: true
            Layout.preferredHeight: win.narrow ? 120 : 200
            color: "#0f1a2e"; radius: 10

CSS 미디어 쿼리와 같은 발상입니다. 다만 QML 은 바인딩이라 columns: 에 조건식만 적어 두면 끝납니다. 위젯에서 같은 걸 하려면 resizeEvent 를 잡아 레이아웃을 다시 짜야 했습니다.
Row 와 RowLayout 이 헷갈릴 때 판별법 — 자식이 스스로 크기를 아는가? 아이콘·고정 버튼처럼 크기가 이미 정해졌으면 Row 가 가볍고 빠릅니다. 입력칸처럼 남는 공간을 나눠 가져야 하면 RowLayout 입니다. Layout.* 을 쓰고 싶어졌다면 그건 RowLayout 이 필요하다는 신호입니다.
정리 — 앵커는 변끼리 잇는 접착제, Row/Column은 줄 세우기, RowLayout/ColumnLayout은 공간 배분입니다. 레이아웃 안에서는 앵커 대신 Layout.* 만 쓰세요.

파이썬 ↔ QML — 두 세계를 잇는 다리

QML 은 화면만 담당하고, pandas·requests·파일 처리는 전부 파이썬이 합니다. 그래서 둘을 어떻게 잇는가가 QML 앱의 핵심 설계입니다. 방법은 두 가지인데, 하나는 옛 방식이고 하나가 현재 권장입니다.

파이썬은 값을 바꾸고 알리기만 합니다 — 화면을 직접 고치는 코드는 한 줄도 없습니다.

backend/bridge.py — 노출용 클래스

from PySide6.QtCore import QObject, Property, Signal, Slot, QUrl
from PySide6.QtQml import QmlElement

QML_IMPORT_NAME = "MyTool.Backend"   # ① QML 에서 import 할 이름
QML_IMPORT_MAJOR_VERSION = 1

@QmlElement                          # ② 이 한 줄로 QML 타입이 됩니다
class Bridge(QObject):

    statusChanged = Signal()                 # ③ notify 용
    progressChanged = Signal()
    finished = Signal(int, arguments=["rows"])   # ④ QML 로 보내는 알림

    def __init__(self, parent=None):
        super().__init__(parent)
        self._status = "준비됨"
        self._progress = 0.0

    # ---------- QML 이 읽는 값 ----------
    def _get_status(self) -> str:
        return self._status

    def _set_status(self, v: str):
        if self._status == v:
            return                    # ★ 같은 값이면 신호를 쏘지 않습니다
        self._status = v
        self.statusChanged.emit()

    status = Property(str, _get_status, _set_status, notify=statusChanged)

    progress = Property(float,
                        lambda self: self._progress,
                        notify=progressChanged)

    # ---------- QML 이 부르는 함수 ----------
    @Slot(QUrl)                       # 파일 대화상자는 QUrl 을 줍니다
    def run(self, file_url: QUrl):
        path = file_url.toLocalFile()
        self._set_status(f"처리 중: {path}")
        rows = self._work(path)       # 실제로는 워커 스레드로
        self._set_status("완료")
        self.finished.emit(rows)

    @Slot(result=str)                 # 반환값이 있으면 result= 로 타입 명시
    def version(self) -> str:
        return "1.4.0"

from PySide6.QtCore import QThread, QObject, Signal, Slot

class Worker(QObject):
    progress = Signal(float)
    done = Signal(int)

    @Slot(str)
    def run(self, path: str):
        total = 100
        for i in range(total):
            ...                       # 무거운 작업

    def __init__(self, parent=None):
        super().__init__(parent)
        self._thread = QThread(self)
        self._worker = Worker()
        self._worker.moveToThread(self._thread)   # ★ 워커를 옮깁니다
        self._worker.progress.connect(self._on_progress)
        self._worker.done.connect(self._on_done)
        self._thread.start()

    @Slot(str)
    def start(self, path: str):
        # 큐 연결이라 워커 스레드에서 실행됩니다

워커 스레드에서 QML 을 직접 만지면 안 됩니다 — 시그널로만 알리세요. 위젯과 같은 규칙이지만 QML 은 씬 그래프까지 얽혀 있어 크래시가 더 잘 납니다. 시그널·슬롯 연결은 스레드를 넘어갈 때 자동으로 큐 방식이 되므로 안전합니다.
파이썬 함수를 QML 에서 부를 때 타입을 꼭 적으세요 — @Slot(str, int, result=bool) 처럼 인자와 반환 타입을 명시해야 QML 이 값을 제대로 변환합니다. 타입 없이 @Slot() 만 쓰면 인자가 넘어가지 않고 조용히 None 이 됩니다. 반대로 QML 의 var(배열·객체)를 받을 때는 @Slot("QVariantList"), @Slot("QVariantMap") 를 씁니다.
정리 — 파이썬은 @QmlElement 붙인 QObject 하나만 노출하고, QML 은 @Property 를 읽고 @Slot 을 부르고 Signal 을 듣습니다. notify= 를 빠뜨리면 화면이 안 바뀝니다.

roleNames 가 핵심입니다

from PySide6.QtCore import (QAbstractListModel, Qt, QModelIndex,
                            QByteArray, Slot)
from PySide6.QtQml import QmlElement

QML_IMPORT_NAME = "MyTool.Backend"
QML_IMPORT_MAJOR_VERSION = 1

@QmlElement
class RowModel(QAbstractListModel):
    NameRole = Qt.ItemDataRole.UserRole + 1
    DeptRole = Qt.ItemDataRole.UserRole + 2
    AmtRole  = Qt.ItemDataRole.UserRole + 3

    def __init__(self, parent=None):
        super().__init__(parent)
        self._rows = []

    # ① 몇 행인가
    def rowCount(self, parent=QModelIndex()) -> int:
        return 0 if parent.isValid() else len(self._rows)

    # ② 한 칸의 값
    def data(self, index, role=Qt.ItemDataRole.DisplayRole):
        if not index.isValid():
            return None
        r = self._rows[index.row()]
        if role == RowModel.NameRole: return r["name"]
        if role == RowModel.DeptRole: return r["dept"]
        if role == RowModel.AmtRole:  return r["amt"]
        return None

    # ③ ★ QML 이 쓸 이름표 — 위젯 모델에는 없던 부분
    def roleNames(self):
        return {
            RowModel.NameRole: QByteArray(b"name"),
            RowModel.DeptRole: QByteArray(b"dept"),
            RowModel.AmtRole:  QByteArray(b"amt"),
        }

    # ④ 통째로 갈아 끼우기
    @Slot("QVariantList")
    def setRows(self, rows):
        self.beginResetModel()
        self._rows = list(rows)
        self.endResetModel()
위젯 모델과 다른 점은 roleNames() 하나뿐입니다. 위젯은 열 번호로 값을 꺼냈지만 QML 델리게이트는 name·amt 같은 이름으로 꺼냅니다.

import pandas as pd

@QmlElement
class FrameModel(QAbstractListModel):
    def __init__(self, parent=None):
        super().__init__(parent)
        self._df = pd.DataFrame()
        self._roles = {}

    def rowCount(self, parent=QModelIndex()) -> int:
        return 0 if parent.isValid() else len(self._df)

    def data(self, index, role=Qt.ItemDataRole.DisplayRole):
        col = self._roles.get(role)
        if col is None or not index.isValid():
            return None
        v = self._df.iat[index.row(), self._df.columns.get_loc(col)]
        # numpy 타입은 QML 이 모릅니다 — 파이썬 기본형으로
        return v.item() if hasattr(v, "item") else str(v)

    def roleNames(self):
        return {k: QByteArray(v.encode())
                for k, v in self._roles.items()}

    @Slot(str)
    def loadExcel(self, path: str):
        df = pd.read_excel(path)
        self.beginResetModel()
        self._df = df
        base = Qt.ItemDataRole.UserRole + 1
        self._roles = {base + i: str(c)
                       for i, c in enumerate(df.columns)}
        self.endResetModel()

Effects · 셰이더 — 위젯으로는 못 하던 것들

블러 · 그림자 · 색 입히기는 MultiEffect 하나로 끝납니다. 그보다 더 나가고 싶으면 GPU 셰이더를 직접 쓸 수도 있습니다. 다만 전부 GPU 를 먹으므로 어디까지 써도 되는지를 같이 봅니다.

효과는 화면을 한 번 더 그리는 일입니다 — 큰 영역 하나에만 쓰세요.

직접 셰이더 쓰기 — .frag 와 qsb 🔴 실무

Qt 6 은 셰이더를 미리 컴파일합니다

# wave.frag — GLSL 로 씁니다 (Vulkan 스타일)
#version 440

layout(location = 0) in vec2 qt_TexCoord0;
layout(location = 0) out vec4 fragColor;

layout(std140, binding = 0) uniform buf {
    mat4  qt_Matrix;
    float qt_Opacity;
    float time;          // QML 에서 넘길 값
};
layout(binding = 1) uniform sampler2D source;

void main() {
    vec2 uv = qt_TexCoord0;
    uv.x += sin(uv.y * 20.0 + time) * 0.01;
    fragColor = texture(source, uv) * qt_Opacity;
}
# .frag → .qsb 로 컴파일 (모든 그래픽 API 용 바이트코드를 한 파일에)
uv run pyside6-rcc --help      # 참고: 리소스는 rcc
pyside6-qsb --qt6 -o wave.frag.qsb wave.frag

# 개발 중에는 검증 옵션을 켜 두면 오류를 빨리 봅니다
pyside6-qsb --qt6 --glsl "100 es,120,150" -o wave.frag.qsb wave.frag
Qt 5 와 달리 셰이더 소스를 문자열로 넣을 수 없습니다 — Qt 6 은 Vulkan·D3D·Metal 을 함께 지원하려고 중간 바이트코드(.qsb) 를 씁니다. 빌드 단계에 qsb 컴파일이 반드시 들어가야 합니다.

언제 셰이더까지 가야 하나 — 솔직히 업무 도구에서는 거의 없습니다. 물결·파티클·글리치 같은 연출이 제품의 핵심일 때만 의미가 있습니다. 먼저 MultiEffect 로 되는지 확인하세요.

Canvas — HTML 캔버스와 같은 API

Canvas {
    id: cv
    width: 300; height: 160
    property var values: [12, 34, 20, 55, 41, 62, 48]

    onPaint: {
        const ctx = getContext("2d")
        ctx.reset()
        ctx.strokeStyle = "#60a5fa"
        ctx.lineWidth = 2
        ctx.beginPath()
        const step = width / (values.length - 1)
        const max = Math.max(...values)
        for (let i = 0; i < values.length; i++) {
            const x = i * step
            const y = height - (values[i] / max) * (height - 12)
            i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y)
        }
        ctx.stroke()
    }
    onValuesChanged: requestPaint()      // ★ 다시 그리라고 알려야 합니다
}
Canvas 는 CPU 로 그립니다 — 웹 개발자에게 익숙하지만 Qt Quick 에서는 느린 축입니다. 매 프레임 다시 그려야 하는 것에는 쓰지 마세요. 가능하면 Shape 나 도형 조합으로 바꾸는 편이 낫습니다.
정리하면 — 정적인 벡터는 Shape, 효과는 MultiEffect, 정말 특별한 픽셀 연산만 ShaderEffect, Canvas 는 최후입니다.
GPU 부담을 줄이는 실전 규칙 넷
① 반투명 레이어를 세 겹 이상 쌓지 마세요 — 같은 픽셀을 반복해 칠합니다(오버드로).
② Image 에는 언제나 sourceSize 를 주세요. 4000px 사진은 GPU 메모리 64MB 를 먹습니다.
③ 움직이지 않는 복잡한 아이템에는 layer.enabled: true 로 텍스처에 구워 두면 재사용됩니다.
④ QSG_RENDER_TIMING=1 환경 변수로 어느 단계가 오래 걸리는지 실제로 재 보세요.
정리 — 블러·그림자·색 입히기는 MultiEffect 하나로 끝납니다. 벡터는 Shape, 픽셀 연산이 꼭 필요할 때만 ShaderEffect + .qsb 이고, 무엇이든 큰 영역 하나에만 거세요.

main.py — 개발/배포를 함께 다루기

import os, sys
from pathlib import Path
from PySide6.QtGui import QGuiApplication
from PySide6.QtQml import QQmlApplicationEngine
from PySide6.QtQuickControls2 import QQuickStyle

import backend.bridge              # @QmlElement 등록

DEV = os.environ.get("MYTOOL_DEV") == "1"

if not DEV:
    import rc_resources            # noqa: F401  가상 파일 시스템 등록

def qml_url() -> str:
    if DEV:
        return str(Path(__file__).parent / "ui" / "main.qml")
    return "qrc:/ui/main.qml"

if __name__ == "__main__":
    app = QGuiApplication(sys.argv)
    app.setOrganizationName("우리회사")
    app.setApplicationName("업무도구")

    QQuickStyle.setStyle("Material")      # load 前에

    engine = QQmlApplicationEngine()
    engine.addImportPath("qrc:/ui" if not DEV
                         else str(Path(__file__).parent / "ui"))
    engine.load(qml_url())

    if not engine.rootObjects():
        print("QML 로드 실패", file=sys.stderr)
        sys.exit(-1)
    sys.exit(app.exec())

QQuickWidget — 위젯 안에 QML 넣기

import sys
from PySide6.QtWidgets import (QApplication, QMainWindow, QWidget,
                               QVBoxLayout, QTableView, QSplitter)
from PySide6.QtQuickWidgets import QQuickWidget
from PySide6.QtCore import QUrl, Qt

class Main(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("표는 위젯 · 대시보드는 QML")

        table = QTableView()              # ① 기존 위젯 자산 그대로

        quick = QQuickWidget()            # ② QML 을 위젯처럼
        quick.setResizeMode(
            QQuickWidget.ResizeMode.SizeRootObjectToView)
        quick.rootContext().setContextProperty("bridge", self.bridge)
        quick.setSource(QUrl("qrc:/ui/dashboard.qml"))

        split = QSplitter(Qt.Orientation.Vertical)
        split.addWidget(quick)            # 위: 움직이는 대시보드
        split.addWidget(table)            # 아래: 촘촘한 표
        split.setSizes([260, 480])
        self.setCentralWidget(split)

if __name__ == "__main__":
    app = QApplication(sys.argv)          # ★ QGui 가 아니라 QApplication
    w = Main(); w.show()
    sys.exit(app.exec())
