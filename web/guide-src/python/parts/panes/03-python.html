# 파이썬 — 파일 하나가 곧 실행 단위
def greet(name):
    if name:
        print(f"안녕, {name}")
    else:
        print("이름이 없네요")

greet("민준")
// Java — 클래스로 감싸야 하고, 중괄호와 세미콜론 필요
public class Main {
    static void greet(String name) {
        if (name != null) {
            System.out.println("안녕, " + name);
        }
    }
    public static void main(String[] a) { greet("민준"); }
}

#   소스 → 토큰 → AST → 바이트코드 → 평가 루프
# 들여쓰기는 '스타일'이 아니라 1단계에서 진짜 토큰이 됩니다.

import tokenize, io
소스 = "def f():\n    return 1\n"
for t in tokenize.generate_tokens(io.StringIO(소스).readline):
    print(f"{tokenize.tok_name[t.type]:10} {t.string!r}")

# NAME       'def'
# ...
# NEWLINE    '\n'
# INDENT     '    '      ← ★ 들여쓰기가 토큰으로 만들어집니다

# 해결 — 들어오는 문자열은 전부 NFC 로 통일
def 정규화(s: str) -> str:
    return unicodedata.normalize("NFC", s)

# 비교·검색·중복체크용 키는 한 단계 더
def 키(s: str) -> str:
    s = unicodedata.normalize("NFKC", s)   # 전각①·ｱ 등을 표준형으로
    return s.strip().casefold()            # lower() 보다 강력 (ß→ss)

쉽게 말하면 — match는 “이 데이터가 이런 모양이니?”라고 묻고, 맞으면 안쪽 값까지 한 번에 꺼내 줍니다. if event["type"] == "click": x = event["x"] … 하며 파고들던 코드가 한 눈에 보이는 표로 바뀝니다.

def handle(event):
    match event:
        # dict 모양 + 안쪽 값을 x, y 로 바로 꺼냄
        case {"type": "click", "x": x, "y": y}:
            return f"클릭 ({x}, {y})"

        # 조건(guard) 을 덧붙일 수 있습니다
        case {"type": "key", "code": code} if code < 32:
            return "제어 문자"

        # 리스트 모양 — 나머지는 *rest 로
        case ["move", *rest]:
            return f"이동 {rest}"

        # 클래스 모양 (dataclass·NamedTuple 과 궁합이 좋습니다)
        case Point(x=0, y=0):
            return "원점"

        # 타입 검사 + 이름 붙이기
        case str() as s:
            return s.upper()

        case 1 | 2 | 3:          # 여러 값 중 하나
            return "작은 수"

        case _:                  # 나머지 전부 (default)
            return "모름"

class 장바구니:

        self.상품들 = list(상품들 or [])

        return len(self.상품들)

        return self.상품들[i]

    def __iter__(self):                # for 상품 in 바구니
        return iter(self.상품들)

    def __contains__(self, 이름):       # "사과" in 바구니

        return 장바구니(self.상품들 + 다른.상품들)

연차별로 막히는 지점이 다릅니다

  • ~1년 — 문법이 아니라 에러 읽기에서 막힙니다. 트레이스백 맨 아랫줄과 내 파일이 나오는 줄, 이 둘만 보는 습관
  • 1~3년 — 돌아가는데 고치기 무서운 코드가 쌓입니다. 해법은 테스트와 함수 분리. “바꿔도 되는지”를 사람이 아니라 테스트가 답하게
  • 3~7년 — 코드보다 구조와 경계가 문제가 됩니다. 의존 방향, 설정 분리, 실패했을 때의 동작을 먼저 설계
  • 7년~ — 기술 선택이 팀의 비용이 됩니다. “내가 잘 쓰는 것”이 아니라 “팀이 6개월 뒤에도 고칠 수 있는 것”으로 고르게 됩니다
  • 공통 — 오래 가는 사람은 측정하고 결정합니다. 느리다고 느끼면 프로파일러를, 메모리가 는다면 tracemalloc을 먼저 켭니다

# 파이썬이 느린 구간은 '파이썬으로 안 도는 코드'로 대체하는 게 정석입니다
#   ① numpy · polars      — 배열·표 연산을 C/Rust 레벨로
#   ② Cython · mypyc      — 기존 파이썬 코드를 그대로 컴파일
#   ③ PyO3(Rust) · pybind11(C++)
#                          — 병목 함수만 네이티브 모듈로 만들어 import

# 예 — Rust 로 만든 함수를 파이썬에서 그냥 import 해서 씁니다
from 내확장 import 무거운계산      # maturin 으로 빌드한 네이티브 모듈
무거운계산(데이터)                 # 순수 파이썬 대비 수십~수백 배

#  ※ 이 방식(파이썬에서 부를 모듈을 C++·Rust 로 만들기)은
#    C++ · Rust 가이드에서 처음부터 다룹니다.
판단 기준 — 네이티브 확장은 빌드·배포·디버깅 비용이 함께 옵니다. ① 알고리즘 ② I/O 횟수 ③ 벡터화까지 다 해 본 뒤, 프로파일러가 특정 함수 하나를 계속 가리킬 때 꺼내는 카드입니다.
