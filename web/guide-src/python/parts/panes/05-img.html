shape 읽는 법

import cv2, numpy as np
img = cv2.imread("photo.jpg")

img.shape        # (1080, 1920, 3)
#                  높이   너비   채널
h, w, c = img.shape

img.dtype        # uint8  -> 0~255
img.size         # 전체 값 개수 = h*w*c
img[0, 0]        # 왼쪽 위 픽셀 -> array([B, G, R])
img[100:200, 50:150]   # 잘라내기 = NumPy 슬라이싱!
좌표 순서가 반대 — 사람은 “가로×세로”로 말하지만 배열은 [행, 열] = [y, x] 순서입니다. img[y, x]이지 img[x, y]가 아닙니다. 여기서 대부분의 버그가 시작됩니다.
OpenCV 5 (2026-07 출시) 주의 — 이 탭에서 쓰는 기본 API (imread · cvtColor · resize · threshold 등)는 5에서도 그대로입니다. 다만 Haar cascade · HOG 기술자 · 전통 ML 모듈이 contrib 패키지로 분리됐고, 레거시 C API가 제거됐습니다. 당장은 4.x로 고정하는 편이 안전합니다 — uv add "opencv-python>=4.14,<5"

기본 사용

from PIL import Image

img = Image.open("photo.jpg")

img.size        # (1920, 1080)  ← (가로, 세로) 순서!
img.width  img.height
img.mode        # "RGB" / "RGBA" / "L"(흑백) / "P"(팔레트)
img.format      # "JPEG"
img.info        # EXIF 등 부가정보

img.show()                     # 기본 뷰어로 열기
img.save("out.png")            # 확장자로 포맷 자동 결정
img.save("out.jpg", quality=90, optimize=True)

# 모드 변환
img.convert("RGB")     # RGBA -> RGB (JPEG 저장 전 필수)
img.convert("L")       # 흑백
img.convert("RGBA")    # 투명 채널 추가
PNG를 JPEG로 저장할 때 — PNG는 투명(RGBA)을 가질 수 있는데 JPEG는 투명을 지원하지 않아 OSError가 납니다. 저장 전에 img.convert("RGB")를 꼭 거치세요.

from PIL import Image
from PIL.Image import Resampling, Transpose

img = Image.open("sample.jpg")

img.resize((800, 600))                      # 비율 무시
img.resize((800, 600), Resampling.LANCZOS)  # 고품질 축소

# 비율 유지 축소 (원본을 직접 수정, 확대는 안 됨)
img.thumbnail((800, 800))

# 비율 유지 계산해서 리사이즈
w, h = img.size
new_w = 800
new_h = int(h * new_w / w)
img = img.resize((new_w, new_h), Resampling.LANCZOS)

# ImageOps 유틸
from PIL import ImageOps
ImageOps.fit(img, (300,300))       # 중앙 크롭 + 리사이즈
ImageOps.pad(img, (300,300), color="black")  # 여백 채우기
ImageOps.contain(img, (300,300))   # 안쪽에 맞추기
리샘플링 선택 — 축소는 LANCZOS(가장 선명), 확대는 BICUBIC, 속도가 중요하면 NEAREST. 기본값은 BICUBIC입니다.

from PIL import Image
from PIL.Image import Transpose

img = Image.open("sample.jpg")

OpenCV에서 한글 넣기

# cv2.putText 는 한글을 못 씁니다 -> Pillow 를 거칩니다
import cv2, numpy as np
from PIL import Image, ImageDraw, ImageFont

def put_korean(cv_img, text, pos, size=30, color=(255,255,255)):
    pil = Image.fromarray(cv2.cvtColor(cv_img, cv2.COLOR_BGR2RGB))
    d = ImageDraw.Draw(pil)
    font = ImageFont.truetype("C:/Windows/Fonts/malgun.ttf", size)
    d.text(pos, text, font=font, fill=color)
    return cv2.cvtColor(np.array(pil), cv2.COLOR_RGB2BGR)

img = cv2.imread("photo.jpg")
img = put_korean(img, "검출 완료", (30, 30))

색공간 · 이진화(Threshold)

슬라이더를 움직여 임계값이 픽셀을 어떻게 흑백으로 가르는지 보세요.

threshold 시뮬레이터

이진화

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# 고정 임계값
_, bw = cv2.threshold(gray, 128, 255, cv2.THRESH_BINARY)
_, bw = cv2.threshold(gray, 128, 255, cv2.THRESH_BINARY_INV)

# 자동 임계값 (Otsu) — 값 고민이 필요 없음 ★

                      cv2.THRESH_BINARY + cv2.THRESH_OTSU)

# 적응형 — 조명이 불균일한 문서 스캔에 필수 ★

    cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
    cv2.THRESH_BINARY, blockSize=11, C=2)

HSV로 특정 색만 뽑기

# RGB보다 HSV가 '색 범위' 지정에 훨씬 유리합니다
hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)

# 파란색 범위 (H:0~179, S:0~255, V:0~255)
lower = np.array([100, 120, 70])
upper = np.array([130, 255, 255])
mask = cv2.inRange(hsv, lower, upper)      # 흰=해당색

result = cv2.bitwise_and(img, img, mask=mask)

# 마스크 다듬기 (노이즈 제거 / 구멍 메우기)
k = np.ones((5,5), np.uint8)
mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, k)   # 점 노이즈 제거
mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, k)  # 구멍 메움
mask = cv2.dilate(mask, k, iterations=1)           # 팽창
mask = cv2.erode(mask, k, iterations=1)            # 침식
업무 자동화 활용 — 화면에서 “특정 색 버튼”을 찾을 때 HSV 마스킹이 이미지 매칭보다 훨씬 빠르고 해상도 변화에도 강합니다.

템플릿(“확인” 버튼 이미지)을 좌상단부터 한 칸씩 밀며 유사도를 계산합니다. 가장 높은 위치가 매칭 결과입니다.

템플릿 매칭

import cv2, numpy as np

screen = cv2.imread("screen.png", cv2.IMREAD_GRAYSCALE)
tpl    = cv2.imread("button.png", cv2.IMREAD_GRAYSCALE)
h, w = tpl.shape

res = cv2.matchTemplate(screen, tpl, cv2.TM_CCOEFF_NORMED)
min_v, max_v, min_loc, max_loc = cv2.minMaxLoc(res)

if max_v >= 0.85:                 # 신뢰도 임계값
    x, y = max_loc
    center = (x + w//2, y + h//2)
    print(f"찾음! 유사도 {max_v:.2f} 위치 {center}")
    cv2.rectangle(screen, max_loc, (x+w, y+h), 255, 2)

# 여러 개 한꺼번에 찾기
locs = np.where(res >= 0.85)
for pt in zip(*locs[::-1]):
    cv2.rectangle(screen, pt, (pt[0]+w, pt[1]+h), 255, 2)

실전 파이프라인 — 수만 장을 처리하기

상품 사진 3만 장의 썸네일을 만드는 일은 이미지 처리 문제가 아니라 처리량 문제입니다. 한 장을 다루는 법과 수만 장을 다루는 법은 코드가 다릅니다.

from PIL import Image, ImageOps

def make_thumb(src: str, dst: str, box=(640, 640)) -> None:
    with Image.open(src) as im:
        # ① EXIF 회전 반영 — 안 하면 아이폰 사진이 옆으로 눕는다
        im = ImageOps.exif_transpose(im)

        # ② 큰 원본은 draft 로 빠르게 축소 (JPEG 디코딩 자체를 줄인다)
        im.draft("RGB", box)

        # ③ 비율 유지 축소 — thumbnail 은 원본보다 크게 만들지 않는다
        im.thumbnail(box, Image.Resampling.LANCZOS)

        # ④ 색 프로파일 — 무시하면 웹에서 색이 달라 보인다
        im = im.convert("RGB")

        im.save(dst, "WEBP", quality=82, method=4)

# 폭탄 이미지 방어 — 압축을 풀면 수십 GB 가 되는 파일이 있다
Image.MAX_IMAGE_PIXELS = 80_000_000
사용자가 올린 이미지는 입력값입니다. 확장자를 믿지 말고 실제 포맷을 확인(Image.open 이 던지는 예외 처리)하고, 픽셀 수 상한을 두세요. 디컴프레션 폭탄은 서버 메모리를 한 방에 날립니다.

수만 장을 — 프로세스로 나누기

from concurrent.futures import ProcessPoolExecutor, as_completed
from pathlib import Path
import os

def run_batch(root: Path, out: Path, workers: int | None = None):
    files = [p for p in root.rglob("*") if p.suffix.lower() in
             {".jpg", ".jpeg", ".png", ".heic"}]
    workers = workers or max(1, (os.cpu_count() or 4) - 1)
    done = failed = 0

    # 이미지 처리는 CPU 작업 → 스레드가 아니라 프로세스
    with ProcessPoolExecutor(max_workers=workers) as ex:
        futs = {ex.submit(make_thumb, str(f), str(out / f"{f.stem}.webp")): f
                for f in files}
        for fu in as_completed(futs):
            try:
                fu.result()
                done += 1
            except Exception as e:                # 한 장 실패로 전체가 멈추면 안 된다
                failed += 1
                log.warning("skip %s: %s", futs[fu], e)
            if (done + failed) % 500 == 0:
                log.info("%d/%d", done + failed, len(files))

    log.info("완료 %d · 실패 %d", done, failed)
  • 스레드로는 안 빨라집니다 — 이미지 디코딩은 CPU 작업이라 GIL 에 묶입니다
  • 워커 = 코어 수 − 1. 전부 쓰면 PC 가 먹통이 됩니다
  • 이미 만든 결과는 건너뛰기 — 중간에 끊겨도 이어서 돌 수 있어야 합니다
  • 메모리가 부족하면 파일 목록을 청크로 잘라 순차 실행하세요
더 빨라야 한다면 Pillow-SIMD 나 ffmpeg·libvips(pyvips) 를 검토하세요 — libvips 는 대용량에서 Pillow 보다 몇 배 빠르고 메모리도 적게 씁니다.

import cv2, numpy as np

def preprocess(path: str) -> np.ndarray:
    img = cv2.imread(path)

    # ① 회색조 + 크기 정규화 (너무 작으면 글자가 뭉갠다)
    g = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    if g.shape[1] < 1000:
        g = cv2.resize(g, None, fx=2, fy=2, interpolation=cv2.INTER_CUBIC)

    # ② 그림자·조명 얼룩 제거 — 휴대폰 촬영본의 최대 적
    bg = cv2.medianBlur(g, 31)
    g  = cv2.divide(g, bg, scale=255)

    # ③ 적응형 이진화 (밝기가 균일하지 않아도 견딘다)
    bw = cv2.adaptiveThreshold(g, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
                               cv2.THRESH_BINARY, 31, 15)

    # ④ 기울기 보정 — 1~2도만 틀어져도 인식률이 떨어진다
    coords = np.column_stack(np.where(bw < 255))
    angle  = cv2.minAreaRect(coords)[-1]
    angle  = -(90 + angle) if angle < -45 else -angle
    h, w   = bw.shape
    M      = cv2.getRotationMatrix2D((w // 2, h // 2), angle, 1.0)
    return cv2.warpAffine(bw, M, (w, h),
                          flags=cv2.INTER_CUBIC,
                          borderMode=cv2.BORDER_REPLICATE)
