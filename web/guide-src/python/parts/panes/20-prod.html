프로젝트 구조 — src 레이아웃과 패키지 경계

구조는 취향이 아니라 "어디를 고치면 어디가 깨지는가"를 정하는 설계입니다. 처음 맡은 저장소에서 가장 먼저 보는 것도, 새 저장소를 만들 때 가장 먼저 정하는 것도 폴더 모양입니다. 이 섹션은 src 레이아웃 · 도메인/어댑터 경계 · __init__.py 를 가볍게 두는 법 · 순환 import 끊기를 다룹니다.

# src/billing/domain/order.py
from dataclasses import dataclass
from decimal import Decimal
from typing import Protocol


@dataclass(frozen=True, slots=True)
class Order:
    id: str
    amount: Decimal
    currency: str = "KRW"


class PaymentGateway(Protocol):          # "이런 모양이면 된다"
    async def charge(self, order: Order, idem_key: str) -> str: ...


class OrderRepository(Protocol):
    async def get(self, order_id: str) -> Order | None: ...
    async def mark_paid(self, order_id: str, tx_id: str) -> None: ...
# src/billing/services/checkout.py — 유스케이스
from billing.domain.errors import NotFound
from billing.domain.order import OrderRepository, PaymentGateway


async def checkout(order_id: str, repo: OrderRepository,
                   pg: PaymentGateway) -> str:
    order = await repo.get(order_id)
    if order is None:
        raise NotFound(f"order={order_id}")
    tx_id = await pg.charge(order, idem_key=f"charge-{order.id}")
    await repo.mark_paid(order.id, tx_id)
    return tx_id

# src/billing/adapters/toss_pg.py
import httpx

from billing.domain.order import Order


class TossGateway:                       # Protocol 을 상속하지 않아도 됩니다
    def __init__(self, client: httpx.AsyncClient) -> None:
        self._client = client

    async def charge(self, order: Order, idem_key: str) -> str:
        resp = await self._client.post(
            "/v1/payments",
            json={"orderId": order.id, "amount": str(order.amount)},
            headers={"Idempotency-Key": idem_key},
        )
        resp.raise_for_status()
        return resp.json()["paymentKey"]

설정과 비밀 — pydantic-settings 하나로 모으기

os.environ["DB_URL"] 이 코드 곳곳에 흩어져 있으면 무엇을 설정해야 서비스가 뜨는지 아무도 모릅니다. 설정은 한 클래스에 모으고, 타입으로 검증하고, 기동할 때 한 번에 실패하게 만드는 것이 실무의 기본입니다. 그리고 그 설정 안의 비밀번호가 로그·예외·에러 리포트로 새지 않게 막는 것까지가 이 섹션입니다.

설정 우선순위 레이어 — 생성자 인자 > 환경변수 > .env > 비밀 파일 > 기본값. 운영 값은 ②·④ 에서만 와야 합니다.

설정 클래스 하나 — 이게 "설정 문서"가 됩니다

# src/billing/settings.py
from functools import lru_cache
from typing import Literal

from pydantic import BaseModel, PostgresDsn, SecretStr
from pydantic_settings import BaseSettings, SettingsConfigDict


class DbSettings(BaseModel):            # 중첩 묶음은 BaseModel
    url: PostgresDsn
    pool_size: int = 10
    password: SecretStr


class Settings(BaseSettings):
    model_config = SettingsConfigDict(
        env_prefix="BILLING_",
        env_file=(".env", ".env.local"),   # 뒤에 적은 파일이 이김
        env_nested_delimiter="__",          # BILLING_DB__URL → db.url
        secrets_dir="/run/secrets",         # 없으면 경고만 내고 건너뜀
        extra="ignore",
        hide_input_in_errors=True,          # 검증 실패 메시지에 값 숨김
    )

    env: Literal["local", "dev", "prod"] = "local"
    db: DbSettings
    pg_api_key: SecretStr                   # 기본값 없음 = 필수
    pg_base_url: str = "https://api.tosspayments.com"
    http_timeout_s: float = 5.0
    debug: bool = False


@lru_cache                                  # 프로세스당 한 번만 읽기
def get_settings() -> Settings:
    return Settings()

타입 힌트 실무 도입 — strict 를 모듈 단위로 넓혀 가기

한 단계 더 — 실전에서 쓰는 타입 네 가지 🟡 중급

Protocol · TypedDict

from typing import NotRequired, Protocol, ReadOnly, TypedDict


# ① Protocol — "이 메서드만 있으면 된다" (상속 불필요)
class Notifier(Protocol):
    def send(self, to: str, text: str) -> None: ...

class SlackNotifier:
    def send(self, to: str, text: str) -> None:
        ...

def alert(n: Notifier) -> None:
    n.send("#billing-alert", "정산 실패")

alert(SlackNotifier())          # ✅ 구조가 맞으므로 통과


# ② TypedDict — 외부에서 오는 dict 의 "모양" 에 이름 붙이기
class PgWebhook(TypedDict):
    paymentKey: ReadOnly[str]   # 3.13 — 읽기 전용 키
    status: str
    approvedAt: NotRequired[str]  # 없을 수도 있는 키

def handle(evt: PgWebhook) -> None:
    if "approvedAt" in evt:       # 좁히기 — 이 안에선 str
        print(evt["approvedAt"])
    evt["paymentKey"] = "x"       # ❌ mypy: ReadOnly 키 대입
TypedDict 와 pydantic 의 차이 — TypedDict 는 검사기만 보는 주석이라 실행 중 검증이 없습니다. 외부 입력을 믿을 수 없으면 pydantic(PROD 11), 이미 검증된 dict 에 이름만 붙이면 TypedDict 입니다.

제네릭(3.12+ 문법) · NewType · TypeIs

from collections.abc import Sequence
from typing import NewType, TypeIs


# ③ 제네릭 — 3.12 부터 TypeVar 선언 없이 [T]
def first[T](xs: Sequence[T]) -> T | None:
    return xs[0] if xs else None

class Page[T]:
    def __init__(self, items: list[T], next_cursor: str | None) -> None:
        self.items = items
        self.next_cursor = next_cursor

page: Page[int] = Page([1, 2], None)

type JSON = dict[str, JSON] | list[JSON] | str | int | float | bool | None


# ④ NewType — 같은 str 이라도 섞이지 않게
OrderId = NewType("OrderId", str)
UserId = NewType("UserId", str)

def cancel(order_id: OrderId) -> None: ...

uid = UserId("u-1")
cancel(uid)                    # ❌ mypy: UserId 는 OrderId 가 아님


# 3.13 — TypeIs 로 사용자 정의 좁히기
def is_str_list(v: list[object]) -> TypeIs[list[str]]:
    return all(isinstance(x, str) for x in v)
NewType 이 실무에서 값을 하는 곳 — 인자가 (user_id: str, order_id: str) 처럼 같은 타입 둘이면 호출하는 쪽에서 순서를 바꿔 넣어도 아무도 못 잡습니다. NewType 은 런타임 비용이 0 이고 이 사고를 검사기가 잡게 만듭니다. 타입 이론은 → 🧬 전문가 탭 · 타입 시스템 심화.

품질 자동화 — ruff · pre-commit · CI 를 한 줄로 잇기

코드 리뷰에서 "여기 import 순서가…" "공백이…" 라는 말이 나오면 사람의 시간을 기계 일에 쓰고 있는 것입니다. 형식과 흔한 버그는 기계가, 설계와 의도는 사람이 봅니다. 이 섹션은 그 분업을 커밋할 때(pre-commit) → 푸시할 때(CI) 두 관문으로 만드는 설정을 통째로 보여 줍니다.

품질 파이프라인 — 몇 초짜리 검사를 앞에 두면 몇 분짜리 테스트를 헛돌리지 않습니다.

예외 설계 — 어디서 잡고 어디서 흘리나

운영 로그에 같은 트레이스백이 세 번 찍혀 있고, 정작 원인인 타임아웃은 except Exception: pass 에 묻혀 있습니다. 예외 처리가 엉키는 이유는 문법이 아니라 "이 층이 이 예외에 대해 무엇을 할 수 있나"를 정하지 않았기 때문입니다. 이 섹션은 도메인 예외 계층 · 경계에서 번역하기 · 재시도 가능 여부 · raise from 을 한 벌로 묶습니다.

스윔레인 — 예외는 경계(어댑터)에서 번역되고 진입점에서 한 번만 기록됩니다. 가운데는 흘려보냅니다.

import logging
import httpx
from tenacity import (
    retry, retry_if_exception, stop_after_attempt, stop_after_delay,
    wait_random_exponential, before_sleep_log, RetryCallState,
)

logger = logging.getLogger(__name__)
RETRY_STATUS = {429, 502, 503, 504}

def is_retryable(exc: BaseException) -> bool:
    if isinstance(exc, (httpx.TimeoutException, httpx.NetworkError)):
        return True
    if isinstance(exc, httpx.HTTPStatusError):
        return exc.response.status_code in RETRY_STATUS
    return getattr(exc, "retryable", False)      # 도메인 예외 (PROD 05)

_jitter = wait_random_exponential(multiplier=0.5, max=20)

def wait_retry_after(state: RetryCallState) -> float:
    exc = state.outcome.exception() if state.outcome else None
    if isinstance(exc, httpx.HTTPStatusError):
        ra = exc.response.headers.get("Retry-After")
        if ra and ra.isdigit():
            return min(float(ra), 60.0)          # 서버가 정해 준 시간 우선
    return _jitter(state)

@retry(
    retry=retry_if_exception(is_retryable),
    wait=wait_retry_after,
    stop=stop_after_attempt(5) | stop_after_delay(30),
    before_sleep=before_sleep_log(logger, logging.WARNING),
    reraise=True,                                # 마지막 원래 예외를 그대로
)
async def get_rate(client: httpx.AsyncClient, cur: str) -> dict:
    r = await client.get(f"/v1/rates/{cur}")
    r.raise_for_status()
    return r.json()

async def handle(key: str, body: bytes, work) -> tuple[int, dict]:
    h = hashlib.sha256(body).hexdigest()
    row = await db.fetchrow(
        "INSERT INTO idempotency_keys (key, request_hash) VALUES ($1, $2) "
        "ON CONFLICT (key) DO NOTHING RETURNING key", key, h)
    if row is None:                       # 이미 본 키
        prev = await db.fetchrow("SELECT * FROM idempotency_keys WHERE key=$1", key)
        if prev["request_hash"] != h:
            return 422, {"error": "idempotency_key_reused"}
        if prev["status_code"] is None:
            return 409, {"error": "in_progress"}
        return prev["status_code"], prev["response"]
    status, resp = await work()
    await db.execute("UPDATE idempotency_keys SET status_code=$2, response=$3 "
                     "WHERE key=$1", key, status, json.dumps(resp))
    return status, resp

# src/billing/batch/settle.py
import logging
from datetime import date

import psycopg

log = logging.getLogger(__name__)
JOB, CHUNK = "settle", 1_000

def run(conn: psycopg.Connection, target: date) -> int:
    """conn 은 autocommit=True — transaction() 블록이 진짜 커밋 단위가 됨."""
    row = conn.execute(
        "SELECT last_id FROM batch_checkpoint WHERE job=%s AND target=%s",
        (JOB, target)).fetchone()
    last_id, total = (row[0] if row else 0), 0
    while True:
        with conn.transaction():                      # 청크 하나 = 커밋 하나
            rows = conn.execute(
                "SELECT id, merchant_id, amount FROM payments "
                "WHERE paid_on=%s AND id > %s ORDER BY id LIMIT %s",
                (target, last_id, CHUNK)).fetchall()
            if not rows:
                break
            with conn.cursor() as cur:
                cur.executemany(UPSERT_SETTLEMENT,    # 왼쪽 upsert SQL
                                [(r[0], r[1], r[2], target) for r in rows])
            last_id = rows[-1][0]
            conn.execute(                              # 같은 트랜잭션에서 기록
                "INSERT INTO batch_checkpoint (job, target, last_id) "
                "VALUES (%s, %s, %s) ON CONFLICT (job, target) "
                "DO UPDATE SET last_id=EXCLUDED.last_id, updated_at=now()",
                (JOB, target, last_id))
        total += len(rows)
        log.info("청크 커밋 last_id=%s 누적=%s", last_id, total)
    return total

# src/billing/batch/cli.py
import logging
import sys
from datetime import date, datetime, timedelta
from zoneinfo import ZoneInfo

import httpx
import psycopg

from billing.batch.settle import JOB, run
from billing.batch.lock import try_lock      # 왼쪽 ① advisory lock 을 함수로
from billing.settings import get_settings

log = logging.getLogger(__name__)

def parse_target(arg: str) -> date:
    if arg == "yesterday":                    # 스케줄러용 — KST 기준 어제
        return datetime.now(ZoneInfo("Asia/Seoul")).date() - timedelta(days=1)
    return date.fromisoformat(arg)            # 재처리용 — 날짜를 명시

def notify(text: str) -> None:
    url = get_settings().slack_webhook_url.get_secret_value()
    try:
        httpx.post(url, json={"text": text}, timeout=5)
    except httpx.HTTPError:
        log.exception("알림 전송 실패")       # 알림 실패가 배치를 죽이지 않게

def main() -> None:
    target = parse_target(sys.argv[1])
    s = get_settings()
    with psycopg.connect(str(s.db.url), autocommit=True,
                         password=s.db.password.get_secret_value()) as conn:
        if not try_lock(conn, f"{JOB}:{target}"):
            log.warning("이미 실행 중 — 건너뜀 target=%s", target)
            return                            # 종료 코드 0
        try:
            n = run(conn, target)
        except Exception:
            log.exception("정산 실패 target=%s", target)
            notify(f":x: 정산 실패 {target} — 체크포인트부터 재실행 가능")

import csv
from collections.abc import Iterator

import psycopg

def iter_orders(conn: psycopg.Connection, target: date) -> Iterator[tuple]:
    # 이름 있는 커서 = 서버 측 커서 → 전부 메모리에 올리지 않음
    with conn.cursor(name="export_orders") as cur:
        cur.itersize = 5_000
        cur.execute(
            "SELECT id, user_id, amount, paid_at FROM orders "
            "WHERE paid_on = %s ORDER BY id", (target,))
        yield from cur                     # 한 행씩 흘려보냄

def export_orders(rows: Iterator[tuple], path: Path) -> int:
    n = 0
    with path.open("w", newline="", encoding="utf-8") as f:
        w = csv.writer(f)
        w.writerow(["id", "user_id", "amount", "paid_at"])
        for row in rows:                   # DB 타입 그대로 — 검증 · 파싱 없음
            w.writerow(row)                # datetime · Decimal 은 str() 로 기록
            n += 1
    return n

# gunicorn.conf.py — uv add gunicorn uvicorn-worker
import os

def _cpu_limit() -> int:
    # 컨테이너 CPU 제한(cgroup v2)을 우선 — os.cpu_count() 는 노드 전체 코어
    try:
        quota, period = open("/sys/fs/cgroup/cpu.max").read().split()
        if quota != "max":
            return max(1, int(int(quota) / int(period)))
    except OSError:
        pass
    return os.process_cpu_count() or 1        # 3.13+ — affinity 반영

bind = "0.0.0.0:8000"
workers = int(os.getenv("WEB_CONCURRENCY", _cpu_limit()))   # async 워커: 코어당 1
worker_class = "uvicorn_worker.UvicornWorker"
timeout = 30                    # 워커가 이 시간 동안 응답(하트비트)이 없으면 재시작
                                #   ※ async 워커에선 "요청 시간 제한"이 아닙니다 — 루프가 막혔을 때만 걸림
graceful_timeout = 25           # SIGTERM 후 진행 중 요청을 마칠 시간
keepalive = 5
max_requests = 2000             # 느린 누수 완화 — 근본 해결은 PROD 09
max_requests_jitter = 200       # 워커들이 동시에 재시작하지 않게

# src/billing/api/schemas.py
from datetime import datetime
from decimal import Decimal
from typing import Annotated, Literal

from pydantic import (BaseModel, BeforeValidator, ConfigDict, Field,
                      field_validator, model_validator)

Currency = Literal["KRW", "USD", "JPY"]
Money = Annotated[Decimal, Field(gt=0, max_digits=12, decimal_places=2)]

def _strip_upper(v: object) -> object:
    return v.strip().upper() if isinstance(v, str) else v   # " krw" → "KRW"

class RefundRequest(BaseModel):
    model_config = ConfigDict(extra="forbid",       # 모르는 필드 = 422
                              str_strip_whitespace=True,
                              frozen=True)

    order_id: Annotated[str, Field(pattern=r"^ord_[0-9A-Za-z]{12}$")]
    amount: Money
    currency: Annotated[Currency, BeforeValidator(_strip_upper)] = "KRW"
    reason: Annotated[str, Field(min_length=2, max_length=200)]
    requested_at: datetime

    @field_validator("requested_at")
    @classmethod
    def must_be_aware(cls, v: datetime) -> datetime:
        if v.tzinfo is None:
            raise ValueError("시간대가 없는 시각은 받지 않습니다")
        return v

    @model_validator(mode="after")
    def krw_has_no_decimals(self) -> "RefundRequest":
        if self.currency == "KRW" and self.amount != self.amount.to_integral():
            raise ValueError("KRW 는 소수점 금액이 없습니다")
        return self
extra="forbid" 를 기본으로 — 기본값(ignore)이면 클라이언트가 amout 처럼 오타를 내도 조용히 버려지고 기본값이 들어갑니다. 받는 쪽에서 오타를 422 로 돌려줘야 연동 초기에 바로 잡힙니다.

# src/billing/api/metrics.py — uv add prometheus-client
import time

from fastapi import FastAPI, Request
from prometheus_client import Counter, Histogram, make_asgi_app

REQS = Counter("http_requests_total", "요청 수",
               ["route", "method", "status"])
LAT = Histogram("http_request_duration_seconds", "요청 지연",
                ["route", "method"],
                buckets=(.025, .05, .1, .25, .5, 1, 2.5, 5))

def install(app: FastAPI) -> None:
    @app.middleware("http")
    async def _metrics(request: Request, call_next):
        start, status = time.perf_counter(), 500
        try:
            resp = await call_next(request)
            status = resp.status_code
            return resp
        finally:
            route = request.scope.get("route")
            path = getattr(route, "path", "unmatched")  # /orders/{order_id}
            REQS.labels(path, request.method, str(status)).inc()
            LAT.labels(path, request.method).observe(time.perf_counter() - start)

    app.mount("/metrics", make_asgi_app())
# 배치 — 끝날 때 "마지막 성공 시각"을 Pushgateway 로
from prometheus_client import CollectorRegistry, Gauge, push_to_gateway

                ["batch"], registry=reg)
last_ok.labels("settle").set_to_current_time()
push_to_gateway("pushgateway:9091", job="billing-batch", registry=reg)

# prometheus/rules/billing.yml
groups:
  - name: billing
    rules:
      - alert: BillingHighErrorRate
        expr: |
          sum(rate(http_requests_total{job="billing",status=~"5.."}[5m]))
            / sum(rate(http_requests_total{job="billing"}[5m])) > 0.02
        for: 10m
        labels: {severity: page}
        annotations:
          summary: "billing 5xx {{ $value | humanizePercentage }}"
          runbook_url: "https://wiki.example.com/runbooks/billing#5xx"

      - alert: BillingSlowP99
        expr: |
          histogram_quantile(0.99, sum by (le) (
            rate(http_request_duration_seconds_bucket{job="billing"}[5m]))) > 1.5
        for: 15m
        labels: {severity: page}
        annotations:
          runbook_url: "https://wiki.example.com/runbooks/billing#slow"

      - alert: BillingSettleMissing
        expr: time() - batch_last_success_timestamp_seconds{batch="settle"} > 26 * 3600
        labels: {severity: ticket}
        annotations:
          runbook_url: "https://wiki.example.com/runbooks/billing#settle"

실무에선 이렇게 — 장애 첫 15분과 런북 템플릿 🔴 실무

장애 첫 15분 — 순서대로

분할 일
0~2선언 — 채널에 "billing 5xx 확인 중, 담당 ○○" 한 줄
2~5범위 — 전체인가 일부 라우트인가, 언제부터인가 (대시보드)
5~7최근 변경 — 배포 · 설정 · 마이그레이션 · 외부 연동 공지
7~10완화 — 최근 배포면 롤백 · 기능 플래그 끔 · 스케일 아웃
10~15증거 확보 — py-spy dump · 에러 로그 샘플 · 지표 스크린샷
이후영향 멈춘 뒤 원인 분석 → 사후 검토 문서
# 증거부터 — 재시작하면 사라집니다
kubectl logs deploy/billing-api --since=15m | grep -E '"level":"(ERROR|CRITICAL)"' | head -50
kubectl top pod -l app=billing-api
kubectl exec deploy/billing-api -- py-spy dump --pid 1 > dump-$(date +%H%M).txt
완화가 원인 분석보다 먼저입니다. 원인을 찾느라 40분을 쓰는 동안 결제가 계속 실패하는 것이 가장 비싼 장애 대응입니다. 다만 재시작 · 롤백 전에 증거 한 장(스택 덤프 · 로그 샘플)은 남겨 두세요 — 재시작하면 재현이 안 되는 문제가 많습니다.
