match 문 — 모양으로 분기하기 🟡 중급

쉽게 말하면 — match 는 “값이 같은가”보다 “이 모양으로 생겼는가”를 묻습니다. 모양이 맞으면 그 안의 값을 이름에 바로 담아 줍니다. 그래서 JSON · 명령어 · 이벤트처럼 모양이 여러 가지인 데이터를 다룰 때 강합니다.
match 는 위에서부터 처음 맞는 case 하나만 실행합니다. 넓은 패턴을 위에 두면 아래 case 는 영영 안 탑니다.

패턴 종류 한 장

def handle(cmd):
    match cmd:
        # 리터럴 — 값 그대로 비교
        case "quit" | "exit":           # | 는 '또는'
            return "bye"

        # 시퀀스 — 길이와 순서로
        case ["go", direction]:
            return f"move {direction}"
        case ["drop", *items]:          # 나머지 전부
            return f"drop {len(items)}"

        # 매핑 — 필요한 키만 보면 됨 (나머지 키는 무시)
        case {"op": "add", "a": a, "b": b}:
            return a + b

        # 클래스 — 타입 검사 + 속성 꺼내기
        case Point(x=0, y=0):
            return "원점"
        case Point(x=x, y=y) if x == y:  # 가드 — 추가 조건
            return "대각선"

        # 캡처 + 타입 검사
        case int(n) | float(n):
            return n * 2

        case _:
            raise ValueError(f"모르는 명령: {cmd!r}")

실무 예 — JSON 이벤트 라우팅

def route(ev: dict) -> str:
    match ev:
        case {"type": "order", "status": "paid", "id": oid}:
            return f"배송 준비 {oid}"
        case {"type": "order", "status": "cancelled" | "refunded"
              as st, "id": oid}:

        case {"type": "user", "action": "signup",
              "email": str(email)} if "@" in email:
            return f"환영 메일 {email}"
        case {"type": t}:
            return f"무시: {t}"

            return "형식 오류"

route({"type": "order", "status": "paid", "id": 7})
# '배송 준비 7'
가장 흔한 실수 — 상수 이름. case RED: 처럼 점 없는 이름은 비교가 아니라 캡처라서 무엇이든 맞고 RED 에 값을 담아 버립니다. 상수는 case Color.RED: 처럼 점이 있는 이름으로 써야 비교가 됩니다.

클래스 패턴과 __match_args__ 🔴 실무

from dataclasses import dataclass

@dataclass
class Point:
    x: int
    y: int
# dataclass 는 __match_args__ = ("x", "y") 를 자동으로 만듦

match Point(3, 3):
    case Point(0, 0):          # 위치로 써도 됨 (__match_args__ 덕분)
        print("원점")
    case Point(x, y) if x == y:
        print("대각선", x)

# 일반 클래스는 직접 적어야 위치 패턴이 됩니다
class Rect:
    __match_args__ = ("w", "h")
    def __init__(self, w, h): self.w, self.h = w, h

match Rect(2, 2):
    case Rect(w, h) if w == h: print("정사각형")

from itertools import groupby
from operator import itemgetter

rows = [("서울", 3), ("부산", 1), ("서울", 2)]

# ❌ 정렬 안 하면 '연속된 것'만 묶음 → 서울이 두 그룹
[(k, list(g)) for k, g in groupby(rows, key=itemgetter(0))]
# [('서울', [...]), ('부산', [...]), ('서울', [...])]

# ✅ 같은 키로 정렬 후 묶기
rows.sort(key=itemgetter(0))
for city, grp in groupby(rows, key=itemgetter(0)):
    print(city, sum(n for _, n in grp))
# 부산 1 / 서울 5

# 정렬하기 싫다면 그냥 dict 로 모으기 (대개 이쪽이 낫다)
from collections import defaultdict
by_city = defaultdict(list)
for city, n in rows:
    by_city[city].append(n)

ADV 04

이터레이터 프로토콜과 제너레이터 심화

for x in obj 는 마법이 아니라 두 메서드의 약속입니다 — iter(obj) 로 반복자를 받고, next() 를 StopIteration 이 날 때까지 부릅니다. 제너레이터는 이 약속을 yield 한 줄로 자동 구현해 주는 함수입니다.

제너레이터는 yield 에서 멈추고 next 가 올 때 그 줄부터 다시 시작합니다. 다 끝나면 StopIteration 을 던지고, for 는 그걸 “끝” 신호로 씁니다.

직접 구현 vs 제너레이터

# ① 클래스로 직접 — 프로토콜이 다 보임
class Countdown:
    def __init__(self, n): self.n = n
    def __iter__(self):            # 반복 가능(iterable)
        return CountdownIter(self.n)

class CountdownIter:
    def __init__(self, n): self.n = n
    def __iter__(self): return self   # 반복자는 자기 자신을 돌려줌
    def __next__(self):
        if self.n <= 0:
            raise StopIteration
        self.n -= 1
        return self.n + 1

# ② 제너레이터 — 위 두 클래스와 같은 일
class Countdown:
    def __init__(self, n): self.n = n
    def __iter__(self):
        n = self.n
        while n > 0:
            yield n
            n -= 1

list(Countdown(3))   # [3, 2, 1]
list(Countdown(3))   # [3, 2, 1] — __iter__ 가 매번 새 반복자
iterable vs iterator — 반복 가능 객체(list · Countdown)는 여러 번 돌 수 있고, 반복자(제너레이터 · iter(list))는 한 번만 돕니다. 컬렉션 클래스를 만들 땐 __iter__ 를 제너레이터로 쓰면 둘 다 공짜로 해결됩니다.

제너레이터 파이프라인 — 10GB 로그도 메모리 몇 KB

def read_lines(path):
    with open(path, encoding="utf-8") as f:
        for line in f:                 # 파일도 한 줄씩 지연
            yield line.rstrip("\n")

def only_errors(lines):
    for ln in lines:
        if " ERROR " in ln:
            yield ln

def parse(lines):
    for ln in lines:
        ts, _, msg = ln.split(" ", 2)
        yield {"ts": ts, "msg": msg}

# 조립 — 아직 아무것도 읽지 않았습니다
pipe = parse(only_errors(read_lines("app.log")))

# 여기서 한 줄씩 흘러감 (앞 5건만 보고 멈춤)
from itertools import islice
for ev in islice(pipe, 5):
    print(ev)

yield from · send · return 값 🔴 실무

yield from — 다른 반복에 위임

# 트리를 깊이 우선으로 펼치기
def walk(node):
    yield node.value
    for child in node.children:
        yield from walk(child)     # 자식 제너레이터의 값을 그대로 흘림

# yield from 은 for-yield 보다 하는 일이 많습니다
#  ① 값 전달  ② send/throw/close 를 안쪽까지 전달
#  ③ 안쪽 제너레이터의 return 값을 받아 옴
def inner():
    yield 1
    yield 2
    return "done"                  # StopIteration.value 로 전달

def outer():
    result = yield from inner()    # result == "done"
    yield f"inner 가 {result}"

list(outer())   # [1, 2, 'inner 가 done']

send — 값을 안으로 넣기 (코루틴의 조상)

def running_avg():
    total = count = 0
    avg = None
    while True:
        x = yield avg          # 바깥이 send 한 값이 x 로 들어옴
        total += x; count += 1
        avg = total / count

g = running_avg()
next(g)          # ★ 첫 yield 까지 먼저 진행 (프라이밍)
g.send(10)       # 10.0
g.send(20)       # 15.0
g.close()        # GeneratorExit 를 안에서 발생시켜 종료

# 실무 감각
#  - send 는 asyncio 이전 코루틴의 원리였고, 지금 직접 쓸 일은 드뭅니다
#  - 상태를 가진 계산기는 보통 클래스가 더 읽기 쉽습니다
#  - 하지만 라이브러리 코드(pytest fixture · contextmanager)에서
#    yield 가 '중간에 멈추는 지점'으로 쓰이는 걸 알아보는 데 필요합니다
상황list 로 반환제너레이터로 yield
결과를 여러 번 순회 · 길이 필요✅❌ 한 번만 돔
결과가 크거나 끝이 없음❌ 메모리 폭발✅
앞 몇 개만 쓰고 멈출 수 있음전부 계산함✅ 필요한 만큼만
예외를 호출 시점에 바로 보고 싶음✅ 호출할 때 실행⚠ 첫 next() 때까지 아무것도 안 함
정리 — for 는 iter() + next() + StopIteration 의 약속이고, 제너레이터는 그 약속을 yield 로 자동 구현합니다. 큰 데이터는 제너레이터를 이어 붙인 파이프라인으로 흘리세요.
ADV 05

스코프 · 클로저 · 늦은 바인딩

파이썬이 이름을 찾는 순서는 L → E → G → B 네 칸뿐입니다. 이 순서와 “함수는 값이 아니라 변수를 기억한다”는 한 가지만 알면 UnboundLocalError, 람다 루프 버그, nonlocal 이 전부 한 번에 풀립니다.

함수 안 어디서든 대입이 한 번 있으면 그 이름은 함수 시작부터 Local 로 취급됩니다 — 대입 전에 읽으면 UnboundLocalError.

UnboundLocalError · global · nonlocal

count = 0
def bump():
    count += 1        # ❌ UnboundLocalError
    # 대입(+=)이 있으니 count 는 Local 인데, 아직 값이 없음

def bump():
    global count      # 모듈 변수를 쓰겠다 (가능하면 피하기)
    count += 1

# 클로저 — 바깥 함수의 변수를 기억하는 안쪽 함수
def make_counter():
    n = 0
    def inc():
        nonlocal n    # 바깥(Enclosing) n 을 다시 묶겠다
        n += 1
        return n
    return inc

c = make_counter()
c(); c()          # 1, 2
c.__closure__[0].cell_contents   # 2 — 기억 중인 값

# 변경(mutate)은 선언이 필요 없음 — 대입이 아니라서
def make_log():
    items = []
    def add(x):
        items.append(x)    # ✅ nonlocal 없이 동작
        return items
    return add

클래스 본문은 스코프가 아닙니다 — 클래스 안 메서드에서 클래스 변수를 그냥 이름으로 읽으면 NameError. 반드시 self.x 나 type(self).x 로 읽습니다. 클래스 본문은 E 단계에 끼지 않습니다.

클로저를 언제 쓰나 — 작은 상태 + 함수 하나 🟡 중급

# 설정을 굳힌 함수 공장
def make_formatter(unit: str, digits: int = 1):
    def fmt(v: float) -> str:
        return f"{v:.{digits}f}{unit}"
    return fmt

won = make_formatter("원", 0)
pct = make_formatter("%")
won(12500)    # '12500원'
pct(3.14159)  # '3.1%'

# 간단한 메모이제이션 (실무에선 functools.cache)
def memo(fn):
    seen = {}
    def wrapper(x):
        if x not in seen:
            seen[x] = fn(x)
        return seen[x]
    return wrapper

ADV 06

데코레이터 실무 — 인자 · wraps · 클래스 데코레이터

@deco 는 f = deco(f) 의 줄임말일 뿐입니다. 이 한 줄을 기억하면 인자를 받는 데코레이터가 왜 함수 세 겹인지, 여러 개를 쌓으면 왜 아래부터 감싸는지 전부 설명됩니다.

데코레이터 순서가 결과를 바꿉니다. @timer 가 바깥이면 재시도 시간까지 합쳐 재고, 안쪽이면 한 번 시도 시간만 잽니다.

기본형 — 반드시 wraps

import functools, time

def timer(fn):
    @functools.wraps(fn)          # ★ 이름·docstring·시그니처 보존
    def wrapper(*args, **kwargs):
        t0 = time.perf_counter()
        try:
            return fn(*args, **kwargs)
        finally:                  # 예외가 나도 시간은 찍음
            ms = (time.perf_counter() - t0) * 1000
            print(f"{fn.__name__}: {ms:.1f}ms")
    return wrapper

@timer
def load(path): ...

load.__name__        # 'load' (wraps 없으면 'wrapper')
load.__wrapped__     # 원래 함수 — 테스트에서 데코레이터 우회
wraps 를 빼면 모든 함수 이름이 wrapper 가 돼서 로그 · 프로파일러 · pytest 출력 · FastAPI 문서가 전부 엉킵니다. FastAPI 처럼 시그니처를 읽는 프레임워크는 wraps 가 없으면 인자 주입이 아예 깨집니다.

인자를 받는 데코레이터 — 세 겹

def retry(times=3, on=(ConnectionError,), delay=0.5):
    def deco(fn):                         # ② 함수를 받음
        @functools.wraps(fn)
        def wrapper(*args, **kwargs):     # ③ 실제 호출
            for attempt in range(1, times + 1):
                try:
                    return fn(*args, **kwargs)
                except on as e:
                    if attempt == times:
                        raise
                    time.sleep(delay * 2 ** (attempt - 1))
        return wrapper
    return deco                           # ① 설정을 받음

@retry(times=5, on=(TimeoutError,))   # retry(...) 가 먼저 실행 → deco 반환
def fetch(url): ...

# 풀어 쓰면
fetch = retry(times=5, on=(TimeoutError,))(fetch)

ADV 07

컨텍스트 매니저 — with 를 직접 만들기

with 는 “들어갈 때 준비, 나올 때 무조건 정리”를 보장하는 문법입니다. 파일 · 락 · DB 트랜잭션 · 임시 디렉터리 · 타이머 — “짝이 맞아야 하는 두 동작”이 보이면 전부 컨텍스트 매니저 후보입니다.

__exit__ 가 True 를 돌려주면 예외가 사라집니다. 의도한 게 아니면 아무것도 돌려주지 마세요(None) — 예외가 정상적으로 위로 올라갑니다.

클래스로 — 프로토콜이 다 보임

import time

class Timer:
    def __enter__(self):
        self.t0 = time.perf_counter()
        return self                 # as t 에 들어갈 값
    def __exit__(self, exc_type, exc, tb):
        self.ms = (time.perf_counter() - self.t0) * 1000
        # return 생략 = None → 예외는 그대로 올라감

with Timer() as t:
    work()
print(f"{t.ms:.1f}ms")             # with 밖에서도 결과 사용

# 트랜잭션 — 성공하면 커밋, 실패하면 롤백
class Tx:
    def __init__(self, conn): self.conn = conn
    def __enter__(self): return self.conn.cursor()
    def __exit__(self, et, e, tb):

@contextmanager — 제너레이터로 짧게

from contextlib import contextmanager
import os

@contextmanager
def cd(path):
    old = os.getcwd()
    os.chdir(path)              # ── __enter__ 부분
    try:
        yield path              # ── as 로 넘길 값 · 본문 실행 지점
    finally:
        os.chdir(old)           # ── __exit__ 부분 (예외가 나도)

with cd("/tmp"):
    ...
# (3.11+ 에는 contextlib.chdir 가 이미 있습니다)

@contextmanager
def transaction(conn):
    cur = conn.cursor()
    try:
        yield cur
    except Exception:
        conn.rollback()
        raise                   # ★ 다시 던져야 호출자가 압니다
    else:
        conn.commit()
try/finally 를 빼먹는 실수 — yield 를 try 로 감싸지 않으면 본문에서 예외가 날 때 yield 줄에서 예외가 터져 정리 코드가 실행되지 않습니다.

from decimal import Decimal
from functools import total_ordering

@total_ordering                 # __eq__ + __lt__ 만 쓰면 <= > >= 자동
class Money:
    __slots__ = ("amount", "cur")

    def __init__(self, amount, cur="KRW"):
        self.amount = Decimal(str(amount))   # float 오차 방지
        self.cur = cur

    def __repr__(self):         # 개발자용 — 다시 만들 수 있는 모양
        return f"Money({str(self.amount)!r}, {self.cur!r})"

    def __str__(self):          # 사용자용
        return f"{self.amount:,} {self.cur}"

    def __eq__(self, other):
        if not isinstance(other, Money):
            return NotImplemented      # ★ False 가 아니라 NotImplemented
        return (self.amount, self.cur) == (other.amount, other.cur)

    def __hash__(self):         # __eq__ 를 쓰면 반드시 같이
        return hash((self.amount, self.cur))

    def __lt__(self, other):
        if not isinstance(other, Money) or other.cur != self.cur:
            return NotImplemented
        return self.amount < other.amount

다중 상속 · MRO · super() 🔴 실무

쉽게 말하면 — 파이썬은 상속 관계를 한 줄로 줄 세운 목록(MRO, Method Resolution Order)을 만들어 두고, 메서드를 찾을 때 그 줄을 앞에서부터 봅니다. super() 는 “내 부모”가 아니라 “그 줄에서 내 다음 칸”을 부릅니다.
Left 안의 super() 가 부모 Base 가 아니라 형제 Right 를 부릅니다. 모든 클래스가 super() 를 부르는 한, 공통 조상은 한 번만 실행됩니다.
class Base:
    def __init__(self, **kw):
        print("Base"); super().__init__(**kw)   # object 까지 넘김

class Left(Base):
    def __init__(self, left=None, **kw):
        print("Left"); super().__init__(**kw)

class Right(Base):
    def __init__(self, right=None, **kw):
        print("Right"); super().__init__(**kw)

class Child(Left, Right):
    def __init__(self, **kw):
        print("Child"); super().__init__(**kw)

Child(left=1, right=2)     # Child Left Right Base — Base 한 번
[c.__name__ for c in Child.__mro__]
# ['Child', 'Left', 'Right', 'Base', 'object']
협력적 다중 상속의 규칙 — ① 모든 클래스가 super() 를 부르고 ② 자기 인자만 꺼낸 뒤 나머지는 **kw 로 넘깁니다. 한 클래스라도 Base.__init__(self) 처럼 이름으로 직접 부르면 줄이 끊겨 형제가 실행되지 않습니다.

import json

    def __repr__(self):

        return f"{type(self).__name__}({kv})"

ABC — 구현 누락을 생성 시점에

from abc import ABC, abstractmethod

class Exporter(ABC):
    @abstractmethod
    def write(self, rows: list[dict]) -> bytes: ...

    @property
    @abstractmethod
    def ext(self) -> str: ...

    # 공통 구현 — 템플릿 메서드
    def save(self, rows, path):
        data = self.write(rows)
        with open(f"{path}.{self.ext}", "wb") as f:
            f.write(data)

class CsvExporter(Exporter):
    ext = "csv"
    def write(self, rows):
        ...

class BadExporter(Exporter):
    ext = "x"                  # write 를 빠뜨림

Exporter()        # TypeError: Can't instantiate abstract class
BadExporter()     # TypeError: ... without an implementation
                  #            for abstract method 'write'

__init_subclass__ — 상속만 하면 자동 등록

class Exporter(ABC):
    registry: dict[str, type["Exporter"]] = {}

    def __init_subclass__(cls, /, fmt: str | None = None, **kw):
        super().__init_subclass__(**kw)
        if fmt:                          # 클래스 문장 키워드로 받은 값
            Exporter.registry[fmt] = cls

    @abstractmethod
    def write(self, rows): ...

class CsvExporter(Exporter, fmt="csv"):
    def write(self, rows): ...

class JsonExporter(Exporter, fmt="json"):
    def write(self, rows): ...

def export(rows, fmt):
    try:
        cls = Exporter.registry[fmt]
    except KeyError:
        raise ValueError(f"지원 형식: {list(Exporter.registry)}") from None
    return cls().write(rows)

# if fmt == "csv": ... elif fmt == "json": ... 가 사라집니다.
# 새 형식 = 새 클래스 하나. 기존 코드 수정 0줄.
메타클래스로 하던 일의 대부분(등록 · 검증 · 속성 강제)은 __init_subclass__ 로 됩니다. 메타클래스가 정말 필요한 경우는 🧬 전문가 탭 DEEP 06.

__getattr__ — 없는 속성만 위임

class LoggedClient:
    """모든 메서드 호출을 로그로 남기는 래퍼 (상속 대신 위임)"""
    def __init__(self, inner):
        self._inner = inner

    def __getattr__(self, name):        # ★ 일반 조회가 실패했을 때만 호출
        attr = getattr(self._inner, name)
        if callable(attr):
            def logged(*a, **kw):
                print(f"→ {name}{a}")
                return attr(*a, **kw)
            return logged
        return attr

디스크립터 + __set_name__ — 검증 필드 재사용

class Positive:
    def __set_name__(self, owner, name):   # 클래스 만들 때 이름을 알려 줌
        self.name = "_" + name
    def __get__(self, obj, objtype=None):
        if obj is None: return self
        return getattr(obj, self.name)
    def __set__(self, obj, value):
        if value <= 0:
            raise ValueError(f"{self.name[1:]} 는 양수여야 합니다")
        setattr(obj, self.name, value)

class Product:
    price = Positive()       # property 를 매번 쓰는 대신
    stock = Positive()       # 같은 규칙을 한 줄로 재사용
    def __init__(self, price, stock):
        self.price, self.stock = price, stock

Product(1000, 0)   # ValueError: stock 는 양수여야 합니다
# property · classmethod 도 내부적으로 디스크립터입니다 (DEEP 05)

    except* ValueError as eg:

def validate(row: dict):

import asyncio, httpx

async def fetch(client, url):
    r = await client.get(url, timeout=10)   # 기다리는 동안 다른 작업 실행
    return url, r.status_code

async def main(urls):
    sem = asyncio.Semaphore(20)             # 동시 20개로 제한
    async with httpx.AsyncClient() as client:
        async def limited(u):
            async with sem:
                return await fetch(client, u)
        async with asyncio.TaskGroup() as tg:
            tasks = [tg.create_task(limited(u)) for u in urls]
    return [t.result() for t in tasks]

results = asyncio.run(main(urls))
