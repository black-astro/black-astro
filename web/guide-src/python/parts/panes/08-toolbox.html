TOOLBOX 01

파일·폴더 다루기 🟢 기초 · 설치 불필요

자동화의 절반은 파일 옮기기입니다. 경로는 pathlib, 복사/이동은 shutil — 이 조합이 표준입니다.

pathlib — 경로의 표준

from pathlib import Path

p = Path("D:/보고서") / "2026" / "매출.xlsx"   # OS 무관 경로 조합

p.exists()      # 있는지
p.is_file()     # 파일인지 / p.is_dir()
p.name          # "매출.xlsx"
p.stem          # "매출"
p.suffix        # ".xlsx"
p.parent        # D:/보고서/2026
p.stat().st_size / 1024   # KB

# 폴더 만들기 (있어도 에러 안 남 — 이 형태로 외우세요)
Path("logs").mkdir(parents=True, exist_ok=True)

# 파일 목록
list(Path("data").glob("*.csv"))        # 이 폴더만
list(Path("data").glob("**/*.csv"))     # 하위 폴더까지
for f in Path("data").iterdir(): ...    # 전부

# 읽기/쓰기 한 줄
text = p.read_text(encoding="utf-8")
p.write_text("내용", encoding="utf-8")

# 이름 바꾸기 / 삭제
p.rename(p.with_name("매출_백업.xlsx"))
p.unlink(missing_ok=True)               # 파일 삭제

shutil — 복사·이동·통삭제

import shutil

shutil.copy("a.txt", "backup/a.txt")     # 파일 복사
shutil.copy2("a.txt", "backup/")         # 수정시각까지 보존
shutil.copytree("프로젝트", "백업/프로젝트")  # 폴더 통째 복사
shutil.move("a.txt", "정리됨/a.txt")      # 이동 (드라이브 넘어도 OK)
shutil.rmtree("임시폴더")                 # 폴더 통째 삭제 ⚠️ 복구 불가
shutil.disk_usage("D:/")                 # 디스크 용량

실전 — 다운로드 폴더 자동 정리

from pathlib import Path
import shutil

규칙 = {".pdf": "문서", ".xlsx": "엑셀", ".png": "이미지",
        ".jpg": "이미지", ".zip": "압축"}

down = Path.home() / "Downloads"
for f in down.iterdir():
    if not f.is_file():
        continue
    폴더 = down / 규칙.get(f.suffix.lower(), "기타")
    폴더.mkdir(exist_ok=True)
    shutil.move(str(f), str(폴더 / f.name))
    print(f"{f.name} → {폴더.name}/")
os.path는 이제 안 써도 됩니다 — 옛 코드에서 보이는 os.path.join, os.listdir는 전부 pathlib로 대체됐습니다. 새 코드는 pathlib로 통일하세요. 단 os.environ(환경변수)과 os.getpid() 같은 것은 여전히 os 담당입니다.
파일 이동은 한 줄 shutil.move("보고서.xlsx", "정리됨/보고서.xlsx")

날짜·시간 계산 🟢 기초 · 설치 불필요

“어제 날짜 파일명”, “지난주 데이터만” — 자동화마다 반드시 나옵니다.

기본기

from datetime import datetime, date, timedelta

now = datetime.now()          # 2026-07-31 14:30:22.123
today = date.today()          # 2026-07-31

# 날짜 -> 문자열 (strftime = string format time)
now.strftime("%Y-%m-%d")            # "2026-07-31"
now.strftime("%Y%m%d_%H%M%S")       # "20260731_143022"
now.strftime("%Y년 %m월 %d일")       # 한글도 OK

# 문자열 -> 날짜 (strptime = string parse time)
d = datetime.strptime("2026-07-31", "%Y-%m-%d")

# 날짜 연산 — timedelta
어제   = today - timedelta(days=1)
다음주 = today + timedelta(weeks=1)
차이   = date(2026, 12, 25) - today
차이.days                            # 147

now.year  now.month  now.day
now.weekday()        # 월=0 … 일=6

실전 레시피

from datetime import date, timedelta

today = date.today()

# 오늘 날짜가 든 파일명
파일명 = f"매출리포트_{today:%Y%m%d}.xlsx"

# 이번 주 월요일
월요일 = today - timedelta(days=today.weekday())

# 지난달 1일 ~ 말일
이달1일 = today.replace(day=1)
지난달말 = 이달1일 - timedelta(days=1)
지난달1일 = 지난달말.replace(day=1)

# 30일 지난 파일 삭제 (로그 청소)
from pathlib import Path
import time
기준 = time.time() - 30 * 86400
for f in Path("logs").glob("*.log"):
    if f.stat().st_mtime < 기준:
        f.unlink()

# 시간대가 필요하면 (표준 zoneinfo)
from zoneinfo import ZoneInfo
datetime.now(ZoneInfo("Asia/Seoul"))
포맷 기호 5개만 외우세요 — %Y 연(4자리) %m 월 %d 일 %H 시(24h) %M 분. f-string 안에서 f"{today:%Y%m%d}"처럼 바로 쓸 수 있습니다.
TOOLBOX 03

CSV · JSON 🟢 기초 · 설치 불필요

데이터 교환의 양대 포맷. 표는 CSV, 구조화된 설정·API 응답은 JSON입니다.

csv

import csv

# 읽기 — DictReader 를 쓰세요 (컬럼명으로 접근)
with open("사원.csv", encoding="utf-8-sig", newline="") as f:
    for row in csv.DictReader(f):
        print(row["이름"], row["매출"])   # 값은 전부 문자열!

# 쓰기
rows = [{"이름": "김민준", "매출": 320},
        {"이름": "이서연", "매출": 210}]
with open("out.csv", "w", encoding="utf-8-sig", newline="") as f:
    w = csv.DictWriter(f, fieldnames=["이름", "매출"])
    w.writeheader()
    w.writerows(rows)
2가지 필수 옵션 — newline=""이 없으면 Windows에서 빈 줄이 한 줄씩 낍니다. encoding="utf-8-sig"가 없으면 엑셀에서 한글이 깨집니다. 표 계산이 필요하면 이 시점에서 pandas(pd.read_csv)로 넘어가는 게 낫습니다 → 탭 3.

json

import json
from pathlib import Path

# 문자열 <-> 파이썬 객체
data = json.loads('{"이름": "김민준", "점수": [90, 85]}')
data["점수"][0]                    # 90 — dict/list 그대로

s = json.dumps(data, ensure_ascii=False, indent=2)
# ensure_ascii=False 없으면 한글이 \uae40 으로 저장됨!

# 파일 <-> 객체
data = json.loads(Path("config.json").read_text(encoding="utf-8"))
Path("out.json").write_text(
    json.dumps(data, ensure_ascii=False, indent=2),
    encoding="utf-8")

# 안전하게 꺼내기 (키가 없을 수도 있을 때)
이름 = data.get("이름", "무명")
도시 = data.get("주소", {}).get("도시", "미상")   # 중첩도 안전
설정 파일 패턴 — 자동화 스크립트의 대상 경로·옵션을 코드에 하드코딩하지 말고 config.json으로 빼두면, 코드 수정 없이 동작을 바꿀 수 있습니다.

압축 — zipfile · tarfile 🟢 기초 · 설치 불필요

백업 만들기, 받은 압축 풀기. 폴더 통째면 shutil 한 줄이 제일 쉽습니다.

가장 쉬운 방법 — shutil

import shutil

# 폴더 통째로 압축 (zip / gztar / tar 지원)
shutil.make_archive("백업_20260731", "zip", "D:/프로젝트")
#  -> 백업_20260731.zip 생성

# 통째로 해제 (형식 자동 감지)
shutil.unpack_archive("백업_20260731.zip", "복원폴더")

zipfile — 세밀한 제어

import zipfile
from pathlib import Path

# 원하는 파일만 골라 담기
with zipfile.ZipFile("리포트.zip", "w",
                     zipfile.ZIP_DEFLATED) as z:   # DEFLATED=실제 압축
    for f in Path("output").glob("*.xlsx"):
        z.write(f, arcname=f.name)    # arcname: 압축 내 이름

# 목록 보고 풀기
with zipfile.ZipFile("받은파일.zip") as z:
    print(z.namelist())
    z.extractall("풀린폴더")
    z.extract("특정파일.csv", "풀린폴더")   # 하나만

tarfile — 리눅스 서버와 주고받기

import tarfile

# tar.gz 만들기
with tarfile.open("logs.tar.gz", "w:gz") as t:
    t.add("logs", arcname="logs")

# 풀기 (3.12+ 는 filter="data" 로 안전하게)
with tarfile.open("logs.tar.gz") as t:
    t.extractall("복원", filter="data")
모르는 압축 파일 주의 — 외부에서 받은 zip/tar 안에 ../../ 경로가 숨어 있으면 엉뚱한 위치에 파일이 풀릴 수 있습니다(Zip Slip 공격). 신뢰할 수 없는 파일은 filter="data"(tar) 를 쓰고, zip은 풀기 전에 namelist()로 경로를 확인하세요.
한글 파일명이 깨질 때 — Windows에서 만든 zip은 cp949로 인코딩된 경우가 많습니다. zipfile.ZipFile(p, metadata_encoding="cp949") (3.11+) 로 열면 해결됩니다.

로그 — logging 🟢 기초 · 설치 불필요

새벽에 혼자 돌다 죽은 자동화의 사인은 로그에만 남습니다. print를 졸업하는 순간 운영이 시작됩니다.

복붙용 표준 설정

import logging
from pathlib import Path

Path("logs").mkdir(exist_ok=True)
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
    handlers=[
        logging.FileHandler("logs/app.log", encoding="utf-8"),
        logging.StreamHandler(),        # 콘솔에도 같이
    ],
)
log = logging.getLogger(__name__)

log.debug("개발 중에만 보는 상세")     # 기본 level 에선 안 찍힘
log.info("작업 시작: 파일 32건")
log.warning("응답 느림 (3.2s)")
log.error("저장 실패: %s", path)

try:
    위험한_작업()
except Exception:
    log.exception("처리 실패")   # ★ 스택트레이스까지 자동 기록

로그 파일이 무한히 커지지 않게

from logging.handlers import RotatingFileHandler, TimedRotatingFileHandler

# 5MB 넘으면 새 파일, 3개까지 보관
RotatingFileHandler("logs/app.log", maxBytes=5*1024*1024,
                    backupCount=3, encoding="utf-8")

# 자정마다 새 파일, 14일 보관
TimedRotatingFileHandler("logs/app.log", when="midnight",
                         backupCount=14, encoding="utf-8")
print와의 결정적 차이 — 로그는 ① 시각이 자동으로 붙고 ② 파일에 남고 ③ 레벨로 걸러지고 ④ log.exception이 에러 위치까지 기록합니다. 자동화 스크립트에서 print는 “화면에 흘러가고 끝”이라 새벽 장애 원인을 알 수 없습니다.
Spring 비유 — 정확히 SLF4J + Logback입니다. getLogger(__name__)은 클래스별 Logger 선언, basicConfig는 logback.xml에 해당합니다.
TOOLBOX 06

로컬 DB — sqlite3 🟢 기초 · 설치 불필요

서버 없이 파일 하나가 DB입니다. “처리한 목록 기억하기”, “수집 데이터 쌓기”에 CSV보다 훨씬 안전합니다.

기본 사용

import sqlite3

conn = sqlite3.connect("app.db")     # 파일이 없으면 생성
conn.row_factory = sqlite3.Row       # 컬럼명으로 접근 가능하게
cur = conn.cursor()

cur.execute("""
    CREATE TABLE IF NOT EXISTS 처리내역 (
        id     INTEGER PRIMARY KEY AUTOINCREMENT,
        파일명  TEXT UNIQUE,
        상태    TEXT,
        시각    TEXT DEFAULT CURRENT_TIMESTAMP
    )
""")

# 넣기 — 반드시 ? 플레이스홀더! (SQL 인젝션 방지)
cur.execute("INSERT INTO 처리내역(파일명, 상태) VALUES (?, ?)",
            ("매출_0731.xlsx", "완료"))
cur.executemany("INSERT INTO 처리내역(파일명, 상태) VALUES (?, ?)",
                [("a.csv", "완료"), ("b.csv", "실패")])
conn.commit()                        # ★ 잊으면 저장 안 됨

# 조회
for row in cur.execute("SELECT * FROM 처리내역 WHERE 상태 = ?", ("완료",)):
    print(row["파일명"], row["시각"])

conn.close()

실전 패턴 — 중복 처리 방지

# 자동화의 단골 요구: "이미 처리한 파일은 건너뛰기"
def 이미_처리했나(conn, 파일명):
    r = conn.execute("SELECT 1 FROM 처리내역 WHERE 파일명 = ?",
                     (파일명,)).fetchone()
    return r is not None

def 처리_기록(conn, 파일명, 상태="완료"):
    with conn:                    # with = 자동 commit/rollback
        conn.execute(
            "INSERT OR REPLACE INTO 처리내역(파일명, 상태) VALUES (?, ?)",
            (파일명, 상태))

# pandas 와도 바로 연결됩니다
import pandas as pd
df = pd.read_sql("SELECT * FROM 처리내역", conn)
df.to_sql("백업", conn, if_exists="replace", index=False)
f-string으로 SQL 만들지 마세요 — f"WHERE 이름='{name}'"은 따옴표 하나에 깨지고 보안 구멍입니다. 값은 항상 ?로 넘기세요.
어디까지 sqlite로 버틸까 — 단일 PC, 수백만 행, 동시 쓰기가 적다면 충분합니다. 여러 사람이 동시에 쓰는 시스템이 되면 그때 MariaDB/PostgreSQL로 넘어가면 됩니다.

엑셀 자동화 🟢 기초 · uv add openpyxl

한국 사무 자동화의 최종 목적지는 대부분 엑셀입니다. 용도별로 도구가 다릅니다.

도구이럴 때특징
pandas데이터 읽고·계산하고·저장가장 흔한 90% (→ 탭 3)
openpyxl서식·수식·셀 단위 제어기존 양식 파일 채우기의 정석
XlsxWriter차트·조건부서식 넣어 새로 만들기쓰기 전용, 서식 기능 최강
xlwings열려 있는 엑셀을 실시간 조작엑셀 설치 필요, 매크로 대체
Polars수백만 행이 느릴 때pandas보다 수~수십 배 빠름

openpyxl — 양식 채우기

# uv add openpyxl
from openpyxl import load_workbook
from openpyxl.styles import Font, PatternFill

wb = load_workbook("월간보고_양식.xlsx")
ws = wb["요약"]                    # 시트 선택 (wb.active 도 가능)

ws["B2"] = "2026년 7월"            # 셀에 값
ws.cell(row=5, column=3, value=1250)
ws["B3"] = "=SUM(C5:C20)"          # 수식도 그대로

# 서식
ws["B2"].font = Font(bold=True, size=14, color="1D4ED8")

def 리포트_발송(받는사람, 제목, 본문, 첨부경로=None):

    msg["From"] = "bot@company.com"

      <p>{본문}</p>
      <p style="color:#888">자동 발송 메일입니다.</p>

            maintype="application", subtype="octet-stream",

import time

from watchdog.events import FileSystemEventHandler

class 새파일처리(FileSystemEventHandler):
    def on_created(self, event):
        if event.is_directory or not event.src_path.endswith(".xlsx"):
            return
        time.sleep(1)          # 복사가 끝날 시간을 잠깐 줌 ★
        print("새 파일:", event.src_path)
        엑셀_처리(event.src_path)

obs = Observer()

obs.start()
print("감시 시작… Ctrl+C 로 종료")
try:
    while True:
        time.sleep(1)
except KeyboardInterrupt:

재시도·장애 대응 — tenacity 🟡 중급 · uv add tenacity

네트워크는 반드시 한 번은 끊깁니다. “3번 더 시도”를 데코레이터 한 줄로 답니다.

tenacity — 표준 선택

# uv add tenacity
import requests
from tenacity import (retry, stop_after_attempt,
                      wait_exponential, retry_if_exception_type)

@retry(
    stop=stop_after_attempt(4),               # 최대 4회
    wait=wait_exponential(min=1, max=30),     # 1s→2s→4s→8s… 점점 길게
    retry=retry_if_exception_type(            # 이 예외들만 재시도
        (requests.ConnectionError, requests.Timeout)),
)
def api_호출(url):
    r = requests.get(url, timeout=10)
    r.raise_for_status()
    return r.json()

# 실패 과정을 로그로 남기기
from tenacity import before_sleep_log
import logging
log = logging.getLogger(__name__)

@retry(stop=stop_after_attempt(3),
       wait=wait_exponential(min=2),
       before_sleep=before_sleep_log(log, logging.WARNING))
def 불안한_작업(): ...

backoff · 그리고 원칙

# uv add backoff — 더 짧게 쓰고 싶을 때
import backoff, requests

@backoff.on_exception(backoff.expo, requests.RequestException,
                      max_tries=4, max_time=60)
def api_호출(url):
    return requests.get(url, timeout=10).json()
재시도하면 안 되는 것 — 인증 실패(401), 잘못된 요청(400), “결제·전송” 같이 두 번 실행되면 사고인 작업. 재시도는 일시적 오류(타임아웃·연결 끊김·503)에만 거세요. 위 예제가 예외 타입을 좁게 지정하는 이유입니다.
왜 대기 시간을 점점 늘리나(지수 백오프) — 상대 서버가 과부하일 때 즉시 재시도하면 부하를 더 얹어 상황을 악화시킵니다. 1초→2초→4초로 물러나며 기다리는 것이 서로에게 최선이라 업계 표준이 됐습니다.
지수 백오프 재시도 wait_exponential(min=1) # 1s → 2s → 4s … 성공
점점 길게빨강=실패 후 대기,
초록=성공. 간격이 2배씩

병렬·동시·비동기 🔴 고급 · 설치 불필요

“파일 100개 다운로드가 너무 느려요” — 동시에 하면 됩니다. 단, 작업의 종류에 따라 도구가 다릅니다.

무엇이 느린가부터 진단
기다리느라 느림 (I/O)
다운로드 · API · 파일 읽기
디스크 · DB 응답 대기
→ ThreadPoolExecutor
→ asyncio
계산하느라 느림 (CPU)
이미지 변환 · 대량 연산
압축 · 암호화
→ ProcessPoolExecutor

한눈 시각화 — 순차 vs 동시

순차 실행 for url in urls: 다운로드(url)
8초8건 × 1초
하나 끝나야 다음
스레드 4개 동시 ThreadPoolExecutor(4) .map(다운로드, urls)
≈2초4레인이 동시에
기다림이 겹쳐짐

concurrent.futures — 먼저 이걸로

from concurrent.futures import (ThreadPoolExecutor,
                                ProcessPoolExecutor, as_completed)

# I/O 작업: 스레드 8개로 동시에 다운로드
def 다운로드(url):
    r = requests.get(url, timeout=30)
    return url, len(r.content)

with ThreadPoolExecutor(max_workers=8) as ex:
    results = list(ex.map(다운로드, url목록))

# 끝나는 순서대로 처리 + 개별 실패 처리
with ThreadPoolExecutor(max_workers=8) as ex:
    futs = {ex.submit(다운로드, u): u for u in url목록}
    for f in as_completed(futs):
        try:
            url, size = f.result()
            print(f"완료 {url} ({size:,}B)")
        except Exception as e:
            print(f"실패 {futs[f]}: {e}")

# CPU 작업: 프로세스로 (반드시 main 가드 안에서!)
if __name__ == "__main__":
    with ProcessPoolExecutor() as ex:
        results = list(ex.map(이미지_변환, 파일목록))

asyncio — 수백 개 동시 I/O

import asyncio, httpx

async def fetch(client, url, sem):
    async with sem:                      # 동시 10개로 제한
        r = await client.get(url, timeout=10)
        return url, r.status_code

async def main():
    sem = asyncio.Semaphore(10)
    async with httpx.AsyncClient() as client:
        tasks = [fetch(client, u, sem) for u in url목록]
        return await asyncio.gather(*tasks,
                                    return_exceptions=True)

results = asyncio.run(main())
GIL 한 줄 요약 — 파이썬 스레드는 계산을 동시에 못 합니다 (한 번에 한 스레드만 실행). 그래서 CPU 작업은 스레드를 늘려도 안 빨라지고 프로세스를 써야 합니다. I/O 대기는 GIL을 놓고 기다리므로 스레드로 충분합니다.
초보자 경로 — ① 일단 순차로 완성 → ② 느리면 ThreadPoolExecutor.map 한 줄 → ③ 수백 개 이상 동시가 필요할 때만 asyncio. asyncio는 코드 전체가 async로 물들기 때문에 마지막 카드입니다.

백그라운드 작업 · 워크플로 🔴 고급 · 개념 위주

“요청은 바로 응답하고 무거운 일은 뒤에서”, “A 끝나면 B, 실패하면 알림” — 규모가 커지면 등장하는 도구들입니다. 지금 당장은 몰라도 됩니다.

작업 큐 — Celery · RQ · Dramatiq

요청을 큐(대기줄)에 넣으면 별도의 워커 프로세스가 꺼내 처리합니다. 웹 서비스에서 “보고서 생성 요청 → 즉시 접수 응답 → 완료되면 메일” 같은 구조에 쓰입니다. Redis 같은 브로커가 필요합니다. 

# RQ — 셋이 중 가장 단순 (uv add rq / Redis 필요)
from redis import Redis
from rq import Queue

q = Queue(connection=Redis())
job = q.enqueue(보고서_생성, "2026-07")   # 큐에 넣고 즉시 리턴
job.get_status()                          # queued/started/finished

# 워커는 별도 터미널에서:  uv run rq worker
도구특징
RQ가장 단순. Redis만 있으면 시작
Dramatiq단순+견고, 재시도 내장
Celery기능 최다·사실상 표준, 설정 복잡

워크플로 — Prefect · Airflow · Dagster

“매일 02시: 수집 → 정제 → 적재 → 리포트, 3단계 실패 시 거기부터 재시도, 전 과정을 웹 UI로 모니터링” — 이런 데이터 파이프라인 관리 도구입니다. 

# Prefect — 파이썬 함수에 데코레이터만 (uv add prefect)
from prefect import flow, task

@task(retries=3, retry_delay_seconds=60)
def 수집(): ...

@task
def 정제(raw): ...

@flow(log_prints=True)
def 일일_파이프라인():
    raw = 수집()
    데이터 = 정제(raw)
    적재(데이터)
도구특징
Prefect가장 파이썬답고 진입 쉬움
Airflow업계 표준, 채용시장 최다, 무거움
Dagster데이터 자산 중심의 현대적 설계
언제 필요해지나 — 작업 스케줄러 + 스크립트 몇 개로 굴리다가 ① 단계 간 의존성이 복잡해지고 ② 실패 지점부터 재실행이 필요하고 ③ 실행 현황을 팀이 봐야 할 때. 그 전까지는 탭의 13·14번 조합이면 충분합니다 — 과한 도구는 오히려 짐입니다.
