shutil — 복사·이동·통삭제

shutil.move("a.txt", "정리됨/a.txt")      # 이동 (드라이브 넘어도 OK)
shutil.rmtree("임시폴더")                 # 폴더 통째 삭제 ⚠️ 복구 불가
shutil.disk_usage("D:/")                 # 디스크 용량

시간대 — 자동화가 조용히 틀리는 대표적인 자리

쓰는 것무엇인가 · 언제
datetime.now()시간대 정보가 없습니다(naive) — 서버 설정에 따라 값이 달라집니다
datetime.now(ZoneInfo("Asia/Seoul"))한국 시각 + 시간대 정보 포함(aware) — 표시·업무 기준일에
datetime.now(timezone.utc)저장·비교·로그에는 항상 이것
date.today()서버 시간대 기준 날짜 — 서버가 UTC 면 한국 새벽 9시 이전엔 어제가 나옵니다
“어제 데이터”가 새벽에만 하루 밀리는 버그가 여기서 나옵니다. 클라우드 서버는 대부분 UTC 로 돌기 때문에, 한국 시각 새벽 0시~오전 9시 사이에 date.today() 를 부르면 한국 기준 어제 날짜가 나옵니다. 낮에 테스트할 땐 멀쩡하고 새벽 배치에서만 틀리므로 원인을 찾기가 어렵습니다.
업무 날짜가 필요하면 반드시 시간대를 명시하세요.

안전한 기본형

from datetime import datetime, date, timedelta, timezone
from zoneinfo import ZoneInfo

KST = ZoneInfo("Asia/Seoul")

# ✅ 업무 기준 "오늘" — 서버 시간대와 무관하게 한국 날짜
업무일 = datetime.now(KST).date()

# ✅ 저장·로그·비교는 UTC 로
생성시각 = datetime.now(timezone.utc)

# ✅ 화면에 보여줄 때만 변환
생성시각.astimezone(KST).strftime("%Y-%m-%d %H:%M")

# ✗ 시간대가 다르면 비교 자체가 예외입니다
naive = datetime.now()
aware = datetime.now(timezone.utc)
# naive < aware   → TypeError: can't compare offset-naive and offset-aware

# DB 에서 읽은 값이 naive 라면 시간대를 붙여 줍니다
저장값 = datetime(2026, 8, 7, 3, 0)          # UTC 로 저장된 값이라 가정
저장값 = 저장값.replace(tzinfo=timezone.utc)  # ★ 변환이 아니라 "표시"
저장값.astimezone(KST)                        # → 2026-08-07 12:00 KST
replace(tzinfo=) 와 astimezone() 은 완전히 다릅니다. replace 는 “이 숫자는 이 시간대였다”고 이름표만 붙이고, astimezone 은 실제로 시각을 변환합니다. 둘을 바꿔 쓰면 정확히 9시간이 어긋납니다 — 한국에서 가장 흔한 시간 버그입니다.

자동화에서 자주 쓰는 기간 계산

from datetime import date, timedelta, datetime, timezone
from zoneinfo import ZoneInfo
import calendar

KST = ZoneInfo("Asia/Seoul")
오늘 = datetime.now(KST).date()

# 지난달 1일 ~ 말일 (월간 정산에 매번 나옵니다)
이달1일   = 오늘.replace(day=1)
지난달말   = 이달1일 - timedelta(days=1)
지난달1일  = 지난달말.replace(day=1)

# 이번 달 말일 — calendar 로 안전하게 (윤년·30/31일 자동)
_, 말일 = calendar.monthrange(오늘.year, 오늘.month)
이달말 = 오늘.replace(day=말일)

# 이번 주 월요일 ~ 일요일
월 = 오늘 - timedelta(days=오늘.weekday())
일 = 월 + timedelta(days=6)

# 영업일만 (주말 제외) N일 전
def 영업일_전(d: date, n: int) -> date:
    while n > 0:
        d -= timedelta(days=1)
        if d.weekday() < 5:      # 월~금
            n -= 1
    return d

# 기간을 하루씩 순회
def 날짜범위(시작: date, 끝: date):
    d = 시작
    while d <= 끝:
        yield d
        d += timedelta(days=1)

for d in 날짜범위(지난달1일, 지난달말):
    처리(f"data_{d:%Y%m%d}.csv")

# ISO 형식 파싱 (API 응답에 가장 흔합니다)
datetime.fromisoformat("2026-08-07T03:00:00+00:00")
datetime.fromisoformat("2026-08-07T03:00:00Z")   # Python 3.11+ 부터 Z 지원
timedelta(months=1) 은 없습니다. 한 달의 길이가 일정하지 않기 때문입니다. “한 달 전”이 필요하면 위처럼 replace(day=1) 과 calendar.monthrange 로 직접 계산하거나, uv add python-dateutil 후 relativedelta(months=1) 를 쓰세요. 1월 31일에서 한 달 뒤가 며칠인지를 팀에서 먼저 합의하는 편이 좋습니다.

def 리소스경로(상대: str) -> Path:
    """개발 중에도, exe 로 묶인 뒤에도 동작하는 경로"""

무엇이 느린가부터 진단

한눈 시각화 — 순차 vs 동시

순차 실행 for url in urls: 다운로드(url)
8초8건 × 1초
하나 끝나야 다음
스레드 4개 동시 ThreadPoolExecutor(4) .map(다운로드, urls)
≈2초4레인이 동시에
기다림이 겹쳐짐

concurrent.futures — 먼저 이걸로

from concurrent.futures import (ThreadPoolExecutor,
                                ProcessPoolExecutor, as_completed)

# I/O 작업: 스레드 8개로 동시에 다운로드
def 다운로드(url):
    r = requests.get(url, timeout=30)
    return url, len(r.content)

with ThreadPoolExecutor(max_workers=8) as ex:
    results = list(ex.map(다운로드, url목록))

# 끝나는 순서대로 처리 + 개별 실패 처리
with ThreadPoolExecutor(max_workers=8) as ex:
    futs = {ex.submit(다운로드, u): u for u in url목록}
    for f in as_completed(futs):
        try:
            url, size = f.result()
            print(f"완료 {url} ({size:,}B)")
        except Exception as e:
            print(f"실패 {futs[f]}: {e}")

# CPU 작업: 프로세스로 (반드시 main 가드 안에서!)
if __name__ == "__main__":
    with ProcessPoolExecutor() as ex:
        results = list(ex.map(이미지_변환, 파일목록))

asyncio — 수백 개 동시 I/O

import asyncio, httpx

async def fetch(client, url, sem):
    async with sem:                      # 동시 10개로 제한
        r = await client.get(url, timeout=10)
        return url, r.status_code

async def main():
    sem = asyncio.Semaphore(10)
    async with httpx.AsyncClient() as client:
        tasks = [fetch(client, u, sem) for u in url목록]
        return await asyncio.gather(*tasks,
                                    return_exceptions=True)

results = asyncio.run(main())
GIL 한 줄 요약 — 파이썬 스레드는 계산을 동시에 못 합니다 (한 번에 한 스레드만 실행). 그래서 CPU 작업은 스레드를 늘려도 안 빨라지고 프로세스를 써야 합니다. I/O 대기는 GIL을 놓고 기다리므로 스레드로 충분합니다.
기초 경로 — ① 일단 순차로 완성 → ② 느리면 ThreadPoolExecutor.map 한 줄 → ③ 수백 개 이상 동시가 필요할 때만 asyncio. asyncio는 코드 전체가 async로 물들기 때문에 마지막 카드입니다.
고민 순서는 늘 같습니다 — 먼저 재고, 무엇을 기다리는지 확인하고, 그다음에 고릅니다.
