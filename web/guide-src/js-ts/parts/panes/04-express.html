// ① 인증 — req 에 사용자 정보를 붙임
export const requireAuth: RequestHandler = (req, res, next) => {
  const token = req.headers.authorization?.replace("Bearer ", "");
  if (!token) return next(new UnauthorizedError());
  try {
    req.user = jwt.verify(token, env.JWT_SECRET) as JwtPayload;
    next();
  } catch {
    next(new UnauthorizedError("토큰이 유효하지 않습니다"));
  }
};

// ② 권한 — 매개변수를 받는 미들웨어는 "함수를 반환"하는 형태
export const requireRole = (...roles: string[]): RequestHandler =>
  (req, res, next) =>
    roles.includes(req.user?.role ?? "")
      ? next()
      : next(new ForbiddenError());

// ③ async 래퍼 (Express 4 에서 필수, 5 에서는 불필요)
export const asyncHandler = (fn: RequestHandler): RequestHandler =>
  (req, res, next) => Promise.resolve(fn(req, res, next)).catch(next);

router.get("/:id", asyncHandler(async (req, res) => {
  const user = await service.find(req.params.id);   // 던져도 안전
  res.json(user);
}));

미들웨어로 연결

type Target = "body" | "query" | "params";

export const validate =
  (schema: z.ZodTypeAny, target: Target = "body"): RequestHandler =>
  (req, res, next) => {
    const result = schema.safeParse(req[target]);
    if (!result.success) {
      return res.status(400).json({
        message: "입력값이 올바르지 않습니다",
        errors: result.error.issues.map(i => ({
          field: i.path.join("."),
          message: i.message,
        })),
      });
    }
    if (target === "query") {
      // ★ Express 5 에서 req.query 는 getter 전용 — 재할당하면 TypeError
      Object.defineProperty(req, "query", { value: result.data });
    } else {
      req[target] = result.data;      // ★ 변환·기본값이 적용된 값으로 교체
    }
    next();
  };

// 사용
router.post("/", validate(CreateUserSchema), async (req, res) => {
  const dto = req.body as CreateUserDto;    // ✅ 검증·변환 완료된 값
  res.status(201).json(await service.create(dto));
});

// errors.ts — 도메인 에러 정의
export class AppError extends Error {
  constructor(
    message: string,
    public status = 500,
    public code = "INTERNAL_ERROR",
    public details?: unknown,
  ) { super(message); this.name = this.constructor.name; }
}
export class NotFoundError extends AppError {
  constructor(what = "리소스") { super(`${what}를 찾을 수 없습니다`, 404, "NOT_FOUND"); }
}
export class UnauthorizedError extends AppError {
  constructor(m = "인증이 필요합니다") { super(m, 401, "UNAUTHORIZED"); }
}
export class ForbiddenError extends AppError {
  constructor(m = "권한이 없습니다") { super(m, 403, "FORBIDDEN"); }
}
export class ConflictError extends AppError {
  constructor(m: string) { super(m, 409, "CONFLICT"); }
}
// middleware/error.ts — ★ 앱에서 마지막에 등록
export const errorHandler: ErrorRequestHandler = (err, req, res, _next) => {
  const requestId = req.headers["x-request-id"];

  if (err instanceof AppError) {
    logger.warn({ err, requestId, path: req.path }, "처리된 오류");
    return res.status(err.status).json({
      code: err.code, message: err.message, details: err.details, requestId,
    });
  }

  // 라이브러리 에러 변환
  if (err instanceof z.ZodError) {
    return res.status(400).json({ code: "VALIDATION_ERROR", errors: err.issues });
  }
  if (err.code === "P2002") {             // Prisma 유니크 위반
    return res.status(409).json({ code: "DUPLICATE", message: "이미 존재합니다" });
  }

  // 예상 못 한 에러 — ★ 내부 정보를 노출하지 마세요
  logger.error({ err, requestId, path: req.path }, "예상치 못한 오류");
  res.status(500).json({
    code: "INTERNAL_ERROR", message: "서버 오류가 발생했습니다", requestId,
  });
};

// 검증 미들웨어
export const requireAuth: RequestHandler = (req, _res, next) => {
  const token = req.headers.authorization?.split(" ")[1];
  if (!token) return next(new UnauthorizedError());
  try {
    const payload = jwt.verify(token, env.JWT_SECRET, { issuer: "my-api" });
    req.user = { id: Number(payload.sub), role: payload.role };
    next();
  } catch (e) {
    next(e instanceof jwt.TokenExpiredError
      ? new UnauthorizedError("토큰이 만료되었습니다")
      : new UnauthorizedError());
  }
};

  // 새 토큰 발급…

  const res = clients.get(userId);
  if (!res) return;
  res.write(`event: ${event}\n`);
  res.write(`id: ${Date.now()}\n`);              // 재연결 시 이어받기용
  res.write(`data: ${JSON.stringify(data)}\n\n`); // 빈 줄 두 개가 끝 표시
}

중계 서버 — 요청을 소켓에 실어 보낸다

import express from 'express';
import { WebSocketServer } from 'ws';
import { randomUUID } from 'node:crypto';

const app = express();
const agents  = new Map<string, WebSocket>();          // tenant → 소켓
const waiting = new Map<string, (r: TunnelRes) => void>(); // reqId → 응답 대기

const wss = new WebSocketServer({ noServer: true });

wss.on('connection', (ws, req, tenant: string) => {
  agents.set(tenant, ws);

  ws.on('message', (buf) => {
    const msg = JSON.parse(buf.toString());
    waiting.get(msg.id)?.(msg);                        // 기다리던 요청에 응답 전달
    waiting.delete(msg.id);
  });

  // 죽은 연결 청소 — TCP 는 끊겨도 한동안 살아 있는 척한다
  let alive = true;
  ws.on('pong', () => { alive = true; });
  const iv = setInterval(() => {
    if (!alive) return ws.terminate();
    alive = false; ws.ping();
  }, 30_000);

  ws.on('close', () => { clearInterval(iv); agents.delete(tenant); });
});

// 외부 요청 → 해당 테넌트의 에이전트로
app.all('/t/:tenant/*', async (req, res) => {
  const ws = agents.get(req.params.tenant);
  if (!ws || ws.readyState !== ws.OPEN)
    return res.status(503).json({ error: 'agent offline' });

  const id = randomUUID();
  const body = await readBody(req);                    // 상한을 두고 읽는다

  const answer = new Promise<TunnelRes>((ok, ng) => {
    waiting.set(id, ok);
    setTimeout(() => { waiting.delete(id); ng(new Error('timeout')); }, 30_000);
  });

  ws.send(JSON.stringify({
    id, method: req.method,
    path: '/' + (req.params[0] ?? ''),
    headers: stripHopByHop(req.headers),
    body: body.toString('base64'),
  }));

  try {
    const r = await answer;
    res.status(r.status).set(r.headers).send(Buffer.from(r.body, 'base64'));
  } catch { res.status(504).json({ error: 'agent timeout' }); }
});

사내 에이전트 — 받아서 로컬로 넘긴다

import WebSocket from 'ws';

const LOCAL = 'http://127.0.0.1:3000';
let backoff = 1_000;

function connect() {
  const ws = new WebSocket(`wss://relay.example.com/agent`, {
    headers: { authorization: `Bearer ${process.env.AGENT_TOKEN}` },
  });

  ws.on('open', () => { backoff = 1_000; log.info('tunnel up'); });

  ws.on('message', async (buf) => {
    const m = JSON.parse(buf.toString());
    try {
      // 로컬 서비스로 그대로 전달 (허용 경로만!)
      const r = await fetch(LOCAL + m.path, {
        method: m.method,
        headers: m.headers,
        body: ['GET', 'HEAD'].includes(m.method)
          ? undefined : Buffer.from(m.body, 'base64'),
        signal: AbortSignal.timeout(25_000),
      });
      const body = Buffer.from(await r.arrayBuffer());
      ws.send(JSON.stringify({
        id: m.id, status: r.status,
        headers: Object.fromEntries(r.headers),
        body: body.toString('base64'),
      }));
    } catch (e) {
      ws.send(JSON.stringify({ id: m.id, status: 502, headers: {}, body: '' }));
    }
  });

  // 끊기면 지수 백오프로 재연결 — 중계 서버 재시작 때 몰리는 것을 막는다
  const retry = () => {
    const jitter = Math.random() * 1_000;
    setTimeout(connect, Math.min(backoff, 60_000) + jitter);
    backoff *= 2;
  };
  ws.on('close', retry);
  ws.on('error', () => ws.close());
}

직접 만들 때 반드시 챙길 것

  • 인증은 연결 시점에 — 토큰 검증 후에만 agents 에 등록. 테넌트 id 를 클라이언트가 자기 입으로 말하게 두면 안 됩니다
  • 경로 화이트리스트 — 에이전트가 로컬의 아무 데나 요청을 전달하면 사내망 전체가 노출됩니다(SSRF)
  • 본문 크기 상한 — 메모리에 통째로 올리는 구조라 10MB 정도로 제한하고, 큰 파일은 별도 스트리밍 경로로
  • 동시 요청 상한 — 에이전트당 p-limit 으로 묶지 않으면 사내 서버가 먼저 쓰러집니다
  • hop-by-hop 헤더 제거 — connection·upgrade· transfer-encoding 을 그대로 넘기면 프록시가 오작동합니다
이 구조는 그 자체로 “원격 제어 채널”입니다. 토큰이 새면 사내망 침투 경로가 됩니다. 토큰은 장비별로 발급하고 회수(revoke) 목록을 서버가 확인하게 만드세요.

  • HTTP 만 통과시키면 되는 좁은 요구

Range 응답 — 동영상 스트리밍의 최소 조건

import { stat, createReadStream } from 'node:fs';

app.get('/media/:id', async (req, res) => {
  const file = await resolveSafe(req.params.id);   // 경로 조작 방어
  const { size } = await stat(file);
  const range = req.headers.range;

  if (!range) {
    res.set({ 'Content-Length': size, 'Accept-Ranges': 'bytes',
              'Content-Type': 'video/mp4' });
    return createReadStream(file).pipe(res);
  }

  // "bytes=1048576-" 형태를 파싱한다
  const m = /^bytes=(\d*)-(\d*)$/.exec(range);
  if (!m) return res.status(416).set('Content-Range', `bytes */${size}`).end();

  const start = m[1] ? Number(m[1]) : 0;
  const end   = m[2] ? Math.min(Number(m[2]), size - 1) : size - 1;
  if (start >= size || start > end)
    return res.status(416).set('Content-Range', `bytes */${size}`).end();

  res.status(206).set({
    'Content-Range':  `bytes ${start}-${end}/${size}`,
    'Accept-Ranges':  'bytes',
    'Content-Length': end - start + 1,
    'Content-Type':   'video/mp4',
  });
  createReadStream(file, { start, end }).pipe(res);
});
206 과 Content-Range 가 없으면 브라우저는 탐색 바를 끌 수 없게 만듭니다(처음부터 다시 받습니다). “재생은 되는데 탐색이 안 돼요”의 원인은 거의 항상 이것입니다.

// ★ 서명 검증에는 '가공되지 않은 원문'이 필요하다.
//   express.json() 이 먼저 파싱해 버리면 검증할 수 없다.
app.post('/webhooks/pay',
  express.raw({ type: 'application/json', limit: '1mb' }),
  async (req, res) => {
    const sig = req.get('x-signature') ?? '';
    const ts  = Number(req.get('x-timestamp') ?? 0);

    // 재전송 공격 방어 — 5분 넘은 것은 거부
    if (Math.abs(Date.now() / 1000 - ts) > 300) return res.sendStatus(400);

    const mine = crypto.createHmac('sha256', process.env.WEBHOOK_SECRET!)
                       .update(`${ts}.`).update(req.body).digest();
    const theirs = Buffer.from(sig, 'hex');

    // 길이가 다르면 timingSafeEqual 이 예외를 던진다
    if (mine.length !== theirs.length ||
        !crypto.timingSafeEqual(mine, theirs)) return res.sendStatus(401);

    const event = JSON.parse(req.body.toString());

    // 같은 이벤트가 여러 번 온다 — 결제사는 재시도를 보장한다
    const fresh = await db.webhookEvent.create({ data: { id: event.id } })
                          .catch(() => null);
    if (!fresh) return res.sendStatus(200);        // 이미 처리함

    // ★ 무거운 일은 큐에 넣고 즉시 200 — 늦게 답하면 결제사가 재시도한다
    await queue.add('pay.event', event);
    res.sendStatus(200);
  });
=== 로 서명을 비교하지 마세요. 문자열 비교는 다른 첫 글자에서 즉시 멈추므로 걸린 시간으로 정답을 한 글자씩 알아낼 수 있습니다(타이밍 공격). 반드시 crypto.timingSafeEqual 을 쓰세요.

웹훅을 못 받았을 때 — 폴링 보완

// 웹훅은 '반드시 온다'가 아니라 '보통 온다' 입니다.
// 서버 재배포 중이거나 장애가 나면 그대로 유실됩니다.
// 그래서 상용 서비스는 폴링을 하나 더 둡니다.

// 5분마다: 결제 요청은 있는데 아직 결론이 없는 건을 직접 조회
cron.schedule('*/5 * * * *', async () => {
  const stuck = await db.order.findMany({
    where: { status: 'PENDING', createdAt: { lt: minutesAgo(3) } },
    take: 100,
  });
  for (const o of stuck) {
    const r = await pg.getPayment(o.paymentKey);    // 결제사에 직접 물어본다
    if (r.status === 'DONE')     await markPaid(o.id, r);
    if (r.status === 'CANCELED') await markCanceled(o.id, r);
  }
});
“웹훅이 안 왔어요”는 결제 연동에서 가장 흔한 장애입니다. 원인은 대개 배포 중 다운타임·타임아웃·5xx 응답 입니다. 웹훅을 유일한 진실 통로로 삼지 말고 폴링을 항상 곁들이세요.
