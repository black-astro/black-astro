utilityProcess — 기본으로 쓸 것

import { utilityProcess, MessageChannelMain } from "electron";

const { port1, port2 } = new MessageChannelMain();

const child = utilityProcess.fork(
  path.join(__dirname, "workers/indexer.js"),
  [],                                  // argv
  {
    serviceName: "indexer",            // ★ 작업 관리자에 이 이름으로 보입니다
    stdio: "pipe",
    execArgv: ["--max-old-space-size=512"],   // 워커만 메모리 제한
  }
);

// 포트를 넘겨 주면 이후 통신은 메인을 거치지 않습니다
child.postMessage({ type: "init" }, [port1]);

child.on("exit", (code) => {
  log.warn("indexer 종료", code);
  if (code !== 0) respawn();           // ★ 죽으면 되살리기
});

child.stdout?.on("data", (d) => log.info("[indexer]", d.toString()));
serviceName 을 꼭 지정하세요. 지정하지 않으면 작업 관리자에 Electron Helper 로만 보여 사용자도 개발자도 어느 프로세스가 CPU 를 먹는지 알 수 없습니다. Chrome 의 작업 관리자(Shift+Esc 개념)처럼 앱 안에 app.getAppMetrics() 기반 진단 화면을 두면 지원 문의가 크게 줄어듭니다.

NATIVE 03

고성능 IPC

ipcRenderer.invoke 로는 감당이 안 되는 고빈도·대용량 구간을 다룹니다.

기본 IPC 의 비용

ipcRenderer.invoke 의 실제 경로

렌더러 → [구조화 복제 직렬화] → 메인 IPC 큐
       → 메인 스레드에서 처리 → [다시 직렬화] → 렌더러

비용이 커지는 지점
 ① 구조화 복제 — 객체가 크고 깊을수록 급증
 ② 메인 스레드 경유 — 메인이 바쁘면 전부 대기
 ③ 초당 수백 회 호출 시 큐가 밀림

체감 기준 (대략)
 · 작은 객체 · 초당 수십 회      → 그냥 invoke 로 충분
 · 수 MB 데이터 · 초당 수백 회   → MessagePort 로
 · 오디오·비디오 프레임 (연속)   → 공유 메모리 · 네이티브로
가장 흔한 실수 — 렌더러가 1초에 60번 invoke("state:get") 을 호출하는 폴링 구조입니다. 메인이 밀어 주는(push) 구조로 바꾸고, 변경이 잦으면 합치기(coalescing) 를 적용하세요 — 16ms 안에 들어온 갱신은 한 번만 보냅니다.

MessagePort — 메인을 건너뛰기

// 메인 — 워커와 렌더러를 직접 연결해 줍니다
import { MessageChannelMain } from "electron";

const { port1, port2 } = new MessageChannelMain();

worker.postMessage({ type: "port" }, [port1]);          // 워커에 한쪽
mainWindow.webContents.postMessage("port:worker", null, [port2]);  // 렌더러에 반대쪽

// ★ 이후 워커 ↔ 렌더러 통신은 메인 스레드를 전혀 거치지 않습니다
// 프리로드 — 포트를 받아 렌더러에 전달
ipcRenderer.on("port:worker", (e) => {
  const port = e.ports[0];
  port.start();
  contextBridge.exposeInMainWorld("workerPort", {
    send: (msg: unknown) => port.postMessage(msg),
    onMessage: (cb: (m: unknown) => void) => {
      port.onmessage = (ev) => cb(ev.data);
    },
  });
});

② C++ 코드 — 동기 함수

#include <napi.h>
#include <vector>
#include <cstdint>

// TypedArray 를 복사 없이 그대로 받습니다
Napi::Value SumSquares(const Napi::CallbackInfo& info) {
  Napi::Env env = info.Env();

  if (info.Length() < 1 || !info[0].IsTypedArray()) {
    Napi::TypeError::New(env, "Float32Array 가 필요합니다")
        .ThrowAsJavaScriptException();
    return env.Null();
  }

  Napi::Float32Array arr = info[0].As<Napi::Float32Array>();
  const float* data = arr.Data();      // ★ JS 힙을 직접 참조 (복사 0)
  size_t len = arr.ElementLength();

  double sum = 0.0;
  for (size_t i = 0; i < len; ++i) sum += static_cast<double>(data[i]) * data[i];

  return Napi::Number::New(env, sum);
}

Napi::Object Init(Napi::Env env, Napi::Object exports) {
  exports.Set("sumSquares", Napi::Function::New(env, SumSquares));
  return exports;
}

③ 비동기 — AsyncWorker

class HashWorker : public Napi::AsyncWorker {
 public:
  HashWorker(Napi::Env env, std::vector<uint8_t> data)
      : Napi::AsyncWorker(env),
        deferred_(Napi::Promise::Deferred::New(env)),
        data_(std::move(data)) {}

  Napi::Promise GetPromise() { return deferred_.Promise(); }

  // ★ 워커 스레드에서 실행 — 여기서 JS 객체를 만지면 안 됩니다
  void Execute() override {
    result_ = heavy_hash(data_.data(), data_.size());
  }

  // ★ 다시 JS 스레드 — 여기서만 JS 값을 다룹니다
  void OnOK() override {
    Napi::HandleScope scope(Env());
    deferred_.Resolve(Napi::String::New(Env(), result_));
  }
  void OnError(const Napi::Error& e) override {
    deferred_.Reject(e.Value());
  }

 private:
  Napi::Promise::Deferred deferred_;
  std::vector<uint8_t> data_;
  std::string result_;
};

Napi::Value HashAsync(const Napi::CallbackInfo& info) {
  auto buf = info[0].As<Napi::Buffer<uint8_t>>();
  // ★ 워커 스레드에서 쓸 데이터는 반드시 복사해 둡니다
  std::vector<uint8_t> copy(buf.Data(), buf.Data() + buf.Length());

  auto* w = new HashWorker(info.Env(), std::move(copy));
  w->Queue();
  return w->GetPromise();
}

④ C++ → JS 콜백 (ThreadSafeFunction)

// 오디오 콜백처럼 "네이티브 스레드가 JS 를 부르는" 경우
Napi::ThreadSafeFunction tsfn;

Napi::Value Start(const Napi::CallbackInfo& info) {
  tsfn = Napi::ThreadSafeFunction::New(
      info.Env(),
      info[0].As<Napi::Function>(),   // JS 콜백
      "AudioCallback",
      0,      // 큐 무제한
      1);     // 스레드 1개가 사용

  std::thread([] {
    while (running) {
      float level = capture_level();
      // ★ 다른 스레드에서 안전하게 JS 호출
      tsfn.BlockingCall([level](Napi::Env env, Napi::Function cb) {
        cb.Call({ Napi::Number::New(env, level) });
      });
    }
    tsfn.Release();                   // ★ 반드시 해제
  }).detach();

  return info.Env().Undefined();
}

class LevelProcessor extends AudioWorkletProcessor {
  constructor() {
    super();
    // ★ 생성자에서 미리 할당 — process() 안에서는 할당 금지
    this._sum = 0; this._count = 0;
  }

  process(inputs, outputs) {
    const input = inputs[0][0];        // 첫 입력의 첫 채널
    if (!input) return true;

    let peak = 0;
    for (let i = 0; i < input.length; i++) {
      const a = Math.abs(input[i]);
      if (a > peak) peak = a;
      outputs[0][0][i] = input[i];     // 그대로 통과
    }

    // ★ 매 블록마다 postMessage 하지 말 것 — 초당 375회가 됩니다
    if (++this._count % 16 === 0) this.port.postMessage(peak);

    return true;                       // false 를 반환하면 노드가 정리됩니다
  }
}

// 음성 활동 감지 — "말하는 중" 표시와 대역폭 절약에 씁니다
class VAD extends AudioWorkletProcessor {
  constructor() {
    super();
    this.speaking = false;
    this.hold = 0;                       // ★ 히스테리시스 (깜빡임 방지)
  }
  process(inputs) {
    const x = inputs[0][0];
    if (!x) return true;

    let energy = 0;
    for (let i = 0; i < x.length; i++) energy += x[i] * x[i];
    const rms = Math.sqrt(energy / x.length);
    const db = 20 * Math.log10(rms + 1e-9);

    // 켜는 문턱과 끄는 문턱을 다르게 (-45 / -50)
    if (db > -45) { this.hold = 25; }     // 약 0.7초 유지
    else if (this.hold > 0) this.hold--;

    const now = this.hold > 0;
    if (now !== this.speaking) {
      this.speaking = now;
      this.port.postMessage({ speaking: now });
    }
    return true;
  }

WebGPU 는 Electron 에서 이미 쓸 수 있습니다. Chromium 기반이라 브라우저 지원 여부를 신경 쓸 필요가 없다는 것이 데스크탑 앱의 큰 장점입니다 — 데이터 시각화·이미지 처리를 GPU 컴퓨트 셰이더로 돌리면 네이티브 애드온 없이도 큰 성능을 얻을 수 있습니다.

  //   NODE_OPTIONS=--require=악성.js 주입 차단

// JS 예외는 Sentry 로, C++ 크래시는 minidump 로 — 둘 다 필요합니다
import { crashReporter, app } from "electron";

crashReporter.start({
  productName: "MyApp",
  companyName: "MyCompany",
  submitURL: "https://crash.example.com/submit",
  uploadToServer: true,
  compress: true,
  extra: {                                  // ★ 원인 좁히기용 컨텍스트
    version: app.getVersion(),
    channel: releaseChannel,
  },
});

// 렌더러가 죽었을 때
app.on("render-process-gone", (_e, wc, details) => {
  log.error("렌더러 종료", details.reason, details.exitCode);
  // reason: crashed · oom · killed · launch-failed ...
  if (details.reason === "oom") notifyUser("메모리가 부족해 창을 다시 엽니다");
  recreateWindow(wc);
});

// 자식 프로세스(GPU·유틸리티)가 죽었을 때
app.on("child-process-gone", (_e, d) => {
  metrics.increment(`child_gone.${d.type}.${d.reason}`);
});
네이티브 애드온을 쓰기 시작하면 크래시의 성격이 바뀝니다. JS 예외는 스택이 나오지만, C++ 세그폴트는 프로세스가 즉사합니다. crashReporter 의 minidump 와 빌드 시 생성한 심볼 파일을 함께 보관해야 나중에 스택을 복원할 수 있습니다 — 심볼을 버리면 덤프는 쓸모없는 바이너리가 됩니다.
