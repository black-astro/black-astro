리액트의 전부는 한 문장입니다 — 화면 = f(상태). DOM 을 직접 만지는 대신 상태를 바꾸면 화면이 따라옵니다. 이 탭은 왜 다시 그려지는가를 축으로 훅과 성능을 정리했습니다. 여기까지 마치면 리렌더 원인을 스스로 추적하고 불필요한 최적화를 걷어낼 수 있습니다.

REACT 01

리액트가 하는 일

직접 DOM 을 고치던 코드가 왜 감당이 안 되는지부터 봅니다.

직접 고치면

// 장바구니에 담을 때마다 손봐야 할 곳
count++;
document.querySelector("#badge").textContent = count;
document.querySelector("#total").textContent = price * count;
document.querySelector("#btn").disabled = count === 0;
document.querySelector("#empty").hidden = count > 0;
// … 화면이 늘수록 '고쳐야 할 자리'가 같이 늘어납니다
// 하나라도 빠뜨리면 화면끼리 값이 어긋납니다

리액트로

function Cart(){
  const [count, setCount] = useState(0);

  return (
    <>
      <span>{count}</span>
      <span>{price * count}</span>
      <button disabled={count === 0}>주문</button>
      {count === 0 && <p>비어 있습니다</p>}
      <button onClick={() => setCount(count + 1)}>담기</button>
    </>
  );
}
// count 하나만 바꾸면 관련된 화면이 '전부 알아서' 맞춰집니다
핵심 전환 — "무엇을 고칠까"가 아니라 "지금 상태가 무엇이고, 그 상태면 화면이 어떻게 생겼나"를 적습니다. 어떻게 바꿀지는 리액트가 계산합니다.
리액트를 배우는 것은 이 순환에서 내가 개입하는 지점이 setState 하나뿐임을 받아들이는 과정입니다.
REACT 02

JSX — HTML 처럼 생긴 JS

HTML 이 아니라 함수 호출입니다. 그래서 JS 규칙을 따릅니다.

const name = "현우";
const items = ["가", "나"];

const el = (
  <div className="card" onClick={handle}>   {/* class ✗ → className */}
    <h2>{name}님</h2>                        {/* {} 안은 JS 식 */}
    <label htmlFor="id">아이디</label>        {/* for ✗ → htmlFor */}
    <input id="id" defaultValue="hi" />      {/* 닫는 / 필수 */}

    {/* 조건 */}
    {name ? <p>환영합니다</p> : null}
    {items.length > 0 && <p>{items.length}개</p>}

    {/* 반복 */}
    <ul>{items.map((v) => <li key={v}>{v}</li>)}</ul>

    {/* 스타일은 객체 · 단위는 문자열로 */}
    <p style={{ marginTop: 8, color: "red" }}>주의</p>
  </div>
);
함정설명
{0 && <div/>}화면에 0 이 찍힙니다. {count > 0 && …} 로 불리언을 만들어 쓰세요
루트가 둘 이상<>…</>(Fragment)로 감쌉니다
주석JSX 안에서는 {/* … */}
if 문JSX 안에는 식만 들어갑니다. 삼항이나 미리 변수로 빼세요
공백·줄바꿈줄바꿈은 무시됩니다. 띄우려면 {" "}

JSX 의 정체 — 변환 결과를 한 번 보면 규칙이 이해됩니다

// 이렇게 쓰면
<div className="card">
  <h2>{name}님</h2>
</div>

// 빌드 도구가 이렇게 바꿉니다 (그냥 함수 호출)
jsx("div", {
  className: "card",
  children: jsx("h2", { children: [name, "님"] }),
});
  • class 대신 className — JS 예약어와 충돌하기 때문
  • 속성이 camelCase — 객체의 키이기 때문 (onClick·tabIndex)
  • if 를 못 쓰는 이유 — 인자 자리에는 식만 올 수 있기 때문
  • 루트가 하나여야 하는 이유 — 함수는 하나만 반환하기 때문

조건 분기 패턴 정리

// 셋 중 상황에 맞는 것을 — 전부 같은 일입니다
// ① 삼항 — 둘 중 하나
{loggedIn ? <Profile /> : <LoginBtn />}

// ② && — 있거나 없거나
{error && <Alert msg={error} />}

// ③ 변수로 미리 — 분기가 3개 이상이면 이게 깨끗
let body;
if (isPending)      body = <Skeleton />;
else if (error)     body = <Alert msg={error} />;
else if (!data.length) body = <Empty />;
else                body = <List rows={data} />;

return <section>{body}</section>;
로딩 → 에러 → 빈 목록 → 정상. 네 상태를 이 순서로 분기하는 습관을 들이면 "데이터 없을 때 터지는 화면"이 사라집니다.
REACT 03

컴포넌트와 props

props 는 부모가 준 값입니다. 자식이 바꿀 수 없습니다.

기본

function Badge({ text, tone = "info", onClose }){
  return (
    <span className={`badge ${tone}`}>
      {text}
      {onClose && <button onClick={onClose}>✕</button>}
    </span>
  );
}

// 사용
<Badge text="신규" />
<Badge text="마감" tone="warn" onClose={() => hide()} />

// children — 태그 사이에 넣은 것
function Card({ title, children }){
  return <div className="card"><h3>{title}</h3>{children}</div>;
}
<Card title="공지"><p>내용</p></Card>

규칙

규칙이유
컴포넌트 이름은 대문자로소문자는 <div> 같은 HTML 태그로 취급됩니다
props 를 수정 금지부모의 값입니다. 바꾸려면 부모가 준 함수를 호출하세요
순수하게 유지같은 props 면 같은 결과. 렌더 중 바깥 변수를 고치면 안 됩니다
컴포넌트 안에 컴포넌트 정의 ✗렌더마다 새 타입이 되어 상태가 초기화됩니다
// ✗ 이러면 입력값이 매번 사라집니다
function Page(){
  function Row(){ return <input />; }   // 렌더마다 새로 만들어짐
  return <Row />;
}
// ✓ 바깥으로 빼세요

언제 쪼개는가 — 파일 크기가 기준이 아닙니다

  • 따로 다시 그려져야 할 때 — 검색 입력마다 페이지 전체가 렌더되면, 입력창을 자식으로 분리해 렌더 범위를 좁힙니다
  • 두 군데 이상에서 쓸 때 — 버튼·배지·모달
  • 이름을 붙일 수 있을 때 — "이 덩어리는 주문 요약이다" 싶으면 <OrderSummary /> 로. JSX 가 문장처럼 읽히게 됩니다
  • 반대로, 한 번 쓰는 40줄짜리를 습관적으로 쪼개지 마세요 — props 배선만 늘어납니다
// 쪼갠 뒤의 목표 — 부모가 '문장'으로 읽힌다
function CheckoutPage(){
  return (
    <main>
      <OrderSummary items={items} />
      <AddressForm value={addr} onChange={setAddr} />
      <PayButton disabled={!valid} onPay={submit} />
    </main>
  );
}

props 설계 요령

// ✗ 낱개로 다 받기 — 부모·자식이 같이 수정됩니다
<UserCard name={u.name} age={u.age} avatar={u.avatar} … />
// ✓ 도메인 객체는 통째로
<UserCard user={u} />

// ✗ 불리언 스위치가 늘어나는 신호
<Button isPrimary isLarge isOutline … />
// ✓ 종류는 문자열 하나로
<Button variant="primary" size="lg" />

// 이벤트는 on+동사, 값+변경 함수는 짝으로
<Modal open={open} onClose={() => setOpen(false)} />
<Select value={v} onChange={setV} />

// 나머지는 그대로 흘려보내기
function Input({ label, ...rest }){
  return <label>{label}<input {...rest} /></label>;
}
props 인터페이스가 곧 컴포넌트의 공개 API 입니다. HTML 표준(value/onChange·open/onClose)과 같은 모양을 유지하면 처음 보는 사람도 쓸 수 있습니다.
REACT 04

state — 바뀌는 값

state 를 바꾸면 그 컴포넌트를 다시 그립니다.

기본과 함정

const [n, setN] = useState(0);

// ✗ 연속 호출이 한 번만 먹습니다
setN(n + 1); setN(n + 1);        // n=0 이면 둘 다 1 → 결과 1

// ✓ 이전 값을 받아서
setN((p) => p + 1); setN((p) => p + 1);   // 결과 2

// ✗ 직접 수정 — 같은 객체라 '안 바뀐 걸로' 봅니다
list.push(x); setList(list);

// ✓ 새 배열·새 객체로
setList([...list, x]);
setList(list.filter((v) => v.id !== id));
setList(list.map((v) => v.id === id ? { ...v, done:true } : v));
setUser({ ...user, name: "새 이름" });
상태를 바꿔도 그 줄에서 값이 바로 안 바뀝니다. setN(1); console.log(n) 은 이전 값을 찍습니다 — 새 값은 다음 렌더부터 보입니다.

state 로 두지 말아야 할 것

// ✗ props 를 그대로 state 에 복사 (동기화가 어긋납니다)
const [name, setName] = useState(props.name);

// ✗ 계산으로 나오는 값을 state 로
const [total, setTotal] = useState(0);
useEffect(() => setTotal(items.length), [items]);

// ✓ 그냥 렌더할 때 계산하세요
const total = items.length;
const visible = items.filter((v) => !v.done);
판단 기준 — 다른 값으로 계산할 수 있으면 state 가 아닙니다. state 가 늘수록 서로 어긋날 경우의 수가 늘어납니다. 입력값·서버 응답·열림 여부처럼 계산으로 못 얻는 것만 state 로 두세요.
"setState 했는데 값이 그대로"는 버그가 아니라 스냅샷이라는 설계입니다 — 한 렌더 안에서 값은 상수입니다.
REACT 05

언제 다시 그리는가

"왜 이게 또 렌더되지?"의 답은 세 가지뿐입니다.

렌더는 state 가 바뀐 지점에서 아래로만 번집니다 — 성능 최적화의 절반은 이 그림에서 상태를 더 아래로 내리는 것입니다.
다시 그리는 조건
  1. 자기 state 가 바뀜
  2. 부모가 다시 그려짐          ← 가장 많이 놓치는 것
  3. 구독 중인 context 값이 바뀜

★ props 가 안 바뀌어도 부모가 그려지면 자식도 그려집니다
   (React.memo 로 감싸야 멈춥니다)
// 렌더 흐름 확인 — 개발 중에만
console.log("render:", Component.name);

// React 19 컴파일러를 켜면 아래 최적화 대부분이 자동입니다
// 켜기 전에는 '측정한 뒤에만' 손대세요
오해사실
렌더 = DOM 갱신다릅니다. 렌더는 "무엇을 그릴지 계산"이고, 바뀐 부분만 DOM 에 반영됩니다
렌더가 많으면 무조건 느리다대부분은 싸다. 진짜 문제는 렌더 안의 무거운 계산입니다
개발 모드에서 두 번 도는 건 버그StrictMode 가 일부러 두 번 실행해 부작용을 드러냅니다. 배포본은 한 번
측정 먼저 — React DevTools 의 Profiler 로 "무엇이 왜 렌더됐는지"를 보고 나서 최적화하세요. 추측으로 memo 를 뿌리면 코드만 복잡해지고 느려집니다.
REACT 06

useEffect — 대부분 필요 없다

화면 밖 세계와 동기화할 때만 씁니다.

쓰지 말아야 할 때

// ✗ 계산 결과를 state 에 넣기
useEffect(() => { setFull(first + " " + last); }, [first, last]);
// ✓ 그냥 계산
const full = first + " " + last;

// ✗ props 가 바뀌면 state 초기화
useEffect(() => { setSel(null); }, [userId]);
// ✓ key 로 컴포넌트를 새로 만들기
<Profile key={userId} … />

// ✗ 이벤트에 대한 반응을 effect 로
useEffect(() => { if (submitted) toast(); }, [submitted]);
// ✓ 이벤트 핸들러 안에서 바로
function onSubmit(){ save(); toast(); }

써야 할 때

useEffect(() => {
  const c = new AbortController();

  fetch(`/api/user/${id}`, { signal: c.signal })
    .then((r) => r.json())
    .then(setUser)
    .catch((e) => { if (e.name !== "AbortError") setErr(e); });

  return () => c.abort();      // ★ 정리 — 떠날 때 취소
}, [id]);

// 구독 · 타이머 · 바깥 라이브러리
useEffect(() => {
  const on = () => setW(innerWidth);
  addEventListener("resize", on);
  return () => removeEventListener("resize", on);
}, []);
정리 함수를 빠뜨리면 타이머·구독이 쌓여 메모리 누수와 "사라진 컴포넌트에 setState" 경고가 납니다. 무언가를 시작했으면 반드시 끝내는 코드를 같이 적으세요.

useRef — 렌더와 무관한 저장소 · DOM 손잡이

// ① DOM 을 직접 만져야 할 때 (포커스 · 스크롤 · 재생)
function SearchBox(){
  const inputRef = useRef(null);
  useEffect(() => { inputRef.current.focus(); }, []);  // 열리면 포커스
  return <input ref={inputRef} />;
}

// ② "바뀌어도 다시 그릴 필요 없는 값" 보관
function Timer(){
  const idRef = useRef(null);           // 타이머 id — 화면과 무관
  const start = () => { idRef.current = setInterval(tick, 1000); };
  const stop  = () => clearInterval(idRef.current);
}
ref.current 를 바꿔도 화면은 안 바뀝니다. 화면에 보여야 하는 값이면 state, 아니면 ref — 이 구분이 "값은 바뀌는데 화면이 안 바뀌어요"를 예방합니다.

경쟁 상태 — effect 로 fetch 할 때의 대표 사고

// id 가 1 → 2 로 빠르게 바뀌면:
//   요청1(느림) · 요청2(빠름) 순서로 응답이 오면
//   화면에는 ★ 요청1(옛날 것) 이 남습니다

useEffect(() => {
  let alive = true;                    // 이 렌더의 생존 플래그
  fetch(`/api/user/${id}`)
    .then(r => r.json())
    .then(d => { if (alive) setUser(d); });  // 죽은 요청은 무시
  return () => { alive = false; };     // id 가 바뀌면 이전 것 무효화
}, [id]);
AbortController 는 요청 자체를 취소하고, 생존 플래그는 응답을 무시합니다 — 둘 다 결과는 같습니다. 이 코드를 매번 쓰기 싫다는 것이 12번(서버 데이터)에서 쿼리 라이브러리로 넘어가는 이유입니다.
effect 는 바깥 세계와 동기화하는 자리입니다 — 계산이나 이벤트 처리 자리가 아닙니다.
REACT 07

리스트와 key

key 를 index 로 주면 입력값이 엉뚱한 줄로 옮겨갑니다.

{/* ✗ 순서가 바뀌거나 중간이 지워지면 어긋납니다 */}
{items.map((it, i) => <Row key={i} item={it} />)}

{/* ✓ 그 데이터를 가리키는 고유 값 */}
{items.map((it) => <Row key={it.id} item={it} />)}
index 를 key 로 썼을 때 실제로 벌어지는 일

  [0] 사과  ← 체크함
  [1] 바나나
  [2] 포도

  '사과'를 지우면 →  [0] 바나나  [1] 포도
  리액트는 "0번은 그대로네" 라고 판단해
  ★ 사과에 있던 체크·입력값이 '바나나' 에 남습니다
key 를 일부러 바꾸는 기법 — <Form key={userId} /> 처럼 주면 userId 가 바뀔 때 컴포넌트를 통째로 새로 만듭니다. "props 바뀌면 state 초기화" 를 effect 없이 해결하는 정석입니다.
index 를 써도 되는 경우는 순서가 절대 안 바뀌고, 추가·삭제도 없고, 각 항목이 상태를 갖지 않을 때뿐입니다. 조건이 하나라도 흔들리면 쓰지 마세요.
key 는 스타일도 성능 팁도 아니라 "이 행의 상태를 어디로 옮길까"에 대한 답입니다.
REACT 08

폼 다루기

입력 하나하나를 state 로 두는 게 항상 정답은 아닙니다.

제어 컴포넌트

const [form, setForm] = useState({ email:"", pw:"" });

const onChange = (e) =>
  setForm((p) => ({ ...p, [e.target.name]: e.target.value }));

<input name="email" value={form.email} onChange={onChange} />
<input name="pw" type="password" value={form.pw} onChange={onChange} />

{/* 값에 따라 즉시 반응해야 할 때 좋습니다 (실시간 검증·미리보기) */}
value 를 주고 onChange 를 안 주면 입력이 안 되는 칸이 됩니다. 읽기 전용이면 readOnly, 초기값만 주려면 defaultValue.

비제어 · 폼 액션

{/* 글자마다 렌더하지 않아 큰 폼에 유리합니다 */}
function Login(){
  const ref = useRef(null);
  const submit = (e) => {
    e.preventDefault();
    const data = Object.fromEntries(new FormData(ref.current));
    login(data);
  };
  return (

커스텀 훅

반복되는 로직을 빼는 도구입니다. 화면을 빼는 게 아닙니다.

// 입력이 멈춘 뒤에만 값을 넘겨 준다 (검색창 필수)
function useDebounced(value, ms = 300){
  const [v, setV] = useState(value);
  useEffect(() => {
    const t = setTimeout(() => setV(value), ms);
    return () => clearTimeout(t);
  }, [value, ms]);
  return v;
}

// 로컬 저장소와 묶인 상태
function useLocal(key, init){
  const [v, setV] = useState(() => {
    try { return JSON.parse(localStorage.getItem(key)) ?? init; }
    catch { return init; }
  });
  useEffect(() => {
    try { localStorage.setItem(key, JSON.stringify(v)); } catch {}
  }, [key, v]);
  return [v, setV];
}

// 사용
const q = useDebounced(input, 400);
const [theme, setTheme] = useLocal("theme", "dark");

이름은 use 로 시작린터가 규칙을 검사할 수 있게

최상위에서만이른 return 뒤에 훅을 두면 같은 문제가 생깁니다
eslint-plugin-react-hooks 를 반드시 켜세요. 의존성 배열 누락과 훅 순서 위반은 사람 눈으로 잡기 어렵습니다.
