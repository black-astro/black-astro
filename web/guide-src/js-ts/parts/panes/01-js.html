JS 01

자바스크립트를 실행하는 법

같은 문법인데 어디서 도느냐에 따라 쓸 수 있는 기능이 완전히 다릅니다.

실행 환경 세 가지

환경있는 것 / 없는 것
브라우저window·document·localStorage 있음 / 파일 읽기 없음
Node.jsfs·process·path 있음 / DOM 없음 (document is not defined)
Deno · BunNode 호환을 지향하는 대안 런타임. 실무 표준은 아직 Node
# 1) 브라우저 콘솔 — F12 를 누르고 바로 입력 (가장 빠름)

# 2) Node — 파일로 실행
node app.js
node --watch app.js            # 저장하면 자동 재시작 (Node 18+)

# 3) 한 줄만 실행
node -e "console.log(1 + 1)"

# 4) TypeScript 를 바로 실행 (Node 22+ 실험 기능)
node --experimental-strip-types app.ts
“이 코드가 어디서 도는가”를 늘 의식하세요. React Native 는 브라우저가 아니라서 document 가 없고, Electron 은 한 앱 안에서 두 환경이 동시에 돕니다(⚡ Electron 탭 e01). 에러의 상당수가 이 구분을 놓쳐서 납니다.

프로젝트의 시작 — package.json

mkdir my-app && cd my-app
npm init -y                    # package.json 생성
npm pkg set type=module        # ★ ESM 으로 고정 (j09 참고)
{
  "name": "my-app",
  "type": "module",            // "import" 문법을 쓰겠다는 선언
  "scripts": {
    "dev": "node --watch src/index.js",
    "start": "node src/index.js"
  },
  "dependencies": {},          // 실행에 필요한 것
  "devDependencies": {}        // 개발할 때만 필요한 것 (테스트·린트)
}
node_modules 는 절대 커밋하지 마세요. 대신 package-lock.json 은 반드시 커밋합니다 — 그래야 내 PC 와 서버가 정확히 같은 버전을 씁니다. (자세한 내용은 🧰 실전 도구 탭 o01)
JS 02

변수 · 타입 · 형변환 함정

자바스크립트의 나쁜 평판은 대부분 이 절에서 나옵니다. 규칙만 알면 피할 수 있습니다.

변수 선언 — 결론부터

const name = "김현우";     // ★ 기본값. 재할당 금지
let count = 0;             // 값이 바뀌어야 할 때만
var old = 1;               // ❌ 쓰지 마세요

// const 는 "재할당 금지"이지 "변경 금지"가 아닙니다
const user = { name: "김" };
user.name = "이";          // ✅ 됩니다 (객체 내부는 바뀜)
user = {};                 // ❌ TypeError: Assignment to constant

// 진짜로 못 바꾸게 하려면
const frozen = Object.freeze({ name: "김" });
frozen.name = "이";        // 조용히 무시됨 (strict 모드면 에러)
var 를 쓰면 안 되는 이유 — 블록을 무시하고 함수 전체가 범위이며, 선언 전에 써도 에러가 안 나고 undefined 가 됩니다(호이스팅). 반복문 안에서 쓰면 모든 콜백이 같은 값을 보게 됩니다(j04).

타입 7 + 1

typeof "글자"        // "string"
typeof 42            // "number"   (정수·실수 구분 없음)
typeof 10n           // "bigint"   (아주 큰 정수)
typeof true          // "boolean"
typeof undefined     // "undefined" (값을 아직 안 넣음)
typeof Symbol()      // "symbol"
typeof null          // "object"   ← ★ 언어의 유명한 버그
typeof {}            // "object"   (배열·함수·날짜도 전부 객체)
typeof (() => {})    // "function"

// 배열인지 확인은 typeof 로 안 됩니다
Array.isArray([1,2])          // true  ★ 이걸 쓰세요
[1,2] instanceof Array        // true (프레임 넘나들면 실패할 수 있음)

형변환 — 실무에서 진짜 겪는 것들

// ① == 는 타입을 멋대로 맞춥니다. === 만 쓰세요
0 == "0"          // true   ← 이래서 사고가 납니다
0 === "0"         // false  ✅
null == undefined // true   (이 한 가지만 == 가 편리)
null === undefined// false

// ② 숫자 변환
Number("42")      // 42
Number("42px")    // NaN
parseInt("42px")  // 42     ← 입력값 파싱에는 이쪽
Number("")        // 0      ★ 빈 문자열이 0 이 되는 함정
Number(null)      // 0
Number(undefined) // NaN

// ③ NaN 은 자기 자신과도 다릅니다
NaN === NaN               // false
Number.isNaN(value)       // ★ 이걸로 확인
Number.isInteger(3.0)     // true

// ④ 소수 계산 (모든 언어 공통 부동소수점 문제)
0.1 + 0.2 === 0.3         // false !
(0.1 + 0.2).toFixed(2)    // "0.30"
// ★ 돈 계산은 "원 단위 정수"로 하거나 decimal.js 를 쓰세요

falsy — 조건문에서 false 로 취급되는 값

false, 0, -0, 0n, "", null, undefined, NaN   // ← 이 8개가 전부
// 나머지는 전부 truthy: "0", "false", [], {}, function(){}

// 그래서 이런 버그가 납니다
if (count) { ... }          // ❌ count 가 0 이면 안 들어감
if (count !== undefined)    // ✅ 의도한 검사

// 개수·금액처럼 0 이 유효한 값이면 반드시 명시적으로 비교하세요
const qty = input ?? 1;     // ✅ null/undefined 일 때만 1
const bad = input || 1;     // ❌ input 이 0 이어도 1 이 됨
?? 와 || 의 차이가 실무에서 제일 자주 나오는 함정입니다. “값이 없을 때 기본값”이 목적이라면 거의 항상 ?? 가 정답입니다.
JS 03

함수 · 화살표 함수 · this

this 는 “어떻게 호출했는가”로 정해집니다. 이 한 문장이 전부입니다.

함수를 만드는 방법

// 1) 함수 선언문 — 호이스팅됨 (선언 전에 호출 가능)
function add(a, b) { return a + b; }

// 2) 함수 표현식
const add2 = function (a, b) { return a + b; };

// 3) 화살표 함수 — ★ 요즘 기본
const add3 = (a, b) => a + b;           // 한 줄이면 return 생략
const double = n => n * 2;               // 인자 1개면 괄호 생략 가능
const makeUser = name => ({ name });     // ★ 객체 반환은 괄호로 감싸기

// 기본값 · 나머지 매개변수
function greet(name, greeting = "안녕", ...rest) {
  console.log(greeting, name, rest);
}
greet("김현우");                   // 안녕 김현우 []
greet("김현우", "반가워", 1, 2);   // 반가워 김현우 [1, 2]
화살표 함수가 만능은 아닙니다. this·arguments·new 가 없으므로 객체의 메서드나 생성자로는 쓸 수 없습니다. addEventListener 에서 this(이벤트 대상)가 필요할 때도 마찬가지입니다.

this 가 정해지는 규칙 — 위에서부터

const obj = {
  name: "객체",
  normal() { return this.name; },          // 호출한 주체 = obj
  arrow:  () => this?.name,                // ★ 바깥 스코프의 this (obj 아님)
};

obj.normal();          // "객체"   — 점 앞의 것이 this
obj.arrow();           // undefined — 화살표는 this 를 안 만듦

const f = obj.normal;  // ★ 떼어내는 순간 주체가 사라짐
f();                   // undefined (strict) — 흔한 버그

// 해결책 3가지
const g = obj.normal.bind(obj);   // ① 묶어 두기
obj.normal.call(obj);             // ② 호출 시 지정
setTimeout(() => obj.normal(), 0);// ③ ★ 화살표로 감싸기 (가장 흔함)

실무에서 이 함정을 만나는 자리

class Timer {
  seconds = 0;

  // ❌ 1초 뒤 this 가 사라짐
  startBad() {
    setInterval(function () { this.seconds++; }, 1000);
  }

  // ✅ 화살표는 바깥(인스턴스)의 this 를 그대로 씀
  startGood() {
    setInterval(() => { this.seconds++; }, 1000);
  }

  // ✅ 클래스 필드로 정의해도 안전 (React 이벤트 핸들러에서 자주)
  handleClick = () => { this.seconds = 0; };
}
외우는 요령 — 일반 함수의 this 는 “누가 불렀나”, 화살표 함수의 this 는 “어디에 쓰였나”. 콜백 안에서는 거의 항상 화살표가 정답입니다.
JS 04

스코프 · 클로저

면접 단골이자, 리액트 훅·모듈 패턴·이벤트 핸들러가 전부 이것 위에 서 있습니다.

클로저 = 함수 + 태어난 곳의 기억

function makeCounter() {
  let count = 0;                    // 바깥 함수의 지역 변수

  return {
    increment() { return ++count; },  // ★ count 를 계속 기억한다
    get value()  { return count; },
  };
}

const c = makeCounter();
c.increment();   // 1
c.increment();   // 2
c.value;         // 2
c.count;         // undefined — 밖에서는 못 건드림 (= 진짜 캡슐화)

// makeCounter 는 이미 끝났는데도 count 가 살아 있습니다.
// "함수가 자기 스코프를 들고 다니는 것" — 이게 클로저입니다.

가장 유명한 함정

var — 전부 3 이 찍힘
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log(i));
}
// 3, 3, 3
// var 는 함수 스코프라
// i 가 하나뿐
let — 0, 1, 2
for (let i = 0; i < 3; i++) {
  setTimeout(() => console.log(i));
}
// 0, 1, 2
// let 은 반복마다
// 새 바인딩을 만듦

실무에서 쓰는 클로저

// ① 디바운스 — 마지막 호출만 실행 (검색창 입력)
function debounce(fn, ms = 300) {
  let timer;                              // ★ 호출 간에 유지되는 상태
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), ms);
  };
}
const search = debounce((q) => fetch(`/api?q=${q}`), 300);

// ② 스로틀 — 일정 간격으로만 실행 (스크롤·리사이즈)
function throttle(fn, ms = 200) {
  let last = 0;
  return (...args) => {
    const now = Date.now();
    if (now - last < ms) return;
    last = now;
    fn(...args);
  };
}

// ③ 한 번만 실행 (초기화 보장)
function once(fn) {
  let done = false, result;
  return (...args) => {
    if (!done) { done = true; result = fn(...args); }
    return result;
  };
}

// ④ 간단한 메모이제이션 (비싼 계산 캐시)
function memoize(fn) {
  const cache = new Map();
  return (arg) => {
    if (!cache.has(arg)) cache.set(arg, fn(arg));
    return cache.get(arg);
  };
}
클로저는 메모리 누수의 흔한 원인이기도 합니다. 큰 객체를 붙잡은 클로저가 이벤트 리스너로 남아 있으면 영원히 해제되지 않습니다. 리스너는 반드시 removeEventListener 로 정리하세요 (🧬 초고급 탭 d04 에서 실제로 찾는 방법을 다룹니다).
JS 05

객체 · 프로토타입

자바스크립트에는 클래스가 원래 없었습니다. 지금도 내부는 프로토타입입니다.

객체 다루기 실전

const user = { id: 1, name: "김현우", role: "dev" };

// 순회 — 목적에 맞는 것을 고르세요
Object.keys(user);     // ["id", "name", "role"]
Object.values(user);   // [1, "김현우", "dev"]
Object.entries(user);  // [["id",1], ["name","김현우"], ...]

// 배열 ↔ 객체 변환
Object.fromEntries([["a",1],["b",2]]);        // { a:1, b:2 }
Object.fromEntries(new URLSearchParams(qs));  // 쿼리스트링 → 객체

// 동적 키 · 계산된 속성
const key = "score";
const obj = { [key]: 10, [`${key}Max`]: 100 };  // { score:10, scoreMax:100 }

// 속성 존재 확인
"name" in user            // true  (프로토타입까지 확인)
Object.hasOwn(user, "name")  // true  ★ 자기 것만 (ES2022, 권장)

복사 — 얕은 복사와 깊은 복사

const orig = { name: "김", tags: ["a", "b"] };

// 얕은 복사 — 1단계만 복사됨
const shallow = { ...orig };
shallow.tags.push("c");        // ★ orig.tags 도 바뀝니다!

// 깊은 복사 — ES2022 내장 (라이브러리 불필요)
const deep = structuredClone(orig);
deep.tags.push("c");           // orig 는 그대로 ✅

// ❌ 옛날 방식 — 함수·Date·undefined 가 사라집니다
const bad = JSON.parse(JSON.stringify(orig));
상태 관리 버그의 절반은 얕은 복사입니다. React·Zustand 에서 “값을 바꿨는데 화면이 안 바뀐다”거나 “안 건드린 값이 같이 바뀐다”면 거의 여기입니다.

프로토타입 사슬

const arr = [1, 2, 3];

arr.map(...)   // arr 에는 map 이 없습니다
               //   → Array.prototype 에서 찾음 ✅

// 사슬: arr → Array.prototype → Object.prototype → null

Object.getPrototypeOf(arr) === Array.prototype;   // true
arr.hasOwnProperty("map");                        // false
Array.prototype.hasOwnProperty("map");            // true
// 프로토타입을 직접 쓰는 경우 (라이브러리 내부에서만)
function Animal(name) { this.name = name; }
Animal.prototype.speak = function () {
  return `${this.name} 이(가) 소리를 냅니다`;
};
const dog = new Animal("멍멍이");
dog.speak();

// 위와 정확히 같은 것을 class 문법으로
class Animal2 {
  constructor(name) { this.name = name; }
  speak() { return `${this.name} 이(가) 소리를 냅니다`; }
}
// ★ class 는 프로토타입을 감싼 "문법 설탕"입니다
내장 프로토타입을 수정하지 마세요. Array.prototype.myUtil = ... 같은 코드는 다른 라이브러리와 충돌하고, for...in 순회를 오염시킵니다. 프로토타입 오염은 실제 보안 취약점이기도 합니다(🧬 초고급 d13).

JS 06

배열 메서드 완전정복

실무 코드의 상당 부분이 배열 변형입니다. 여기가 손에 붙으면 속도가 달라집니다.

목적별 지도

하려는 일메서드
모양 바꾸기map
걸러 내기filter
하나로 합치기reduce
하나 찾기find / findIndex / findLast
있는지 확인some(하나라도) / every(전부) / includes
정렬toSorted (원본 유지) / sort (원본 변경)
평탄화flat / flatMap
그룹으로 묶기Object.groupBy ES2024
const orders = [
  { id: 1, user: "김", amount: 15000, status: "paid" },
  { id: 2, user: "이", amount:  8000, status: "pending" },
  { id: 3, user: "김", amount: 32000, status: "paid" },
];

// 체이닝 — 읽는 순서대로 흘러갑니다
const total = orders
  .filter(o => o.status === "paid")     // 결제된 것만
  .map(o => o.amount)                    // 금액만
  .reduce((sum, n) => sum + n, 0);       // 합계  → 47000

// 사용자별로 묶기 (ES2024)
Object.groupBy(orders, o => o.user);
// { 김: [주문1, 주문3], 이: [주문2] }

// 예전 방식 (지원 범위가 넓음)
orders.reduce((acc, o) => {
  (acc[o.user] ??= []).push(o);          // ★ ??= 로 초기화
  return acc;
}, {});

원본을 바꾸는 메서드 — 사고의 원인

// ❌ 원본을 바꿉니다 (React 상태에 쓰면 화면이 안 바뀜)
arr.sort()      arr.reverse()   arr.splice()
arr.push()      arr.pop()       arr.shift()   arr.unshift()

// ✅ 새 배열을 반환합니다 (ES2023 — 이제 짝이 다 생겼습니다)
arr.toSorted()   arr.toReversed()   arr.toSpliced()   arr.with(i, v)

const scores = [3, 1, 2];
const sorted = scores.toSorted((a, b) => a - b);   // scores 는 그대로 ✅
const changed = scores.with(0, 99);                 // [99, 1, 2]
sort() 는 기본이 문자열 정렬입니다. [10, 9, 1].sort() 는 [1, 10, 9] 가 됩니다. 숫자는 반드시 비교 함수를 주세요 — (a, b) => a - b.

자주 쓰는 조합

// 중복 제거
[...new Set([1, 2, 2, 3])];                    // [1, 2, 3]

// 객체 배열 중복 제거 (id 기준)
const uniq = [...new Map(list.map(o => [o.id, o])).values()];

// 두 배열의 교집합 · 차집합
const setB = new Set(b);
const inter = a.filter(x => setB.has(x));
const diff  = a.filter(x => !setB.has(x));

// 청크 나누기 (100개씩 처리 — 대량 API 호출에 필수)
const chunk = (arr, n) =>
  Array.from({ length: Math.ceil(arr.length / n) },
             (_, i) => arr.slice(i * n, i * n + n));

// 여러 기준으로 정렬
list.toSorted((a, b) =>
  a.grade - b.grade || a.name.localeCompare(b.name));  // ★ 한글 정렬
배열 안에서 includes 로 검사하는 반복문은 느립니다. arr.filter(x => other.includes(x)) 는 O(n×m) 입니다. 한쪽을 Set 으로 바꾸면 O(n+m) 이 됩니다. 데이터가 만 건을 넘으면 체감이 확 달라집니다.

클래스 · 상속

기본형

class User {
  // public 필드
  role = "user";
  // ★ 진짜 private (ES2022) — 밖에서 접근하면 문법 에러
  #password;
  // static — 인스턴스 없이 클래스에서 바로
  static count = 0;

  constructor(name, password) {
    this.name = name;
    this.#password = password;
    User.count++;
  }

  // getter / setter — 속성처럼 쓰이는 메서드
  get displayName() { return `${this.name}(${this.role})`; }
  set displayName(v) { this.name = v.trim(); }

  // 인스턴스 메서드
  checkPassword(input) { return this.#password === input; }

  // 정적 팩토리 — 이름 있는 생성자 대용
  static fromJSON(json) { return new User(json.name, json.pw); }
}

const u = new User("김현우", "1234");
u.displayName;         // "김현우(user)"  ← 괄호 없이 호출
u.#password;           // ❌ SyntaxError — 밖에서는 접근 불가
# 은 관례가 아니라 진짜 강제입니다. 예전의 _password 는 “건드리지 말자”는 약속일 뿐이었지만, #password 는 언어 차원에서 막습니다. TypeScript 의 private 은 컴파일 때만 검사하므로, 런타임 보호가 필요하면 # 를 쓰세요.

class Admin extends User {

    super(name, password);         // ★ this 를 쓰기 전에 반드시 먼저

  }

  get displayName() {
    return `👑 ${super.displayName}`;    // 부모 것도 쓸 수 있음
  }

// 커스텀 에러 — 실무에서 가장 자주 쓰는 상속
class AppError extends Error {
  constructor(message, code, status = 400) {
    super(message);
    this.name = this.constructor.name;   // "AppError"
    this.code = code;
    this.status = status;
    Error.captureStackTrace?.(this, this.constructor);  // Node 전용
  }
}

class NotFoundError extends AppError {
  constructor(what) { super(`${what} 을(를) 찾을 수 없습니다`, "NOT_FOUND", 404); }
}

try {
  throw new NotFoundError("주문");
} catch (e) {
  if (e instanceof NotFoundError) res.status(e.status).json({ code: e.code });
}
상속을 남용하지 마세요. 3단계를 넘어가면 “이 메서드가 어디서 왔지?”를 찾는 데 시간이 더 듭니다. 대부분은 합성(composition) — 필요한 객체를 받아서 쓰는 것이 더 유연합니다. 상속은 에러 클래스와 프레임워크가 요구할 때 정도로 충분합니다.

Node 진영에서 가장 많은 시간을 잡아먹는 주제입니다. 규칙을 한 번에 정리합니다.

CommonJS — 옛 Node 방식

혼용 규칙
  ESM 에서 CJS 가져오기   ✅ 됨 (default 로 들어옴)
  CJS 에서 ESM 가져오기   ❌ require 불가 → await import() 만
  ★ 그래서 "라이브러리가 ESM 전용"이면 CJS 프로젝트가 막힙니다

· Cannot use import statement outside a module → "type": "module" 이 빠짐

· ERR_MODULE_NOT_FOUND → ESM 인데 확장자(.js)를 안 적음

실무 규칙

  • 새 프로젝트는 무조건 ESM — 표준이고, 최신 라이브러리가 ESM 전용으로 가고 있습니다
  • 내부 경로에는 별칭을 쓰세요 — ../../../utils 는 리팩터링 지옥입니다
  • 배럴 파일(index.js 에서 몰아 export)은 편하지만 번들 크기와 순환 참조 문제를 만듭니다. 라이브러리에서는 피하세요
  • 순환 참조는 ESM 에서 undefined 로 조용히 나타납니다 — A가 B를, B가 A를 import 하면 한쪽이 초기화 전에 읽힙니다

싱글 스레드인데 동시에 일하는 것처럼 보이는 이유. 면접 최다 출제 주제입니다.

한 장으로 보는 구조

        ┌─────────────────┐
        │   콜 스택        │  ← 지금 실행 중인 코드 (하나뿐)
        └────────┬────────┘
                 │ 비었을 때만 아래에서 가져온다
    ┌────────────┴────────────┐
    │ ① 마이크로태스크 큐      │  Promise.then, queueMicrotask
    │    (전부 비울 때까지)     │  ★ 우선순위 높음
    └────────────┬────────────┘
    ┌────────────┴────────────┐
    │ ② 매크로태스크 큐        │  setTimeout, setInterval,
    │    (한 번에 하나)         │  I/O 완료, 이벤트 콜백
    └─────────────────────────┘

핵심 규칙
  · 스택이 비어야 큐를 본다
  · 마이크로태스크를 전부 비운 뒤에 매크로태스크 하나
  · 그래서 Promise 가 setTimeout(0) 보다 항상 먼저

  await null;
  console.log("6");                    // await 뒤는 마이크로태스크
})();

console.log("7");

// 출력: 1 → 5 → 7 → 3 → 4 → 6 → 2
//       └── 동기 ──┘  └ 마이크로 ┘  └매크로

Promise · async/await

비동기 코드를 동기처럼 읽히게 만드는 문법. 실무 코드의 절반이 여기입니다.

기본 사용법

// async 함수는 항상 Promise 를 반환합니다
async function getUser(id) {
  const res = await fetch(`/api/users/${id}`);
  if (!res.ok) throw new Error(`HTTP ${res.status}`);   // ★ fetch 는 4xx/5xx 에 안 던집니다
  return res.json();
}

// 호출하는 쪽
try {
  const user = await getUser(1);
} catch (e) {
  console.error("실패", e);
}

// 최상위에서도 await 가능 (ESM 한정)
const data = await getUser(1);
fetch 는 404·500 에서 에러를 던지지 않습니다. 네트워크 자체가 실패했을 때만 reject 됩니다. res.ok 확인을 빼먹으면 에러 페이지 HTML 을 JSON 으로 파싱하려다 엉뚱한 곳에서 터집니다. axios 는 반대로 4xx 에 던집니다.

순차 vs 병렬 — 성능 차이가 큽니다

순차 — 300ms
const a = await getA(); //100ms
const b = await getB(); //100ms
const c = await getC(); //100ms
// 서로 관계없는데
// 줄 서서 기다림
병렬 — 100ms
const [a, b, c] =
  await Promise.all([
    getA(), getB(), getC()
  ]);
// 동시에 출발

Promise 조합 메서드 — 목적이 다릅니다

메서드동작
all전부 성공해야 성공. 하나라도 실패하면 즉시 reject
allSettled전부 끝날 때까지 기다림. 성공/실패를 각각 알려 줌
race가장 먼저 끝난 것 하나 (성공이든 실패든) — 타임아웃 구현에
any가장 먼저 성공한 것 하나 (여러 미러 서버 중 하나)
// 실패해도 나머지는 처리해야 할 때 — 알림 발송 등
const results = await Promise.allSettled(users.map(sendMail));
const failed = results.filter(r => r.status === "rejected");
console.log(`${failed.length}건 실패`);

// 타임아웃 붙이기 (직접 구현)
const withTimeout = (p, ms) => Promise.race([
  p,
  new Promise((_, rej) =>
    setTimeout(() => rej(new Error("시간 초과")), ms)),
]);
await withTimeout(fetch(url), 3000);

// ★ 요즘은 내장 기능으로 (AbortSignal)
await fetch(url, { signal: AbortSignal.timeout(3000) });

흔한 실수

// ❌ forEach 는 await 를 기다려 주지 않습니다
items.forEach(async (item) => { await save(item); });
console.log("완료");     // 저장 끝나기 전에 찍힘

// ✅ 순서대로 저장해야 한다면
for (const item of items) await save(item);

// ✅ 동시에 저장해도 된다면
await Promise.all(items.map(save));

// ⚠️ 단, 1,000개를 한꺼번에 보내면 상대 서버가 죽습니다
//    → 동시 실행 수 제한이 필요합니다 (🟩 Node 탭 n11)
await 를 안 붙인 Promise 는 조용히 사라집니다. save(item); 처럼 쓰면 에러가 나도 아무도 모르고, Node 에서는 프로세스가 죽을 수도 있습니다. 의도적으로 기다리지 않을 때는 void save(item) 또는 .catch(log) 를 반드시 붙이세요.

    super(message);
    this.name = "ValidationError";
    this.field = field;

제너레이터 기본

function* range(start, end, step = 1) {
  for (let i = start; i < end; i += step) yield i;   // ★ 여기서 멈췄다 재개
}

for (const n of range(0, 5)) console.log(n);   // 0 1 2 3 4
[...range(0, 3)];                              // [0, 1, 2]

// 무한 수열도 가능합니다 (필요한 만큼만 계산되므로)
function* ids() { let i = 1; while (true) yield i++; }
const gen = ids();
gen.next().value;   // 1
gen.next().value;   // 2

// 값을 주고받기
function* dialog() {
  const name = yield "이름이?";
  yield `안녕하세요 ${name}님`;
}
const d = dialog();
d.next().value;        // "이름이?"
d.next("김현우").value; // "안녕하세요 김현우님"

// 페이지네이션 API 를 "그냥 반복문"처럼 쓰기
async function* fetchAllUsers() {
  let page = 1;
  while (true) {
    const res = await fetch(`/api/users?page=${page}`);
    const { items, hasNext } = await res.json();
    yield* items;                    // 한 건씩 흘려보냄
    if (!hasNext) return;
    page++;
  }
}

  await processUser(user);
