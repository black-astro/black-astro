Node.js 연결

비동기라 await 를 빠뜨리면 조용히 망가집니다.

// PostgreSQL — pg
import pg from 'pg';

const pool = new pg.Pool({
  connectionString: process.env.DATABASE_URL,
  max: 10,
  idleTimeoutMillis: 30_000,
  connectionTimeoutMillis: 3_000,
  application_name: 'shop-api',
});

const { rows } = await pool.query(
  'SELECT id, name FROM member WHERE grade = $1', ['GOLD']);   // ★ $1 바인딩
console.log(rows);

// 트랜잭션은 커넥션 하나를 잡고 진행합니다
const client = await pool.connect();
try {
  await client.query('BEGIN');
  await client.query('UPDATE stock SET qty = qty - 1 WHERE id = $1 AND qty >= 1', [10]);
  await client.query('INSERT INTO orders (member_id) VALUES ($1)', [1]);
  await client.query('COMMIT');
} catch (e) {
  await client.query('ROLLBACK');
  throw e;
} finally {
  client.release();        // ★ 반드시 반납 — 빠뜨리면 커넥션 누수
}
// MySQL — mysql2 (프로미스 버전)
import mysql from 'mysql2/promise';

const pool = mysql.createPool({
  host: '127.0.0.1', user: 'app', password: process.env.DB_PW,
  database: 'shop', charset: 'utf8mb4',

Node 에서 가장 흔한 사고 두 가지
① await 누락 — 쿼리가 끝나기 전에 다음 코드가 실행되고, 에러가 unhandled rejection 으로 조용히 사라집니다

SECTION 05

SQL 인젝션 — 바인딩

원인은 하나입니다 — 문자열을 이어 붙여 쿼리를 만드는 것. 파라미터 바인딩을 쓰면 원천 차단됩니다.

바인딩이 안전한 이유는 값을 검사해서가 아니라 값이 문법이 될 기회 자체를 없애기 때문입니다.
✗ 문자열 연결 — 즉시 뚫립니다
uid = request.args.get("id")
cur.execute(f"SELECT * FROM member WHERE id = {uid}")

# 공격자가 id 에 이렇게 넣으면
#   1 OR 1=1
#   1; DROP TABLE member; --
#   1 UNION SELECT password FROM admin --
✓ 파라미터 바인딩
cur.execute("SELECT * FROM member WHERE id = %s", (uid,))

# DB 가 쿼리 구조와 값을 분리해 처리하므로
# 값 안에 무엇이 들어와도 '값'으로만 취급됩니다
왜 안전한가 — 바인딩을 쓰면 DB가 쿼리 구조를 먼저 파싱해 확정하고 나중에 값을 채웁니다. 값 안의 ' 나 ; 가 구문으로 해석될 여지 자체가 없습니다. "특수문자를 이스케이프한다"는 접근은 빠뜨리는 경우가 반드시 생기므로 권장되지 않습니다.

바인딩할 수 없는 것 — 테이블·컬럼 이름

# ✗ 이건 바인딩이 안 됩니다 (구조는 파라미터가 될 수 없음)
cur.execute("SELECT * FROM member ORDER BY %s", (sort_col,))    # 동작 안 함

# ✓ 화이트리스트로 검증
ALLOWED = {"id", "name", "joined_at"}
if sort_col not in ALLOWED:
    raise ValueError("허용되지 않은 정렬 기준")
cur.execute(f"SELECT * FROM member ORDER BY {sort_col}")        # 검증했으므로 안전

# 정렬 방향도 마찬가지
direction = "DESC" if desc else "ASC"

IN 절에 목록 넣기

# ✗ 문자열 join 은 위험
ids_str = ",".join(ids)
cur.execute(f"SELECT * FROM member WHERE id IN ({ids_str})")

# ✓ 자리표시자를 개수만큼 만들기
ph = ",".join(["%s"] * len(ids))
cur.execute(f"SELECT * FROM member WHERE id IN ({ph})", tuple(ids))

# ✓ PostgreSQL 은 배열로 더 깔끔하게
cur.execute("SELECT * FROM member WHERE id = ANY(%s)", (ids,))
ORM 을 써도 안전하지 않은 경우가 있습니다.
· raw() · text() · createQuery() 에 문자열을 조립해 넣는 경우
· 정렬·필터 컬럼명을 사용자 입력에서 그대로 받는 경우
· LIKE 패턴을 조립하는 경우 (%·_ 이스케이프 필요)
"ORM 을 쓰니 안전하다"는 착각이 실제 사고로 이어집니다.
추가 방어선 — ① 앱 계정에 DDL 권한을 주지 않기 (인젝션이 성공해도 DROP 불가) ② 에러 메시지를 사용자에게 노출하지 않기 (스키마 정보가 새어 나갑니다) ③ 쿼리 타임아웃 (시간 기반 블라인드 인젝션을 어렵게) ④ WAF·정적 분석 도구로 문자열 조립 패턴 탐지
SECTION 06

ORM을 쓸 것인가

"ORM 이냐 생쿼리냐"는 잘못된 질문입니다. 어디에 무엇을 쓸지가 옳은 질문입니다.

상황권장
단순 CRUDORM — 반복 코드가 사라집니다
도메인 로직이 있는 저장ORM — 객체 상태 관리가 편합니다
복잡한 조회·통계생쿼리 또는 쿼리 빌더 — ORM 으로 짜면 읽기도 튜닝하기도 어렵습니다
대량 배치생쿼리 · 전용 대량 적재(a10)
리포트·집계생쿼리 — 윈도우 함수·CTE 를 그대로 씁니다
성능이 중요한 API측정 후 결정. 필요한 컬럼만 조회하는 프로젝션이 핵심
언어ORM쿼리 빌더 · 경량
PythonSQLAlchemy ORM · Django ORMSQLAlchemy Core
JavaJPA/HibernateQueryDSL · MyBatis · jOOQ
Node/TSPrisma · TypeORMKysely · Drizzle · knex

ORM 이 만드는 전형적 문제

  • N+1 쿼리 — 지연 로딩이 반복문 안에서 터집니다(a08). 가장 흔한 성능 문제
  • SELECT * — 엔티티 전체를 가져오느라 필요 없는 컬럼·LOB 까지 읽습니다
  • 예상 못 한 UPDATE — 변경 감지(dirty checking)로 의도하지 않은 쿼리가 나갑니다
  • 복잡한 쿼리의 가독성 붕괴 — 메서드 체인 20줄보다 SQL 10줄이 명확한 경우가 많습니다
  • 실행되는 SQL 을 모름 — 로그를 반드시 켜세요
# 개발 환경에서는 반드시 SQL 로그를 켜세요
# Spring Boot
logging.level.org.hibernate.SQL=DEBUG
logging.level.org.hibernate.orm.jdbc.bind=TRACE

# SQLAlchemy
create_engine(url, echo=True)

# Prisma
new PrismaClient({ log: ['query'] })

# Django
LOGGING = {'loggers': {'django.db.backends': {'level': 'DEBUG'}}}
실무의 현실적 조합 — 기본은 ORM, 무거운 조회는 생쿼리. 한 프로젝트에서 섞어 쓰는 것이 이상한 일이 아닙니다. 자바에서는 JPA + MyBatis(또는 QueryDSL), 파이썬에서는 SQLAlchemy ORM + Core 조합이 흔합니다.
ORM 은 SQL 을 대체하지 않습니다 — 반복되는 SQL 을 줄여 주는 도구입니다.
SECTION 07

코드에서 트랜잭션 다루기

경계를 어디서 열고 어디서 닫는가가 핵심입니다. 잘못 잡으면 롤백이 안 됩니다. 

# Python — 컨텍스트 매니저로 경계를 명확히
from contextlib import contextmanager

@contextmanager
def transaction(conn):
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise                      # ★ 반드시 다시 던지세요 — 삼키면 호출자가 성공으로 오해합니다

with transaction(conn) as tx:
    tx.execute("UPDATE stock SET qty = qty - 1 WHERE id = %s AND qty >= 1", (10,))
    if tx.rowcount == 0:
        raise ValueError("재고 부족")        # 예외 → 자동 롤백
    tx.execute("INSERT INTO orders (member_id) VALUES (%s)", (1,))
// Spring — 선언적 트랜잭션
@Service
public class OrderService {

    @Transactional                                    // 서비스 계층에 경계를 둡니다
    public Order create(Long memberId, Long productId) {
        int updated = stockRepository.decrease(productId);
        if (updated == 0) throw new OutOfStockException();   // 런타임 예외 → 롤백
        return orderRepository.save(new Order(memberId));
    }

    @Transactional(readOnly = true)                   // 읽기 전용 힌트 (최적화 · 실수 방지)
    public List<Order> findByMember(Long memberId) { ... }
}
Spring @Transactional 의 3대 함정
① 자기 호출(self-invocation) — 같은 클래스 안에서 this.method() 로 부르면 프록시를 거치지 않아 트랜잭션이 안 걸립니다
② 체크 예외는 기본적으로 롤백하지 않습니다 — @Transactional(rollbackFor = Exception.class) 를 명시하세요
③ 예외를 잡아 삼키면 롤백되지 않습니다 (catch 후 로깅만 하고 끝내는 코드)

경계를 정하는 원칙

  • 서비스(유스케이스) 계층에 — 컨트롤러나 리포지토리가 아니라
  • 짧게 — 외부 API 호출·파일 IO·사용자 입력 대기를 안에 두지 마세요
  • 중첩을 의식적으로 — 전파 속성(REQUIRED·REQUIRES_NEW)이 의도대로인지 확인. REQUIRES_NEW 는 별도 커넥션을 쓰므로 데드락 위험이 있습니다
  • 읽기 전용 표시 — 실수로 쓰는 것을 막고 복제본 라우팅에 활용할 수 있습니다
// Node — 트랜잭션 헬퍼로 반납 누락을 막습니다
async function withTransaction(pool, fn) {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    const result = await fn(client);
    await client.query('COMMIT');
    return result;
  } catch (e) {
    await client.query('ROLLBACK');
    throw e;
  } finally {
    client.release();
  }
}

await withTransaction(pool, async (c) => {
  await c.query('UPDATE stock SET qty = qty - 1 WHERE id = $1 AND qty >= 1', [10]);
  await c.query('INSERT INTO orders (member_id) VALUES ($1)', [1]);
});
"가끔 느려요"의 흔한 원인은 트랜잭션 안에 들어간 네트워크 호출입니다.

N+1 문제와 배치 조회

목록 100건을 불러오는데 쿼리가 101번 나갑니다. 실무 성능 문제 1위입니다.

N+1 은 쿼리 튜닝으로 못 고칩니다. 고칠 대상은 쿼리가 아니라 왕복 횟수입니다.
✗ N+1
orders = db.query("SELECT * FROM orders LIMIT 100")   # 1번

for o in orders:                                     # 100번
    member = db.query(
        "SELECT * FROM member WHERE id = %s", o.member_id)
    print(member.name)

# 총 101번. 각 쿼리가 2ms 면 202ms.
# 네트워크 왕복이 5ms 면 500ms 를 넘습니다
✓ 배치 조회 — 2번
orders = db.query("SELECT * FROM orders LIMIT 100")

ids = list({o.member_id for o in orders})
members = db.query(
    "SELECT * FROM member WHERE id = ANY(%s)", (ids,))
by_id = {m.id: m for m in members}

for o in orders:
    print(by_id[o.member_id].name)

# 총 2번

ORM 에서의 해법

# SQLAlchemy
from sqlalchemy.orm import selectinload, joinedload

# ✗ 지연 로딩 → N+1
orders = session.query(Order).limit(100).all()
for o in orders: print(o.member.name)

# ✓ selectinload : IN 절로 한 번에 (컬렉션에 유리)
orders = session.query(Order).options(selectinload(Order.member)).limit(100).all()

# ✓ joinedload : 조인 한 번으로 (단일 연관에 유리)
orders = session.query(Order).options(joinedload(Order.member)).limit(100).all()
// JPA — 페치 조인
@Query("SELECT o FROM Order o JOIN FETCH o.member WHERE o.status = :s")
List<Order> findWithMember(@Param("s") String status);

// 또는 @EntityGraph
@EntityGraph(attributePaths = {"member", "items"})
List<Order> findByStatus(String status);

// 컬렉션은 batch size 로 (IN 절로 묶어 가져옵니다)
// application.properties:
//   spring.jpa.properties.hibernate.default_batch_fetch_size=100
// Prisma
const orders = await prisma.order.findMany({
  take: 100,
  include: { member: true },        // 연관을 함께 로드
});
페치 조인의 함정 — 컬렉션(1:N)을 조인하면 행이 곱해집니다. 주문 100건에 각 품목 5개면 500행이 되고, 페이징이 의도대로 동작하지 않습니다 (JPA는 메모리에서 페이징하며 경고를 냅니다).
1:N 은 조인이 아니라 batch_fetch_size·selectinload 로 푸세요.
N+1 을 찾는 법 — 개발 환경에서 SQL 로그를 켜고 화면 하나를 열었을 때 나가는 쿼리 수를 세어 보세요. 같은 형태의 쿼리가 수십 번 반복되면 N+1 입니다. 테스트에서 쿼리 수를 검증하는 도구(datasource-proxy·pytest 픽스처)를 붙이면 회귀를 자동으로 막을 수 있습니다.

SECTION 11

재시도 · 타임아웃 · 누수

네트워크는 반드시 끊깁니다. 페일오버도 일어납니다. 이를 가정하지 않은 코드는 장애 때 서비스 전체를 멈춥니다.

모든 재시도가 동시에 몰리면 그것이 2차 장애입니다 — 백오프에 지터(무작위 흔들기)를 반드시 넣으세요.

재시도해도 되는 오류와 안 되는 오류

판단오류
재시도 O 연결 실패·타임아웃 · 데드락 · 직렬화 충돌(SERIALIZABLE) · 페일오버 중 일시 오류 · 락 대기 타임아웃
재시도 X 문법 오류 · 제약조건 위반 · 권한 없음 · 데이터 오류 — 다시 해도 같은 결과이므로 재시도는 낭비입니다
import time, random
import psycopg

RETRYABLE = ("40001", "40P01", "08000", "08003", "08006")   # 직렬화·데드락·연결 오류

def with_retry(fn, attempts=3, base=0.1):
    for i in range(attempts):
        try:
            return fn()
        except psycopg.Error as e:
            code = getattr(e, "sqlstate", None)
            if code not in RETRYABLE or i == attempts - 1:
                raise
            # 지수 백오프 + 지터 (동시에 몰려 재시도하는 것을 막습니다)
            time.sleep(base * (2 ** i) + random.uniform(0, base))
    raise RuntimeError("unreachable")
재시도는 멱등할 때만 안전합니다. INSERT 를 재시도하면 중복 데이터가 생길 수 있습니다.
· 멱등 키를 두고 UNIQUE 제약으로 중복을 막거나
· ON CONFLICT DO NOTHING 으로 처리하거나
· 조회 → 없으면 삽입이 아니라 UPSERT 로 (🎓 e08)

타임아웃 — 계층마다 설정

-- Oracle

커넥션 누수 잡기

-- PostgreSQL : 오래 idle in transaction 인 세션
SELECT pid, usename, application_name, state,
       now() - state_change AS 지속, query
FROM   pg_stat_activity
WHERE  state = 'idle in transaction' AND now() - state_change > interval '1 min';

-- MySQL
SELECT * FROM information_schema.processlist WHERE command = 'Sleep' AND time > 60;

-- 애플리케이션 이름별 커넥션 수 (어느 앱이 새고 있나)
SELECT application_name, count(*) FROM pg_stat_activity GROUP BY 1 ORDER BY 2 DESC;

테스트에서 DB 다루기

"테스트는 SQLite 로, 운영은 PostgreSQL 로"는 운영에서만 터지는 버그를 만듭니다.

왜 DB를 바꿔 테스트하면 안 되나
· 타입·제약 동작이 다릅니다 (SQLite 는 타입을 강제하지 않습니다)
· 방언 SQL 이 테스트를 통과하지 못하거나, 반대로 운영에서만 실패합니다
· 격리 수준·락 동작이 완전히 다릅니다
· 함수·인덱스 종류가 다릅니다
결과적으로 초록불인데 배포하면 깨지는 상황이 반복됩니다.

테스트 데이터 격리 전략

전략장단점
트랜잭션 롤백 가장 빠르고 깔끔합니다. 단, 테스트 대상이 스스로 커밋하면 쓸 수 없습니다
매번 TRUNCATE 확실하지만 느립니다. 외래키 순서를 고려해야 합니다
테스트마다 새 스키마 완전 격리 · 병렬 실행 가능. 준비 비용이 있습니다
컨테이너 재기동 가장 느립니다. 스위트 단위로만
CI 에서 느리다면 — ① 컨테이너를 세션 스코프로 한 번만 띄우고 ② 테스트마다 트랜잭션 롤백으로 격리하고 ③ tmpfs 에 데이터 디렉터리를 두어 디스크 IO를 없애고 ④ fsync=off 같은 테스트 전용 설정을 씁니다(운영에는 절대 금지).
무엇을 테스트할 것인가 — ORM 이 SELECT 를 잘 만드는지는 테스트할 필요가 없습니다(라이브러리를 믿으세요). 우리가 짠 제약·트랜잭션 경계·동시성 처리를 테스트하세요. 특히 재고 차감·중복 방지·롤백 같은 정합성 로직은 반드시 테스트 대상입니다.
