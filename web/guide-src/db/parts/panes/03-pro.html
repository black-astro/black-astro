SECTION 01

윈도우 함수 — 개념과 문법

GROUP BY는 행을 합쳐 없앱니다. 윈도우 함수는 행을 그대로 두고 옆에 계산 결과를 붙입니다. 이 차이 하나가 리포트 쿼리의 난이도를 완전히 바꿉니다.

고르는 기준은 하나입니다 — 결과에 개별 행이 남아야 하면 윈도우 함수, 요약만 필요하면 GROUP BY.
GROUP BY — 행이 사라집니다
SELECT category, AVG(price)
FROM   product
GROUP  BY category;

-- 카테고리별 1행씩만 남습니다
-- 개별 상품 이름은 볼 수 없습니다
OVER — 행이 그대로 남습니다
SELECT name, category, price,
       AVG(price) OVER (PARTITION BY category)
         AS 카테고리평균
FROM   product;

-- 상품 5개가 그대로 나오면서
-- 각 행에 자기 카테고리 평균이 붙습니다
name category price 카테고리평균 무선 키보드 주변기기 49000 36000 게이밍 마우스 주변기기 32000 36000 USB-C 허브 주변기기 27000 36000 27인치 모니터 디스플레이 289000 289000

문법 구조

함수() OVER (
    PARTITION BY 그룹기준     -- 생략하면 전체가 한 덩어리
    ORDER BY 정렬기준         -- 순위·누적에 필요
    ROWS BETWEEN … AND …      -- 계산 범위(프레임) · e03 에서 자세히
)
부분뜻없으면
PARTITION BY어떤 단위로 나눠 계산할지전체를 하나의 그룹으로 봅니다
ORDER BY그룹 안의 순서순위·LAG·누적을 쓸 수 없습니다
ROWS/RANGE몇 번째 행까지 계산에 넣을지ORDER BY가 있으면 처음~현재행이 기본(누적)

쓸 수 있는 함수

  • 집계형 — SUM · AVG · COUNT · MIN · MAX 를 그대로 OVER와 함께 씁니다
  • 순위형 — ROW_NUMBER · RANK · DENSE_RANK · NTILE (e02)
  • 이동형 — LAG · LEAD · FIRST_VALUE · LAST_VALUE · NTH_VALUE (e03)
  • 분포형 — PERCENT_RANK · CUME_DIST · PERCENTILE_CONT
지원 버전 — MySQL 8.0+ · MariaDB 10.2+ · SQLite 3.25+ · PostgreSQL·Oracle은 오래전부터 지원합니다. MySQL 5.7 이하를 쓰는 레거시라면 이 탭의 많은 문법이 동작하지 않습니다 — 변수(@rownum) 트릭 대신 업그레이드를 검토하는 편이 낫습니다.
실무에서 가장 자주 쓰는 형태 세 가지
① 그룹 안 순위 — ROW_NUMBER() OVER (PARTITION BY x ORDER BY y DESC)
② 전체 대비 비중 — amount * 100.0 / SUM(amount) OVER ()
③ 직전 값과 비교 — LAG(amount) OVER (ORDER BY month)
SECTION 02

순위 RANK 3형제

셋 다 등수를 매기지만 동점을 처리하는 방식이 다릅니다. 이것만 구분하면 됩니다. 

-- 동점을 보여주기 위해 '마우스패드 XL'(49000)을 하나 추가했다고 가정합니다
SELECT name, price,
       ROW_NUMBER() OVER (ORDER BY price DESC) AS rn,
       RANK()       OVER (ORDER BY price DESC) AS rnk,
       DENSE_RANK() OVER (ORDER BY price DESC) AS drnk
FROM   product;

마우스패드 XL49000322

함수동점일 때쓰는 곳
ROW_NUMBER()무조건 1,2,3,4 — 동점도 임의로 갈라놓습니다페이징, 그룹별 대표 행 1건 뽑기
RANK()동점은 같은 등수, 다음은 건너뜁니다 (1,2,2,4)올림픽식 순위, 성적 등수
DENSE_RANK()동점은 같은 등수, 다음은 바로 다음 (1,2,2,3)"상위 3개 가격대" 처럼 등급이 필요할 때

가장 많이 쓰는 패턴 — 그룹별 상위 N

-- 카테고리별 최고가 상품 1개씩
SELECT category, name, price
FROM  (SELECT category, name, price,
              ROW_NUMBER() OVER (PARTITION BY category ORDER BY price DESC) AS rn
       FROM   product) t
WHERE  rn = 1;

-- 회원별 최근 주문 3건
SELECT member_id, id AS order_no, ordered_at
FROM  (SELECT member_id, id, ordered_at,
              ROW_NUMBER() OVER (PARTITION BY member_id ORDER BY ordered_at DESC) AS rn
       FROM   orders) t
WHERE  rn <= 3;
WHERE 절에 윈도우 함수를 직접 쓸 수 없습니다. WHERE ROW_NUMBER() OVER (…) = 1 은 에러입니다 — 윈도우 함수는 WHERE보다 나중에 계산되기 때문입니다(q18). 서브쿼리나 CTE로 한 번 감싼 뒤 바깥에서 거르세요. (PostgreSQL의 QUALIFY 는 지원하지 않습니다 — Snowflake·BigQuery 문법입니다)

NTILE — 몇 등분으로 나누기

-- 구매액 기준 4등분해 등급 매기기
SELECT name, 총구매액,
       NTILE(4) OVER (ORDER BY 총구매액 DESC) AS 분위
FROM   v_member_sales;
-- 분위 1 = 상위 25%
SECTION 03

LAG · LEAD · 누적합 프레임

전월 대비 증감, 누적 매출, 이동평균 — 시계열 리포트의 핵심이 전부 여기서 나옵니다.

LAG · LEAD — 앞뒤 행의 값 가져오기

SELECT 월, 매출,
       LAG(매출)  OVER (ORDER BY 월) AS 전월매출,
       매출 - LAG(매출) OVER (ORDER BY 월) AS 증감,
       ROUND( (매출 - LAG(매출) OVER (ORDER BY 월)) * 100.0
              / NULLIF(LAG(매출) OVER (ORDER BY 월), 0), 1) AS 증감률
FROM   monthly_sales
ORDER  BY 월;
월 매출 전월매출 증감 증감률 2026-01 1200000 (NULL) (NULL) (NULL) 2026-02 1500000 1200000 300000 25.0 2026-03 1350000 1500000 -150000 -10.0
-- 기본값을 주면 첫 행의 NULL 을 없앨 수 있습니다
LAG(매출, 1, 0) OVER (ORDER BY 월)     -- 1칸 앞, 없으면 0
LEAD(매출, 1, 0) OVER (ORDER BY 월)    -- 1칸 뒤

프레임 — 어디부터 어디까지 계산할지

프레임뜻
ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW 누적합 — 처음부터 현재 행까지 (ORDER BY 가 있으면 기본값)
ROWS BETWEEN 6 PRECEDING AND CURRENT ROW 7일 이동평균 — 앞 6행 + 현재 행
ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING 현재부터 끝까지 (잔여 합계)
ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING 전체 — PARTITION BY 만 쓴 것과 같습니다
-- 누적 매출
SELECT 일자, 매출,
       SUM(매출) OVER (ORDER BY 일자
                       ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS 누적매출
FROM   daily_sales;

-- 7일 이동평균 (그래프에서 노이즈를 걷어낼 때)
SELECT 일자, 매출,
       ROUND(AVG(매출) OVER (ORDER BY 일자
                             ROWS BETWEEN 6 PRECEDING AND CURRENT ROW), 0) AS 이동평균7일
FROM   daily_sales;

-- 전체 대비 비중
SELECT category, 매출,
       ROUND(매출 * 100.0 / SUM(매출) OVER (), 1) AS 비중
FROM   category_sales;
ROWS 와 RANGE 는 다릅니다. ROWS 는 물리적 행 개수로, RANGE 는 값이 같은 행을 한 덩어리로 셉니다. 같은 날짜가 여러 행 있을 때 RANGE 는 그 전부를 포함하므로 결과가 달라집니다. 헷갈리면 ROWS 를 쓰세요 — 의도가 명확합니다.
FIRST_VALUE 의 함정 — LAST_VALUE(x) OVER (ORDER BY y) 는 기본 프레임이 "처음~현재"라 마지막 값이 아니라 현재 값이 나옵니다. 진짜 마지막을 원하면 프레임을 ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING 으로 명시하세요.
SECTION 04

CTE — 쿼리를 문단으로

WITH 는 쿼리에 이름 붙인 문단을 만듭니다. 3중으로 중첩된 서브쿼리를 위에서 아래로 읽히게 펴 줍니다.

✗ 중첩 서브쿼리 — 안쪽부터 읽어야 함
SELECT * FROM (
  SELECT t.*, RANK() OVER (ORDER BY 구매액 DESC) rk
  FROM (
    SELECT m.id, m.name,
           SUM(oi.qty*oi.unit_price) 구매액
    FROM member m
    JOIN orders o ON o.member_id=m.id
    JOIN order_item oi ON oi.order_id=o.id
    GROUP BY m.id, m.name
  ) t
) x
WHERE x.rk <= 3;
✓ CTE — 위에서 아래로 읽힘
WITH 구매액 AS (
  SELECT m.id, m.name,
         SUM(oi.qty*oi.unit_price) AS 금액
  FROM   member m
  JOIN   orders o      ON o.member_id = m.id
  JOIN   order_item oi ON oi.order_id = o.id
  GROUP  BY m.id, m.name
),
순위 AS (
  SELECT *, RANK() OVER (ORDER BY 금액 DESC) AS rk
  FROM   구매액
)
SELECT * FROM 순위 WHERE rk <= 3;

CTE 를 여러 개 쓰고 서로 참조하기

WITH 유효주문 AS (
  SELECT * FROM orders WHERE status <> 'CANCELED'
),
회원별 AS (
  SELECT member_id, COUNT(*) AS 건수
  FROM   유효주문                      -- ← 앞의 CTE 를 그대로 참조
  GROUP  BY member_id
),
평균 AS (
  SELECT AVG(건수) AS 평균건수 FROM 회원별
)
SELECT m.name, r.건수, a.평균건수
FROM   회원별 r
JOIN   member m ON m.id = r.member_id
CROSS  JOIN 평균 a
WHERE  r.건수 > a.평균건수;

CTE 로 데이터를 바꾸기 (PostgreSQL)

-- 지운 행을 그대로 아카이브 테이블로 옮기기 — 한 트랜잭션에서
WITH deleted AS (
  DELETE FROM orders
  WHERE  ordered_at < '2024-01-01'
  RETURNING *
)
INSERT INTO orders_archive SELECT * FROM deleted;
이 문법은 PostgreSQL 전용입니다(WITH 안의 DML + RETURNING). Oracle·MySQL·SQLite 에서는 두 문장으로 나눠 트랜잭션으로 묶으세요.
성능 주의 — CTE 가 항상 공짜는 아닙니다. PostgreSQL은 12 버전부터 CTE를 인라인으로 펼쳐 최적화합니다(그 전에는 항상 별도 실행). WITH x AS MATERIALIZED (…) 로 강제 저장을, NOT MATERIALIZED 로 인라인을 지정할 수 있습니다. 같은 CTE를 여러 번 참조하면 DB에 따라 그만큼 반복 실행될 수 있으니 실행계획으로 확인하세요.
SECTION 05

재귀 CTE — 계층 탐색

부모 → 자식 → 손자를 끝까지 따라가는 문법입니다. 조직도·카테고리 트리·댓글 스레드·부품 전개(BOM)가 전부 이 패턴입니다.

재귀 CTE 는 층수를 미리 몰라도 됩니다 — 더 나올 자식이 없을 때 스스로 멈춥니다.
-- 예제 데이터
CREATE TABLE category (
  id        INTEGER PRIMARY KEY,
  name      VARCHAR(50),
  parent_id INTEGER REFERENCES category(id)
);
INSERT INTO category VALUES
 (1,'전체',NULL), (2,'전자제품',1), (3,'주변기기',2),
 (4,'키보드',3),  (5,'마우스',3),   (6,'의류',1);
WITH RECURSIVE tree AS (
    -- ① 시작점 (앵커) — 최상위부터
    SELECT id, name, parent_id, 1 AS depth,
           CAST(name AS VARCHAR(200)) AS path
    FROM   category
    WHERE  parent_id IS NULL

    UNION ALL

    -- ② 반복 — 이전 결과의 자식을 계속 붙인다
    SELECT c.id, c.name, c.parent_id, t.depth + 1,
           CAST(t.path || ' > ' || c.name AS VARCHAR(200))
    FROM   category c
    JOIN   tree t ON t.id = c.parent_id
)
SELECT depth, path FROM tree ORDER BY path;
depth path 1 전체 2 전체 > 의류 2 전체 > 전자제품 3 전체 > 전자제품 > 주변기기 4 전체 > 전자제품 > 주변기기 > 마우스 4 전체 > 전자제품 > 주변기기 > 키보드

구조는 항상 같습니다

①
앵커 — 출발점재귀 없이 한 번만 실행됩니다. "최상위" 또는 "내가 찾는 그 노드".
▼
UNION ALL
둘을 잇습니다UNION(중복 제거)을 쓰면 느려지고, 순환 데이터에서 결과가 달라질 수 있습니다.
▼
②
재귀 — 이전 결과를 참조CTE 이름(tree)을 자기 안에서 조인합니다. 더 붙일 게 없으면 자동으로 멈춥니다.

역방향 — 특정 노드의 조상 찾기

WITH RECURSIVE up AS (
    SELECT id, name, parent_id FROM category WHERE id = 4      -- '키보드'에서 출발
    UNION ALL
    SELECT c.id, c.name, c.parent_id
    FROM   category c
    JOIN   up u ON c.id = u.parent_id                          -- 부모를 따라 올라간다
)
SELECT * FROM up;
-- 키보드 → 주변기기 → 전자제품 → 전체
무한 루프를 조심하세요. 데이터에 순환(A의 부모가 B, B의 부모가 A)이 있으면 영원히 돕니다.
· 깊이 제한을 걸어 두세요: WHERE t.depth < 20
· PostgreSQL 14+ 는 CYCLE id SET is_cycle USING path 로 순환을 자동 감지합니다
· Oracle 은 CONNECT BY NOCYCLE (o09) 이 있습니다

SECTION 06

ROLLUP · CUBE · GROUPING SETS

소계와 총계를 한 번의 쿼리로 뽑습니다. UNION을 세 번 쓰던 보고서 쿼리가 한 줄이 됩니다. 

-- ROLLUP : 계층적 소계 (지역 → 지역+등급 → 총계)
SELECT city, grade, COUNT(*) AS 인원
FROM   member
GROUP  BY ROLLUP (city, grade);

부산SILVER1상세

-- CUBE : 모든 조합 (지역별 · 등급별 · 지역+등급 · 총계)
SELECT city, grade, COUNT(*) FROM member GROUP BY CUBE (city, grade);

-- GROUPING SETS : 원하는 조합만 콕 집어서
SELECT city, grade, COUNT(*)
FROM   member
GROUP  BY GROUPING SETS ((city), (grade), ());   -- 지역별 · 등급별 · 총계

-- NULL 이 '소계'인지 '진짜 NULL'인지 구분하기
SELECT CASE WHEN GROUPING(city) = 1 THEN '전체' ELSE city END AS 지역,
       COUNT(*) AS 인원
FROM   member
GROUP  BY ROLLUP (city);

SECTION 07

PIVOT — 행을 열로

세로로 긴 표를 가로로 펼칩니다. "월별 매출을 12개 컬럼으로" 같은 엑셀형 리포트에 씁니다.

어디서나 되는 방법 — CASE 집계

SELECT category,
       SUM(CASE WHEN 월 = '01' THEN 매출 ELSE 0 END) AS "1월",
       SUM(CASE WHEN 월 = '02' THEN 매출 ELSE 0 END) AS "2월",
       SUM(CASE WHEN 월 = '03' THEN 매출 ELSE 0 END) AS "3월",
       SUM(매출) AS 합계
FROM   monthly_category_sales
GROUP  BY category;
category 1월 2월 3월 합계 주변기기 320000 410000 380000 1110000 디스플레이 578000 289000 0 867000
이 방법이 실무 표준입니다. 네 DB 어디서나 동작하고, 이식성이 가장 좋습니다. 아래의 전용 문법은 편하지만 DB에 종속됩니다.

Oracle 전용 — PIVOT / UNPIVOT

SELECT * FROM (
    SELECT category, 월, 매출 FROM monthly_category_sales
)
PIVOT (
    SUM(매출) FOR 월 IN ('01' AS "1월", '02' AS "2월", '03' AS "3월")
);

-- 반대로 가로를 세로로
SELECT * FROM sales_wide
UNPIVOT (매출 FOR 월 IN ("1월", "2월", "3월"));

PostgreSQL — crosstab 확장

CREATE EXTENSION IF NOT EXISTS tablefunc;

SELECT * FROM crosstab(
  'SELECT category, 월, 매출 FROM monthly_category_sales ORDER BY 1,2',
  'SELECT DISTINCT 월 FROM monthly_category_sales ORDER BY 1'
) AS ct(category TEXT, m01 NUMERIC, m02 NUMERIC, m03 NUMERIC);
피벗의 근본적 한계 — SQL의 결과 컬럼은 실행 전에 개수가 정해져야 합니다. "데이터에 있는 월만큼 컬럼을 만들어 줘"는 순수 SQL로 불가능합니다. 동적 피벗이 필요하면 ① 애플리케이션에서 컬럼을 만들거나 ② SQL 문자열을 조립해 실행하거나(인젝션 주의) ③ 세로 형태로 내보내고 화면에서 피벗하세요 — ③이 가장 안전합니다.
SECTION 08

UPSERT 4종 대조

"있으면 수정, 없으면 입력"을 한 문장으로 처리합니다. DB마다 문법이 완전히 다른 대표적인 영역입니다.

PostgreSQL · SQLite
INSERT INTO stock (product_id, qty)
VALUES (10, 5)
ON CONFLICT (product_id)
DO UPDATE SET qty = stock.qty + EXCLUDED.qty;

-- 있으면 무시하고 넘어가기
INSERT INTO stock (product_id, qty)
VALUES (10, 5)
ON CONFLICT DO NOTHING;
MySQL · MariaDB
INSERT INTO stock (product_id, qty)
VALUES (10, 5)
ON DUPLICATE KEY UPDATE
    qty = qty + VALUES(qty);

-- MySQL 8.0.20+ 권장 문법
INSERT INTO stock (product_id, qty)
VALUES (10, 5) AS new
ON DUPLICATE KEY UPDATE qty = qty + new.qty;
Oracle · 표준 SQL
MERGE INTO stock s
USING (SELECT 10 AS pid, 5 AS q FROM dual) x
   ON (s.product_id = x.pid)
WHEN MATCHED THEN
    UPDATE SET s.qty = s.qty + x.q
WHEN NOT MATCHED THEN
    INSERT (product_id, qty) VALUES (x.pid, x.q);
SQLite (구버전 대안)
-- 3.24 미만에서 쓰던 방식
INSERT OR REPLACE INTO stock (product_id, qty)
VALUES (10, 5);

-- ⚠️ REPLACE 는 기존 행을 DELETE 후 INSERT 라
-- 적지 않은 컬럼이 기본값으로 날아갑니다
INSERT OR REPLACE · REPLACE INTO 를 조심하세요. 이들은 "수정"이 아니라 삭제 후 재삽입입니다. ① 지정하지 않은 컬럼이 DEFAULT 값으로 초기화되고, ② 자동 증가 PK가 새 번호로 바뀌며, ③ ON DELETE CASCADE 가 걸린 자식 행이 같이 삭제될 수 있습니다. 반드시 ON CONFLICT DO UPDATE 를 쓰세요.

UPSERT 가 성립하려면 제약이 있어야 합니다

-- 이 UNIQUE 제약이 있어야 "충돌"이라는 판단이 가능합니다
ALTER TABLE stock ADD CONSTRAINT uq_stock_product UNIQUE (product_id);

-- 여러 컬럼 조합이 기준일 때
INSERT INTO daily_stat (dt, category, cnt)
VALUES (DATE '2026-08-05', '주변기기', 1)
ON CONFLICT (dt, category) DO UPDATE SET cnt = daily_stat.cnt + 1;

왜 UPSERT를 써야 하나 — 동시성

✗ 조회 후 분기 — 경합에서 깨집니다
# 두 요청이 동시에 오면
row = db.query("SELECT … WHERE product_id=10")
if row:
    db.execute("UPDATE …")
else:
    db.execute("INSERT …")   # ← 둘 다 여기로 와서
                             #    유니크 제약 위반 or 중복 행
✓ UPSERT — DB가 원자적으로 처리
db.execute("""
  INSERT INTO stock (product_id, qty) VALUES (%s, %s)
  ON CONFLICT (product_id)
  DO UPDATE SET qty = stock.qty + EXCLUDED.qty
""", (10, 5))
# 동시에 들어와도 한 쪽은 INSERT, 한 쪽은 UPDATE
MERGE 는 표준 SQL 문법입니다. Oracle·SQL Server·PostgreSQL 15+ 가 지원합니다. 다만 동시성 보장이 DB마다 다르므로(일부 구현에서 경합 시 오류 발생 가능) 단순한 UPSERT라면 각 DB의 전용 문법이 더 안전합니다.
SECTION 09

LATERAL · APPLY

각 행마다 딸린 조회를 따로 실행하는 조인입니다. "회원별 최근 주문 3건"처럼 행마다 다른 상위 N이 필요할 때 씁니다. 

-- PostgreSQL · Oracle 12c+ : LATERAL
SELECT m.name, o.id AS order_no, o.ordered_at
FROM   member m
CROSS JOIN LATERAL (
    SELECT id, ordered_at
    FROM   orders
    WHERE  member_id = m.id           -- ← 바깥 m 을 참조할 수 있습니다
    ORDER  BY ordered_at DESC
    LIMIT  3
) o;

-- 주문이 없는 회원도 보고 싶다면
SELECT m.name, o.id, o.ordered_at
FROM   member m
LEFT JOIN LATERAL (
    SELECT id, ordered_at FROM orders
    WHERE  member_id = m.id ORDER BY ordered_at DESC LIMIT 3
) o ON TRUE;
핵심은 "바깥을 참조할 수 있다"는 점입니다. 보통 서브쿼리는 FROM 절에서 바깥 테이블의 컬럼을 볼 수 없지만, LATERAL 을 붙이면 왼쪽에서 이미 읽은 행의 값을 쓸 수 있습니다.

DB별 대응

DB문법
PostgreSQLCROSS JOIN LATERAL · LEFT JOIN LATERAL … ON TRUE (9.3+)
OracleCROSS APPLY · OUTER APPLY · LATERAL() (12c+)
MySQLLATERAL (8.0.14+) — MariaDB는 미지원
SQLite✗ — 윈도우 함수로 대체

윈도우 함수로도 됩니다 — 어느 쪽이 빠를까

-- 어디서나 되는 대안
SELECT name, order_no, ordered_at
FROM  (SELECT m.name, o.id AS order_no, o.ordered_at,
              ROW_NUMBER() OVER (PARTITION BY m.id ORDER BY o.ordered_at DESC) AS rn
       FROM   member m JOIN orders o ON o.member_id = m.id) t
WHERE  rn <= 3;
선택 기준 — 윈도우 함수 방식은 모든 주문을 읽어 정렬한 뒤 3건만 남깁니다. LATERAL 방식은 회원마다 인덱스로 3건만 찍어 옵니다. 왼쪽 행 수가 적고 오른쪽이 매우 크면 LATERAL 이 압도적으로 빠릅니다. 반대로 왼쪽이 수만 건이면 왕복이 늘어 윈도우 함수가 나을 수 있습니다 — 실행계획으로 확인하세요.
SECTION 10

JSON을 SQL로 다루기

스키마가 자주 바뀌는 데이터(설정·로그·외부 API 응답)를 컬럼 하나에 통째로 저장하고 검색합니다.

PostgreSQL — JSONB (가장 강력)
-- 값 꺼내기
SELECT data->>'name'          AS name,      -- 텍스트로
       data->'tags'->>0       AS first_tag,
       (data->>'price')::int  AS price
FROM   item;

-- 포함 검색 (GIN 인덱스 사용 가능)
SELECT * FROM item WHERE data @> '{"brand":"LG"}';

-- 경로 존재 여부
SELECT * FROM item WHERE data ? 'discount';

CREATE INDEX idx_item_data ON item USING GIN (data);
MySQL / MariaDB
SELECT JSON_EXTRACT(data, '$.name')   AS name,
       data->>'$.name'                AS name2,
       JSON_UNQUOTE(data->'$.price')  AS price
FROM   item;

SELECT * FROM item
WHERE  JSON_CONTAINS(data, '"LG"', '$.brand');

-- 인덱스는 생성 컬럼을 통해서
ALTER TABLE item
  ADD COLUMN brand VARCHAR(50)
  GENERATED ALWAYS AS (data->>'$.brand') STORED,
  ADD INDEX idx_brand (brand);
Oracle
SELECT JSON_VALUE(data, '$.name')  AS name,
       JSON_QUERY(data, '$.tags')  AS tags
FROM   item;

-- 표로 펼치기
SELECT t.* FROM item i,
  JSON_TABLE(i.data, '$'
    COLUMNS (name VARCHAR2(50) PATH '$.name',
             price NUMBER      PATH '$.price')) t;

-- 21c+ 는 JSON 전용 타입 지원
SQLite (JSON1 · 기본 내장)
SELECT json_extract(data, '$.name')  AS name,
       data ->> '$.price'            AS price
FROM   item;

-- 배열 펼치기
SELECT i.id, j.value AS tag
FROM   item i, json_each(i.data, '$.tags') j;

-- 표현식 인덱스
CREATE INDEX idx_brand
  ON item (json_extract(data, '$.brand'));
JSON 컬럼을 남용하지 마세요. 자주 조회·조건에 쓰는 값은 반드시 정식 컬럼으로 빼야 합니다. JSON 안에 넣으면 ① 인덱스 설계가 어렵고 ② 제약조건을 못 걸고 ③ 타입 검증이 없어 애플리케이션이 전부 책임지게 됩니다.
기준 — WHERE·JOIN·ORDER BY 에 쓰이면 정식 컬럼, 그냥 보관만 하면 JSON.
PostgreSQL 의 JSON vs JSONB — JSON 은 텍스트 그대로 저장(입력 순서·공백 유지, 파싱 매번), JSONB 는 이진 형태로 저장(파싱 완료, 인덱스 가능, 키 순서 미보존). 거의 항상 JSONB 를 쓰세요.

정규식 — 패턴이 복잡할 때

전문 검색 — 본문에서 단어 찾기

SECTION 12

집합 기반 사고

SQL을 잘 쓰는 사람과 못 쓰는 사람의 차이는 문법 암기량이 아니라 사고방식입니다. "한 건씩"이 아니라 "한꺼번에" 생각하는 훈련입니다.

✗ 절차형 사고 — 10만 번 왕복
members = db.query("SELECT id FROM member")
for m in members:                       # 10만 번
    orders = db.query(
        "SELECT * FROM orders WHERE member_id=%s", m.id)
    total = sum(o.amount for o in orders)
    db.execute(
        "UPDATE member SET total=%s WHERE id=%s",
        total, m.id)                    # 또 10만 번

# 왕복 20만 번 · 수십 분
✓ 집합 사고 — 한 번
UPDATE member m
SET    total = t.합계
FROM  (SELECT member_id, SUM(amount) AS 합계
       FROM   orders GROUP BY member_id) t
WHERE  t.member_id = m.id;

-- 왕복 1번 · 수 초

사고 전환 훈련 — 이렇게 바꿔 생각합니다

절차형으로 생각하면집합으로 바꾸면
"각 회원을 돌면서 주문을 세고…"GROUP BY member_id
"이전 값을 변수에 저장해 두었다가 비교"LAG() OVER (ORDER BY …)
"카운터를 1씩 늘려 등수를 매김"ROW_NUMBER() OVER (…)
"있는지 확인하고 없으면 INSERT"INSERT … ON CONFLICT
"부모를 찾아 while 로 올라가며 반복"WITH RECURSIVE
"리스트를 만들어 IN 절에 넣기"서브쿼리 또는 EXISTS 로 한 번에
"조건별로 세 번 쿼리해 합치기"SUM(CASE WHEN … END) 조건부 집계
다만 무조건 SQL로 몰아넣는 것도 답이 아닙니다. ① 비즈니스 규칙이 자주 바뀌는 로직, ② 외부 시스템 호출이 필요한 흐름, ③ 테스트가 중요한 복잡한 계산은 애플리케이션 코드가 낫습니다. 기준 — 데이터를 줄이고 모으는 일은 SQL, 판단하고 흐름을 만드는 일은 코드.

한 번에 하되, 너무 크면 잘라서

"한 번에"가 항상 옳은 것도 아닙니다. 수천만 건을 단일 트랜잭션으로 바꾸면 락이 오래 걸리고 로그가 폭증해 오히려 장애가 됩니다. 집합으로 생각하되 청크로 나눠 실행하는 것이 실무의 정답입니다 → 다음 섹션.

SECTION 13

대량 배치 UPDATE · DELETE

수천만 건을 한 문장으로 바꾸면 락 대기·복제 지연·로그 폭증이 동시에 터집니다. 운영 중에는 반드시 잘라서 돌립니다.

한 방에 치면 벌어지는 일
① 대상 행 전체에 락이 걸려 서비스 쿼리가 줄줄이 대기
② UNDO/롤백 세그먼트·WAL 이 디스크를 가득 채움
③ 복제본이 따라오지 못해 복제 지연이 수십 분
④ 중간에 실패하면 전부 롤백 — 몇 시간이 날아갑니다

기본형 — PK 범위로 자르기

-- PostgreSQL / MySQL : 반복 실행 (영향 행이 0이 될 때까지)
UPDATE orders
SET    status = 'ARCHIVED'
WHERE  id IN (
    SELECT id FROM orders
    WHERE  status = 'PAID' AND ordered_at < '2024-01-01'
    ORDER  BY id
    LIMIT  10000
);

-- 삭제도 같은 방식
DELETE FROM log_table
WHERE  id IN (SELECT id FROM log_table
              WHERE created_at < '2025-01-01'
              ORDER BY id LIMIT 10000);
-- PostgreSQL : 진행 상황을 돌려받으며 청크 처리
WITH target AS (
    SELECT id FROM orders
    WHERE  ordered_at < '2024-01-01' AND status = 'PAID'
    ORDER  BY id
    LIMIT  10000
    FOR UPDATE SKIP LOCKED          -- 다른 세션이 잡은 행은 건너뜀
)
UPDATE orders o
SET    status = 'ARCHIVED'
FROM   target t
WHERE  o.id = t.id
RETURNING o.id;

셸 스크립트로 반복 돌리기

#!/usr/bin/env bash
# 10000건씩 지우고 0.5초 쉬기 — 복제 지연을 만들지 않는 속도로
while : ; do
  n=$(psql -qtAX -d shop -c "
        DELETE FROM log_table
        WHERE id IN (SELECT id FROM log_table
                     WHERE created_at < '2025-01-01'
                     ORDER BY id LIMIT 10000)
        RETURNING 1" | wc -l)
  echo "$(date +%T)  삭제 $n 건"
  [ "$n" -eq 0 ] && break
  sleep 0.5                      # ← 이 쉬는 시간이 핵심입니다
done

더 좋은 방법 — 아예 다시 만들기

-- 전체의 80% 이상을 지워야 한다면 DELETE 보다 이쪽이 훨씬 빠릅니다
CREATE TABLE orders_new (LIKE orders INCLUDING ALL);   -- PostgreSQL
INSERT INTO orders_new SELECT * FROM orders WHERE ordered_at >= '2025-01-01';

BEGIN;
ALTER TABLE orders     RENAME TO orders_old;
ALTER TABLE orders_new RENAME TO orders;
COMMIT;

-- 확인 후
DROP TABLE orders_old;

쿼리 안티패턴 12선

실제로 장애를 만든 패턴들입니다. 코드 리뷰에서 이것만 걸러도 사고가 크게 줄어듭니다.

#안티패턴왜 나쁜가 · 어떻게

3앞에 % 붙은 LIKE 풀스캔 확정 → 전문 검색 또는 역방향 인덱스
4암묵적 형변환
WHERE varchar_col = 123		 DB가 컬럼을 형변환하며 인덱스가 죽습니다 → 타입을 맞춰 비교
5NOT IN (서브쿼리) NULL 하나에 결과가 0행 → NOT EXISTS
6깊은 OFFSET 페이징 뒤 페이지가 기하급수적으로 느려짐 → 커서 페이징 (⚙️ t09)
7코드에서 반복 쿼리 (N+1) 목록 100건에 쿼리 101번 → 조인 · IN 절 · 배치 조회 (🔌 a08)

특히 자주 나오는 세 가지 실제 수정 예

✗ 날짜 함수로 인덱스 사망
SELECT * FROM orders
WHERE  DATE_FORMAT(ordered_at,'%Y-%m') = '2026-01';
✓ 범위로
SELECT * FROM orders
WHERE  ordered_at >= '2026-01-01'
  AND  ordered_at <  '2026-02-01';
✗ 존재 확인에 COUNT
SELECT COUNT(*) FROM orders WHERE member_id = 1;
-- 결과가 0보다 크면 주문이 있다고 판단
✓ EXISTS — 첫 행에서 멈춤
SELECT EXISTS (
  SELECT 1 FROM orders WHERE member_id = 1
);
✗ DISTINCT 로 중복 덮기
SELECT DISTINCT m.id, m.name
FROM   member m
JOIN   orders o ON o.member_id = m.id;
-- 주문 수만큼 중복돼서 DISTINCT 를 붙임
✓ EXISTS 로 의도를 명확히
SELECT m.id, m.name
FROM   member m
WHERE  EXISTS (SELECT 1 FROM orders o
               WHERE o.member_id = m.id);
