C++ 에는 std::stack·std::queue 가 있지만 이것들은 컨테이너가 아니라 어댑터입니다. 안에 든 진짜 저장소(vector·deque)를 고르는 것이 성능을 정하고, pop() 이 값을 돌려주지 않는 이유, 빈 상태에서 top() 이 미정의 동작인 이유를 알아야 사고가 안 납니다. 이 탭은 2026년 현업 기준으로 무엇을 어떤 타입으로 쓰는지, std::deque 가 메모리에 어떻게 놓이는지, 표준에 없는 스레드 안전 큐를 어떻게 만들고 무엇을 가져다 쓰는지까지 다룹니다. 끝나면 BFS·단조 덱·레이트 리미터·undo/redo 를 C++ 관용구로 바로 짤 수 있습니다.

SQ 01

한 장 지도 — C++ 에서 무엇으로 쓰나

자바는 Stack 을 버리고 ArrayDeque 로 갔고, 파이썬은 list 와 deque 를 씁니다. C++ 의 답은 조금 다릅니다 — 스택은 대개 std::vector 그 자체, 큐는 std::queue(안은 deque), 그리고 스레드 사이 큐는 표준에 아예 없습니다. 먼저 지도를 펴고 시작합니다.

C++ 의 기본값은 스택 = vector, 큐 = std::queue(deque). 동시성 큐는 직접 만들거나 가져옵니다.

#include <deque>
#include <functional>
#include <print>
#include <queue>
#include <vector>

int main() {
    // ① 스택 — vector 의 뒤쪽만 씁니다
    std::vector<int> st;
    for (int x : {1, 2, 3}) st.push_back(x);
    std::println("stack top={} size={}", st.back(), st.size());
    st.pop_back();
    std::println("after pop {}", st);

    // ② 큐 — std::queue (안은 deque)
    std::queue<int> q;
    for (int x : {1, 2, 3}) q.push(x);
    std::println("queue front={} back={}", q.front(), q.back());
    q.pop();
    std::println("after pop front={}", q.front());

    // ③ 덱 — 양쪽 끝
    std::deque<int> d{2, 3};
    d.push_front(1);
    d.push_back(4);
    std::println("deque {}", d);

    // ④ 우선순위 큐 — 기본은 '최대' 힙
    std::priority_queue<int> maxq;
    std::priority_queue<int, std::vector<int>, std::greater<>> minq;
    for (int x : {5, 1, 8, 3}) { maxq.push(x); minq.push(x); }
    std::println("max top={} min top={}", maxq.top(), minq.top());
}

한 단계 더 — 어댑터란 무엇인가 🟡 중급

std::stack 의 진짜 정의 — 거의 이것뿐입니다

// 표준 라이브러리 구현을 줄여 옮긴 것 (libstdc++ · libc++ · MSVC 모두 이 모양)
template <class T, class Container = std::deque<T>>
class stack {
protected:
    Container c;                 // ★ 진짜 저장소. protected 라 상속하면 보입니다
public:
    bool empty() const           { return c.empty(); }
    size_t size() const          { return c.size(); }
    T&   top()                   { return c.back(); }      // 빈 상태면 UB
    void push(const T& v)        { c.push_back(v); }
    void push(T&& v)             { c.push_back(std::move(v)); }
    template <class... A>
    decltype(auto) emplace(A&&... a) { return c.emplace_back(std::forward<A>(a)...); }
    void pop()                   { c.pop_back(); }        // ★ 반환값 없음
};
// 반복자도, clear() 도, operator[] 도 없습니다 — 일부러 '막아 둔' 것입니다
어댑터 = 기능을 빼는 래퍼 — std::stack 은 성능을 더해 주지 않습니다. “이 변수는 뒤에서만 넣고 뺀다”는 약속을 타입으로 강제하는 것이 존재 이유입니다. 팀 코드에서 의도를 드러내고 싶으면 쓰고, 순회·디버그 출력이 필요하면 vector 를 그대로 씁니다.

SQ 02

스택 — pop() 은 왜 값을 안 돌려주나

자바·파이썬에서 온 사람이 C++ 스택에서 처음 막히는 곳이 auto x = s.pop(); 입니다. 컴파일 에러가 납니다. 이것은 실수가 아니라 예외 안전성 때문에 일부러 고른 설계입니다. 그 이유를 알면 C++ 컨테이너 API 전체가 읽힙니다.

읽기(top)와 지우기(pop)를 나눈 것이 요점입니다 — 실패해도 데이터가 사라지지 않습니다.

std::stack 전체 API — 실행해서 확인

#include <print>
#include <stack>
#include <string>
#include <vector>

int main() {
    std::stack<std::string, std::vector<std::string>> s;   // 하부를 vector 로 ★

    s.push("a");                    // 복사(여기선 임시 → 이동)
    s.emplace(3, 'b');              // string(3, 'b') == "bbb" 를 제자리 생성
    auto& t = s.emplace("c");       // C++17 부터 emplace 는 참조를 돌려줍니다
    t += "!";                       // 그 참조로 바로 수정

    std::println("size={} top={}", s.size(), s.top());

    while (!s.empty()) {            // ★ 비었는지 먼저 — 빈 top() 은 UB
        std::string v = std::move(s.top());   // ① 꺼내 오고
        s.pop();                              // ② 지운다
        std::print("{} ", v);
    }
    std::println("| empty={}", s.empty());
}
// 출력
// size=3 top=c!
// c! bbb a | empty=true
std::move(s.top()) 다음 pop() — 이것이 C++ 의 “pop 하면서 값 받기” 관용구입니다. 옮긴 뒤 껍데기만 남은 원소를 바로 지우므로 복사 비용이 0이고 안전합니다.

vector 로 쓰는 스택 — 현업에서 제일 흔한 모양

#include <optional>
#include <print>
#include <ranges>
#include <vector>

// "값을 돌려주는 pop" 이 꼭 필요하면 optional 로 직접 — 빈 경우도 표현됩니다
template <class T>
std::optional<T> pop_value(std::vector<T>& st) {
    if (st.empty()) return std::nullopt;
    std::optional<T> v{std::move(st.back())};   // 여기서 던지면 원본은 남음
    st.pop_back();                               // 성공한 뒤에만 제거
    return v;
}

int main() {
    std::vector<int> st;
    st.reserve(64);                     // 깊이를 알면 미리 — 재할당 0회
    for (int i = 1; i <= 4; ++i) st.push_back(i * 10);

    // 순회 순서: vector 는 바닥 → 꼭대기
    std::println("bottom→top {}", st);
    std::println("top→bottom {}", st | std::views::reverse);

    while (auto v = pop_value(st)) std::print("{} ", *v);
    std::println("| left={}", st.size());
    std::println("{}", pop_value(st).has_value());
}
// 출력
// bottom→top [10, 20, 30, 40]
// top→bottom [40, 30, 20, 10]
// 40 30 20 10 | left=0
// false
자바에서 온 사람에게 — 자바 Stack 은 순회하면 바닥부터, ArrayDeque 는 꼭대기부터 나와서 헷갈렸습니다. C++ 은 단순합니다 — std::stack 은 순회 자체가 안 되고, vector 는 늘 바닥 → 꼭대기입니다. 꼭대기부터 보려면 std::views::reverse.

실무에선 이렇게 — 재귀를 명시적 스택으로 🔴 실무

깊은 트리 순회 — 스택 오버플로를 피합니다

#include <memory>
#include <print>
#include <vector>

struct Node {
    int val;
    std::vector<std::unique_ptr<Node>> kids;
    explicit Node(int v) : val(v) {}

    // ★ 소멸도 명시적 스택으로 — 기본 소멸자는 자식을 '재귀로' 지웁니다
    ~Node() {
        std::vector<std::unique_ptr<Node>> st;
        for (auto& k : kids) if (k) st.push_back(std::move(k));
        while (!st.empty()) {
            std::unique_ptr<Node> n = std::move(st.back());
            st.pop_back();
            for (auto& k : n->kids) if (k) st.push_back(std::move(k));
        }   // n 은 자식을 다 떼인 채(null 만 남음) 소멸 → 재귀 깊이 1
    }
};

// 재귀 버전은 깊이 10만이면 기본 스레드 스택(1~8MB)을 넘길 수 있습니다
long long sum_iterative(const Node& root) {
    long long total = 0;                       // long 은 Windows 에서 32비트 — 넘칩니다
    std::vector<const Node*> st{&root};      // 힙에 있는 '내 스택'
    while (!st.empty()) {
        const Node* n = st.back();
        st.pop_back();
        total += n->val;
        for (const auto& k : n->kids) st.push_back(k.get());
    }
    return total;
}

int main() {
    Node root(1);
    Node* cur = &root;
    for (int i = 2; i <= 100'000; ++i) {          // 깊이 10만짜리 일자 트리
        cur->kids.push_back(std::make_unique<Node>(i));
        cur = cur->kids.back().get();
    }
    std::println("{}", sum_iterative(root));
}   // root 소멸 — ~Node 가 반복으로 풀어서 스택 오버플로 없음
// 출력
// 5000050000
순회만 고치면 반쪽입니다 — ~Node 를 직접 쓰지 않으면 root 가 사라질 때 unique_ptr 소멸자가 재귀로 10만 단계를 내려가 Windows(1MB 스택)에서 바로 죽습니다. 깊은 연결 구조(긴 리스트·일자 트리)는 소멸도 명시적 스택으로 풉니다.

SQ 03

큐(FIFO) — std::queue 와 하부 컨테이너

std::queue 는 뒤에 넣고 앞에서 뺍니다. 그래서 하부 컨테이너는 pop_front 가 있어야 하고, vector 는 끼울 수 없습니다. 자바의 offer/poll, 파이썬의 append/popleft 에 대응하는 것과, C++ 에는 “실패하면 예외” 버전이 없다는 점을 정리합니다.

하부 컨테이너 규칙은 “필요한 멤버 함수가 있는가” 하나입니다 — 큐에 vector 는 못 끼웁니다.

std::queue API — 실행해서 확인

#include <print>
#include <queue>
#include <string>
#include <vector>

struct Job {
    int id;
    std::string name;
};

int main() {
    std::queue<Job> q;
    q.push({1, "resize"});
    q.emplace(2, "upload");           // Job{2,"upload"} 제자리 생성 (C++20 집합체 () 초기화)
    q.push(Job{3, "notify"});

    std::println("size={} front={} back={}", q.size(), q.front().id, q.back().id);

    while (!q.empty()) {
        Job j = std::move(q.front());  // ① 앞에서 꺼내 오고
        q.pop();                       // ② 지운다
        std::println("run #{} {}", j.id, j.name);
    }

    // C++23: 범위에서 바로 만들기 · 한꺼번에 넣기
    std::vector<int> src{7, 8, 9};
    std::queue<int> q2(std::from_range, src);
    q2.push_range(std::vector{10, 11});
    std::println("q2 front={} back={} size={}", q2.front(), q2.back(), q2.size());
}
// 출력
// size=3 front=1 back=3
// run #1 resize
// run #2 upload
// run #3 notify
// q2 front=7 back=11 size=5

SQ 04

덱 — std::deque 는 원형 배열이 아닙니다

자바 ArrayDeque·러스트 VecDeque 는 원형 배열, 파이썬 deque 는 블록의 이중 연결 리스트입니다. C++ std::deque 는 다릅니다 — 고정 크기 블록 여러 개 + 그 블록을 가리키는 포인터 맵입니다. 이 구조 때문에 양 끝 삽입에서 원소 주소가 절대 안 바뀌고, 대신 블록 크기가 구현마다 극단적으로 다릅니다.

deque 는 “짧은 배열들의 배열”입니다 — 양 끝이 O(1) 인 대가로 인덱스 접근이 간접 2번입니다.

주소가 안 바뀐다 — deque 만의 성질을 실행해서 확인

#include <deque>

    // ❌ it 는 무효 — 양 끝 삽입이라도 '반복자'는 전부 무효 (*it 는 UB)

    int* q = &v[1];

RingBuffer<T, N> — std::array 기반 전체 구현

#include <array>
#include <cstddef>
#include <optional>
#include <print>
#include <utility>

template <class T, std::size_t N>
class RingBuffer {
    static_assert(N > 0 && (N & (N - 1)) == 0, "N 은 2의 거듭제곱");
    std::array<T, N> buf_{};               // 힙 할당 0 — 객체 안에 통째로
    std::size_t head_ = 0, tail_ = 0;     // 계속 증가하는 카운터
    static constexpr std::size_t slot(std::size_t i) { return i & (N - 1); }
public:
    bool empty() const noexcept { return head_ == tail_; }
    bool full()  const noexcept { return tail_ - head_ == N; }
    std::size_t size() const noexcept { return tail_ - head_; }

    bool try_push(T v) {                   // 가득 차면 거절
        if (full()) return false;
        buf_[slot(tail_++)] = std::move(v);
        return true;
    }
    void push_overwrite(T v) {             // 가득 차면 가장 오래된 것을 밀어냄
        if (full()) ++head_;
        buf_[slot(tail_++)] = std::move(v);
    }
    std::optional<T> try_pop() {
        if (empty()) return std::nullopt;
        return std::move(buf_[slot(head_++)]);
    }
    T& operator[](std::size_t i) { return buf_[slot(head_ + i)]; }  // 0 = 가장 오래된 것
};

int main() {
    RingBuffer<int, 4> r;
    for (int i = 1; i <= 6; ++i) r.push_overwrite(i);   // 1, 2 는 밀려남
    for (std::size_t i = 0; i < r.size(); ++i) std::print("{} ", r[i]);
    std::println("| full={}", r.full());
    std::println("try_push(7)={}", r.try_push(7));
    auto v = r.try_pop();
    std::println("pop={} size={}", *v, r.size());
}
// 출력
// 3 4 5 6 | full=true
// try_push(7)=false
// pop=3 size=3

우선순위 큐 — less 를 넣었는데 최대 힙인 이유

std::priority_queue 는 기본이 최대 힙입니다. 자바·파이썬·C# 이 전부 최소 힙이라 옮겨 올 때 가장 많이 틀립니다. 비교자를 읽는 법, 여러 키로 정렬하기, 요소를 고칠 수 없다는 제약과 지연 삭제, 그리고 std::ranges::push_heap 으로 vector 를 직접 힙으로 다루는 법까지 갑니다.

힙은 “부모가 자식보다 크다”만 지킵니다 — 그래서 정렬도 아니고 FIFO 도 아닙니다.

여러 키 + 같은 우선순위는 먼저 넣은 것부터

#include <print>
#include <queue>
#include <string>
#include <tuple>
#include <vector>

struct Task {
    std::string name;
    int priority;      // 클수록 먼저
    int deadline;      // 작을수록 먼저
    unsigned seq;      // 넣은 순서 — 동점이면 먼저 넣은 것 먼저 ★
};

// 비교자는 "a 가 b 보다 '뒤에' 나와야 하면 true"
struct Later {
    bool operator()(const Task& a, const Task& b) const {
        return std::tuple(-a.priority, a.deadline, a.seq)
             > std::tuple(-b.priority, b.deadline, b.seq);
    }
};

int main() {
    std::priority_queue<Task, std::vector<Task>, Later> pq;
    unsigned seq = 0;
    pq.push({"backup", 1, 50, seq++});
    pq.push({"alert",  3, 10, seq++});
    pq.push({"email",  2, 30, seq++});
    pq.push({"report", 2, 20, seq++});
    pq.push({"ping",   3, 10, seq++});   // alert 와 완전 동점 → seq 로 가름

    while (!pq.empty()) {
        const Task& t = pq.top();         // 참조는 pop 전까지만 유효
        std::print("{}({}/{}) ", t.name, t.priority, t.deadline);
        pq.pop();
    }
    std::println("");
}
// 출력
// alert(3/10) ping(3/10) report(2/20) email(2/30) backup(1/50)
seq 를 빼면 alert 와 ping 의 순서는 정해져 있지 않습니다 — 힙은 안정 정렬이 아닙니다. “같은 우선순위면 들어온 순서대로”가 요구사항이면 단조 증가 번호를 마지막 키로 넣는 것이 표준 해법입니다.

외우는 법 — 비교자는 sort 와 같은 “작다” 관계이고, 힙은 그 순서의 마지막(가장 ‘큰’ 것)을 꼭대기에 둡니다. 그래서 less → 큰 것이 위, greater → 작은 것이 위입니다.

실무에선 이렇게 — 요소를 못 고친다 → 지연 삭제 🔴 실무

다익스트라 — decrease-key 대신 하나 더 넣기

#include <functional>
#include <limits>
#include <print>
#include <queue>
#include <utility>
#include <vector>

int main() {
    using P = std::pair<int, int>;              // (거리, 정점) — 거리 먼저 비교
    std::vector<std::vector<P>> adj(5);         // adj[u] = {(v, w), ...}
    auto edge = [&](int u, int v, int w) { adj[u].push_back({v, w}); };
    edge(0, 1, 4); edge(0, 2, 1); edge(2, 1, 2);
    edge(1, 3, 1); edge(2, 3, 5); edge(3, 4, 3);

    const int INF = std::numeric_limits<int>::max();
    std::vector<int> dist(5, INF);
    std::priority_queue<P, std::vector<P>, std::greater<>> pq;   // 최소 힙
    dist[0] = 0;
    pq.emplace(0, 0);
    int stale = 0;

    while (!pq.empty()) {
        auto [d, u] = pq.top();                 // 값으로 복사한 뒤
        pq.pop();                               // 지운다
        if (d > dist[u]) { ++stale; continue; } // ★ 낡은 항목 — 건너뜀
        for (auto [v, w] : adj[u])
            if (d + w < dist[v]) {
                dist[v] = d + w;
                pq.emplace(dist[v], v);         // 고치지 않고 새로 넣음
            }
    }
    std::println("dist={} stale={}", dist, stale);
}
// 출력
// dist=[0, 3, 1, 4, 7] stale=2
왜 고칠 수 없나 — top() 은 const& 이고 중간 원소에는 접근 자체가 안 됩니다. 값을 바꾸면 힙 조건이 깨지기 때문입니다. 낡은 항목을 남겨 두고 꺼낼 때 버리는 것(지연 삭제)이 다익스트라·타이머·작업 취소의 정석이고, 큐 크기는 최대 간선 수까지 늘 수 있습니다.

#include <algorithm>
#include <functional>
#include <print>
#include <string>
#include <vector>

struct Player { std::string name; int score; };

int main() {
    std::vector<Player> all{{"ann", 50}, {"bob", 10}, {"cid", 90}, {"dan", 30},
                            {"eve", 70}, {"fay", 80}, {"gus", 20}};
    const std::size_t k = 3;
    std::vector<Player> heap;            // 크기 k 의 '최소' 힙 — 꼭대기가 현재 k 등
    heap.reserve(k + 1);

    for (const auto& p : all) {
        heap.push_back(p);
        std::ranges::push_heap(heap, std::greater{}, &Player::score);   // 투영 ★
        if (heap.size() > k) {
            std::ranges::pop_heap(heap, std::greater{}, &Player::score); // 최솟값을 끝으로
            heap.pop_back();                                           // 버림
        }
    }
    std::ranges::sort_heap(heap, std::greater{}, &Player::score);      // 점수 내림차순
    for (const auto& p : heap) std::print("{}:{} ", p.name, p.score);
    std::println("");
}
// 출력
// cid:90 fay:80 eve:70

스레드 안전 큐 — 표준에 없으니 제대로 만듭니다

자바에는 BlockingQueue, 파이썬에는 queue.Queue, C# 에는 Channel 이 있습니다. C++ 표준 라이브러리에는 동시성 큐가 하나도 없습니다(표준화 제안은 오래 논의 중입니다). 그래서 mutex + condition_variable 블로킹 큐를 정확하게 짜는 법 — 가짜 깨어남, 배압, jthread 로 종료 — 을 익히고, 성능이 필요할 때 무엇을 가져다 쓰는지 정리합니다.

블로킹 큐의 전부는 조건 두 개(not_empty · not_full)와 닫힘 플래그입니다.

BlockingQueue<T> — 유한 용량 · 닫기 · stop_token

#include <condition_variable>
#include <cstddef>
#include <deque>
#include <mutex>
#include <optional>
#include <stop_token>

template <class T>
class BlockingQueue {
public:
    explicit BlockingQueue(std::size_t cap) : cap_(cap) {}

    // 가득이면 자리가 날 때까지 기다림(배압). 닫혔거나 중지 요청이면 false
    bool push(T v, std::stop_token st = {}) {
        std::unique_lock lk(m_);
        not_full_.wait(lk, st, [&] { return closed_ || q_.size() < cap_; });
        if (closed_ || q_.size() >= cap_) return false;   // 닫힘 또는 중지
        q_.push_back(std::move(v));
        lk.unlock();                 // 풀고 깨우면 깬 쪽이 바로 락을 잡습니다
        not_empty_.notify_one();
        return true;
    }

    // 비었으면 기다림. 닫히고 다 비었거나 중지 요청이면 nullopt
    std::optional<T> pop(std::stop_token st = {}) {
        std::unique_lock lk(m_);
        not_empty_.wait(lk, st, [&] { return closed_ || !q_.empty(); });
        if (q_.empty()) return std::nullopt;
        T v = std::move(q_.front());
        q_.pop_front();
        lk.unlock();
        not_full_.notify_one();
        return v;
    }

    bool try_push(T v) {             // 기다리지 않는 버전
        { std::lock_guard lk(m_);
          if (closed_ || q_.size() >= cap_) return false;
          q_.push_back(std::move(v)); }
        not_empty_.notify_one();
        return true;
    }

    void close() {                   // 더는 안 넣음 — 소비자는 남은 걸 비우고 끝
        { std::lock_guard lk(m_); closed_ = true; }
        not_empty_.notify_all();
        not_full_.notify_all();
    }

private:
    std::mutex m_;
    std::condition_variable_any not_empty_, not_full_;   // stop_token 을 받는 쪽
    std::deque<T> q_;
    std::size_t cap_;
    bool closed_ = false;
};

생산자 1 · 소비자 2 — 실행해서 확인

#include <atomic>
#include <print>
#include <thread>
#include <vector>
// + 위의 BlockingQueue

int main() {
    BlockingQueue<int> q(4);                 // 최대 4개 — 넘치면 생산자가 기다림
    std::atomic<long long> sum{0};
    std::atomic<int> count{0};

    std::vector<std::jthread> consumers;
    for (int i = 0; i < 2; ++i)
        consumers.emplace_back([&](std::stop_token st) {   // jthread 가 토큰을 넘겨 줌
            while (auto v = q.pop(st)) { sum += *v; ++count; }
        });

    std::jthread producer([&] {
        for (int i = 1; i <= 100; ++i) q.push(i);
        q.close();                           // 다 넣었으면 닫기 — 소비자 종료 신호
    });

    producer.join();                         // 넣기 + 닫기 끝
    consumers.clear();                       // 각 jthread: request_stop() → join()
    std::println("count={} sum={}", count.load(), sum.load());
}
// 출력 (어느 소비자가 무엇을 꺼냈는지는 매번 다르지만 합계는 같습니다)

request_stop() 이 와도 남은 원소는 비워집니다 — 술어를 받는 wait(lk, st, pred) 는 중지 요청이 있어도 마지막에 술어를 한 번 더 보고 돌려주므로, 큐에 원소가 있으면 꺼냅니다. “즉시 버리고 끝”이 필요하면 pop 에서 st.stop_requested() 를 먼저 보세요.

비동기 · 채널 — 표준 채널은 아직 없습니다

코틀린 Channel, C# System.Threading.Channels, 파이썬 asyncio.Queue, 러스트 tokio::sync::mpsc 에 해당하는 것이 C++ 표준에는 없습니다. C++20 이 준 것은 코루틴 문법뿐이고 스케줄러·채널은 라이브러리 몫입니다. 현업에서 쓰는 Asio 채널과, 의존성 없이 만드는 단일 스레드 작업 큐를 봅니다.

// 최근 Boost · -std=c++20 이상
#include <boost/asio.hpp>
#include <boost/asio/experimental/channel.hpp>
#include <print>

namespace asio = boost::asio;
using Chan = asio::experimental::channel<void(boost::system::error_code, int)>;

asio::awaitable<void> producer(Chan& ch) {
    for (int i = 1; i <= 3; ++i)          // 버퍼가 차면 여기서 '중단' (스레드는 안 멈춤)
        co_await ch.async_send(boost::system::error_code{}, i, asio::use_awaitable);
    ch.close();                           // 받는 쪽에 '끝' 을 알림
}

asio::awaitable<void> consumer(Chan& ch) {
    for (;;) {
        auto [ec, v] = co_await ch.async_receive(asio::as_tuple(asio::use_awaitable));
        if (ec) { std::println("closed"); co_return; }   // 닫히고 비었음
        std::println("got {}", v);
    }
}

int main() {
    asio::io_context io;                  // 이벤트 루프 — 스레드 하나
    Chan ch(io, 2);                       // 버퍼 2칸 (0 이면 랑데부)
    asio::co_spawn(io, producer(ch), asio::detached);
    asio::co_spawn(io, consumer(ch), asio::detached);
    io.run();
}
// 출력
// got 1
// got 2
// got 3
// closed

#include <deque>
#include <functional>
#include <memory>
#include <print>

class EventLoop {
    std::deque<std::move_only_function<void()>> jobs_;   // C++23
public:
    void post(std::move_only_function<void()> f) { jobs_.push_back(std::move(f)); }
    void run() {
        while (!jobs_.empty()) {
            auto job = std::move(jobs_.front());   // ① 꺼내 오고
            jobs_.pop_front();                     // ② 지운 뒤
            job();                                 // ③ 실행 — 안에서 post 해도 안전
        }
    }
};

int main() {
    EventLoop loop;
    auto data = std::make_unique<int>(42);
    loop.post([&loop, p = std::move(data)] {      // unique_ptr 캡처 → std::function 불가
        std::println("A {}", *p);
        loop.post([] { std::println("C (A 가 넣음)"); });
    });
    loop.post([] { std::println("B"); });
    loop.run();
}
// 출력
// A 42
// B
// C (A 가 넣음)
실행 전에 꺼내 두는 것이 요점입니다 — jobs_.front()() 로 바로 부르면, 작업이 실행 도중 큐를 비우거나(clear) run() 을 다시 부르는 순간 실행 중인 자기 자신이 소멸합니다. 꺼내 둔 지역 변수는 큐가 어떻게 바뀌든 끝까지 살아 있습니다. std::move_only_function C++23 이 없는 환경이면 std::function 으로 바꾸고 캡처를 복사 가능하게 합니다.

알고리즘 패턴 — C++ 관용구로 다섯 개

괄호 검사 · 단조 스택 · BFS · 슬라이딩 윈도우 최댓값 · 0-1 BFS 는 코딩 테스트와 실무 양쪽에서 반복해 나옵니다. 이론(CS 가이드 s03·s09)은 넘어가고, 여기서는 C++ 로 쓸 때의 타입 선택과 관용구 — vector 스택, 인덱스를 넣는 덱, 구조적 바인딩, 넣을 때 방문 표시 — 에 집중합니다. 전부 복붙해서 돕니다.

단조 덱 — 뒤에서는 쓸모없어진 값을, 앞에서는 만료된 값을 버리면 앞이 곧 답입니다.

① 괄호 검사 — vector<char> 스택

#include <print>
#include <string_view>
#include <vector>

bool balanced(std::string_view s) {
    std::vector<char> st;
    for (char c : s) {
        switch (c) {
            case '(': st.push_back(')'); break;   // ★ 닫는 짝을 미리 넣어 두면
            case '[': st.push_back(']'); break;   //   비교가 한 줄로 끝남
            case '{': st.push_back('}'); break;
            case ')': case ']': case '}':
                if (st.empty() || st.back() != c) return false;  // 빈 확인 먼저
                st.pop_back();
                break;
            default: break;                       // 다른 글자는 무시
        }
    }
    return st.empty();                            // 안 닫힌 게 남으면 실패
}

int main() {
    for (std::string_view s : {"([]{})", "([)]", "((", "a[b(c)]d", ")("})
        std::println("{:10} {}", s, balanced(s));
}
// 출력
// ([]{})     true
// ([)]       false
// ((         false
// a[b(c)]d   true
// )(         false

② 단조 스택 — 다음 큰 수

#include <print>
#include <vector>

// res[i] = i 오른쪽에서 처음 나오는 a[i] 보다 큰 값 (없으면 -1)
std::vector<int> next_greater(const std::vector<int>& a) {
    std::vector<int> res(a.size(), -1);
    std::vector<std::size_t> st;          // 답을 아직 못 찾은 '인덱스' — 값은 내림차순
    st.reserve(a.size());
    for (std::size_t i = 0; i < a.size(); ++i) {
        while (!st.empty() && a[st.back()] < a[i]) {
            res[st.back()] = a[i];        // a[i] 가 그들의 '다음 큰 수'
            st.pop_back();
        }
        st.push_back(i);
    }
    return res;                           // 남은 인덱스는 -1 그대로
}

int main() {
    std::println("{}", next_greater({2, 1, 2, 4, 3}));
}
// 출력
// [4, 2, 4, -1, -1]
스택에 값이 아니라 인덱스를 넣는 것이 C++ 관용구입니다 — 결과를 그 자리에 써야 하고, 값은 a[i] 로 언제든 볼 수 있습니다. 주식 스팬·히스토그램 최대 직사각형·빗물 받기가 전부 이 틀입니다.

③ 격자 BFS — std::queue + 넣을 때 방문 표시

#include <array>
#include <print>
#include <queue>
#include <string>
#include <utility>
#include <vector>

int main() {
    const std::vector<std::string> g{"S..#", "##.#", "....", "#.#G"};
    const int R = 4, C = 4;
    std::vector dist(R, std::vector<int>(C, -1));        // -1 = 아직 안 감
    constexpr std::array<std::pair<int, int>, 4> dirs{{{1, 0}, {-1, 0}, {0, 1}, {0, -1}}};

    std::queue<std::pair<int, int>> q;
    dist[0][0] = 0;
    q.push({0, 0});
    while (!q.empty()) {
        auto [r, c] = q.front();                         // 복사해 두고
        q.pop();                                         // 지운다
        for (auto [dr, dc] : dirs) {
            int nr = r + dr, nc = c + dc;
            if (nr < 0 || nr >= R || nc < 0 || nc >= C) continue;
            if (g[nr][nc] == '#' || dist[nr][nc] != -1) continue;
            dist[nr][nc] = dist[r][c] + 1;               // ★ 넣을 때 표시 → 중복 삽입 없음
            q.push({nr, nc});
        }
    }
    std::println("S→G = {}", dist[3][3]);
    for (const auto& row : dist) std::println("{}", row);
}
// 출력
// S→G = 6
// [0, 1, 2, -1]
// [-1, -1, 3, -1]
// [6, 5, 4, 5]
// [-1, 6, -1, 6]

④ 슬라이딩 윈도우 최댓값 — 인덱스를 넣는 deque

#include <deque>
#include <print>
#include <vector>

std::vector<int> window_max(const std::vector<int>& a, std::size_t k) {
    std::deque<std::size_t> dq;           // 인덱스 — 값은 앞 → 뒤 내림차순 유지
    std::vector<int> out;
    for (std::size_t i = 0; i < a.size(); ++i) {
        while (!dq.empty() && a[dq.back()] <= a[i])
            dq.pop_back();                // 나보다 작으면 영영 최대가 못 됨
        dq.push_back(i);
        if (dq.front() + k <= i)          // 창 밖 (i - k 는 부호 없는 뺄셈이라 피함)
            dq.pop_front();
        if (i + 1 >= k) out.push_back(a[dq.front()]);
    }
    return out;
}

int main() {
    std::println("{}", window_max({5, 3, 4, 1, 2, 6}, 3));
    std::println("{}", window_max({1, 3, -1, -3, 5, 3, 6, 7}, 3));
}
// 출력
// [5, 4, 4, 6]
// [3, 3, 5, 5, 6, 7]
dq.front() <= i - k 로 쓰면 안 됩니다 — i < k 일 때 size_t 뺄셈이 거대한 수로 감겨 조건이 참이 되고, 아직 창 안인 원소를 버립니다. 뺄셈을 덧셈으로 옮기는 습관이 부호 없는 인덱스의 기본기입니다.

실무에선 이렇게 — 0-1 BFS · 덱의 양 끝을 다 쓰는 알고리즘 🔴 실무

⑤ 벽 부수기 최소 횟수 — 비용 0 은 앞에, 1 은 뒤에

#include <array>
#include <deque>
#include <print>
#include <string>
#include <utility>
#include <vector>

int main() {
    const std::vector<std::string> g{"S.#.", "###.", "..##", ".#.G"};
    const int R = 4, C = 4, INF = 1'000'000;
    std::vector dist(R, std::vector<int>(C, INF));
    constexpr std::array<std::pair<int, int>, 4> dirs{{{1, 0}, {-1, 0}, {0, 1}, {0, -1}}};

    std::deque<std::pair<int, int>> dq;
    dist[0][0] = 0;
    dq.push_back({0, 0});
    while (!dq.empty()) {
        auto [r, c] = dq.front();
        dq.pop_front();
        for (auto [dr, dc] : dirs) {
            int nr = r + dr, nc = c + dc;
            if (nr < 0 || nr >= R || nc < 0 || nc >= C) continue;
            int w = (g[nr][nc] == '#') ? 1 : 0;           // 벽을 부수면 1
            if (dist[r][c] + w < dist[nr][nc]) {
                dist[nr][nc] = dist[r][c] + w;
                if (w == 0) dq.push_front({nr, nc});      // ★ 비용 0 → 앞
                else        dq.push_back({nr, nc});       // ★ 비용 1 → 뒤
            }
        }
    }
    std::println("부순 벽 최소 = {}", dist[3][3]);
}
// 출력
// 부순 벽 최소 = 2
간선 가중치가 0 과 1 뿐이면 다익스트라의 힙이 필요 없습니다 — 덱의 앞은 “현재 거리”, 뒤는 “현재 + 1” 이 되어 덱이 저절로 정렬 상태를 유지합니다. O(V + E) 로, priority_queue 판(O(E log V))보다 빠릅니다.

SQ 09

성능과 메모리 — 복잡도가 같아도 열 배 차이

vector·deque·list 는 모두 “끝에 넣고 빼기 O(1)” 입니다. 그런데 실제 시간은 열 배 넘게 벌어집니다. 차이를 만드는 것은 할당 횟수와 캐시 지역성입니다. 여기에 이동 의미론·emplace·큰 객체 저장법·pmr 할당자까지 묶어 “같은 코드를 싸게 만드는 법”을 정리합니다.

복잡도는 같아도 할당 횟수가 순위를 정합니다 — 연결 리스트는 거의 언제나 꼴찌입니다.

직접 재 보기 — 의존성 없는 측정 코드

// g++ -std=c++23 -O2 bench.cpp && ./a.out
#include <chrono>
#include <deque>
#include <list>
#include <print>
#include <vector>

template <class C>
double bench_stack(int n) {
    auto t0 = std::chrono::steady_clock::now();
    C c;
    long long sum = 0;
    for (int round = 0; round < 10; ++round) {      // 10 × n 번 push/pop
        for (int i = 0; i < n; ++i) c.push_back(i);
        while (!c.empty()) { sum += c.back(); c.pop_back(); }
    }
    auto t1 = std::chrono::steady_clock::now();
    if (sum == 42) std::println("?");               // 최적화로 통째 삭제되는 것 방지
    return std::chrono::duration<double, std::milli>(t1 - t0).count();
}

int main() {
    const int n = 1'000'000;
    std::println("vector {:7.1f} ms", bench_stack<std::vector<int>>(n));
    std::println("deque  {:7.1f} ms", bench_stack<std::deque<int>>(n));
    std::println("list   {:7.1f} ms", bench_stack<std::list<int>>(n));
}

// deque     17.6 ms
// list     118.4 ms

실무에선 이렇게 — pmr 로 할당을 없애기 🔴 실무

프레임마다 쓰고 버리는 BFS 큐 — 할당 0회

#include <array>
#include <cstddef>
#include <deque>
#include <memory_resource>
#include <queue>
#include <vector>

int reachable(const std::vector<std::vector<int>>& adj, int start) {
    std::array<std::byte, 64 * 1024> arena;               // 스택 위 64 KB
    std::pmr::monotonic_buffer_resource pool{arena.data(), arena.size()};

    std::queue<int, std::pmr::deque<int>> q{std::pmr::deque<int>{&pool}};
    std::pmr::vector<char> seen(adj.size(), 0, &pool);

    int count = 0;
    seen[start] = 1;
    q.push(start);
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        ++count;
        for (int v : adj[u])
            if (!seen[v]) { seen[v] = 1; q.push(v); }
    }
    return count;
}   // pool 소멸 = 한꺼번에 반환. 64 KB 를 넘을 때만 upstream(new) 호출
monotonic_buffer_resource 는 해제를 무시합니다 — pop_front 가 블록을 돌려줘도 재사용하지 않으므로 오래 사는 큐에 쓰면 계속 자랍니다. 함수·프레임 단위로 쓰고 버릴 때만 쓰고, 오래 사는 큐는 unsynchronized_pool_resource(🚀 h04)를 씁니다.

흔한 실수 모음 — 컴파일은 되는데 터지는 것들

C++ 스택·큐의 사고는 대부분 컴파일러가 잡아 주지 않는 곳에서 납니다 — 빈 컨테이너 접근, pop 뒤에 남은 참조, 반복자 무효화, 비교자 방향, 이미 사라진 객체를 가리키는 원소. 실제 코드리뷰에서 자주 나오는 순서대로 나쁜 예와 고친 예를 나란히 둡니다.

무효화 규칙은 “원소가 움직이는가”로 외웁니다 — vector 는 움직이고, list 는 안 움직이고, deque 는 원소는 안 움직이지만 지도(맵)가 움직입니다.
❌ ① pop 한 뒤에 front() 의 참조를 씀
while (!q.empty()) {

#include <chrono>
#include <cstddef>
#include <deque>
#include <print>

class SlidingWindowLimiter {
    using Clock = std::chrono::steady_clock;
    std::deque<Clock::time_point> hits_;   // 허용한 요청의 시각
    std::size_t limit_;
    Clock::duration window_;
public:
    SlidingWindowLimiter(std::size_t limit, Clock::duration window)
        : limit_(limit), window_(window) {}

    bool allow(Clock::time_point now) {
        while (!hits_.empty() && now - hits_.front() >= window_)
            hits_.pop_front();                 // 창 밖 — 만료
        if (hits_.size() >= limit_) return false;
        hits_.push_back(now);
        return true;
    }
};

int main() {
    using namespace std::chrono_literals;
    SlidingWindowLimiter lim(3, 1000ms);
    const std::chrono::steady_clock::time_point t0{};   // 시험용 기준 시각
    for (int ms : {0, 200, 400, 600, 900, 1000, 1250, 1300, 2000})
        std::println("{:>4} ms {}", ms,
                     lim.allow(t0 + std::chrono::milliseconds(ms)) ? "허용" : "거절");
}

//    0 ms 허용
//  200 ms 허용
//  400 ms 허용
//  600 ms 거절
//  900 ms 거절
// 1000 ms 허용
// 1250 ms 허용
// 1300 ms 거절
// 2000 ms 허용

② 재시도 큐 — 다음 시도 시각 기준 최소 힙

#include <print>
#include <queue>
#include <string>
#include <utility>
#include <vector>

struct Retry {
    long long due_ms;     // 다음 시도 시각
    unsigned seq;         // 같은 시각이면 먼저 넣은 것부터
    std::string name;
    int attempt;          // 몇 번째 시도 (1부터)
};
struct LaterDue {
    bool operator()(const Retry& a, const Retry& b) const {
        return std::pair(a.due_ms, a.seq) > std::pair(b.due_ms, b.seq);
    }
};

bool try_send(const std::string& name, int attempt) {   // 가짜 외부 호출
    if (name == "A") return attempt >= 3;               // 세 번째에 성공
    if (name == "B") return true;                       // 바로 성공
    return false;                                       // C 는 계속 실패
}

int main() {
    std::priority_queue<Retry, std::vector<Retry>, LaterDue> pq;
    unsigned seq = 0;
    for (const char* n : {"A", "B", "C"}) pq.push({0, seq++, n, 1});
    const int max_attempts = 3;

    while (!pq.empty()) {
        Retry r = pq.top();             // 실제 서비스라면 due 까지 타이머 대기
        pq.pop();
        if (try_send(r.name, r.attempt)) {
            std::println("t={:4} {} #{} ok", r.due_ms, r.name, r.attempt);
        } else if (r.attempt >= max_attempts) {
            std::println("t={:4} {} #{} give up → 데드레터", r.due_ms, r.name, r.attempt);
        } else {
            long long delay = 100LL << (r.attempt - 1);      // 100 · 200 · 400 ...
            std::println("t={:4} {} #{} fail → retry at {}",
                         r.due_ms, r.name, r.attempt, r.due_ms + delay);
            pq.push({r.due_ms + delay, seq++, r.name, r.attempt + 1});
        }
    }
}
// 출력
// t=   0 A #1 fail → retry at 100
// t=   0 B #1 ok
// t=   0 C #1 fail → retry at 100
// t= 100 A #2 fail → retry at 300
// t= 100 C #2 fail → retry at 300
// t= 300 A #3 ok
// t= 300 C #3 give up → 데드레터

③ 프레임 이벤트 큐 — vector 두 개를 교환

#include <print>
#include <string>
#include <utility>
#include <vector>

struct Event { std::string name; };

class EventBus {
    std::vector<Event> pending_, processing_;
public:
    void post(Event e) { pending_.push_back(std::move(e)); }

    template <class F>
    void dispatch(F&& handle) {
        std::swap(pending_, processing_);      // 이번 프레임 몫을 확정 (포인터 교환)
        for (auto& e : processing_) handle(e); // 처리 중 post → pending_ 로 (다음 프레임)
        processing_.clear();                   // 용량은 유지 → 프레임마다 할당 0
    }
};

int main() {
    EventBus bus;
    bus.post({"hit"});
    bus.post({"jump"});
    for (int frame = 1; frame <= 2; ++frame) {
        bus.dispatch([&](const Event& e) {
            std::println("frame {}: {}", frame, e.name);
            if (e.name == "hit") bus.post({"sound"});   // 다음 프레임에 처리
        });
    }
}
// 출력
// frame 1: hit
// frame 1: jump
// frame 2: sound

④ 명령 패턴 undo/redo — 스택 두 개

#include <memory>
#include <print>
#include <string>
#include <vector>

struct Command {
    virtual ~Command() = default;
    virtual void apply(std::string& doc) = 0;
    virtual void revert(std::string& doc) = 0;
};
struct Append : Command {
    std::string text;
    explicit Append(std::string t) : text(std::move(t)) {}
    void apply(std::string& doc) override  { doc += text; }
    void revert(std::string& doc) override { doc.resize(doc.size() - text.size()); }
};

class Editor {
    std::string doc_;
    std::vector<std::unique_ptr<Command>> undo_, redo_;   // 스택 두 개
    static bool move_top(std::vector<std::unique_ptr<Command>>& from,
                         std::vector<std::unique_ptr<Command>>& to) {
        if (from.empty()) return false;
        to.push_back(std::move(from.back()));
        from.pop_back();
        return true;
    }
public:
    void exec(std::unique_ptr<Command> c) {
        c->apply(doc_);
        undo_.push_back(std::move(c));
        redo_.clear();                         // ★ 새 작업을 하면 redo 기록은 버림
    }
    bool undo() { if (!move_top(undo_, redo_)) return false; redo_.back()->revert(doc_); return true; }
    bool redo() { if (!move_top(redo_, undo_)) return false; undo_.back()->apply(doc_);  return true; }
    const std::string& doc() const { return doc_; }
};

int main() {
    Editor e;
    e.exec(std::make_unique<Append>("Hello"));
    e.exec(std::make_unique<Append>(" World"));
    std::println("[{}]", e.doc());
    e.undo(); std::println("[{}]", e.doc());
    e.undo(); std::println("[{}]", e.doc());
    e.redo(); std::println("[{}]", e.doc());
    e.exec(std::make_unique<Append>("!"));
    bool r = e.redo();                         // redo 기록이 비워졌으므로 false
    std::println("[{}] redo={}", e.doc(), r);
}
// 출력
// [Hello World]
// [Hello]
// []

⑤ 최근 N 개 로그 — 장애 때 덤프하는 “블랙박스”

#include <cstddef>
#include <deque>
#include <mutex>
#include <print>
#include <string>

class RecentLog {
    mutable std::mutex m_;
    std::deque<std::string> lines_;
    std::size_t cap_;
public:
    explicit RecentLog(std::size_t cap) : cap_(cap) {}
    void add(std::string s) {
        std::lock_guard g(m_);
        if (lines_.size() == cap_) lines_.pop_front();   // 가장 오래된 것 버림
        lines_.push_back(std::move(s));
    }
    void dump() const {                                  // 크래시 핸들러·/debug 엔드포인트에서
        std::lock_guard g(m_);
        for (const auto& l : lines_) std::println("  {}", l);
    }
};

int main() {
    RecentLog log(3);
    for (const char* s : {"boot", "connect", "query", "timeout", "retry"}) log.add(s);
    log.dump();
}
// 출력
//   query
//   timeout
//   retry
