JVM 01

네 가지 길 — 어디로 갈 것인가를 먼저 정합니다

자바에서 C++ 를 부르는 방법은 크게 넷입니다. 넷 다 “된다”는 점에서는 같지만 호출 한 번의 비용 · 손으로 쓸 코드의 양 · 안드로이드 지원 여부가 전부 다릅니다. 코드를 한 줄 쓰기 전에 이 지도를 먼저 그려 놓아야 나중에 갈아엎지 않습니다.

네 줄 중 어느 줄에 서느냐가 이 탭의 나머지를 결정합니다.

같은 함수 하나, 네 가지 선언

// native/mylib.cpp — C++ 쪽은 어느 길이든 이 한 겹이 출발점입니다
#include <cstdint>

extern "C" {                       // 이름 맹글링을 끕니다 (🔗 C ABI 탭)
#if defined(_WIN32)
  #define API __declspec(dllexport)
#else
  #define API __attribute__((visibility("default")))
#endif

API int32_t mylib_add(int32_t a, int32_t b) { return a + b; }
}
// ① JNI — 이름 규칙이 정해져 있어 C++ 쪽을 따로 써야 합니다
public class Jni { static native int add(int a, int b); }

// ② FFM — 자바 코드 안에서 시그니처를 서술합니다 (Java 22+)
MethodHandle add = Linker.nativeLinker().downcallHandle(
    SymbolLookup.libraryLookup("mylib", Arena.global()).find("mylib_add").orElseThrow(),
    FunctionDescriptor.of(ValueLayout.JAVA_INT, ValueLayout.JAVA_INT, ValueLayout.JAVA_INT));

// ③ JNA — 인터페이스 하나면 끝
public interface MyLib extends Library {
    MyLib I = Native.load("mylib", MyLib.class);
    int mylib_add(int a, int b);
}

// ④ JNR-FFI — JNA 와 거의 같은 모양, 더 빠른 구현
public interface MyLibN { int mylib_add(int a, int b); }
MyLibN n = LibraryLoader.create(MyLibN.class).load("mylib");
눈에 띄는 차이 — ②③④ 는 자바 파일만 고치면 되지만, ① JNI 는 C++ 파일에도 정해진 이름의 함수를 만들어야 합니다. 이 “양쪽을 맞춰야 한다”가 JNI 의 번거로움 전부이자, 반대로 세밀한 제어의 근거이기도 합니다.

JVM 02

JNI 첫 모듈 — 30분이면 자바가 C++ 를 부릅니다

JNI 는 “복잡하다”는 평판과 달리 첫 호출까지는 네 단계뿐입니다. native 메서드를 선언하고, javac -h 로 헤더를 뽑고, 그 시그니처대로 C++ 를 구현하고, 공유 라이브러리로 만들어 System.loadLibrary 로 얹으면 끝입니다.

JNI 는 이름 규칙으로 연결되는 시스템입니다 — 그래서 헤더를 손으로 쓰지 않습니다.

① 자바 쪽 — 이게 전부입니다

// src/main/java/com/example/native_/MathOps.java
package com.example.native_;

public final class MathOps {

    static {                                  // 클래스가 처음 쓰일 때 한 번
        System.loadLibrary("mathops");        // libmathops.so / mathops.dll
    }

    // 본문이 없습니다 — 구현은 C++ 쪽에 있습니다
    public static native int add(int a, int b);
    public static native long sumRange(long from, long to);
    public static native String upper(String s);

    private MathOps() {}
}
static { } 블록에 넣는 이유 — 클래스가 처음 로드될 때 딱 한 번 실행되기 때문입니다. 두 번 부르면 UnsatisfiedLinkError: Native Library ... already loaded in another classloader 가 날 수 있고, 늦게 부르면 첫 호출에서 터집니다. native 메서드를 가진 클래스 안의 static 블록이 정석입니다.

② 헤더 생성 — 손으로 쓰지 마세요

# javac 가 컴파일하면서 JNI 헤더까지 뽑아 줍니다 (JDK 8+)
javac -h native/include \
      -d build/classes \
      src/main/java/com/example/native_/MathOps.java

# 생성된 파일: native/include/com_example_native__MathOps.h
#   · 패키지의 . 은 _ 로
#   · 이름 안의 _ 는 _1 로 (native_ → native_1)
#   · 오버로드가 있으면 뒤에 __타입시그니처 가 붙습니다

이름 규칙을 외울 필요는 없습니다 — 대신 헤더를 절대 손으로 고치지 마세요. 자바 쪽 시그니처를 바꾸면 javac -h 를 다시 돌려 헤더를 새로 뽑고, C++ 구현을 거기에 맞춥니다. 이 순서를 지키는 것만으로 링크 에러의 대부분이 사라집니다.

③ C++ 구현 — JNIEnv* 가 모든 것의 손잡이

// native/src/mathops.cpp
#include <jni.h>
#include <string>
#include <algorithm>
#include "com_example_native__MathOps.h"

extern "C" {                        // ★ 필수 — C++ 맹글링을 끕니다

JNIEXPORT jint JNICALL
Java_com_example_native_1_MathOps_add(JNIEnv*, jclass, jint a, jint b) {
    return a + b;                   // jint 는 그냥 int32_t 입니다
}

JNIEXPORT jlong JNICALL
Java_com_example_native_1_MathOps_sumRange(JNIEnv*, jclass, jlong from, jlong to) {
    long long s = 0;
    for (long long i = from; i <= to; ++i) s += i;
    return static_cast<jlong>(s);
}

JNIEXPORT jstring JNICALL
Java_com_example_native_1_MathOps_upper(JNIEnv* env, jclass, jstring in) {
    const char* c = env->GetStringUTFChars(in, nullptr);   // ① 빌리고
    if (!c) return nullptr;                                // OOM 이면 예외가 이미 걸려 있음
    std::string s(c);
    env->ReleaseStringUTFChars(in, c);                     // ② 반드시 돌려주고
    std::transform(s.begin(), s.end(), s.begin(), ::toupper);
    return env->NewStringUTF(s.c_str());                   // ③ 새 자바 문자열로
}

}   // extern "C"
세 가지만 기억하면 됩니다 — ① 파일 전체를 extern "C" 로 감싼다, ② 첫 인자 JNIEnv* env 로 모든 JNI 기능을 부른다, ③ 빌린 것은 반드시 돌려준다(Get… 에는 짝이 되는 Release… 가 있습니다). env->X() 는 C++ 문법이고, C 로 쓸 때는 (*env)->X(env, …) 입니다.

CMakeLists.txt — 20줄이면 세 OS 공통

    OUTPUT_NAME "mathops"
    CXX_VISIBILITY_PRESET hidden      # JNIEXPORT 붙은 것만 노출

# build/libmathops.so (리눅스) · build/Release/mathops.dll (윈도우)

JVM 03

타입 매핑과 문자열 — Modified UTF-8 이라는 함정

숫자는 그냥 통과합니다. 문제는 문자열입니다. JNI 가 주는 “UTF-8” 은 진짜 UTF-8 이 아닙니다. 이 사실을 모르면 한글은 잘 되는데 이모지에서만 깨지는 버그를 며칠 동안 쫓게 됩니다.

“UTF-8 이라 쓰여 있으니 UTF-8 이겠지”가 JNI 문자열 버그의 출발점입니다.

RAII 로 감싸면 실수가 사라집니다

// jni_util.hpp — 프로젝트마다 하나씩 두게 되는 도우미
#include <jni.h>
#include <string>
#include <string_view>

class JStringUtf {                 // Modified UTF-8 을 잠깐 빌리는 래퍼
public:
    JStringUtf(JNIEnv* env, jstring s) : env_(env), s_(s) {
        if (s_) c_ = env_->GetStringUTFChars(s_, nullptr);
    }
    ~JStringUtf() {                             // ★ 어떤 경로로 나가도 반환됩니다
        if (c_) env_->ReleaseStringUTFChars(s_, c_);
    }
    JStringUtf(const JStringUtf&) = delete;     // 두 번 Release 를 원천 차단
    JStringUtf& operator=(const JStringUtf&) = delete;

    bool ok() const { return c_ != nullptr; }
    std::string_view view() const { return c_ ? std::string_view(c_) : std::string_view{}; }
    std::string str() const { return c_ ? std::string(c_) : std::string{}; }

private:
    JNIEnv* env_;
    jstring s_;
    const char* c_ = nullptr;
};

// 쓰는 쪽 — 예외를 던지든 일찍 return 하든 새지 않습니다
JNIEXPORT jint JNICALL
Java_com_example_native_1_MathOps_countChars(JNIEnv* env, jclass, jstring in) {
    JStringUtf s(env, in);
    if (!s.ok()) return -1;              // OOM — 예외가 이미 걸려 있습니다
    return static_cast<jint>(s.view().size());
}
왜 RAII 인가 — JNI 함수는 중간에 예외 상태를 만들 수 있고, 그때 return 으로 빠져나가면서 Release 를 빼먹기 딱 좋습니다. 소멸자에 넣는 순간 “잊어버릴 수가 없는” 코드가 됩니다. jbyteArray · jintArray 도 같은 패턴으로 하나씩 만들어 두세요(v04).

JVM 04

배열과 제로카피 — 복사를 없애야 이득이 남습니다

8MB 이미지를 C++ 에 넘겨 0.4ms 만에 처리했는데 전체는 3ms 가 걸렸다면, 2.6ms 는 복사입니다. JNI 에서 진짜 성능은 알고리즘이 아니라 경계에서 몇 번 복사되느냐에서 갈립니다.

경계를 넘는 데이터가 크면 DirectByteBuffer 가 사실상 유일한 정답입니다.

배열을 받는 세 가지 방법

import java.nio.ByteBuffer;
import java.nio.ByteOrder;

public final class Image {
    static { System.loadLibrary("imageops"); }

    // ByteBuffer 는 반드시 direct 여야 합니다 — heap buffer 면 주소가 없습니다
    private static native void nGrayscale(ByteBuffer src, ByteBuffer dst, int w, int h);

    public static ByteBuffer grayscale(ByteBuffer rgba, int w, int h) {
        if (!rgba.isDirect()) throw new IllegalArgumentException("direct buffer 여야 합니다");
        ByteBuffer out = ByteBuffer.allocateDirect(w * h)
                                   .order(ByteOrder.nativeOrder());   // ★ 엔디안 맞추기
        nGrayscale(rgba, out, w, h);
        return out;
    }
}
JNIEXPORT void JNICALL
Java_com_example_Image_nGrayscale(JNIEnv* env, jclass,
                                  jobject src, jobject dst, jint w, jint h) {
    auto* s = static_cast<const uint8_t*>(env->GetDirectBufferAddress(src));
    auto* d = static_cast<uint8_t*>(env->GetDirectBufferAddress(dst));
    if (!s || !d) {                          // direct 가 아니면 nullptr 입니다
        env->ThrowNew(env->FindClass("java/lang/IllegalArgumentException"),
                      "direct ByteBuffer required");
        return;
    }
    // 용량 검사는 반드시 — 여기서 안 하면 힙을 넘어 씁니다
    if (env->GetDirectBufferCapacity(src) < (jlong)w * h * 4) { /* throw */ return; }

    const size_t n = (size_t)w * h;
    for (size_t i = 0; i < n; ++i) {         // 복사 없이 그 자리에서
        d[i] = (uint8_t)((s[i*4] * 77 + s[i*4+1] * 150 + s[i*4+2] * 29) >> 8);
    }
}
반드시 용량을 검사하세요 — GetDirectBufferAddress 는 생짜 포인터입니다. 자바 쪽에서 작은 버퍼를 넘겨도 C++ 는 모릅니다. 경계 검사 없는 JNI 함수는 그대로 원격 메모리 손상 취약점이 됩니다. GetDirectBufferCapacity 로 매번 확인하는 것이 규칙입니다.

객체 · 콜백 · 예외 — C++ 가 자바를 부를 때

여기까지는 자바가 C++ 를 불렀습니다. 이제 반대 방향입니다 — 진행률 보고 · 이벤트 통지 · 정렬 비교 함수처럼 C++ 가 자바를 되부르는 순간 클래스 캐싱 · 예외 확인 · 스레드 붙이기라는 세 가지 규칙이 한꺼번에 필요해집니다.

JNI_OnLoad — 캐시는 여기서 한 번만

#include <jni.h>

static JavaVM*   g_vm       = nullptr;   // 스레드 붙일 때 필요합니다
static jclass    g_listener = nullptr;   // 전역 참조
static jmethodID g_onProgress = nullptr; // ★ methodID 는 참조가 아니라 그냥 값
static jmethodID g_onDone     = nullptr;

// 라이브러리가 로드될 때 JVM 이 자동으로 불러 줍니다
JNIEXPORT jint JNICALL JNI_OnLoad(JavaVM* vm, void*) {
    g_vm = vm;
    JNIEnv* env = nullptr;
    if (vm->GetEnv(reinterpret_cast<void**>(&env), JNI_VERSION_1_8) != JNI_OK)
        return JNI_ERR;

    // ★ 여기서 찾아야 하는 이유 — 이 순간에는 "정상적인 클래스로더"가 붙어 있습니다
    jclass local = env->FindClass("com/example/ProgressListener");
    if (!local) return JNI_ERR;
    g_listener = static_cast<jclass>(env->NewGlobalRef(local));
    env->DeleteLocalRef(local);

    g_onProgress = env->GetMethodID(g_listener, "onProgress", "(I)V");
    g_onDone     = env->GetMethodID(g_listener, "onDone",     "(Ljava/lang/String;)V");
    if (!g_onProgress || !g_onDone) return JNI_ERR;

    return JNI_VERSION_1_8;      // ★ 반드시 버전을 돌려줘야 로드가 성공합니다
}

JNIEXPORT void JNICALL JNI_OnUnload(JavaVM* vm, void*) {
    JNIEnv* env = nullptr;
    if (vm->GetEnv(reinterpret_cast<void**>(&env), JNI_VERSION_1_8) == JNI_OK)
        if (g_listener) env->DeleteGlobalRef(g_listener);
}
왜 하필 JNI_OnLoad 인가 — FindClass 는 현재 스레드의 클래스로더를 기준으로 클래스를 찾습니다. C++ 가 직접 만든 스레드에는 애플리케이션 클래스로더가 없어서 ClassNotFoundException 이 납니다 — 자바에서는 멀쩡히 보이는 클래스인데도요. JNI_OnLoad 는 loadLibrary 를 부른 스레드에서 실행되므로 안전합니다.

    for (jint i = 0; i < total; ++i) {
        heavy_step(i);

        if (i % 1000 == 0) {                       // ★ 매번 부르지 않습니다
            env->CallVoidMethod(listener, g_onProgress, (jint)(i * 100 / total));

            // ★★ 콜백 직후에는 반드시 예외 확인 — 자바 쪽이 던졌을 수 있습니다
            if (env->ExceptionCheck()) return -1;   // 예외를 그대로 자바로 전파
        }
    }

    jstring msg = env->NewStringUTF("done");
    env->CallVoidMethod(listener, g_onDone, msg);
    env->DeleteLocalRef(msg);                       // 루프 밖이라도 습관적으로
    if (env->ExceptionCheck()) return -1;
    return 0;
}

// 반환 타입에 따라 함수 이름이 다릅니다
//   CallVoidMethod / CallIntMethod / CallLongMethod / CallObjectMethod ...
//   정적 메서드는 CallStatic...Method(클래스, mid, ...)
//   상위 클래스 구현을 강제로 부르려면 CallNonvirtual...Method
콜백 빈도가 곧 성능입니다 — 진행률을 루프마다 보고하면 경계를 100만 번 넘습니다. 1000번에 한 번 또는 50ms 에 한 번으로 묶으면 사용자 경험은 그대로인데 오버헤드는 1000분의 1이 됩니다. “콜백은 비싸다”가 아니라 “콜백 횟수가 비싸다”입니다.

핸들 패턴 — C++ 객체를 자바 객체에 매달기 🟡 중급

long 필드 하나에 포인터를 담습니다

public final class Encoder implements AutoCloseable {
    static { System.loadLibrary("codec"); }

    private long handle;                   // ★ C++ 포인터가 여기에 들어갑니다

    public Encoder(int w, int h) { handle = nCreate(w, h); }

    public int encode(java.nio.ByteBuffer frame) {
        if (handle == 0) throw new IllegalStateException("closed");
        return nEncode(handle, frame);
    }

    @Override public void close() {        // ★ try-with-resources 로 반드시 해제
        if (handle != 0) { nDestroy(handle); handle = 0; }
    }

    private static native long nCreate(int w, int h);
    private static native int  nEncode(long h, java.nio.ByteBuffer b);
    private static native void nDestroy(long h);
}
struct EncoderImpl { int w, h; /* ... */ };

JNIEXPORT jlong JNICALL
Java_com_example_Encoder_nCreate(JNIEnv*, jclass, jint w, jint h) {
    return reinterpret_cast<jlong>(new EncoderImpl{w, h});
}

JNIEXPORT void JNICALL
Java_com_example_Encoder_nDestroy(JNIEnv*, jclass, jlong h) {
    delete reinterpret_cast<EncoderImpl*>(h);
}

JNIEXPORT jint JNICALL
Java_com_example_Encoder_nEncode(JNIEnv* env, jclass, jlong h, jobject buf) {
    auto* impl = reinterpret_cast<EncoderImpl*>(h);
    if (!impl) return -1;
    auto* p = static_cast<uint8_t*>(env->GetDirectBufferAddress(buf));
    return impl->encode(p);
}
반드시 long 이어야 합니다 — int 에 담으면 64비트 주소가 잘려 32비트 환경에서만 도는 코드가 됩니다. 그리고 finalize() 는 쓰지 마세요(폐기 예정입니다). AutoCloseable + try-with-resources, 보험이 필요하면 java.lang.ref.Cleaner 를 씁니다.

// ★ C++ 예외가 JNI 경계를 넘으면 정의되지 않은 동작입니다 (대개 즉사)
JNIEXPORT jint JNICALL
Java_com_example_Codec_nDecode(JNIEnv* env, jclass, jbyteArray in) try {

catch (const std::bad_alloc&) {

    return -1;
}
catch (const std::exception& e) {
    env->ThrowNew(env->FindClass("java/lang/RuntimeException"), e.what());
    return -1;
}
catch (...) {
    env->ThrowNew(env->FindClass("java/lang/Error"), "unknown native error");
    return -1;
}

// ★ ThrowNew 는 "예외를 걸어 둘 뿐" 함수를 끝내지 않습니다 — return 을 잊지 마세요

// 반대로, 자바가 던진 예외를 C++ 가 확인·처리하는 방법
if (env->ExceptionCheck()) {
    jthrowable t = env->ExceptionOccurred();
    env->ExceptionDescribe();         // 스택트레이스를 stderr 로 (디버깅용)
    env->ExceptionClear();            // ★ 지워야 다른 JNI 함수를 부를 수 있습니다
    /* 정리 작업 ... */
    env->Throw(t);                    // 필요하면 그대로 다시 던지기
}
예외가 걸린 상태에서 JNI 함수를 부르면 안 됩니다 — 허용되는 것은 ExceptionOccurred · ExceptionDescribe · ExceptionClear · DeleteLocalRef 같은 소수뿐입니다. 그 밖의 함수를 부르면 동작이 정의되어 있지 않고, -Xcheck:jni 를 켜면 그 자리에서 “JNI call made with exception pending” 이라고 알려 줍니다.

C++ 스레드에서 자바 부르기 — attach 와 detach 🔴 실무

JNIEnv* 는 스레드마다 다릅니다

// ★ 절대 금지 — JNIEnv* 를 다른 스레드로 넘기기
//   JNIEnv 는 스레드 지역입니다. JavaVM* 만 스레드 사이에서 공유할 수 있습니다

// RAII 붙이기 도우미 — 이 패턴 하나만 있으면 됩니다
class JniAttach {
public:
    explicit JniAttach(JavaVM* vm) : vm_(vm) {
        jint r = vm_->GetEnv(reinterpret_cast<void**>(&env_), JNI_VERSION_1_8);
        if (r == JNI_EDETACHED) {
            JavaVMAttachArgs a{JNI_VERSION_1_8, const_cast<char*>("native-worker"), nullptr};
            // AsDaemon: 이 스레드가 살아 있어도 JVM 종료를 막지 않습니다 ★
            if (vm_->AttachCurrentThreadAsDaemon(
                    reinterpret_cast<void**>(&env_), &a) == JNI_OK)
                attached_ = true;
        }
    }
    ~JniAttach() { if (attached_) vm_->DetachCurrentThread(); }
    JNIEnv* env() const { return env_; }
private:
    JavaVM* vm_; JNIEnv* env_ = nullptr; bool attached_ = false;
};

// C++ 워커 스레드에서 자바 콜백 부르기
void worker(jobject globalListener) {          // ★ 전역 참조여야 합니다
    JniAttach a(g_vm);
    if (!a.env()) return;
    a.env()->CallVoidMethod(globalListener, g_onProgress, 100);
    if (a.env()->ExceptionCheck()) a.env()->ExceptionClear();
}   // ← 소멸자가 detach

자주 나는 사고와 원인

"attempting to use JNIEnv from the wrong thread"
  → JNIEnv* 를 멤버 변수에 저장해 두고 다른 스레드에서 썼습니다

"JNI DetachCurrentThread called on thread not attached"
  → attach 없이 detach. RAII 로 묶으면 생기지 않습니다

프로세스가 종료 시점에 멈춤(hang)
  → AttachCurrentThread 로 붙인 스레드를 detach 안 했습니다.
    JVM 이 그 스레드를 기다립니다 → AsDaemon 을 쓰세요

콜백을 부르는 순간 크래시
  → 리스너를 지역 참조로 저장했습니다. NewGlobalRef 필수

스레드마다 attach/detach 를 반복해서 느림
  → 붙이는 데 수 µs 가 듭니다. 워커 스레드는 시작할 때 한 번 붙이고
    끝날 때 떼는 것이 정석입니다 (스레드 지역 RAII)
가장 안전한 설계 — C++ 가 자바를 되부르지 않게 만드는 것입니다. 결과를 큐에 쌓아 두고 자바 쪽 스레드가 주기적으로 꺼내 가게 하면 attach · 전역 참조 · 예외 전파 문제가 전부 사라집니다. 콜백은 “꼭 필요할 때만” 쓰는 도구로 두세요.
정리 — 콜백의 규칙은 셋입니다. 클래스와 메서드 ID 는 JNI_OnLoad 에서 한 번만 찾고, 콜백 뒤에는 항상 ExceptionCheck 하고, C++ 스레드에서는 attach 를 RAII 로 묶습니다. C++ 객체는 long 핸들 + AutoCloseable 로 자바 객체에 매답니다.

성능과 안정성 — 호출을 줄이고 크래시를 읽습니다

JNI 코드가 느린 이유는 대개 C++ 가 느려서가 아니라 경계를 너무 자주 넘어서입니다. 그리고 JNI 코드가 죽으면 예외가 아니라 JVM 프로세스 전체가 사라집니다 — 그때 남는 hs_err_pid 파일을 읽을 줄 알아야 합니다.

JNI 최적화의 90%는 “호출 횟수 줄이기” 한 가지입니다.

배치 설계 — 나쁜 API 와 좋은 API

경계를 100만 번 넘는 설계
  • native double sample(int i) 를 루프에서 호출
  • 픽셀 하나마다 native int pixel(x, y)
  • 진행률을 반복마다 콜백
  • 결과를 한 건씩 String 으로 반환
한 번에 크게 넘기는 설계
  • native void sampleAll(ByteBuffer out, int n)
  • 이미지 전체를 DirectBuffer 로 한 번
  • 진행률은 50ms 간격으로만
  • 결과를 구조체 배열(DirectBuffer)로 한 번에
// ✗ 이 API 는 아무리 C++ 를 잘 짜도 느립니다
for (int i = 0; i < 1_000_000; i++) sum += Native.sample(i);

// ✓ 같은 계산, 경계는 한 번
ByteBuffer out = ByteBuffer.allocateDirect(1_000_000 * 8)
                           .order(ByteOrder.nativeOrder());
Native.sampleAll(out, 1_000_000);
DoubleBuffer d = out.asDoubleBuffer();      // 자바 쪽에서 그대로 읽습니다
API 를 설계할 때의 질문 — “이 함수는 한 번 호출에 몇 µs 어치 일을 하는가?” 1µs 미만이면 그 함수는 애초에 경계에 두면 안 되는 함수입니다. 위로 한 겹 올려서 덩어리로 묶은 함수를 대신 내보내세요.

RegisterNatives — 이름 규칙에서 벗어나기

// 함수 이름을 마음대로 짓고, JNI_OnLoad 에서 직접 연결합니다
static jint nAdd(JNIEnv*, jclass, jint a, jint b) { return a + b; }
static void nRun(JNIEnv*, jclass, jobject buf)    { /* ... */ }

static const JNINativeMethod kMethods[] = {
    {"add", "(II)I",                    reinterpret_cast<void*>(nAdd)},
    {"run", "(Ljava/nio/ByteBuffer;)V", reinterpret_cast<void*>(nRun)},
};

JNIEXPORT jint JNICALL JNI_OnLoad(JavaVM* vm, void*) {
    JNIEnv* env;
    if (vm->GetEnv((void**)&env, JNI_VERSION_1_8) != JNI_OK) return JNI_ERR;
    jclass c = env->FindClass("com/example/Fast");
    if (!c) return JNI_ERR;
    if (env->RegisterNatives(c, kMethods,
            sizeof(kMethods)/sizeof(kMethods[0])) != JNI_OK) return JNI_ERR;
    return JNI_VERSION_1_8;
}

놓치기 쉬운 성능 요인 넷 🟡 중급

JIT 가 못 넘어가는 벽

① native 메서드는 인라인되지 않습니다
   자바 메서드끼리는 JIT 가 합쳐서 오버헤드를 0 으로 만들지만,
   native 경계에서는 그 최적화가 끊깁니다.
   → 작은 native 함수를 루프에서 부르는 설계가 특히 손해입니다

② 안전 지점(safepoint)과의 관계
   네이티브 코드 실행 중에는 그 스레드가 safepoint 에 들어가지 않습니다.
   → 아주 긴 네이티브 호출은 GC 를 기다리게 만들 수 있습니다
   → 수십 ms 이상 걸리는 작업은 중간에 끊어 돌려주는 설계가 낫습니다

③ 첫 호출은 느립니다
   심볼 탐색 · 클래스 로딩 · 페이지 폴트가 한꺼번에 일어납니다.
   → 기동 직후 워밍업 호출을 한 번 넣으면 첫 요청 지연이 사라집니다
   → JMH 의 @Warmup 이 존재하는 이유이기도 합니다

④ 네이티브 스레드는 JVM 이 모릅니다
   C++ 안에서 만든 스레드는 -XX:ActiveProcessorCount 계산 밖입니다.
   → 컨테이너 CPU 한도를 넘겨 스로틀링이 걸리는 사고가 흔합니다
③번은 실제 장애로 자주 이어집니다 — 배포 직후 첫 요청만 수백 ms 가 걸려 헬스체크가 실패하고 롤링 배포가 멈추는 형태입니다. 기동 시 워밍업 호출 한 줄이면 끝납니다 — 네이티브 라이브러리를 쓰는 서비스의 기본 장비로 두세요.

“C++ 로 바꿨는데 안 빨라졌다”의 진짜 원인 1위 — 여전히 복사가 남아 있는 것입니다(v04). 2위는 호출이 잘게 쪼개져 있는 것이고, 3위는 애초에 병목이 거기가 아니었던 것입니다. 셋 다 프로파일러 한 번이면 구분됩니다.
정리 — 성능은 호출 횟수가 결정하고, 안정성은 규약 준수가 결정합니다. 덩어리로 넘기는 API를 설계하고, 연결은 RegisterNatives 로 깔끔하게 하고, 개발 내내 -Xcheck:jni 를 켜 두면 JNI 는 생각만큼 위험하지 않습니다.
JVM 07

Panama FFM — C++ 파일을 한 줄도 안 쓰고 부릅니다

JNI 의 번거로움은 전부 “C++ 쪽에도 정해진 이름의 함수를 만들어야 한다”에서 나왔습니다. Java 22 에서 정식이 된 FFM(Foreign Function & Memory) API 는 그 요구를 없앱니다 — 이미 있는 extern "C" 함수를 자바 코드만으로 부릅니다.

JNI 의 malloc/free 자리를 Arena 가 구조적으로 대신합니다.

C++ 쪽 — 이것뿐입니다

// native/src/mathffm.cpp — JNI 헤더도, 정해진 함수 이름도 없습니다
#include <cstdint>
#include <cstring>
#include <cmath>

extern "C" {
#if defined(_WIN32)
  #define API __declspec(dllexport)
#else
  #define API __attribute__((visibility("default")))
#endif

API int32_t mathffm_add(int32_t a, int32_t b) { return a + b; }

API double mathffm_norm(const double* v, int32_t n) {   // 배열은 포인터 + 길이
    double s = 0;
    for (int32_t i = 0; i < n; ++i) s += v[i] * v[i];
    return std::sqrt(s);
}

API int32_t mathffm_greet(const char* name, char* out, int32_t cap) {
    int n = std::snprintf(out, (size_t)cap, "Hello, %s!", name);
    return n;                                            // 쓴 길이
}
}   // extern "C"
# 빌드 — JNI 헤더가 필요 없으니 -I 옵션도 없습니다
g++ -std=c++20 -O2 -fPIC -shared src/mathffm.cpp -o libmathffm.so
여기가 FFM 의 핵심 장점입니다 — 이 .so 는 자바를 전혀 모르는 평범한 C 라이브러리입니다. 같은 파일을 파이썬 ctypes 도, Node ffi 도, C# P/Invoke 도 씁니다. JNI 라이브러리는 자바 전용이라 그럴 수 없습니다.

자바 쪽 — Java 25 전체 코드

import java.lang.foreign.*;
import java.lang.invoke.MethodHandle;
import java.nio.file.Path;

public final class MathFfm implements AutoCloseable {

    private final Arena arena = Arena.ofShared();     // 라이브러리 수명 = 앱 수명
    private final MethodHandle add, norm, greet;

    public MathFfm(Path lib) {
        Linker linker = Linker.nativeLinker();
        SymbolLookup lookup = SymbolLookup.libraryLookup(lib, arena);

        add = linker.downcallHandle(
            lookup.find("mathffm_add").orElseThrow(),
            FunctionDescriptor.of(ValueLayout.JAVA_INT,      // 반환
                                  ValueLayout.JAVA_INT,      // a
                                  ValueLayout.JAVA_INT));    // b

        norm = linker.downcallHandle(
            lookup.find("mathffm_norm").orElseThrow(),
            FunctionDescriptor.of(ValueLayout.JAVA_DOUBLE,
                                  ValueLayout.ADDRESS,       // const double*
                                  ValueLayout.JAVA_INT));

        greet = linker.downcallHandle(
            lookup.find("mathffm_greet").orElseThrow(),
            FunctionDescriptor.of(ValueLayout.JAVA_INT,
                                  ValueLayout.ADDRESS, ValueLayout.ADDRESS,
                                  ValueLayout.JAVA_INT));
    }

    public int add(int a, int b) {
        try { return (int) add.invokeExact(a, b); }        // ★ invokeExact
        catch (Throwable t) { throw new RuntimeException(t); }
    }

    public double norm(double[] v) {
        try (Arena tmp = Arena.ofConfined()) {              // ★ 호출 동안만 사는 메모리
            MemorySegment seg = tmp.allocateFrom(ValueLayout.JAVA_DOUBLE, v);
            return (double) norm.invokeExact(seg, v.length);
        } catch (Throwable t) { throw new RuntimeException(t); }
    }

    public String greet(String name) {
        try (Arena tmp = Arena.ofConfined()) {
            MemorySegment in  = tmp.allocateFrom(name);     // 자동으로 널 종단 UTF-8
            MemorySegment out = tmp.allocate(256);
            int n = (int) greet.invokeExact(in, out, 256);
            if (n < 0) throw new IllegalStateException("greet failed");
            return out.getString(0);                        // 널 종단 문자열 읽기
        } catch (Throwable t) { throw new RuntimeException(t); }
    }

    @Override public void close() { arena.close(); }        // 라이브러리도 여기서 정리
}
invokeExact 는 타입이 정확히 맞아야 합니다 — (int) mh.invokeExact(3, 4) 에서 캐스트를 빼거나 long 을 넘기면 실행 시점에 WrongMethodTypeException 이 납니다. 느슨하게 가려면 invoke 를 쓸 수 있지만 변환 비용이 붙습니다 — 뜨거운 경로에는 invokeExact + static final MethodHandle 이 정석입니다.

jextract · 구조체 · 콜백 — 그리고 FFM 이 아직 못 하는 것

함수가 다섯 개면 손으로 써도 됩니다. 헤더가 3천 줄이면 이야기가 다릅니다. jextract 는 C 헤더를 읽어 자바 바인딩을 통째로 생성합니다. 여기에 구조체 레이아웃과 upcall 콜백까지 더하면 FFM 으로 못 할 일이 거의 없습니다 — “거의”가 어디까지인지도 정확히 짚습니다.

upcall — C 가 자바를 되부르게 🔴 실무

자바 메서드를 함수 포인터로 만들기

import java.lang.foreign.*;
import java.lang.invoke.*;

public final class Sorter {

    // ① C 가 부를 자바 메서드 — 인자는 전부 MemorySegment(포인터)
    static int compare(MemorySegment a, MemorySegment b) {
        int x = a.get(ValueLayout.JAVA_INT, 0);
        int y = b.get(ValueLayout.JAVA_INT, 0);
        return Integer.compare(x, y);
    }

    public static void sort(int[] data) throws Throwable {
        Linker linker = Linker.nativeLinker();

        try (Arena arena = Arena.ofConfined()) {
            // ② 자바 메서드 → MethodHandle
            MethodHandle cmp = MethodHandles.lookup().findStatic(
                Sorter.class, "compare",
                MethodType.methodType(int.class, MemorySegment.class, MemorySegment.class));

            // ③ MethodHandle → C 함수 포인터 (upcall stub)
            FunctionDescriptor cmpDesc = FunctionDescriptor.of(
                ValueLayout.JAVA_INT, ValueLayout.ADDRESS, ValueLayout.ADDRESS);
            MemorySegment cmpPtr = linker.upcallStub(cmp, cmpDesc, arena);
            //   ★ arena 가 닫히면 이 포인터는 무효가 됩니다

            // ④ C 표준 qsort 에 그대로 넘깁니다
            MethodHandle qsort = linker.downcallHandle(
                linker.defaultLookup().find("qsort").orElseThrow(),
                FunctionDescriptor.ofVoid(ValueLayout.ADDRESS, ValueLayout.JAVA_LONG,
                                          ValueLayout.JAVA_LONG, ValueLayout.ADDRESS));

            MemorySegment buf = arena.allocateFrom(ValueLayout.JAVA_INT, data);
            qsort.invokeExact(buf, (long) data.length, 4L, cmpPtr);
            MemorySegment.copy(buf, ValueLayout.JAVA_INT, 0, data, 0, data.length);
        }
    }
}
upcall 의 두 가지 함정 — ① stub 의 수명은 arena 에 묶입니다. C 쪽이 포인터를 계속 보관한다면 Arena.ofShared()·global() 로 만들어야 합니다. ② 자바 예외가 C 스택을 통과하면 안 됩니다 — upcall 메서드 안에서 try/catch 로 전부 잡아 에러 코드로 바꿔 돌려주세요.

FFM 이 아직 못 하는 것 ★

한계실제 의미
C++ 클래스 직접 불가맹글링 · vtable · 예외를 모릅니다 — C 래퍼 필수
안드로이드 미지원ART 에 java.lang.foreign 이 없습니다 — JNI 로
Java 21 이하preview 라 --enable-preview 필요 · API 가 다름
템플릿 · 오버로드헤더에 실체화된 함수가 있어야 뽑힙니다
가변 인자(varargs)가능하지만 Linker.Option.firstVariadicArg 필요
성능(검사 비용)경계·수명 검사가 붙습니다 — critical 로 줄입니다
// 뜨거운 경로에서 검사 비용을 줄이는 옵션 (Java 22+)
MethodHandle fast = Linker.nativeLinker().downcallHandle(
    addr,
    FunctionDescriptor.of(ValueLayout.JAVA_INT, ValueLayout.ADDRESS, ValueLayout.JAVA_INT),
    Linker.Option.critical(true));   // allowHeapAccess=true — 힙 배열도 그대로 전달
//   ★ 대신 그 호출 동안 안전장치가 줄어듭니다:
//     · 오래 걸리는 함수에 쓰면 GC 를 방해할 수 있습니다
//     · 아주 짧고(수 µs) 블로킹하지 않는 함수에만 쓰세요
//     · JNI 의 GetPrimitiveArrayCritical 과 같은 성격의 도구입니다

// 가변 인자 — printf 처럼 뒤가 열린 함수
FunctionDescriptor printfDesc = FunctionDescriptor.of(
        ValueLayout.JAVA_INT, ValueLayout.ADDRESS, ValueLayout.JAVA_INT);
MethodHandle printf = Linker.nativeLinker().downcallHandle(
        Linker.nativeLinker().defaultLookup().find("printf").orElseThrow(),
        printfDesc, Linker.Option.firstVariadicArg(1));
C++ 프로젝트라면 결국 같은 결론 — FFM 이든 JNI 든 extern "C" 래퍼 한 겹이 필요합니다. 차이는 그 위에서 자바 코드만 쓰느냐(FFM), C++ 코드도 쓰느냐(JNI)입니다. 🔗 C ABI 탭에서 만든 래퍼가 그대로 재활용됩니다.
정리 — 헤더가 크면 jextract, 구조체는 패딩까지 명시한 StructLayout, 콜백은 upcallStub + arena 수명입니다. FFM 은 C++ 클래스와 안드로이드 두 곳에서 멈추고, 그 자리에는 JNI 가 남습니다.
JVM 09

JNA · JNR — 인터페이스 하나로 끝내는 길

JNI 헤더도, FFM 조립도 없이 자바 인터페이스 하나만 선언하면 C 함수가 불립니다. 대신 호출마다 값을 조립하는 중간 계층이 끼기 때문에 느립니다. 이 거래가 이득인 자리가 분명히 있고, 손해인 자리는 더 분명합니다.

계층이 하나 늘 때마다 수십~수백 ns 가 붙습니다 — 호출 횟수가 판단 기준입니다.

JNA — 코드 0줄 바인딩

// build.gradle.kts:  implementation("net.java.dev.jna:jna:5.x")
import com.sun.jna.*;

public interface MyLib extends Library {

    int    mylib_add(int a, int b);
    double mylib_norm(double[] v, int n);      // 배열이 그냥 통합니다
    String mylib_version();                    // char* → String 자동 변환
    Pointer mylib_create(int w, int h);        // 불투명 포인터
    void   mylib_destroy(Pointer p);
}

package com.example.native_;

import java.io.*;
import java.nio.file.*;

public final class NativeLoader {
    private static boolean loaded = false;

    public static synchronized void load(String lib) {
        if (loaded) return;
        try {
            System.loadLibrary(lib);          // ① 시스템에 이미 있으면 그걸 씁니다
        } catch (UnsatisfiedLinkError ignored) {
            extractAndLoad(lib);              // ② 없으면 jar 에서 꺼냅니다
        }
        loaded = true;
    }

    private static void extractAndLoad(String lib) {
        String res = "/native/" + platform() + "/" + mapName(lib);
        try (InputStream in = NativeLoader.class.getResourceAsStream(res)) {
            if (in == null) throw new UnsatisfiedLinkError("no native for " + platform());

            Path dir = Files.createTempDirectory("nat-");
            dir.toFile().deleteOnExit();
            Path out = dir.resolve(mapName(lib));
            Files.copy(in, out, StandardCopyOption.REPLACING_EXISTING);
            out.toFile().deleteOnExit();

            System.load(out.toAbsolutePath().toString());   // ★ 절대 경로여야 합니다
        } catch (IOException e) {
            throw new UnsatisfiedLinkError("extract failed: " + e);
        }
    }

    private static String mapName(String lib) { return System.mapLibraryName(lib); }

    static String platform() {
        String os = System.getProperty("os.name").toLowerCase();
        String ar = System.getProperty("os.arch").toLowerCase();
        String o = os.contains("win") ? "windows"
                 : (os.contains("mac") || os.contains("darwin")) ? "darwin" : "linux";
        String a = switch (ar) {
            case "amd64", "x86_64" -> "x86-64";
            case "aarch64", "arm64" -> "aarch64";
            default -> ar;
        };
        return o + "-" + a;
    }
    private NativeLoader() {}
}
놓치기 쉬운 세 가지 — ① System.mapLibraryName("foo") 이 OS 규칙대로 이름을 만들어 줍니다(직접 문자열을 붙이지 마세요). ② 윈도우는 로드된 DLL 을 지울 수 없어 deleteOnExit 가 실패합니다 — 정상입니다. ③ 컨테이너의 /tmp 가 noexec 로 마운트되면 로드가 실패합니다 — -Djava.io.tmpdir 로 바꿀 수 있게 열어 두세요.

@RestController
class ImageController {
    private final ImageOps ops;
    private final BufferPool pool;

    ImageController(ImageOps ops, BufferPool pool) { this.ops = ops; this.pool = pool; }

    @PostMapping(value = "/grayscale", produces = MediaType.IMAGE_PNG_VALUE)
    byte[] grayscale(@RequestParam int w, @RequestParam int h,
                     @RequestBody byte[] body) {
        ByteBuffer in = pool.borrow();
        try {
            in.clear(); in.put(body).flip();
            ByteBuffer out = ops.grayscale(in, w, h);
            byte[] r = new byte[out.remaining()];
            out.get(r);
            return r;
        } finally { pool.giveBack(in); }
    }
}

JNIEXPORT void JNICALL

                                        jobject src, jobject dst, jint w, jint h) {
    auto* s = static_cast<const uint8_t*>(env->GetDirectBufferAddress(src));
    auto* d = static_cast<uint8_t*>(env->GetDirectBufferAddress(dst));
    if (!s || !d) { LOGI("direct buffer required"); return; }
    const size_t n = (size_t)w * h;
    for (size_t i = 0; i < n; ++i)
        d[i] = (uint8_t)((s[i*4]*77 + s[i*4+1]*150 + s[i*4+2]*29) >> 8);
}

코틀린 object 안에 두는 이유 — 자바의 static { } 자리를 object 의 init 블록이 대신합니다. 싱글턴이므로 정확히 한 번 실행되고, 그래서 클래스가 아니라 jobject 가 두 번째 인자로 옵니다(object 의 메서드는 인스턴스 메서드입니다). @JvmStatic 을 붙이면 jclass 가 됩니다.

코루틴에서 안전하게 부르기

import kotlinx.coroutines.*
import java.nio.ByteBuffer

class ImageRepository(private val pool: BufferPool) {

    // ★ 네이티브 호출은 "블로킹"입니다 — 반드시 적절한 디스패처로 옮깁니다
    suspend fun grayscale(src: ByteBuffer, w: Int, h: Int): ByteBuffer =
        withContext(Dispatchers.Default) {          // CPU 작업 → Default
            val dst = pool.borrow(w * h)
            ImageOps.grayscale(src, dst, w, h)
            dst
        }

    // 파일·네트워크가 섞이면 IO
    suspend fun decodeFile(path: String): ByteBuffer =
        withContext(Dispatchers.IO) { ImageOps.decode(path) }
}

// ✗ 절대 하면 안 되는 것 — 메인 스레드에서 무거운 네이티브 호출
fun onClick() { ImageOps.grayscale(...) }     // ANR 로 앱이 죽습니다

// ★ 취소는 자동으로 전달되지 않습니다
//   코루틴을 cancel 해도 이미 시작된 C++ 루프는 끝까지 돕니다.
//   중간에 멈춰야 한다면 C++ 쪽에 취소 플래그를 두고 주기적으로 확인하게 합니다:
//     external fun requestCancel(handle: Long)
//     → C++ 루프가 std::atomic<bool> 을 매 프레임 확인
코루틴이 “가벼워서” 생기는 착각 — Dispatchers.Default 의 스레드 수는 코어 수로 제한됩니다. 네이티브 호출이 1초씩 블로킹하면 그 스레드는 그동안 아무것도 못 합니다. 긴 네이티브 작업이 여럿이면 전용 디스패처 (Executors.newFixedThreadPool(2).asCoroutineDispatcher())를 따로 두세요.

실전 — 하나의 C++ 코어를 JNI 와 FFM 으로 두 번 감쌉니다

마지막은 처음부터 끝까지 이어 봅니다. 이미지 리사이즈 C++ 코어 하나를 만들고, JNI + DirectByteBuffer 와 FFM 두 가지로 감싼 다음, JMH 로 순수 자바 구현까지 셋을 나란히 재고, Spring Boot 엔드포인트에 붙여 배포합니다.

① C++ 코어 — 자바를 전혀 모릅니다

// native/src/imageops.cpp
#include <cstdint>
#include <cstddef>

extern "C" {
#if defined(_WIN32)
  #define API __declspec(dllexport)
#else
  #define API __attribute__((visibility("default")))
#endif

// RGBA8888 이미지를 이중선형 보간으로 리사이즈합니다.
// 반환: 0 성공, 음수 실패. 메모리는 호출자가 준비합니다 (소유권 이동 없음)
API int32_t resize_rgba(const uint8_t* src, int32_t sw, int32_t sh,
                        uint8_t* dst, int32_t dw, int32_t dh) {
    if (!src || !dst || sw <= 0 || sh <= 0 || dw <= 0 || dh <= 0) return -1;

    const float fx = static_cast<float>(sw) / dw;
    const float fy = static_cast<float>(sh) / dh;

    for (int32_t y = 0; y < dh; ++y) {
        const float sy = (y + 0.5f) * fy - 0.5f;
        int32_t y0 = (int32_t)(sy < 0 ? 0 : sy);
        int32_t y1 = (y0 + 1 < sh) ? y0 + 1 : y0;
        const float wy = sy - y0;

        for (int32_t x = 0; x < dw; ++x) {
            const float sx = (x + 0.5f) * fx - 0.5f;
            int32_t x0 = (int32_t)(sx < 0 ? 0 : sx);
            int32_t x1 = (x0 + 1 < sw) ? x0 + 1 : x0;
            const float wx = sx - x0;

            for (int c = 0; c < 4; ++c) {
                const float a = src[(y0*(size_t)sw + x0)*4 + c] * (1-wx)
                              + src[(y0*(size_t)sw + x1)*4 + c] * wx;
                const float b = src[(y1*(size_t)sw + x0)*4 + c] * (1-wx)
                              + src[(y1*(size_t)sw + x1)*4 + c] * wx;
                dst[(y*(size_t)dw + x)*4 + c] = (uint8_t)(a*(1-wy) + b*wy + 0.5f);
            }
        }
    }
    return 0;
}

API const char* imageops_version(void) { return "1.0.0"; }
}   // extern "C"
이 함수의 설계가 전부를 결정합니다 — 포인터 + 크기만 받고 메모리를 소유하지 않으며 에러를 코드로 돌려줍니다. 이 세 규칙 덕분에 JNI · FFM · 파이썬 · C# 어디에도 똑같이 붙습니다. 여기에 jni.h 를 한 줄이라도 넣는 순간 자바 전용 코드가 됩니다.

② 자바 인터페이스와 두 구현

public interface ImageOps extends AutoCloseable {
    /** src 는 direct ByteBuffer(RGBA), 결과도 direct ByteBuffer */
    ByteBuffer resize(ByteBuffer src, int sw, int sh, int dw, int dh);
    String version();
    @Override default void close() {}
}
// (a) JNI 구현 — DirectByteBuffer 로 복사 0
public final class JniImageOps implements ImageOps {
    static { NativeLoader.load("imageops_jni"); }

    private static native int  nResize(ByteBuffer src, int sw, int sh,
                                       ByteBuffer dst, int dw, int dh);
    private static native String nVersion();

    @Override public ByteBuffer resize(ByteBuffer src, int sw, int sh, int dw, int dh) {
        ByteBuffer dst = ByteBuffer.allocateDirect(dw * dh * 4)
                                   .order(ByteOrder.nativeOrder());
        int r = nResize(src, sw, sh, dst, dw, dh);
        if (r != 0) throw new IllegalStateException("resize failed: " + r);
        return dst;
    }
    @Override public String version() { return nVersion(); }
}
// JNI 껍데기 — 코어를 부르기만 합니다 (별도 .so 로 빌드)
#include <jni.h>
extern "C" int32_t resize_rgba(const uint8_t*, int32_t, int32_t,
                               uint8_t*, int32_t, int32_t);

extern "C" JNIEXPORT jint JNICALL
Java_com_example_JniImageOps_nResize(JNIEnv* env, jclass,
        jobject src, jint sw, jint sh, jobject dst, jint dw, jint dh) {
    auto* s = static_cast<const uint8_t*>(env->GetDirectBufferAddress(src));
    auto* d = static_cast<uint8_t*>(env->GetDirectBufferAddress(dst));
    if (!s || !d) return -2;
    if (env->GetDirectBufferCapacity(src) < (jlong)sw*sh*4) return -3;   // ★ 경계 검사
    if (env->GetDirectBufferCapacity(dst) < (jlong)dw*dh*4) return -4;
    return resize_rgba(s, sw, sh, d, dw, dh);
}
// (b) FFM 구현 — C++ 껍데기가 아예 없습니다
public final class FfmImageOps implements ImageOps {
    private final Arena arena = Arena.ofShared();
    private final MethodHandle resize;

    public FfmImageOps(Path lib) {
        Linker linker = Linker.nativeLinker();
        SymbolLookup look = SymbolLookup.libraryLookup(lib, arena);
        resize = linker.downcallHandle(
            look.find("resize_rgba").orElseThrow(),
            FunctionDescriptor.of(ValueLayout.JAVA_INT,
                ValueLayout.ADDRESS, ValueLayout.JAVA_INT, ValueLayout.JAVA_INT,
                ValueLayout.ADDRESS, ValueLayout.JAVA_INT, ValueLayout.JAVA_INT));
    }

    @Override public ByteBuffer resize(ByteBuffer src, int sw, int sh, int dw, int dh) {
        ByteBuffer dst = ByteBuffer.allocateDirect(dw * dh * 4)
                                   .order(ByteOrder.nativeOrder());
        // ★ direct ByteBuffer 는 MemorySegment 로 그대로 감쌉니다 — 복사 0
        MemorySegment s = MemorySegment.ofBuffer(src);
        MemorySegment d = MemorySegment.ofBuffer(dst);
        try {
            int r = (int) resize.invokeExact(s, sw, sh, d, dw, dh);
            if (r != 0) throw new IllegalStateException("resize failed: " + r);
            return dst;
        } catch (Throwable t) { throw new RuntimeException(t); }
    }
    @Override public String version() { return "1.0.0"; }
    @Override public void close() { arena.close(); }
}
차이를 눈으로 보세요 — (a)는 C++ 파일이 하나 더 필요하고 따로 빌드해 배포해야 합니다. (b)는 자바 파일 하나가 전부이고 libimageops 는 손대지 않은 순수 C 라이브러리 그대로입니다. 성능은 사실상 같습니다 — 그래서 Java 22+ 서버라면 (b)가 이깁니다.

@Configuration
class ImageConfig {
    @Bean(destroyMethod = "close")
    ImageOps imageOps(@Value("${app.native.impl:ffm}") String impl,
                      @Value("${app.native.lib:/opt/app/libimageops.so}") String lib) {

            case "jni" -> new JniImageOps();

            default    -> new PureJavaImageOps();       // 폴백 ★
        };
    }

@RestController
class ResizeController {
    private final ImageOps ops;
    private final BufferPool pool;
    ResizeController(ImageOps ops, BufferPool pool) { this.ops = ops; this.pool = pool; }

    @PostMapping(value = "/resize", consumes = MediaType.APPLICATION_OCTET_STREAM_VALUE)
    ResponseEntity<byte[]> resize(@RequestParam int sw, @RequestParam int sh,
                                  @RequestParam int dw, @RequestParam int dh,
                                  @RequestBody byte[] body) {
        if ((long) sw * sh * 4 != body.length)
            return ResponseEntity.badRequest().build();

        ByteBuffer in = pool.borrow(body.length);
        try {
            in.clear(); in.put(body).flip();
            ByteBuffer out = ops.resize(in, sw, sh, dw, dh);
            byte[] r = new byte[out.remaining()];
            out.get(r);
            return ResponseEntity.ok(r);
        } finally { pool.giveBack(in); }
    }

    @GetMapping("/health/native")
    String health() { return ops.version(); }   // 기동 직후 로드 실패를 잡습니다
}
