FX 02

첫 창 만들기

JavaFX 앱의 뼈대는 항상 같습니다 — Stage(창) → Scene(장면) → Node(요소).

public class MainApp extends Application {

    @Override
    public void start(Stage stage) {           // 창 하나가 Stage
        Label label = new Label("안녕하세요");
        label.setStyle("-fx-font-size: 20px;");

        Button button = new Button("클릭");
        button.setOnAction(e -> label.setText("클릭했습니다!"));

        VBox root = new VBox(12, label, button);   // 세로 배치, 간격 12
        root.setAlignment(Pos.CENTER);
        root.setPadding(new Insets(20));

        Scene scene = new Scene(root, 400, 300);   // 장면 + 크기
        scene.getStylesheets().add(
                getClass().getResource("/style.css").toExternalForm());

        stage.setTitle("내 첫 프로그램");
        stage.setScene(scene);
        stage.setMinWidth(320);
        stage.getIcons().add(new Image(
                getClass().getResourceAsStream("/icon.png")));
        stage.show();
    }

    @Override
    public void stop() {                        // 종료 시 정리
        // 스레드 종료, 설정 저장 등
    }

    public static void main(String[] args) {
        launch(args);
    }
}

레이아웃

화면 배치는 레이아웃 컨테이너를 겹쳐서 만듭니다. 6개만 알면 대부분 됩니다.

레이아웃배치 방식언제
VBox세로로 쌓기가장 많이 씁니다. 폼·목록
HBox가로로 나열버튼 줄·툴바
BorderPane상·하·좌·우·중앙 5칸메인 창의 기본 틀
GridPane행·열 격자입력 폼 (라벨 + 입력창)
StackPane겹쳐 쌓기로딩 오버레이·배경
AnchorPane모서리에 고정절대 위치가 필요할 때
FlowPane / TilePane넘치면 다음 줄로태그·썸네일 목록
SplitPane드래그로 크기 조절탐색기형 화면

StringProperty status = new SimpleStringProperty("대기 중");

IntegerProperty count = new SimpleIntegerProperty(0);

// 조건부 — 입력이 비면 버튼 비활성화

        nameField.textProperty().isEmpty()
            .or(emailField.textProperty().isEmpty()));

// 진행률 연결
progressBar.progressProperty().bind(task.progressProperty());

// 크기 연결

바인딩을 쓰면 “값 바뀔 때마다 화면 갱신 코드”가 사라집니다. 모델의 값만 바꾸면 UI 는 알아서 따라옵니다. JavaFX 를 쓸 거라면 이 개념을 꼭 익히세요.

FXML · Scene Builder

화면을 코드가 아니라 XML 로 분리합니다. 드래그&드롭 도구로 그릴 수 있습니다.

      spacing="10" prefWidth="600" prefHeight="400">

      <TableColumn fx:id="nameCol" text="고객명" prefWidth="120"/>

컨트롤러

public class OrderController {

    @FXML private TextField searchField;      // fx:id 와 이름이 같아야 함
    @FXML private TableView<OrderRow> orderTable;
    @FXML private TableColumn<OrderRow, String> noCol;
    @FXML private TableColumn<OrderRow, Integer> amountCol;
    @FXML private Label statusLabel;

    private final ObservableList<OrderRow> data = FXCollections.observableArrayList();

    @FXML
    public void initialize() {                 // FXML 로드 후 자동 호출 ★
        noCol.setCellValueFactory(new PropertyValueFactory<>("orderNo"));
        amountCol.setCellValueFactory(new PropertyValueFactory<>("amount"));
        orderTable.setItems(data);
        load();
    }

    @FXML
    private void onSearch() {                  // onAction="#onSearch"
        String keyword = searchField.getText();
        statusLabel.setText(keyword + " 검색 중...");
        load();
    }
}

// 로드

TableView · 데이터 바인딩

업무용 프로그램의 90%는 결국 표입니다. 여기만 잘 다루면 됩니다.

// 1) 모델 — JavaFX 프로퍼티를 쓰면 값 변경이 표에 자동 반영됩니다
public class OrderRow {
    private final StringProperty orderNo = new SimpleStringProperty();
    private final StringProperty customerName = new SimpleStringProperty();
    private final IntegerProperty amount = new SimpleIntegerProperty();
    private final ObjectProperty<LocalDate> orderDate = new SimpleObjectProperty<>();

    public OrderRow(String no, String name, int amount, LocalDate date) {
        this.orderNo.set(no);
        this.customerName.set(name);
        this.amount.set(amount);
        this.orderDate.set(date);
    }

    public StringProperty orderNoProperty() { return orderNo; }      // ★ 이 이름 규칙
    public StringProperty customerNameProperty() { return customerName; }
    public IntegerProperty amountProperty() { return amount; }
    public ObjectProperty<LocalDate> orderDateProperty() { return orderDate; }

    public String getOrderNo() { return orderNo.get(); }
    public int getAmount() { return amount.get(); }
    public void setAmount(int v) { amount.set(v); }    // 값을 바꾸면 표가 자동 갱신 ✅
}

// 2) 컬럼 연결
TableView<OrderRow> table = new TableView<>();

TableColumn<OrderRow, String> noCol = new TableColumn<>("주문번호");
noCol.setCellValueFactory(c -> c.getValue().orderNoProperty());

TableColumn<OrderRow, Number> amountCol = new TableColumn<>("금액");
amountCol.setCellValueFactory(c -> c.getValue().amountProperty());

// 3) 셀 서식 — 금액에 콤마, 조건부 색상
amountCol.setCellFactory(col -> new TableCell<>() {
    @Override
    protected void updateItem(Number value, boolean empty) {
        super.updateItem(value, empty);
        if (empty || value == null) {
            setText(null);
            setStyle("");
        } else {
            setText(String.format("%,d원", value.intValue()));
            setAlignment(Pos.CENTER_RIGHT);
            setStyle(value.intValue() >= 100000
                    ? "-fx-text-fill: #e74c3c; -fx-font-weight: bold;" : "");
        }
    }
});

// 4) 데이터 연결 + 정렬 + 필터
ObservableList<OrderRow> master = FXCollections.observableArrayList(rows);
FilteredList<OrderRow> filtered = new FilteredList<>(master, r -> true);
SortedList<OrderRow> sorted = new SortedList<>(filtered);
sorted.comparatorProperty().bind(table.comparatorProperty());
table.setItems(sorted);
table.getColumns().addAll(noCol, amountCol);

// 검색창과 연결 — 입력하면 즉시 필터링 ★
searchField.textProperty().addListener((obs, old, keyword) ->
    filtered.setPredicate(row ->
        keyword == null || keyword.isBlank()
            || row.getOrderNo().contains(keyword)
            || row.getCustomerName().contains(keyword)));

// 5) 편집 가능하게
table.setEditable(true);
amountCol.setCellFactory(TextFieldTableCell.forTableColumn(
        new NumberStringConverter()));
amountCol.setOnEditCommit(e ->
        e.getRowValue().setAmount(e.getNewValue().intValue()));

// 6) 우클릭 메뉴 · 더블클릭
ContextMenu menu = new ContextMenu(new MenuItem("삭제"));
table.setContextMenu(menu);
table.setRowFactory(tv -> {
    TableRow<OrderRow> row = new TableRow<>();
    row.setOnMouseClicked(e -> {
        if (e.getClickCount() == 2 && !row.isEmpty()) openDetail(row.getItem());
    });
    return row;
});

FX 09

백그라운드 작업

여기가 JavaFX 초보가 가장 많이 실수하는 곳입니다. 화면이 얼어붙는 원인이 전부 여기에 있습니다.

절대 규칙 — 모든 UI 조작은 JavaFX 애플리케이션 스레드에서만 해야 합니다.
· UI 스레드에서 오래 걸리는 작업(DB·파일·API)을 하면 화면이 멈춥니다.
· 반대로 다른 스레드에서 UI 를 건드리면 IllegalStateException: Not on FX application thread 가 납니다.
❌ 화면이 얼어붙습니다
button.setOnAction(e -> {
    List<Order> orders = api.fetchAll();   // 5초 걸림
    table.setItems(FXCollections.observableArrayList(orders));
});
// 5초 동안 창이 하얗게 되고 "응답 없음"
✅ Task 로 분리
Task<List<Order>> task = new Task<>() {
    @Override
    protected List<Order> call() {          // 백그라운드에서 실행
        return api.fetchAll();               // UI 를 건드리지 않는다
    }
};

task.setOnSucceeded(e ->                    // UI 스레드에서 자동 실행 ✅
        table.setItems(FXCollections.observableArrayList(task.getValue())));
task.setOnFailed(e -> showError(task.getException()));

new Thread(task).start();

진행률까지 표시하기

Task<Void> task = new Task<>() {
    @Override
    protected Void call() throws Exception {
        int total = files.size();
        for (int i = 0; i < total; i++) {
            if (isCancelled()) break;              // 취소 지원
            process(files.get(i));
            updateProgress(i + 1, total);          // 진행률
            updateMessage("처리 중: " + files.get(i).getName());
        }
        return null;
    }
};

progressBar.progressProperty().bind(task.progressProperty());
statusLabel.textProperty().bind(task.messageProperty());
cancelButton.setOnAction(e -> task.cancel());

task.setOnSucceeded(e -> {
    progressBar.progressProperty().unbind();       // ★ 바인딩 해제
    statusLabel.textProperty().unbind();
    statusLabel.setText("완료");
});

// 스레드 풀로 관리 (가상 스레드도 가능)
ExecutorService pool = Executors.newVirtualThreadPerTaskExecutor();
pool.submit(task);

FX 10

실행 파일로 배포

사용자가 자바를 설치하지 않아도 돌아가는 프로그램으로 만듭니다.

jpackage — JDK 내장 (권장)

# 1) 실행 가능한 jar 를 먼저 만든다
./gradlew build

# 2) 런타임을 포함한 설치 파일 생성
jpackage \
  --name "주문관리" \
  --app-version 1.0.0 \
  --input build/libs \
  --main-jar myapp-1.0.0.jar \
  --main-class com.myapp.MainApp \
  --icon src/main/resources/icon.ico \
  --type msi \
  --win-shortcut --win-menu \
  --java-options "-Xmx1g -Dfile.encoding=UTF-8" \
  --dest dist

# --type 선택
#   윈도우: msi, exe      맥: dmg, pkg      리눅스: deb, rpm
#   app-image : 설치 없이 폴더째 실행 (USB 배포에 편함)
// Gradle 플러그인으로 자동화 (org.beryx.jlink)
jlink {
    options = listOf("--strip-debug", "--no-header-files", "--no-man-pages")
    launcher {
        name = "주문관리"
        jvmArgs = listOf("-Dfile.encoding=UTF-8")
    }
    jpackage {
        icon = "src/main/resources/icon.ico"
        installerType = "msi"
        installerOptions = listOf("--win-shortcut", "--win-menu")
    }
}
// ./gradlew jpackage

배포 체크리스트

  • 인코딩 — -Dfile.encoding=UTF-8 을 꼭 넣으세요. 안 넣으면 윈도우에서 한글이 깨집니다
  • 아이콘 — 윈도우는 .ico, 맥은 .icns
  • 설정 파일 위치 — 설치 폴더는 쓰기 권한이 없을 수 있습니다. 사용자 설정은 System.getProperty("user.home") 아래에 저장하세요
  • DB 접속 정보를 프로그램에 넣지 마세요 — 디컴파일하면 다 보입니다. 서버 API 를 거치게 만드세요
  • 자동 업데이트가 필요하면 시작 시 버전을 확인해 안내하는 정도로 시작
  • 로그를 파일로 남기세요 — 사용자 PC 에서 무슨 일이 났는지 알 방법이 그것뿐입니다
// 설정 저장 위치

Files.createDirectories(configDir);
Path config = configDir.resolve("config.properties");
