TOOL 01

Lombok — 보일러플레이트 제거

자바의 “장황함” 절반을 없애 줍니다. 다만 아무 데나 쓰면 사고가 납니다.

compileOnly("org.projectlombok:lombok") · annotationProcessor("org.projectlombok:lombok")
테스트에서도 쓰려면 testCompileOnly · testAnnotationProcessor 도 함께. IDE 에는 Lombok 플러그인 + Annotation Processing 활성화가 필요합니다.

Lombok 사고 3대장
① 엔티티에 @Data·@ToString 금지 — 연관 엔티티를 따라가며 무한 순환과 N+1이 납니다. 꼭 쓰려면 @ToString(exclude = "orders").
② 엔티티에 @Setter 금지 — 아무 데서나 상태가 바뀌면 도메인 규칙을 지킬 수 없습니다.
③ @EqualsAndHashCode 전체 필드 사용 금지 — 엔티티는 ID 기준으로만. @EqualsAndHashCode(of = "id")

MDC — 요청 추적 ID 심기 ★

@Component
public class MdcFilter extends OncePerRequestFilter {

    @Override
    protected void doFilterInternal(HttpServletRequest req, HttpServletResponse res,
                                    FilterChain chain) throws ServletException, IOException {
        String traceId = Optional.ofNullable(req.getHeader("X-Trace-Id"))
                .orElse(UUID.randomUUID().toString().substring(0, 8));
        try {
            MDC.put("traceId", traceId);
            res.setHeader("X-Trace-Id", traceId);       // 응답에도 실어 보냄
            chain.doFilter(req, res);
        } finally {
            MDC.clear();          // ★ 반드시 정리 (스레드 풀은 재사용됩니다)
        }
    }
}
// 패턴에 %X{traceId} 를 넣으면 모든 로그에 자동으로 찍힙니다
// → 사용자 문의가 오면 traceId 하나로 전 구간 검색 가능

@Service
@Slf4j
public class ExcelReader {

    public List<OrderRow> read(InputStream in) throws IOException {
        List<OrderRow> rows = new ArrayList<>();

        try (Workbook wb = WorkbookFactory.create(in)) {     // xls/xlsx 자동 판별
            Sheet sheet = wb.getSheetAt(0);                   // 또는 wb.getSheet("주문")
            DataFormatter fmt = new DataFormatter();          // 셀을 '보이는 대로' 문자열화

            for (int r = 1; r <= sheet.getLastRowNum(); r++) {   // 1행부터 (0은 헤더)
                Row row = sheet.getRow(r);
                if (row == null || isEmpty(row, fmt)) continue;   // 빈 행 건너뛰기

                try {
                    rows.add(new OrderRow(
                            getString(row, 0, fmt),
                            getInt(row, 1),
                            getDate(row, 2)
                    ));
                } catch (Exception e) {
                    log.warn("{}행 파싱 실패: {}", r + 1, e.getMessage());
                    // 실패한 행 번호를 모아 사용자에게 돌려주면 훨씬 친절합니다
                }
            }
        }
        return rows;
    }

    // ── 셀 읽기 유틸 — 이게 실무의 8할입니다 ──────────────────
    private String getString(Row row, int idx, DataFormatter fmt) {
        Cell cell = row.getCell(idx, Row.MissingCellPolicy.RETURN_BLANK_AS_NULL);
        return cell == null ? null : fmt.formatCellValue(cell).trim();
    }

    private Integer getInt(Row row, int idx) {
        Cell cell = row.getCell(idx);
        if (cell == null) return null;
        return switch (cell.getCellType()) {
            case NUMERIC -> (int) cell.getNumericCellValue();
            case STRING  -> {
                String s = cell.getStringCellValue().replaceAll("[^0-9-]", "");
                yield s.isBlank() ? null : Integer.parseInt(s);
            }
            case FORMULA -> (int) cell.getNumericCellValue();
            default      -> null;
        };
    }

    private LocalDate getDate(Row row, int idx) {
        Cell cell = row.getCell(idx);
        if (cell == null) return null;
        // 엑셀 날짜는 내부적으로 '숫자'입니다
        if (cell.getCellType() == CellType.NUMERIC && DateUtil.isCellDateFormatted(cell)) {
            return cell.getLocalDateTimeCellValue().toLocalDate();
        }
        String s = cell.getStringCellValue().trim();
        return s.isBlank() ? null : LocalDate.parse(s.replace("/", "-"));
    }

    private boolean isEmpty(Row row, DataFormatter fmt) {
        for (int c = row.getFirstCellNum(); c < row.getLastCellNum(); c++) {
            Cell cell = row.getCell(c);
            if (cell != null && !fmt.formatCellValue(cell).isBlank()) return false;
        }
        return true;
    }
}

실무에서 반드시 밟는 함정

  • 빈 셀은 null 이 아니라 객체 자체가 없습니다 — MissingCellPolicy 를 지정하세요
  • 숫자가 문자열로 들어있는 경우가 흔합니다 — DataFormatter 로 통일해서 읽으세요

“조회 결과를 엑셀로 받고 싶어요” — 이것도 단골 요청입니다.

@Service
public class ExcelWriter {

    private static final String[] HEADERS = {"주문번호", "고객명", "금액", "상태", "주문일"};

    public void write(List<OrderRes> orders, OutputStream out) throws IOException {
        try (Workbook wb = new XSSFWorkbook()) {
            Sheet sheet = wb.createSheet("주문목록");

            // ── 스타일은 반드시 '한 번만' 만들어 재사용 ★ ──────────
            CellStyle headerStyle = headerStyle(wb);
            CellStyle moneyStyle = moneyStyle(wb);
            CellStyle dateStyle = dateStyle(wb);

            // 헤더
            Row header = sheet.createRow(0);
            for (int i = 0; i < HEADERS.length; i++) {
                Cell c = header.createCell(i);
                c.setCellValue(HEADERS[i]);
                c.setCellStyle(headerStyle);
            }

            // 데이터
            int r = 1;
            for (OrderRes o : orders) {
                Row row = sheet.createRow(r++);
                row.createCell(0).setCellValue(o.orderNo());
                row.createCell(1).setCellValue(o.customerName());

                Cell amount = row.createCell(2);
                amount.setCellValue(o.amount());
                amount.setCellStyle(moneyStyle);

                row.createCell(3).setCellValue(o.status().label());

                Cell date = row.createCell(4);
                date.setCellValue(o.createdAt());
                date.setCellStyle(dateStyle);
            }

            // 마무리
            for (int i = 0; i < HEADERS.length; i++) sheet.autoSizeColumn(i);
            sheet.createFreezePane(0, 1);                     // 헤더 고정
            sheet.setAutoFilter(new CellRangeAddress(0, r - 1, 0, HEADERS.length - 1));

            wb.write(out);
        }
    }

    private CellStyle headerStyle(Workbook wb) {
        CellStyle s = wb.createCellStyle();
        Font f = wb.createFont();
        f.setBold(true);
        f.setColor(IndexedColors.WHITE.getIndex());
        s.setFont(f);
        s.setFillForegroundColor(IndexedColors.DARK_BLUE.getIndex());
        s.setFillPattern(FillPatternType.SOLID_FOREGROUND);
        s.setAlignment(HorizontalAlignment.CENTER);
        s.setBorderBottom(BorderStyle.THIN);
        return s;
    }

    private CellStyle moneyStyle(Workbook wb) {
        CellStyle s = wb.createCellStyle();
        s.setDataFormat(wb.createDataFormat().getFormat("#,##0"));
        return s;
    }

    private CellStyle dateStyle(Workbook wb) {
        CellStyle s = wb.createCellStyle();
        s.setDataFormat(wb.createDataFormat().getFormat("yyyy-mm-dd hh:mm"));
        return s;
    }
}

        "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");
    res.setHeader(HttpHeaders.CONTENT_DISPOSITION,
            "attachment; filename=\"" + encoded + "\"; filename*=UTF-8''" + encoded);

    // 파일을 만들지 않고 응답 스트림에 바로 씁니다 (메모리·디스크 절약)
    try (OutputStream out = res.getOutputStream()) {
        excelWriter.write(orderService.searchAll(cond), out);
        out.flush();
    }
}
한글 파일명이 깨지는 이유 — HTTP 헤더는 기본이 ASCII 입니다. filename*=UTF-8'' 형식을 같이 넣어야 브라우저마다 다르게 동작하는 문제를 피할 수 있습니다.

public void writeLarge(Stream<OrderRes> orders, OutputStream out) throws IOException {

public void readLarge(InputStream in, Consumer<List<String>> rowHandler)
        throws Exception {

    try (OPCPackage pkg = OPCPackage.open(in)) {
        XSSFReader reader = new XSSFReader(pkg);
        SharedStrings sst = reader.getSharedStringsTable();
        StylesTable styles = reader.getStylesTable();

        var handler = new XSSFSheetXMLHandler(styles, sst,
                new XSSFSheetXMLHandler.SheetContentsHandler() {
                    private final List<String> row = new ArrayList<>();

                    @Override public void startRow(int rowNum) { row.clear(); }
                    @Override public void cell(String ref, String value,
                                               XSSFComment c) { row.add(value); }
                    @Override public void endRow(int rowNum) {
                        if (rowNum > 0) rowHandler.accept(List.copyOf(row));
                    }
                }, false);

        XMLReader parser = XMLHelper.newXMLReader();
        parser.setContentHandler(handler);

        Iterator<InputStream> sheets = reader.getSheetsData();
        while (sheets.hasNext()) {
            try (InputStream sheet = sheets.next()) {
                parser.parse(new InputSource(sheet));
            }
        }
    }
}

XMLInputFactory factory = XMLInputFactory.newInstance();

OrderRow current = null;

while (r.hasNext()) {
    switch (r.next()) {
        case XMLStreamConstants.START_ELEMENT -> {
            tag = r.getLocalName();
            if ("order".equals(tag)) {
                current = new OrderRow();
                current.setNo(r.getAttributeValue(null, "no"));
            }
        }
        case XMLStreamConstants.CHARACTERS -> {
            if (current != null && !r.isWhiteSpace()) {
                String v = r.getText().trim();
                switch (tag) {
                    case "customerName" -> current.setName(v);
                    case "amount" -> current.setAmount(Integer.parseInt(v));
                    default -> { }
                }
            }
        }
        case XMLStreamConstants.END_ELEMENT -> {
            if ("order".equals(r.getLocalName()) && current != null) {
                handle(current);            // 한 건 완성 → 즉시 처리
                current = null;
            }
            tag = null;
        }
        default -> { }
    }
}
r.close();
