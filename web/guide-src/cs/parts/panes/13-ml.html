import numpy as np

# 세 축을 그대로 코드로 옮기면 '학습'이 된다
def predict(X, w):            # ① 가설 공간: 선형 함수만 허용
    return X @ w

def loss(X, y, w):            # ② 손실: 제곱 오차
    r = predict(X, w) - y
    return np.mean(r ** 2)

def fit(X, y, lr=0.1, steps=500):
    w = np.zeros(X.shape[1])
    for _ in range(steps):    # ③ 최적화: 기울기 반대 방향으로 조금씩
        g = 2 * X.T @ (predict(X, w) - y) / len(y)
        w -= lr * g
    return w

# 바뀌는 것은 predict 의 모양과 loss 의 식뿐이다.
“모델을 바꾼다”는 말의 정확한 뜻 — 대부분 가설 공간을 바꾸는 것입니다. 성능이 안 나올 때 “데이터가 부족한가 · H 가 좁은가 · 최적화가 실패했나” 세 축 중 어디인지 먼저 판정하면 헛수고가 크게 줄어듭니다.

def kfold_scores(X, y, fit_fn, score_fn, k=5, seed=0):

선형 회귀부터 — 한 번에 푸는 답과 굴러 내려가는 답

선형 회귀는 해석해가 존재하는 유일한 자리입니다. 공식으로 한 번에 푸는 길과 경사하강으로 조금씩 내려가는 길이 같은 점에 도착한다는 사실을 확인하고 나면, 그 뒤 모든 모델이 왜 경사하강만 쓰는지도 보입니다.

모델과 비용 함수

ŷ = w0 + w1x1 + … + wdxd = Xw 비용   J(w) = (1/n) ‖ Xw − y ‖²  — 오차를 제곱해서 더한 뒤 평균낸 값입니다. 
# 왜 하필 '제곱' 인가
#  ① 부호를 없애야 한다 (+3 과 -3 이 상쇄되면 안 됨)
#  ② 절댓값보다 미분이 매끄럽다 (0 에서 꺾이지 않음)
#  ③ 큰 오차에 더 큰 벌점 → 크게 틀린 것부터 고친다
#  ④ 오차가 정규분포라고 가정하면 제곱 오차 최소화 = 최대가능도 추정

# ③ 의 부작용: 이상치 하나가 답을 통째로 끌고 간다
#    → 그럴 땐 Huber 손실 (작을 땐 제곱, 클 땐 절댓값)
손실이 모델의 성격을 정합니다 — 제곱 오차를 최소화하면 조건부 평균을, 절댓값 오차를 최소화하면 조건부 중앙값을 예측하게 됩니다. “이상치에 흔들리지 않는 모델”을 원하면 모델이 아니라 손실을 바꾸는 것이 정답인 경우가 많습니다.

해석해 — 정규방정식

∇J = 0  ⟹  w* = (XᵀX)−1 Xᵀy 유도와 기하학적 의미(직교 투영)는 ➗ 선형대수 탭 15 최소제곱법에 있습니다. 
import numpy as np

X = np.c_[np.ones(len(x)), x]          # 절편을 위한 1 열 추가

w_exact = np.linalg.lstsq(X, y, rcond=None)[0]   # ★ 이렇게 푼다
w_bad   = np.linalg.inv(X.T @ X) @ X.T @ y       # 교과서 공식 그대로 ✗

# inv 를 직접 부르면 안 되는 이유
#  · XᵀX 는 조건수가 X 의 '제곱' — 오차가 제곱으로 증폭된다
#  · lstsq 는 QR/SVD 로 XᵀX 를 만들지 않고 푼다
#  → 수치 안정성은 ➗ 선형대수 탭 13 참고
해석해가 있는데 왜 경사하강을 배우나 — (XᵀX)⁻¹ 계산이 특성 수 d 에 대해 O(d³) 이기 때문입니다. d 가 수만이면 불가능하고, 로지스틱 회귀부터는 해석해가 아예 존재하지 않습니다.

핵심 개념 — 경사하강 — 왜 기울기의 반대인가 🟢 기초

기울기는 가장 가파르게 올라가는 방향입니다 — 그래서 내려가려면 반대로 가고, 얼마나 갈지를 정하는 것이 학습률입니다.

기울기 유도와 구현

∂J/∂w = (2/n) Xᵀ(Xw − y)    w ← w − α · ∂J/∂w “오차를 각 특성에 되돌려 곱한 것” — 그 특성이 오차에 얼마나 기여했는지입니다. 
def gd(X, y, lr=0.05, steps=2000):
    w = np.zeros(X.shape[1]); hist = []
    for t in range(steps):
        err = X @ w - y                 # (n,)
        g   = 2 * X.T @ err / len(y)    # (d,)
        w  -= lr * g
        hist.append(np.mean(err ** 2))
    return w, hist

w_gd, hist = gd(X, y)
np.allclose(w_gd, w_exact, atol=1e-3)   # True — 같은 답에 도착한다

# 비용이 줄어드는지 반드시 찍어 볼 것.
# 늘어나면 → lr 을 1/3 로.  들쭉날쭉하면 → 스케일링을 안 했다.
선형 회귀의 비용 함수는 볼록(convex)합니다 — 국소 최소가 곧 전역 최소이고, 학습률만 적당하면 시작점과 무관하게 같은 답에 갑니다. 신경망은 볼록하지 않아 이 보장이 사라집니다(09절).

“스팸인가 아닌가”에 회귀를 그대로 쓰면 예측이 1.7 이나 −0.3 이 나옵니다. 출력을 0~1 로 눌러 담고, 손실을 확률에 맞게 바꾸는 두 가지 수정만으로 회귀가 분류기가 됩니다. 그 두 수정이 왜 필요한지가 이 절의 전부입니다.

로지스틱 회귀는 선형 점수 z 를 확률로 번역할 뿐, 경계 자체는 여전히 직선(초평면)입니다.

시그모이드와 로그 오즈

p = σ(z) = 1/(1+e−z)  ⟺  z = log( p / (1−p) ) 오즈 = 일어날 확률 / 안 일어날 확률. 그 로그가 곧 선형 점수 z 입니다. 그래서 “로지스틱 회귀는 로그 오즈에 대한 선형 회귀”라고 부릅니다. 
import numpy as np

def sigmoid(z):
    # 그대로 exp(-z) 를 쓰면 z 가 -800 일 때 오버플로가 난다
    out = np.empty_like(z, dtype=float)
    pos, neg = z >= 0, z < 0
    out[pos] = 1 / (1 + np.exp(-z[pos]))
    ez = np.exp(z[neg])
    out[neg] = ez / (1 + ez)            # 수학적으로 동일 · 수치적으로 안전
    return out

# 계수 해석
#  w_j = 1.1  →  x_j 가 1 늘면 오즈가 e^1.1 ≈ 3배
#  선형 회귀와 달리 "확률이 얼마 늘어난다"가 아니라 "오즈가 몇 배"다
계수를 읽을 수 있다는 것이 로지스틱 회귀가 의료·금융에서 아직 쓰이는 이유입니다. exp(w) 가 오즈비이고, 규제 기관에 설명할 수 있는 숫자입니다.

교차 엔트로피 — 확률에 맞는 손실

L = −[ y log p + (1−y) log(1−p) ] 정답이 1 인데 p=0.01 이면 −log 0.01 = 4.6 — 확신에 찬 오답에 큰 벌점을 줍니다. 엔트로피 자체의 정의는 ➗ 선형대수 · 확률 탭 14에 있습니다. 
def bce(y, p, eps=1e-12):
    p = np.clip(p, eps, 1 - eps)        # log(0) = -inf 방지
    return -np.mean(y * np.log(p) + (1 - y) * np.log(1 - p))

def softmax(Z):

복잡도를 키우면 편향은 줄고 분산은 늡니다 — 정규화는 이 저울을 왼쪽으로 되미는 손잡이입니다.

분해식이 말하는 것

E[(y − ĥ(x))²] = 편향[ĥ]² + 분산[ĥ] + σ² 편향 = 평균적으로 얼마나 빗나가는가 (모델이 너무 단순해서) · 분산 = 데이터가 조금 바뀌면 답이 얼마나 흔들리는가 · σ² = 줄일 수 없는 잡음 
import numpy as np
rng = np.random.default_rng(0)

def experiment(deg, trials=200, n=25):
    preds = []
    for _ in range(trials):
        x = rng.uniform(-3, 3, n)
        y = np.sin(x) + rng.normal(0, 0.3, n)      # 참 함수 + 잡음
        w = np.polyfit(x, y, deg)
        preds.append(np.polyval(w, 1.0))           # x=1 에서의 예측
    preds = np.array(preds)
    bias2 = (preds.mean() - np.sin(1.0)) ** 2
    var   = preds.var()
    return bias2, var, bias2 + var

for d in (1, 3, 9, 15):
    print(d, [round(v, 3) for v in experiment(d)])

# 릿지는 해석해가 그대로 있다 — 대각선에 λ 를 더할 뿐
def ridge(X, y, lam):
    d = X.shape[1]
    return np.linalg.solve(X.T @ X + lam * np.eye(d), X.T @ y)

#  → 핵심 연산이 soft-thresholding:
def soft_threshold(w, a):
    return np.sign(w) * np.maximum(np.abs(w) - a, 0)   # 작은 계수를 '0 으로 밀어붙임'

import numpy as np

def gini(y):
    _, cnt = np.unique(y, return_counts=True)
    p = cnt / len(y)
    return 1 - (p ** 2).sum()

def best_split(X, y):
    base, best = gini(y), None
    for j in range(X.shape[1]):                 # 모든 특성에 대해
        for t in np.unique(X[:, j]):            # 모든 임계값 후보를
            L = X[:, j] <= t
            if L.sum() == 0 or L.sum() == len(y):
                continue
            child = (L.mean() * gini(y[L]) +
                     (1 - L.mean()) * gini(y[~L]))
            gain = base - child                 # 정보 이득
            if best is None or gain > best[0]:
                best = (gain, j, t)
    return best      # 탐욕적 — 지금 한 번의 이득만 본다

# 반반 섞인 잎:            gini = 0.5

정답 라벨이 없을 때 할 수 있는 일은 “비슷한 것끼리 묶기”와 “중요한 축만 남기기” 두 가지입니다. 둘 다 “무엇이 비슷한가”를 사람이 미리 정해야 한다는 함정을 안고 있습니다.

k-means 는 할당과 이동을 번갈아 하는 좌표 하강입니다 — 그래서 수렴은 보장되지만 전역 최적은 보장되지 않습니다.

직접 짜 보는 k-means

def kmeans(X, k=3, iters=100, seed=0):
    rng = np.random.default_rng(seed)
    C = X[rng.choice(len(X), k, replace=False)]       # k-means++ 가 더 좋다
    for _ in range(iters):
        d = ((X[:, None, :] - C[None, :, :]) ** 2).sum(-1)
        lab = d.argmin(1)                             # ② 할당
        newC = np.stack([X[lab == j].mean(0) if (lab == j).any() else C[j]
                         for j in range(k)])          # ③ 이동
        if np.allclose(newC, C):
            break
        C = newC
    return C, lab

# 정하고 들어가야 하는 것
#  · k : 몇 개로 나눌지 (데이터는 안 알려 준다)
#  · 거리 : 유클리드 → 구형 군집만 잘 찾는다

import numpy as np
rng = np.random.default_rng(0)

def init(d_in, d_h, d_out):
    # He 초기화 — ReLU 용. 이유는 08절에서
    W1 = rng.normal(0, np.sqrt(2 / d_in), (d_in, d_h))
    W2 = rng.normal(0, np.sqrt(2 / d_h),  (d_h, d_out))
    return [W1, np.zeros(d_h), W2, np.zeros(d_out)]

def forward(P, X):
    W1, b1, W2, b2 = P
    Z1 = X @ W1 + b1
    H  = np.maximum(0, Z1)              # ReLU
    Z2 = H @ W2 + b2
    P2 = np.exp(Z2 - Z2.max(1, keepdims=True))
    P2 /= P2.sum(1, keepdims=True)      # softmax
    return Z1, H, P2

def backward(P, X, y, cache):           # 유도는 08절
    W1, b1, W2, b2 = P
    Z1, H, Pr = cache
    n = len(y)
    dZ2 = Pr.copy(); dZ2[np.arange(n), y] -= 1; dZ2 /= n
    dW2, db2 = H.T @ dZ2, dZ2.sum(0)
    dH  = dZ2 @ W2.T
    dZ1 = dH * (Z1 > 0)                 # ReLU 도함수
    dW1, db1 = X.T @ dZ1, dZ1.sum(0)
    return [dW1, db1, dW2, db2]

P = init(X.shape[1], 64, 10)
for step in range(3000):
    cache = forward(P, X)
    G = backward(P, X, y, cache)
    for p, g in zip(P, G):
        p -= 0.1 * g                    # 09절에서 이 줄을 Adam 으로 바꿉니다
이 40줄이 프레임워크의 전부입니다 — PyTorch 가 해 주는 일은 backward 를 자동으로 만들어 주는 것(08절)과 GPU 로 돌리는 것(➗ 선형대수 탭 02 · 🖥️ 컴퓨터구조 탭 10)뿐입니다.
정리 — 신경망은 선형 변환과 비선형을 번갈아 쌓은 함수입니다. 비선형이 없으면 층이 하나로 접히고, 비선형의 기울기가 작으면 학습 신호가 사라집니다. ReLU 계열이 표준이 된 이유가 바로 그 기울기입니다.

역전파는 새로운 수학이 아니라 고등학교 연쇄 법칙을 “중간 결과를 재사용하는 순서”로 정리한 것입니다. 한 번 손으로 굴려 보면 자동 미분이 무엇을 자동화하는지도 함께 보입니다.

덧셈 노드는 기울기를 그대로 복사하고, 곱셈 노드는 상대편 입력값을 곱해 넘깁니다.

손으로 한 번 굴려 보기

∂L/∂w = (∂L/∂p)·(∂p/∂v)·(∂v/∂u)·(∂u/∂w) 오른쪽부터 하나씩 곱해 가는 것이 전부입니다. 각 항은 노드 하나의 국소 미분입니다. 
x, w, b, y = 2.0, 3.0, 1.0, 1.0

# --- 순전파 (값을 저장해 둔다. 역전파에서 필요하다) ---
u = w * x                 # 6.0
v = u + b                 # 7.0
p = 1 / (1 + np.exp(-v))  # 0.99909
L = -np.log(p)            # 0.00091

# --- 역전파 (오른쪽에서 왼쪽으로) ---
dL_dp = -1 / p                 # -1.00091
dp_dv = p * (1 - p)            #  0.00091
dL_dv = dL_dp * dp_dv          # -0.00091   ← 결국 (p - y) 다
dL_du = dL_dv * 1              # + 노드: 그대로 통과
dL_db = dL_dv * 1              # -0.00091
dL_dw = dL_du * x              # × 노드: 상대편 입력을 곱한다 → -0.00182
dL_dx = dL_du * w              # -0.00273

# 검산 — 수치 미분과 맞춰 본다
eps = 1e-6
num = (loss(w + eps) - loss(w - eps)) / (2 * eps)
assert abs(num - dL_dw) < 1e-7        # ✓ 구현 검증의 표준 방법
기울기 검산(gradient check)은 직접 구현할 때 반드시 하세요. 수치 미분은 느려서 학습에는 못 쓰지만, 한 번 맞춰 보는 용도로는 완벽합니다. 오차가 1e-7 보다 크면 어딘가 틀렸습니다.

class V:                      # 스칼라 자동 미분 (micrograd 식)
    def __init__(self, d, parents=(), back=lambda: None):
        self.d, self.grad = d, 0.0
        self._parents, self._back = parents, back

    def __add__(self, o):
        o = o if isinstance(o, V) else V(o)
        out = V(self.d + o.d, (self, o))
        def back():
            self.grad += out.grad           # 분배 (그대로)
            o.grad    += out.grad
        out._back = back; return out

    def __mul__(self, o):
        o = o if isinstance(o, V) else V(o)
        out = V(self.d * o.d, (self, o))
        def back():
            self.grad += o.d * out.grad     # 교환 (상대편 값)
            o.grad    += self.d * out.grad
        out._back = back; return out

    def backward(self):
        order, seen = [], set()
        def build(v):                       # 위상 정렬 (🔀 알고리즘 탭 06)
            if v in seen: return
            seen.add(v)
            for p in v._parents: build(p)
            order.append(v)
        build(self)
        self.grad = 1.0
        for v in reversed(order):           # 역순으로 국소 미분 적용
            v._back()

최적화 — Adam 이 고친 것을 하나씩 거슬러 올라가기

옵티마이저는 유행이 아니라 앞의 방법이 못 하던 것을 하나씩 메운 계보입니다. SGD → 모멘텀 → RMSProp → Adam → AdamW 로 오면서 각 단계가 정확히 무슨 문제를 고쳤는지를 알면 고를 때 헤매지 않습니다.

import numpy as np

def sgd(w, g, lr=0.1, st={}):
    return w - lr * g

def momentum(w, g, lr=0.1, beta=0.9, st={}):
    st['v'] = beta * st.get('v', 0) + g          # 관성 누적
    return w - lr * st['v']

def rmsprop(w, g, lr=0.01, beta=0.99, eps=1e-8, st={}):
    st['s'] = beta * st.get('s', 0) + (1 - beta) * g**2
    return w - lr * g / (np.sqrt(st['s']) + eps)  # 축마다 보폭 조절

def adam(w, g, lr=1e-3, b1=0.9, b2=0.999, eps=1e-8, st={}):
    st['t'] = st.get('t', 0) + 1
    st['m'] = b1 * st.get('m', 0) + (1 - b1) * g       # 1차 모멘트 (방향)
    st['v'] = b2 * st.get('v', 0) + (1 - b2) * g**2    # 2차 모멘트 (크기)
    mh = st['m'] / (1 - b1 ** st['t'])                 # ★ 편향 보정
    vh = st['v'] / (1 - b2 ** st['t'])                 #   초기 0 편향을 제거
    return w - lr * mh / (np.sqrt(vh) + eps)

def adamw(w, g, lr=1e-3, wd=0.01, **kw):
    return adam(w, g, lr=lr, **kw) - lr * wd * w       # 감쇠를 '따로' 뺀다
편향 보정이 필요한 이유 — m, v 를 0 에서 시작하면 초기 몇 스텝은 실제보다 작게 추정됩니다. 1−β^t 로 나누면 t 가 작을 때 크게 부풀려 이를 상쇄합니다.

스케줄 구현

import math

def lr_at(step, base=3e-4, warmup=2000, total=100_000, min_ratio=0.1):
    if step < warmup:
        return base * step / warmup                  # 선형 워밍업
    t = (step - warmup) / max(1, total - warmup)     # 0 → 1
    cos = 0.5 * (1 + math.cos(math.pi * t))          # 1 → 0
    return base * (min_ratio + (1 - min_ratio) * cos)

# 흔히 쓰는 다른 모양
#  · step decay      : 30 에폭마다 ×0.1 (CNN 고전)
#  · cosine restart  : 주기적으로 다시 올림 (SGDR) — 여러 최소점 탐색
#  · one-cycle       : 올렸다 크게 내림 — 짧은 학습에서 강력
#  · inverse sqrt    : 1/√step — 원조 트랜스포머 논문

# 학습률 탐색(LR range test): 아주 작은 값부터 지수적으로 올리며

# 왜 적은데 더 좋은가 — 가설 공간을 '옳은 방향으로' 좁혔기 때문이다.
#  · 이웃 픽셀끼리 관련 있다 (지역성)
#  · 왼쪽 위의 모서리 검출기는 오른쪽 아래에서도 유용하다 (공유)
# 이 두 가정이 이미지에서는 참이므로, 버린 가설들은 어차피 쓸모없었다.

def conv2d(X, K):                       # 개념 확인용 (실제로는 im2col + GEMM)
    kh, kw = K.shape
    H, W = X.shape[0] - kh + 1, X.shape[1] - kw + 1
    out = np.empty((H, W))
    for i in range(H):
        for j in range(W):
            out[i, j] = (X[i:i+kh, j:j+kw] * K).sum()
    return out

for x in sequence:

# 역전파에서 시간 축을 거슬러 올라가면 (BPTT)
#   dL/dh_0 = dL/dh_T · Π (W_h.T · diag(tanh'))
#   → W_h 의 최대 고유값이 1보다 작으면 지수적으로 0
#     1보다 크면 지수적으로 폭발
# 100 스텝 전의 단어가 지금 예측에 영향을 못 준다 = 장기 의존성 실패

def lstm_step(x, h, c, P):
    z = P['W'] @ np.concatenate([h, x]) + P['b']
    f, i, g, o = np.split(z, 4)
    f, i, o = sigmoid(f), sigmoid(i), sigmoid(o)
    g = np.tanh(g)
    c = f * c + i * g            # ★ 곱하고 '더한다' — 여기가 핵심
    h = o * np.tanh(c)
    return h, c

import numpy as np

def softmax(Z):
    Z = Z - Z.max(-1, keepdims=True)
    e = np.exp(Z); return e / e.sum(-1, keepdims=True)

def attention(X, Wq, Wk, Wv, mask=None):
    Q, K, V = X @ Wq, X @ Wk, X @ Wv        # (n, d_k)
    d_k = Q.shape[-1]
    S = Q @ K.T / np.sqrt(d_k)              # (n, n) 점수 행렬
    if mask is not None:
        S = np.where(mask, S, -np.inf)      # 인과 마스크 → softmax 후 0
    A = softmax(S)                          # 각 행의 합이 1
    return A @ V, A                         # (n, d_v), (n, n)

n = 6
causal = np.tril(np.ones((n, n), dtype=bool))   # 하삼각 = 과거만 허용

# 멀티헤드 — 같은 계산을 h 개로 쪼개서 병렬로
def multihead(X, Wq, Wk, Wv, Wo, h=8):
    n, d = X.shape
    Q = (X @ Wq).reshape(n, h, d // h).transpose(1, 0, 2)   # (h, n, d/h)
    K = (X @ Wk).reshape(n, h, d // h).transpose(1, 0, 2)
    V = (X @ Wv).reshape(n, h, d // h).transpose(1, 0, 2)
    S = Q @ K.transpose(0, 2, 1) / np.sqrt(d // h)
    O = softmax(S) @ V                                       # (h, n, d/h)
    return O.transpose(1, 0, 2).reshape(n, d) @ Wo           # 다시 합쳐 섞는다

# "누가 앞이었나"를 전혀 모른다. 그래서 위치 정보를 따로 넣어야 한다.

# ① 사인/코사인 (원조 트랜스포머) — 학습하지 않는 고정값
def sinusoidal(n, d):
    pos = np.arange(n)[:, None]
    i   = np.arange(0, d, 2)[None, :]
    ang = pos / (10000 ** (i / d))
    pe = np.zeros((n, d))
    pe[:, 0::2], pe[:, 1::2] = np.sin(ang), np.cos(ang)
    return pe                       # 임베딩에 '더한다'

# ② 학습형 절대 위치 — 그냥 표를 학습 (BERT · GPT-2)
#    최대 길이를 넘어가면 쓸 수 없다.

# ③ RoPE (회전 위치 인코딩) — 2026년 LLM 의 사실상 표준
#    Q, K 를 위치 각도만큼 '회전'시킨다. 그러면 내적이 상대 거리에만 의존한다.
def rope(x, pos, base=10000):
    d = x.shape[-1]
    theta = pos[:, None] / (base ** (np.arange(0, d, 2) / d))
    c, s = np.cos(theta), np.sin(theta)
    x1, x2 = x[..., 0::2], x[..., 1::2]
    return np.stack([x1 * c - x2 * s, x1 * s + x2 * c], -1).reshape(x.shape)

모델을 고르는 일은 숫자를 고르는 일입니다. 잘못된 지표를 고르면 잘못된 모델을 고르게 되고, 그보다 더 흔한 사고는 지표 자체가 오염되는 것(데이터 누수)입니다.

지표는 네 칸을 어떻게 요약할지의 선택입니다 — 요약 방식이 바뀌면 순위도 바뀝니다.

직접 계산해 보기

import numpy as np

def confusion(y, pred):
    TP = int(((pred == 1) & (y == 1)).sum())
    FP = int(((pred == 1) & (y == 0)).sum())
    FN = int(((pred == 0) & (y == 1)).sum())
    TN = int(((pred == 0) & (y == 0)).sum())
    return TP, FP, FN, TN

def metrics(y, pred):
    TP, FP, FN, TN = confusion(y, pred)
    P = TP / max(TP + FP, 1)
    R = TP / max(TP + FN, 1)
    return {
        "정밀도": P, "재현율": R,
        "F1": 2 * P * R / max(P + R, 1e-12),
        "특이도": TN / max(TN + FP, 1),
        "정확도": (TP + TN) / len(y),
        "균형정확도": 0.5 * (R + TN / max(TN + FP, 1)),   # 불균형에 강함
        "MCC": (TP*TN - FP*FN) / max(np.sqrt((TP+FP)*(TP+FN)*(TN+FP)*(TN+FN)), 1e-12),
    }

# MCC(매튜 상관계수)는 네 칸을 전부 쓰는 유일한 단일 지표 —
# 불균형 데이터에서 F1 보다 정직하다는 평이 많다.

def curves(y, score):
    order = np.argsort(-score)          # 점수 높은 순
    y = y[order]
    tp = np.cumsum(y)                   # 상위 k 개 중 진짜 양성 수
    fp = np.cumsum(1 - y)
    P = tp / (tp + fp)                  # 정밀도@k
    R = tp / y.sum()                    # 재현율@k
    FPR = fp / (1 - y).sum()
    return R, P, FPR

# ROC-AUC 의 확률적 의미:
#   무작위로 뽑은 양성 하나와 음성 하나에 대해
#   모델이 양성에 더 높은 점수를 줄 확률
#   → 0.5 = 동전 던지기 · 1.0 = 완벽한 순위

# 그래서 ROC-AUC 는 '순위 능력' 지표이지 '판정 능력' 지표가 아니다.
# 임계값을 정해 실제로 판정할 때의 성능은 PR 쪽이 잘 보여 준다.

def boot_ci(y, pred, metric, n=2000):
    rng = np.random.default_rng(0)
    vals = [metric(y[i], pred[i]) for i in
            (rng.integers(0, len(y), len(y)) for _ in range(n))]
    return np.percentile(vals, [2.5, 97.5])

# FGSM — 가장 단순한 공격. '손실을 늘리는 방향'으로 한 걸음.
def fgsm(model, x, y, eps=0.01):
    g = grad_of_loss_wrt_input(model, x, y)     # 입력에 대한 기울기
    return np.clip(x + eps * np.sign(g), 0, 1)  # 부호만 써서 균일하게 흔든다
