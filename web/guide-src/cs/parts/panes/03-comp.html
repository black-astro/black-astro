def turing(rules, tape, state="q0", halt="H", steps=10_000):
    """rules[(state, symbol)] = (write, move, next_state)"""
    tape = list(tape); pos = 0
    for _ in range(steps):
        if state == halt: break
        while pos < 0:  tape.insert(0, " "); pos = 0     # 테이프 확장
        while pos >= len(tape): tape.append(" ")
        sym = tape[pos]
        if (state, sym) not in rules: break
        write, move, state = rules[(state, sym)]
        tape[pos] = write
        pos += 1 if move == "R" else -1
    return "".join(tape).strip()

# 1 의 개수를 하나 늘리는 기계 (단항 증가)
rules = {
    ("q0", "1"): ("1", "R", "q0"),
    ("q0", " "): ("1", "R", "H"),
}
print(turing(rules, "111"))     # 1111

import ast, operator
OPS = {ast.Add: operator.add, ast.Sub: operator.sub,
       ast.Mult: operator.mul, ast.Div: operator.truediv}

def safe_eval(expr):
    def ev(node):
        if isinstance(node, ast.Constant) and isinstance(node.value, (int, float)):
            return node.value
        if isinstance(node, ast.BinOp) and type(node.op) in OPS:
            return OPS[type(node.op)](ev(node.left), ev(node.right))
        raise ValueError("허용되지 않은 식")
    return ev(ast.parse(expr, mode="eval").body)

정규 언어 — 정규식이 못 하는 일은 정해져 있습니다

“정규식으로 HTML 파싱하지 마라”는 관용구가 아니라 수학적 사실입니다. 정규식은 유한 상태 기계와 정확히 같은 힘을 가지고, 상태가 유한하면 중첩을 셀 수 없기 때문입니다.

정규식 = FSM

정규식      a(b|c)*d
            ↕ 완전히 동등
FSM         q0 --a--> q1 --d--> q2(수락)
                      ↺ b, c

정규 언어로 표현할 수 있는 것
  · 고정된 패턴          이메일 형태 · 날짜 형식
  · 반복                  a* · [0-9]+
  · 선택                  (cat|dog)

표현할 수 없는 것 ★
  · 괄호 짝 맞추기        ((()))
  · 중첩 태그             <div><div></div></div>
  · aⁿbⁿ (같은 개수)
  · 회문 (팰린드롬)

이유: '몇 개나 열렸는지' 세려면 무한한 상태가 필요하다.
      FSM 의 상태는 유한하다. 끝.
펌핑 보조정리가 이것을 증명하는 도구입니다 — “정규 언어라면 충분히 긴 문자열의 어떤 부분을 반복해도 여전히 그 언어에 속한다”. 괄호 짝에서 여는 괄호만 반복하면 깨지므로 정규 언어가 아닙니다.

정규식이 못 하는 괄호 짝 · 중첩 구조를 다루는 문법입니다. 모든 프로그래밍 언어의 문법이 여기 속하고, 그래서 파서가 필요합니다.

BNF 로 문법 적기

# 사칙연산 문법 (연산자 우선순위가 구조에 들어 있다)
expr   ::= expr '+' term
         | expr '-' term
         | term

term   ::= term '*' factor
         | term '/' factor
         | factor

import re
TOKEN = re.compile(r'\s*(\d+\.?\d*|[-+*/()])')

class Parser:
    def __init__(self, text):
        self.toks = TOKEN.findall(text)
        self.i = 0
    def peek(self):  return self.toks[self.i] if self.i < len(self.toks) else None
    def eat(self, t=None):
        tok = self.peek()
        if t and tok != t: raise SyntaxError(f"{t} 를 기대했는데 {tok}")
        self.i += 1
        return tok

    def expr(self):                       # expr ::= term (('+'|'-') term)*
        v = self.term()
        while self.peek() in ("+", "-"):
            op = self.eat()
            v = v + self.term() if op == "+" else v - self.term()
        return v

    def term(self):                       # term ::= factor (('*'|'/') factor)*
        v = self.factor()
        while self.peek() in ("*", "/"):
            op = self.eat()
            v = v * self.factor() if op == "*" else v / self.factor()
        return v

    def factor(self):                     # factor ::= NUMBER | '(' expr ')'
        if self.peek() == "(":
            self.eat("("); v = self.expr(); self.eat(")")
            return v
        return float(self.eat())

상환 분석 — 가끔 비싼 일을 어떻게 세는가

list.append() 는 대부분 O(1) 인데 가끔 O(n) 입니다. 그런데도 “평균 O(1)” 이라고 말하는 근거가 상환 분석이고, 이 개념이 지연 스파이크를 설명해 줍니다.

동적 배열이 두 배씩 늘리는 이유

용량이 꽉 차면 2배로 늘리고 전부 복사한다

append 1~4    : 각 O(1)
append 5      : 복사 4개  → O(n)
append 6~8    : 각 O(1)
append 9      : 복사 8개  → O(n)
...

n 번 append 하는 동안의 총 복사 횟수
  1 + 2 + 4 + 8 + ... + n/2 + n  <  2n

→ 총 비용 O(n) / n 번 = 평균 O(1)   ★ 상환 O(1)

★ 만약 '한 칸씩' 늘린다면?
  1 + 2 + 3 + ... + n = n²/2  → 평균 O(n)  ← 훨씬 나쁘다
배수가 왜 2 인가 — 1.5배를 쓰는 구현도 많습니다(C++ 일부 · Java). 2배는 이전에 반납한 공간을 재사용할 수 없고, 1.5배는 복사가 더 잦습니다. 둘 다 상환 O(1) 입니다.

방법아이디어
총합법n 번의 총비용 ÷ n
회계법싼 연산에 미리 요금을 더 받아 비싼 연산 때 씀
포텐셜법자료구조의 “긴장도”를 함수로 정의
회계법으로 본 append
  · append 한 번에 3 코인을 청구한다
    ① 1 코인: 실제 삽입
    ② 1 코인: 나중에 자기를 복사할 때 쓸 저금
    ③ 1 코인: 이미 복사된 앞쪽 원소를 위한 저금
  · 확장이 일어날 때 저금이 항상 충분하다
  → 연산당 상수(3) → 상환 O(1) ∎

from ortools.sat.python import cp_model

workers, days, shifts = range(5), range(7), range(3)
m = cp_model.CpModel()

# 변수: x[w][d][s] = 1 이면 근무자 w 가 d 일 s 교대에 근무
x = {(w,d,s): m.NewBoolVar(f"x{w}_{d}_{s}")
     for w in workers for d in days for s in shifts}

# 제약 ① 각 교대에는 정확히 한 명
for d in days:
    for s in shifts:
        m.AddExactlyOne(x[w,d,s] for w in workers)

# 제약 ② 한 사람은 하루에 최대 한 교대
for w in workers:
    for d in days:
        m.AddAtMostOne(x[w,d,s] for s in shifts)

# 제약 ③ 야간(2) 다음 날 오전(0) 금지
for w in workers:
    for d in days[:-1]:
        m.Add(x[w,d,2] + x[w,d+1,0] <= 1)

# 목표: 근무 수를 최대한 고르게
loads = [sum(x[w,d,s] for d in days for s in shifts) for w in workers]
mx, mn = m.NewIntVar(0, 21, "mx"), m.NewIntVar(0, 21, "mn")
m.AddMaxEquality(mx, loads); m.AddMinEquality(mn, loads)
m.Minimize(mx - mn)

solver = cp_model.CpSolver()
solver.parameters.max_time_in_seconds = 10.0     # ★ 시간 예산
status = solver.Solve(m)

  4. 솔버로 먼저 시도 — 시간 예산을 두고 돌려 보면 “현실 데이터에서는 금방 풀리는” 경우가 흔합니다.

# 실무에서 자주 쓰는 조합: 탐욕으로 시작해 지역 탐색으로 개선
def local_search(solution, cost, neighbors, max_iter=10_000):
    best, best_cost = solution, cost(solution)
    for _ in range(max_iter):
        improved = False
        for cand in neighbors(best):        # 예: 두 원소 교환
            c = cost(cand)
            if c < best_cost:
                best, best_cost, improved = cand, c, True
                break
        if not improved: break              # 지역 최적에 도달
    return best, best_cost

#  · 재시작 (여러 초기해)

#  · 타부 리스트 (최근 이동 금지)
