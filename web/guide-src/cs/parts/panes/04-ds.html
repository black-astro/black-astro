DS 01

배열 vs 연결 리스트 — 캐시가 답을 바꿉니다

교과서는 “중간 삽입이 잦으면 연결 리스트”라고 합니다. 현실에서는 대부분 배열이 이깁니다. 이유는 알고리즘이 아니라 메모리가 어떻게 생겼는가에 있습니다.

복잡도는 같아도 메모리에 어떻게 놓여 있느냐가 실제 속도를 10배까지 가릅니다.

“중간 삽입 O(1)” 의 함정 — 그 위치를 이미 알고 있을 때만 입니다. 보통은 찾아가야 하므로 결국 O(n) 이고, 그 탐색이 캐시 미스를 계속 일으키는 형태입니다.

LRU 캐시 — 해시 + 이중 연결 리스트

class LRU:
    """해시맵으로 O(1) 조회 + 연결 리스트로 O(1) 순서 갱신"""
    def __init__(self, cap):
        self.cap = cap
        self.map = {}                       # key -> node
        self.head = Node(None, None)        # 더미 (최근)
        self.tail = Node(None, None)        # 더미 (오래됨)
        self.head.nxt, self.tail.prv = self.tail, self.head

    def _remove(self, n):
        n.prv.nxt, n.nxt.prv = n.nxt, n.prv     # ★ O(1) — 배열이면 O(n)

    def _push_front(self, n):
        n.nxt, n.prv = self.head.nxt, self.head
        self.head.nxt.prv = n; self.head.nxt = n

    def get(self, k):
        if k not in self.map: return None
        n = self.map[k]; self._remove(n); self._push_front(n)
        return n.val

    def put(self, k, v):
        if k in self.map: self._remove(self.map[k])
        n = Node(k, v); self.map[k] = n; self._push_front(n)
        if len(self.map) > self.cap:
            old = self.tail.prv
            self._remove(old); del self.map[old.key]

class RingBuffer:
    """최근 N 개만 유지 — 로그 · 메트릭 · 오디오 버퍼에 쓴다"""
    def __init__(self, cap):
        self.buf = [None] * cap
        self.cap = cap
        self.head = 0          # 다음에 쓸 위치
        self.size = 0

    def push(self, x):
        self.buf[self.head] = x
        self.head = (self.head + 1) % self.cap      # ★ 모듈러로 순환
        self.size = min(self.size + 1, self.cap)

    def __iter__(self):
        start = (self.head - self.size) % self.cap
        for i in range(self.size):
            yield self.buf[(start + i) % self.cap]

PAIRS = {")": "(", "]": "[", "}": "{"}

def balanced(s):
    stack = []
    for i, ch in enumerate(s):
        if ch in "([{":
            stack.append((ch, i))
        elif ch in PAIRS:
            if not stack or stack[-1][0] != PAIRS[ch]:
                return f"{i}번째 '{ch}' 가 짝이 안 맞음"
            stack.pop()
    if stack:
        ch, i = stack[-1]
        return f"{i}번째 '{ch}' 가 닫히지 않음"
    return None

balanced("a(b[c]{d})e")     # None
balanced("a(b[c)d]")        # "5번째 ')' 가 짝이 안 맞음"

덱 실전 — 슬라이딩 윈도우 최댓값

from collections import deque

def window_max(nums, k):
    """모든 크기 k 윈도우의 최댓값을 O(n) 에 구한다"""
    dq = deque()          # 값이 감소하는 인덱스만 유지
    out = []
    for i, x in enumerate(nums):
        while dq and nums[dq[-1]] <= x:
            dq.pop()                  # 나보다 작은 것은 영원히 쓸모없다
        dq.append(i)
        if dq[0] <= i - k:
            dq.popleft()              # 윈도우 밖으로 나감
        if i >= k - 1:
            out.append(nums[dq[0]])
    return out

window_max([1,3,-1,-3,5,3,6,7], 3)    # [3,3,5,5,6,7]

# 단순하게 하면 O(n·k) — 이 기법으로 O(n)
“단조 덱”이라 부르는 기법입니다. 지표 모니터링의 이동 최댓값, 주식 차트, 스트리밍 집계에서 같은 패턴이 쓰입니다.

상환 O(1) 의 좋은 예

class Queue:
    def __init__(self):
        self.inbox, self.outbox = [], []

    def push(self, x):
        self.inbox.append(x)              # 항상 O(1)

    def pop(self):
        if not self.outbox:
            while self.inbox:             # 통째로 뒤집는다 — O(n)
                self.outbox.append(self.inbox.pop())
        return self.outbox.pop()          # 이후로는 O(1)

# 각 원소는 '들어가고 · 옮겨지고 · 나오고' 정확히 3번만 만져진다
# → n 번 연산에 총 3n → 상환 O(1)  (🧮 계산이론 10)

함수형 언어에서 큐를 만드는 표준 방법이고, 불변 자료구조에서도 상환 O(1) 이 나옵니다. 면접 단골이기도 합니다.

class SimpleHashMap:
    def __init__(self, cap=8):
        self.buckets = [[] for _ in range(cap)]
        self.n = 0

    def _idx(self, k): return hash(k) % len(self.buckets)

    def put(self, k, v):
        b = self.buckets[self._idx(k)]
        for i, (kk, _) in enumerate(b):
            if kk == k: b[i] = (k, v); return       # 갱신
        b.append((k, v)); self.n += 1
        if self.n / len(self.buckets) > 0.75:       # ★ 로드 팩터
            self._resize()

    def _resize(self):
        old = self.buckets
        self.buckets = [[] for _ in range(len(old) * 2)]
        self.n = 0
        for b in old:
            for k, v in b: self.put(k, v)           # 전부 재배치 — O(n)

import bisect, hashlib

class ConsistentHash:
    def __init__(self, nodes, vnodes=150):
        self.ring = []          # (해시, 노드) 정렬 리스트
        self.vnodes = vnodes
        for n in nodes: self.add(n)

    def _h(self, s):
        return int(hashlib.blake2b(s.encode(), digest_size=8).hexdigest(), 16)

    def add(self, node):
        for i in range(self.vnodes):          # ★ 가상 노드로 고르게
            bisect.insort(self.ring, (self._h(f"{node}#{i}"), node))

    def get(self, key):
        h = self._h(key)
        i = bisect.bisect(self.ring, (h,)) % len(self.ring)
        return self.ring[i][1]

# (단순 % N 이면 거의 전부 이동 — 🔢 이산수학 10)

def simhash(tokens, bits=64):
    v = [0] * bits
    for t in tokens:
        h = int(hashlib.blake2b(t.encode(), digest_size=8).hexdigest(), 16)
        for i in range(bits):
            v[i] += 1 if (h >> i) & 1 else -1
    out = 0
    for i, x in enumerate(v):
        if x > 0: out |= 1 << i
    return out

def hamming(a, b): return bin(a ^ b).count("1")

class Node:
    __slots__ = ("key", "val", "left", "right")
    def __init__(self, k, v):
        self.key, self.val = k, v
        self.left = self.right = None

def insert(node, k, v):
    if node is None: return Node(k, v)
    if k < node.key:   node.left  = insert(node.left, k, v)
    elif k > node.key: node.right = insert(node.right, k, v)
    else:              node.val = v
    return node

def search(node, k):
    while node:
        if k == node.key: return node.val
        node = node.left if k < node.key else node.right
    return None

def inorder(node):                  # ★ 중위 순회 = 정렬된 순서
    if node:
        yield from inorder(node.left)
        yield node.key, node.val
        yield from inorder(node.right)

def delete(node, k):
    if node is None: return None
    if k < node.key:   node.left  = delete(node.left, k)
    elif k > node.key: node.right = delete(node.right, k)
    else:
        # ① 자식이 없거나 하나 → 그 자식으로 대체
        if node.left is None:  return node.right
        if node.right is None: return node.left
        # ② 자식이 둘 → 오른쪽 서브트리의 최솟값(중위 후계자)으로 대체
        succ = node.right
        while succ.left: succ = succ.left
        node.key, node.val = succ.key, succ.val
        node.right = delete(node.right, succ.key)
    return node

# 왜 '오른쪽의 최솟값' 인가
#  · 그 값보다 왼쪽은 전부 작고, 오른쪽은 전부 크다 → BST 성질 유지

import random

class SkipList:
    MAXLEVEL, P = 16, 0.5
    def __init__(self):
        self.head = [None] * self.MAXLEVEL
        self.level = 1

    def _rand_level(self):
        lv = 1
        while random.random() < self.P and lv < self.MAXLEVEL: lv += 1
        return lv

    def search(self, key):
        node, cur = None, self.head
        for i in reversed(range(self.level)):
            while cur[i] and cur[i].key < key:
                node, cur = cur[i], cur[i].nxt
        nxt = cur[0]
        return nxt.val if nxt and nxt.key == key else None

B-트리 · B+트리 — 모든 DB 인덱스의 정체

디스크는 바이트 하나를 읽어도 페이지 하나(4~16KB)를 읽습니다. 그 사정에 맞춰 노드를 크고 납작하게 만든 트리가 B-트리이고, 그래서 모든 관계형 DB 의 인덱스가 이 구조입니다.

왜 이진 트리가 아닌가

1억 건 인덱스

이진 트리         : 높이 log₂(1억) ≈ 27
                   → 디스크 접근 27번 × 0.1ms = 2.7ms

★ 핵심: 한 번 읽을 때 '많이' 읽는다

  · 키가 16바이트면 한 노드에 500개

import heapq

def top_k_stream(stream, k):
    h = []                                  # 최소 힙 (가장 작은 것이 위)
    for item in stream:
        if len(h) < k:
            heapq.heappush(h, item)
        elif item > h[0]:                   # 현재 k등보다 크면
            heapq.heapreplace(h, item)      # 교체 (pop+push 보다 빠름)
    return sorted(h, reverse=True)

class Trie:
    def __init__(self):
        self.root = {}
        self.END = "$"

    def insert(self, word):
        node = self.root
        for ch in word:
            node = node.setdefault(ch, {})
        node[self.END] = True

    def search(self, word):
        node = self._walk(word)
        return node is not None and self.END in node

    def starts_with(self, prefix):
        return self._walk(prefix) is not None

    def _walk(self, s):
        node = self.root
        for ch in s:
            if ch not in node: return None
            node = node[ch]
        return node

    def complete(self, prefix, limit=10):
        node = self._walk(prefix)
        if node is None: return []
        out = []
        def dfs(n, acc):
            if len(out) >= limit: return
            if self.END in n: out.append(prefix + acc)
            for ch, child in n.items():
                if ch != self.END: dfs(child, acc + ch)
        dfs(node, "")
        return out

class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.count = n                 # 그룹 개수

    def find(self, x):
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]   # ★ 경로 압축
            x = self.parent[x]
        return x

    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb: return False                          # 이미 같은 그룹
        if self.rank[ra] < self.rank[rb]: ra, rb = rb, ra  # ★ 랭크로 합치기
        self.parent[rb] = ra
        if self.rank[ra] == self.rank[rb]: self.rank[ra] += 1
        self.count -= 1
        return True

    def same(self, a, b): return self.find(a) == self.find(b)

def has_cycle(n, edges):
    uf = UnionFind(n)
    for u, v in edges:
        if not uf.union(u, v):   # 이미 같은 그룹인데 또 연결
            return True          # → 사이클
    return False

# 크루스칼 MST (🔢 이산수학 08)
def kruskal(n, edges):
    uf = UnionFind(n)
    total, used = 0, []
    for w, u, v in sorted(edges):
        if uf.union(u, v):
            total += w; used.append((u, v, w))
    return used, total

# 원소 나열이 필요하면 별도 자료구조를 함께 유지한다
class UnionFindWithMembers(UnionFind):
    def __init__(self, n):
        super().__init__(n)
        self.members = {i: {i} for i in range(n)}

    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb: return False
        if len(self.members[ra]) < len(self.members[rb]): ra, rb = rb, ra
        super().union(ra, rb)
        root = self.find(ra)
        other = rb if root == ra else ra
        self.members[root] |= self.members.pop(other)     # 작은 쪽을 큰 쪽에
        return True

# ★ '작은 쪽을 큰 쪽에 합치기' 를 지키면 총 병합 비용이 O(n log n)
#   (각 원소는 자기가 속한 집합 크기가 2배 될 때만 옮겨진다)
“작은 쪽을 큰 쪽에”는 small-to-large 라 부르는 일반적인 기법입니다. 트리 병합 · 세그먼트 트리 병합 · 맵 합치기에서 반복해 나옵니다. 단순한 규칙 하나로 O(n²) 이 O(n log n) 이 됩니다.

import math, hashlib

class BloomFilter:
    def __init__(self, n, fp_rate=0.01):
        self.m = math.ceil(-n * math.log(fp_rate) / (math.log(2) ** 2))
        self.k = max(1, round(self.m / n * math.log(2)))
        self.bits = bytearray((self.m + 7) // 8)

    def _idx(self, item, i):
        h = hashlib.blake2b(f"{i}:{item}".encode(), digest_size=8).digest()
        return int.from_bytes(h, "big") % self.m

    def add(self, item):
        for i in range(self.k):
            p = self._idx(item, i)
            self.bits[p // 8] |= 1 << (p % 8)

    def __contains__(self, item):
        return all(self.bits[(p := self._idx(item, i)) // 8] & (1 << (p % 8))
                   for i in range(self.k))

bf = BloomFilter(1_000_000, 0.01)

class CountMinSketch:
    def __init__(self, w=2048, d=5):
        self.w, self.d = w, d
        self.t = [[0] * w for _ in range(d)]

    def _h(self, item, i):
        return int(hashlib.blake2b(f"{i}:{item}".encode(),
                                   digest_size=8).hexdigest(), 16) % self.w

    def add(self, item, n=1):
        for i in range(self.d):
            self.t[i][self._h(item, i)] += n

    def count(self, item):
        return min(self.t[i][self._h(item, i)] for i in range(self.d))

class SegTree:
    """구간 합. op 를 바꾸면 min·max·gcd 로 그대로 씁니다."""

    def __init__(self, arr, op=lambda a, b: a + b, e=0):
        self.n, self.op, self.e = len(arr), op, e
        self.t = [e] * (2 * self.n)          # 크기 2n 이면 충분합니다
        self.t[self.n:] = arr                # 뒤 절반이 잎
        for i in range(self.n - 1, 0, -1):   # 부모는 자식 둘을 합친 값
            self.t[i] = op(self.t[2*i], self.t[2*i+1])

    def update(self, i, v):                  # i 번째를 v 로 — O(log n)
        i += self.n
        self.t[i] = v
        while i > 1:                         # 잎에서 뿌리까지 올라가며 갱신
            i //= 2
            self.t[i] = self.op(self.t[2*i], self.t[2*i+1])

    def query(self, l, r):                   # [l, r) 구간 — O(log n)
        res_l = res_r = self.e
        l += self.n; r += self.n
        while l < r:
            if l & 1:                        # l 이 오른쪽 자식이면 자기 것만 쓰고 이동
                res_l = self.op(res_l, self.t[l]); l += 1
            if r & 1:                        # r 도 마찬가지 (오른쪽에서 왼쪽으로)
                r -= 1; res_r = self.op(self.t[r], res_r)
            l //= 2; r //= 2
        return self.op(res_l, res_r)


st = SegTree([1,2,3,4,5,6,7,8])
print(st.query(2, 6))      # 3+4+5+6 = 18
st.update(3, 40)           # 4 → 40

class BIT:
    """구간 합 전용. 세그먼트 트리보다 짧고 빠르지만 min·max 는 못 합니다."""

    def __init__(self, n):
        self.n = n
        self.t = [0] * (n + 1)               # 1-based

    def add(self, i, delta):                 # i 번째에 delta 더하기 — O(log n)
        i += 1
        while i <= self.n:
            self.t[i] += delta
            i += i & (-i)                    # ★ 다음 담당 구간으로
    def _sum(self, i):                       # [0..i] 합
        s, i = 0, i + 1
        while i > 0:
            s += self.t[i]
            i -= i & (-i)                    # ★ 부모 구간으로
        return s

    def query(self, l, r):                   # [l, r] 합
        return self._sum(r) - self._sum(l - 1) if l else self._sum(r)
핵심은 i & (−i) — 가장 낮은 1 비트를 뽑는 연산 12 = 1100₂  →  12 & (−12) = 100₂ = 4

def update(self, node, s, e, l, r, add):
    self._push(node, s, e)                   # 밀린 갱신 먼저 반영
    if r < s or e < l:                       # 겹치지 않음
        return
    if l <= s and e <= r:                    # ★ 완전히 덮임 → 여기서 멈춤
        self.lazy[node] += add
        self._push(node, s, e)
        return
    mid = (s + e) // 2
    self.update(node*2,   s, mid,   l, r, add)
    self.update(node*2+1, mid+1, e, l, r, add)
    self.tree[node] = self.tree[node*2] + self.tree[node*2+1]

def _push(self, node, s, e):
    if self.lazy[node] == 0: return
    self.tree[node] += self.lazy[node] * (e - s + 1)   # 구간 길이만큼
    if s != e:                                          # 잎이 아니면 자식에게 예약
        self.lazy[node*2]   += self.lazy[node]
        self.lazy[node*2+1] += self.lazy[node]
    self.lazy[node] = 0

균형 트리(07절)는 확실하지만 회전 규칙이 복잡하고 동시성에 약합니다. 스킵 리스트는 그 규칙을 통째로 동전 던지기로 대신해 같은 O(log n) 을 얻습니다 — Redis 의 정렬 집합과 LevelDB 의 memtable 이 이것입니다. 지속(persistent) 자료구조는 수정할 때 통째로 복사하지 않고 바뀐 경로만 새로 만드는 방식이고, Git · React · 실행 취소 · MVCC 가 전부 여기에서 나옵니다.

두 주제를 한 절에 묶은 이유 — 둘 다 “정직하게 하면 비싼 일”을 다른 각도로 피하는 기법입니다. 한쪽은 균형 유지 비용을 확률로, 다른 한쪽은 복사 비용을 공유로 없앱니다.

import random

MAX_L = 16

class Node:
    __slots__ = ("key", "next")
    def __init__(self, key, lv):
        self.key = key
        self.next = [None] * lv        # 층마다 다음 노드

class SkipList:
    def __init__(self):
        self.head = Node(None, MAX_L)
        self.level = 1

    def _rand_level(self):
        lv = 1
        while random.random() < 0.5 and lv < MAX_L:   # 동전 던지기
            lv += 1
        return lv

    def _find(self, key):
        """각 층에서 key 직전 노드를 모아 둡니다."""
        prev, cur = [self.head] * MAX_L, self.head
        for i in range(self.level - 1, -1, -1):
            while cur.next[i] and cur.next[i].key < key:
                cur = cur.next[i]          # 같은 층에서 전진
            prev[i] = cur                  # 지나쳤으니 한 층 내려감
        return prev

    def add(self, key):
        prev = self._find(key)
        lv = self._rand_level()
        self.level = max(self.level, lv)
        node = Node(key, lv)
        for i in range(lv):                # 층마다 끼워 넣기
            node.next[i] = prev[i].next[i]
            prev[i].next[i] = node

    def has(self, key):
        n = self._find(key)[0].next[0]
        return n is not None and n.key == key

    def range(self, lo, hi):               # 구간 질의가 자연스럽습니다
        cur = self._find(lo)[0].next[0]
        while cur and cur.key <= hi:
            yield cur.key
            cur = cur.next[0]
높이가 log n 인 이유 층 i 이상에 올라갈 확률 = (1/2)i−1
→ 층 i 의 노드 수 기댓값 = n/2i−1

그 값이 1 이 되는 i = log2 n + 1
한 층에서 옆으로 가는 횟수의 기댓값은 2 로 상수

→ 탐색 기댓값 = 2 · log2 n = O(log n)
전체 노드 수 기댓값 = n(1 + ½ + ¼ + …) = 2n

from typing import NamedTuple, Optional

class Node(NamedTuple):
    key: int
    left: Optional["Node"] = None
    right: Optional["Node"] = None

def insert(node, key):
    """새 트리를 돌려줍니다. 원본은 절대 안 건드립니다."""
    if node is None:
        return Node(key)
    if key < node.key:                       # 왼쪽만 새로 만들고
        return node._replace(left=insert(node.left, key))
    if key > node.key:                       # 오른쪽 가지는 그대로 공유
        return node._replace(right=insert(node.right, key))
    return node

v1 = None
for k in (50, 30, 70, 20, 40):
    v1 = insert(v1, k)

v2 = insert(v1, 35)      # 새로 만든 노드는 경로상 3개뿐

print(v1.left.right.left)          # None  — v1 은 그대로
print(v2.left.right.left.key)      # 35    — v2 에만 있음
