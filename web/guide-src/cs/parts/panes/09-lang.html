어휘 분석 — 글자를 낱말로 자르기

x=1+2 라는 글자 나열을 [x][=][1][+][2] 라는 토큰으로 자르는 단계입니다. 로그 파서 · 설정 파서 · DSL 을 만들 때 그대로 쓰입니다.

렉서가 하는 일

입력:  let x = 42 + y;  // 주석

토큰:  KEYWORD(let)
       IDENT(x)
       OP(=)
       NUMBER(42)
       OP(+)
       IDENT(y)
       SEMI(;)
       (주석과 공백은 버린다)

규칙
 · 최장 일치 (maximal munch)
   "==" 를 만나면 "=" 두 개가 아니라 "==" 하나로
 · 키워드는 식별자 규칙에 먼저 걸리므로
   식별자를 뽑은 뒤 예약어 표에서 확인한다
 · 위치 정보(줄·열)를 함께 담는다 → 에러 메시지에 필요

직접 만들기 — 30줄

import re
from dataclasses import dataclass

@dataclass
class Token:
    kind: str; value: str; line: int; col: int

SPEC = [
    ("NUMBER",  r"\d+(\.\d+)?"),
    ("IDENT",   r"[A-Za-z_]\w*"),
    ("STRING",  r'"[^"\n]*"'),
    ("OP",      r"[+\-*/=<>!]=?|&&|\|\|"),
    ("PUNCT",   r"[(){}\[\],;]"),
    ("SKIP",    r"[ \t]+"),
    ("COMMENT", r"//[^\n]*"),
    ("NEWLINE", r"\n"),
]
MASTER = re.compile("|".join(f"(?P<{k}>{v})" for k, v in SPEC))
KEYWORDS = {"let", "if", "else", "while", "return"}

def tokenize(src):
    line = col_base = 1
    for m in MASTER.finditer(src):
        kind, value = m.lastgroup, m.group()
        if kind == "NEWLINE":
            line += 1; col_base = m.end() + 1; continue
        if kind in ("SKIP", "COMMENT"): continue
        if kind == "IDENT" and value in KEYWORDS: kind = "KEYWORD"
        yield Token(kind, value, line, m.start() - col_base + 2)

list(tokenize("let x = 42 + y;"))

# 상태를 가진 렉서 — 문자열 안의 이스케이프 처리
def lex_string(src, i):
    """i 는 여는 따옴표 위치"""
    out, i = [], i + 1
    while i < len(src):
        c = src[i]
        if c == "\\":
            nxt = src[i+1]
            out.append({"n": "\n", "t": "\t", '"': '"', "\\": "\\"}.get(nxt, nxt))
            i += 2
        elif c == '"':
            return "".join(out), i + 1
        elif c == "\n":
            raise SyntaxError("문자열이 닫히지 않음")
        else:
            out.append(c); i += 1
    raise SyntaxError("문자열이 닫히지 않음")

def error(src, tok, message, hint=None):
    lines = src.splitlines()
    line = lines[tok.line - 1]
    caret = " " * (tok.col - 1) + "^" * max(1, len(tok.value))
    parts = [
        f"error: {message}",
        f"  --> {tok.line}:{tok.col}",
        f"   |",
        f"{tok.line:>3} | {line}",
        f"   | {caret}",
    ]
    if hint: parts.append(f"   = 힌트: {hint}")
    raise SyntaxError("\n".join(parts))

# Pratt 파서 — 우선순위를 표로 관리
PREC = {"||":1, "&&":2, "==":3, "<":4, "+":5, "-":5, "*":6, "/":6}

def parse_expr(toks, min_prec=0):
    left = parse_atom(toks)
    while toks.peek() and toks.peek().value in PREC:
        op = toks.peek().value
        prec = PREC[op]
        if prec < min_prec: break
        toks.next()
        right = parse_expr(toks, prec + 1)      # 좌결합
        left = {"op": op, "left": left, "right": right}
    return left

import ast, pathlib

class DangerFinder(ast.NodeVisitor):
    DANGEROUS = {"eval", "exec", "pickle.loads", "os.system"}

    def __init__(self, path):
        self.path, self.hits = path, []

    def visit_Call(self, node):
        name = ast.unparse(node.func)
        if name in self.DANGEROUS:
            self.hits.append((self.path, node.lineno, name))
        self.generic_visit(node)

for p in pathlib.Path("src").rglob("*.py"):

    f.visit(ast.parse(p.read_text()))
    for path, line, name in f.hits:

class Scope:
    def __init__(self, parent=None):
        self.symbols = {}
        self.parent = parent

    def declare(self, name, info):
        if name in self.symbols:
            raise NameError(f"'{name}' 이 이미 선언됨")
        self.symbols[name] = info

    def lookup(self, name):
        s = self
        while s:                              # ★ 안쪽부터 바깥으로
            if name in s.symbols: return s.symbols[name]
            s = s.parent
        raise NameError(f"'{name}' 을 찾을 수 없음")

# 블록마다 새 스코프
# 이 '체인' 이 클로저와 변수 섀도잉의 근거다
이 구조가 클로저의 정체입니다 — 함수가 만들어질 때 자기 바깥 스코프에 대한 참조를 들고 다닙니다. “왜 반복문 안에서 만든 함수들이 같은 값을 보나”가 여기서 설명됩니다(var vs let).

// 타입 좁히기(narrowing) — 의미 분석기가 흐름을 따라간다
function f(x: string | number | null) {
  if (x === null) return "없음";
  // 여기서 x 는 string | number
  if (typeof x === "string") {
    return x.toUpperCase();      // x 는 string
  }
  return x.toFixed(2);           // x 는 number
}

// 판별 유니온 — 가장 안전한 패턴
type Result =
  | { ok: true;  value: number }
  | { ok: false; error: string };

function handle(r: Result) {
  if (r.ok) return r.value;      // value 에 접근 가능
  return r.error;                // error 에 접근 가능
}

// 사용자 정의 타입 가드
function isUser(x: unknown): x is User {
  return typeof x === "object" && x !== null && "id" in x;
}

// 망라성 검사 — 케이스를 빠뜨리면 컴파일 에러
function area(s: Shape): number {
  switch (s.kind) {
    case "circle": return Math.PI * s.r ** 2;
    case "rect":   return s.w * s.h;
    default:
      const _exhaustive: never = s;   // ★ 새 도형을 추가하면 여기서 잡힌다
      throw new Error("unreachable");
  }
}
never 를 이용한 망라성 검사는 실무에서 아주 유용합니다 — 새로운 케이스를 추가했을 때 처리를 빠뜨린 곳을 컴파일러가 전부 찾아 줍니다. Rust 의 match, Kotlin 의 when 도 같은 보장을 제공합니다.

중간 표현 · 최적화 — -O2 가 하는 일

컴파일러는 소스도 기계어도 아닌 중간 언어로 한번 옮긴 뒤 최적화합니다. 그 덕분에 언어 N개 × 아키텍처 M개 를 N + M 으로 줄일 수 있습니다.

LLVM 구조

프런트엔드는 IR 만 만들면 되고
백엔드는 IR 만 읽으면 된다

SSA (Static Single Assignment)
  각 변수에 딱 한 번만 대입한다
    x = 1;  x = 2;  y = x;
    →  x1 = 1;  x2 = 2;  y1 = x2;
  · "이 값이 어디서 왔나" 가 자명해진다
  · 데이터 흐름 분석이 훨씬 쉬워진다
  · 거의 모든 현대 컴파일러가 SSA 를 쓴다

주요 최적화

최적화내용

상수 전파변수를 아는 값으로 치환

LANG 06

코드 생성 · 레지스터 할당 — 컴파일러의 마지막 큰 결정

IR 을 실제 기계어로 옮기는 단계입니다. 핵심은 “어떤 값을 레지스터에 둘 것인가” — 레지스터는 수십 개뿐인데 변수는 훨씬 많습니다.

그래프 색칠 문제

레지스터 할당 = 그래프 색칠 (🧮 계산이론 11 — NP-완전)

 · 정점 = 변수
 · 간선 = "동시에 살아 있다" (간섭)
 · 색   = 레지스터
 · 같은 색을 인접 정점에 못 준다

레지스터가 부족하면 → 스필(spill)
 · 일부 변수를 스택(메모리)에 저장
 · 필요할 때 다시 로드
 · ★ 느려진다

그래서 실무에서
 · 함수가 너무 크면 스필이 늘어난다
 · 반복문 안에서 변수를 많이 쓰면 마찬가지
 · 인라인을 과하게 하면 레지스터 압박이 커진다

코드 생성이 만드는 차이

// 같은 의미, 다른 코드
// ① 조건 분기
int max_a(int a, int b) {
    if (a > b) return a;
    return b;
}
// → cmp + jump (분기 예측 실패 가능)

// ② 조건부 이동

}
// → cmp + cmov (분기 없음)

// ★ 현대 컴파일러는 대개 알아서 cmov 를 쓴다.

//   예측이 잘 되는 분기는 분기가 더 빠르다.

링커와 로더 — undefined reference의 정체

여러 조각을 하나로 붙이는 단계입니다. C/C++ 개발자가 가장 자주 막히는 곳이고, 동적 라이브러리 문제는 다른 언어에서도 그대로 만납니다.

링커가 하는 일

① 심볼 해석
   각 .o 파일의 "정의되지 않은 심볼(U)" 을
   다른 .o 나 라이브러리의 "정의(T)" 와 연결

② 재배치
   심볼의 최종 주소가 정해지면 참조하는 곳을 전부 수정

③ 섹션 병합
   .text(코드) · .data · .bss · .rodata 를 모아서 배치

확인
  nm main.o | grep " U "     # 필요로 하는 심볼
  nm libfoo.a | grep " T "   # 제공하는 심볼
  nm -C ...                  # C++ 이름 디맹글링

JIT 컴파일 — 실행 중에 다시 컴파일

자주 도는 부분만 골라 기계어로 바꿉니다. 실행 중이라 실제 타입과 분기 통계를 알 수 있어 정적 컴파일러보다 유리한 경우도 있습니다.

동작 흐름

① 처음에는 인터프리터로 실행 (시작이 빠름)
② 호출 횟수·반복 횟수를 센다
③ 임계값을 넘으면 "뜨겁다(hot)" 고 판단
④ 그 메서드를 기계어로 컴파일
⑤ 실행 중 수집한 정보로 공격적으로 최적화
   · 이 호출은 항상 같은 타입 → 타입 검사 생략 + 인라인
   · 이 분기는 항상 참 → 반대쪽 제거
⑥ 가정이 깨지면 역최적화(deoptimization) → 인터프리터로 복귀

★ ⑤ 가 정적 컴파일러가 못 하는 일이다

가비지 컬렉션 — 멈춤 시간이 어디서 오나

“아무도 안 가리키는 메모리”를 찾아 회수합니다. 방식에 따라 멈추는 시간과 처리량이 달라지고, 그것이 p99 지연에 직접 나타납니다.

타입을 안 적었는데 컴파일러가 알아내는 방법

  let f = fun x -> x + 1

 ① 미지수를 둔다: x : α, f : α → β
 ② 제약을 모은다: + 는 int → int → int 이므로 α = int, β = int
 ③ 단일화(unification)로 제약을 푼다
 ④ 결과: f : int → int

일반화(generalization)
  let id = fun x -> x
  제약이 없으므로 α 가 자유롭다 → id : ∀α. α → α (제네릭!)

import re
from dataclasses import dataclass

# ───────── 1. 렉서 ─────────
TOKEN = re.compile(r"""
    (?P<NUM>\d+(?:\.\d+)?)
  | (?P<ID>[A-Za-z_]\w*)
  | (?P<OP><=|>=|==|!=|[-+*/<>=])
  | (?P<PUNCT>[(){};,])
  | (?P<WS>\s+)
  | (?P<COMMENT>\#[^\n]*)
""", re.VERBOSE)
KEYWORDS = {"let", "if", "else", "while", "print"}

def tokenize(src):
    pos, out = 0, []
    while pos < len(src):
        m = TOKEN.match(src, pos)
        if not m: raise SyntaxError(f"알 수 없는 문자: {src[pos]!r} (위치 {pos})")
        pos = m.end()
        kind, val = m.lastgroup, m.group()
        if kind in ("WS", "COMMENT"): continue
        if kind == "ID" and val in KEYWORDS: kind = "KW"
        out.append((kind, val))
    out.append(("EOF", ""))
    return out

# ───────── 2. 파서 (재귀 하강) ─────────
class Parser:
    def __init__(self, toks): self.t, self.i = toks, 0
    def peek(self):  return self.t[self.i]
    def next(self):  self.i += 1; return self.t[self.i - 1]
    def accept(self, val):
        if self.peek()[1] == val: return self.next()
        return None
    def expect(self, val):
        if not self.accept(val):
            raise SyntaxError(f"'{val}' 를 기대했는데 {self.peek()[1]!r}")

    def program(self):
        stmts = []
        while self.peek()[0] != "EOF": stmts.append(self.statement())
        return ("block", stmts)

    def statement(self):
        if self.accept("let"):
            name = self.next()[1]; self.expect("="); e = self.expr(); self.expect(";")
            return ("let", name, e)
        if self.accept("print"):
            e = self.expr(); self.expect(";"); return ("print", e)
        if self.accept("if"):
            self.expect("("); c = self.expr(); self.expect(")")
            then = self.block()
            els = self.block() if self.accept("else") else None
            return ("if", c, then, els)
        if self.accept("while"):
            self.expect("("); c = self.expr(); self.expect(")")
            return ("while", c, self.block())
        name = self.next()[1]; self.expect("="); e = self.expr(); self.expect(";")
        return ("assign", name, e)

    def block(self):
        self.expect("{"); stmts = []
        while not self.accept("}"): stmts.append(self.statement())
        return ("block", stmts)

    # 우선순위: 비교 < 덧셈 < 곱셈 < 단항
    def expr(self):   return self.binary(self.add,  ("<", ">", "<=", ">=", "==", "!="))
    def add(self):    return self.binary(self.mul,  ("+", "-"))
    def mul(self):    return self.binary(self.unary, ("*", "/"))
    def binary(self, sub, ops):
        left = sub()
        while self.peek()[1] in ops:
            op = self.next()[1]
            left = ("bin", op, left, sub())
        return left
    def unary(self):
        if self.accept("-"): return ("neg", self.unary())
        return self.atom()
    def atom(self):
        kind, val = self.next()
        if kind == "NUM": return ("num", float(val))
        if kind == "ID":  return ("var", val)
        if val == "(":
            e = self.expr(); self.expect(")"); return e
        raise SyntaxError(f"예상치 못한 토큰: {val!r}")

# ───────── 3. 평가기 ─────────
import operator
BINOPS = {"+": operator.add, "-": operator.sub, "*": operator.mul,
          "/": operator.truediv, "<": operator.lt, ">": operator.gt,
          "<=": operator.le, ">=": operator.ge,
          "==": operator.eq, "!=": operator.ne}

def evaluate(node, env, out):
    kind = node[0]
    if kind == "block":
        for s in node[1]: evaluate(s, env, out)
    elif kind == "let" or kind == "assign":
        env[node[1]] = evaluate(node[2], env, out)
    elif kind == "print":
        out.append(evaluate(node[1], env, out))
    elif kind == "if":
        if evaluate(node[1], env, out): evaluate(node[2], env, out)
        elif node[3]: evaluate(node[3], env, out)
    elif kind == "while":
        guard = 0
        while evaluate(node[1], env, out):
            evaluate(node[2], env, out)
            guard += 1
            if guard > 1_000_000: raise RuntimeError("반복 한도 초과")  # 🧮 계산이론 06
    elif kind == "bin":
        return BINOPS[node[1]](evaluate(node[2], env, out), evaluate(node[3], env, out))
    elif kind == "neg":  return -evaluate(node[1], env, out)
    elif kind == "num":  return node[1]
    elif kind == "var":
        if node[1] not in env: raise NameError(f"정의되지 않은 변수: {node[1]}")
        return env[node[1]]

def run(src):
    out = []
    evaluate(Parser(tokenize(src)).program(), {}, out)
    return out

# ───────── 실행 ─────────

        let t = a + b;
        a = b;
        b = t;
        n = n - 1;
